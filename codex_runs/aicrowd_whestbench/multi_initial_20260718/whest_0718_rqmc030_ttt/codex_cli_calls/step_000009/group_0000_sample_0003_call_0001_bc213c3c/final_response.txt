from __future__ import annotations

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


_TARGET_FLOP_FRACTION = 0.88
_TAIL_CLIP = 1e-12
_ARRAY_BYTES_LIMIT = 96 * 1024 * 1024
_FLOAT64_ITEMSIZE = 8
_FULL_CV_WIDTH_LIMIT = 1536
_CHUNK_WIDTH = 256
_PRIME_CACHE = {}


def _first_primes(count: int) -> tuple[int, ...]:
    cached = _PRIME_CACHE.get(count)
    if cached is not None:
        return cached

    primes = []
    candidate = 2
    while len(primes) < count:
        ok = True
        for prime in primes:
            if prime * prime > candidate:
                break
            if candidate % prime == 0:
                ok = False
                break
        if ok:
            primes.append(candidate)
        candidate += 1

    result = tuple(primes)
    _PRIME_CACHE[count] = result
    return result


def _lattice_generator(width: int) -> fnp.ndarray:
    roots = fnp.sqrt(fnp.array(_first_primes(width), dtype=fnp.float64))
    return roots - fnp.floor(roots)


def _lattice_uniforms(n_pairs: int, generator: fnp.ndarray, shift: fnp.ndarray) -> fnp.ndarray:
    indices = fnp.arange(n_pairs, dtype=fnp.float64)[:, None]
    values = indices * generator[None, :] + shift[None, :]
    base = values - fnp.floor(values)
    return fnp.concatenate((base, 1.0 - base), axis=0)


def _first_layer_variance(weight: fnp.ndarray) -> fnp.ndarray:
    return fnp.sum(weight * weight, axis=0)


def _exact_first_layer_mean(weight: fnp.ndarray) -> fnp.ndarray:
    return fnp.sqrt(_first_layer_variance(weight)) / fnp.sqrt(2.0 * fnp.pi)


def _exact_pair_abs_var(weight: fnp.ndarray) -> fnp.ndarray:
    variance = _first_layer_variance(weight)
    return variance * (1.0 - 2.0 / fnp.pi) / 4.0


def _sample_count(budget, width: int, depth: int, use_cv: bool) -> int:
    if budget is None:
        budget = 10**12

    lattice_per_sample = 45 * width
    forward_per_sample = depth * (2 * width * width + width)
    mean_per_sample = max(depth - 1, 0) * width
    cv_per_sample = width * width if use_cv and depth > 1 else 0

    fixed = 2 * width * width + 10 * width
    if use_cv and depth > 1 and width <= _FULL_CV_WIDTH_LIMIT:
        fixed += 4 * width * width * width

    tracked = int(_TARGET_FLOP_FRACTION * int(budget))
    per_sample = lattice_per_sample + forward_per_sample + mean_per_sample + cv_per_sample
    n_samples = (tracked - fixed) // max(per_sample, 1)

    max_samples_by_bytes = _ARRAY_BYTES_LIMIT // max(2 * width * _FLOAT64_ITEMSIZE, 1)
    n_samples = int(max(2, min(n_samples, max_samples_by_bytes)))
    return 2 * max(1, n_samples // 2)


def _full_pair_abs_cov(weight: fnp.ndarray) -> fnp.ndarray:
    gram = weight.T @ weight
    scale = fnp.sqrt(fnp.maximum(fnp.diag(gram), 0.0))
    denom = fnp.maximum(scale[:, None] * scale[None, :], 1e-30)
    rho = fnp.clip(gram / denom, -1.0, 1.0)

    arc = fnp.arccos(rho)
    asin = 0.5 * fnp.pi - arc
    cov = denom * (
        fnp.sqrt(fnp.maximum(1.0 - rho * rho, 0.0)) + rho * asin - 1.0
    ) / (2.0 * fnp.pi)

    ridge = fnp.maximum(fnp.mean(fnp.diag(cov)), 1e-12) * 1e-8
    return cov + fnp.eye(weight.shape[1], dtype=fnp.float64) * ridge


def _control_variate_final(
    first_activations: fnp.ndarray,
    final_activations: fnp.ndarray,
    first_mean_exact: fnp.ndarray,
    first_pair_var_exact: fnp.ndarray,
    first_weight: fnp.ndarray,
) -> fnp.ndarray:
    n_samples = first_activations.shape[0]
    width = first_activations.shape[1]
    n_pairs = n_samples // 2

    first_pair = 0.5 * (first_activations[:n_pairs] + first_activations[n_pairs:])
    final_pair = 0.5 * (final_activations[:n_pairs] + final_activations[n_pairs:])

    first_mean_sample = fnp.mean(first_pair, axis=0)
    final_mean = fnp.mean(final_pair, axis=0)
    delta = first_mean_sample - first_mean_exact

    if width <= _FULL_CV_WIDTH_LIMIT:
        try:
            cross = (first_pair.T @ final_pair) / n_pairs
            cross = cross - first_mean_sample[:, None] * final_mean[None, :]
            cov = _full_pair_abs_cov(first_weight)
            beta = fnp.linalg.solve(cov, cross)
            return fnp.maximum(final_mean - delta @ beta, 0.0)
        except Exception:
            pass

    chunks = []
    denom = fnp.maximum(first_pair_var_exact, 1e-18)
    for start in range(0, width, _CHUNK_WIDTH):
        stop = min(width, start + _CHUNK_WIDTH)
        y = final_pair[:, start:stop]
        y_mean = final_mean[start:stop]
        cross = (first_pair.T @ y) / n_pairs
        cross = cross - first_mean_sample[:, None] * y_mean[None, :]
        beta = cross / denom[:, None]
        chunks.append(fnp.maximum(y_mean - delta @ beta, 0.0))

    return fnp.concatenate(chunks, axis=0)


class Estimator(BaseEstimator):
    def setup(self, context):
        self._setup_seed = context.seed

    def teardown(self):
        pass

    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth

        if depth <= 0:
            return fnp.zeros((0, width), dtype=fnp.float64)

        use_cv = depth > 1
        n_samples = _sample_count(budget, width, depth, use_cv)

        rng = fnp.random.default_rng(mlp.seed)
        generator = _lattice_generator(width)
        shift = rng.random(width)

        uniforms = _lattice_uniforms(n_samples // 2, generator, shift)
        uniforms = fnp.clip(uniforms, _TAIL_CLIP, 1.0 - _TAIL_CLIP)
        activations = flops.stats.norm.ppf(uniforms)

        rows = []
        first_activations = None
        first_mean_exact = None
        first_pair_var_exact = None

        for layer_index, weight in enumerate(mlp.weights):
            activations = fnp.maximum(activations @ weight, 0.0)

            if layer_index == 0:
                first_activations = activations
                first_mean_exact = _exact_first_layer_mean(weight)
                first_pair_var_exact = _exact_pair_abs_var(weight)
                rows.append(first_mean_exact)
            else:
                rows.append(fnp.mean(activations, axis=0))

        if use_cv and first_activations is not None:
            rows[-1] = _control_variate_final(
                first_activations,
                activations,
                first_mean_exact,
                first_pair_var_exact,
                mlp.weights[0],
            )

        return fnp.stack(rows, axis=0)
