```python
from __future__ import annotations

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


_TARGET_FLOP_FRACTION = 0.96
_TAIL_CLIP = 1e-12
_ARRAY_BYTES_LIMIT = 96 * 1024 * 1024
_FLOAT64_ITEMSIZE = 8


def _first_primes(count: int) -> tuple[int, ...]:
    primes = []
    candidate = 2
    while len(primes) < count:
        ok = True
        for prime in primes:
            if prime * prime > candidate:
                break
            if candidate % prime == 0:
                ok = False
                break
        if ok:
            primes.append(candidate)
        candidate += 1
    return tuple(primes)


def _lattice_generator(width: int) -> fnp.ndarray:
    roots = fnp.sqrt(fnp.array(_first_primes(width), dtype=fnp.float64))
    return roots - fnp.floor(roots)


def _lattice_uniforms(n_samples: int, generator: fnp.ndarray, shift: fnp.ndarray) -> fnp.ndarray:
    indices = fnp.arange(n_samples, dtype=fnp.float64)[:, None]
    values = indices * generator[None, :] + shift[None, :]
    return values - fnp.floor(values)


def _exact_first_layer_mean(weight: fnp.ndarray) -> fnp.ndarray:
    variance = fnp.sum(weight * weight, axis=0)
    return fnp.sqrt(variance) / fnp.sqrt(2.0 * fnp.pi)


def _sample_count(budget: int, width: int, depth: int) -> int:
    lattice_per_sample = 89 * width
    forward_per_sample = depth * (2 * width * width + width)
    mean_per_sample = max(depth - 1, 0) * width
    fixed = 2 * width * width + 8 * width
    tracked = int(_TARGET_FLOP_FRACTION * int(budget))
    per_sample = lattice_per_sample + forward_per_sample + mean_per_sample
    n_samples = (tracked - fixed) // max(per_sample, 1)
    max_samples_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _FLOAT64_ITEMSIZE, 1)
    return int(max(1, min(n_samples, max_samples_by_bytes)))


class Estimator(BaseEstimator):
    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth
        n_samples = _sample_count(budget, width, depth)

        rng = fnp.random.default_rng(mlp.seed)
        generator = _lattice_generator(width)

        # Antithetic Cranley-Patterson shifts lower odd-order lattice error while
        # keeping the same deterministic budget model.
        half = n_samples // 2
        if half > 0:
            shift = rng.random(width)
            uniforms_a = _lattice_uniforms(half, generator, shift)
            uniforms_b = 1.0 - uniforms_a
            uniforms = fnp.concatenate((uniforms_a, uniforms_b), axis=0)
            if n_samples > 2 * half:
                extra_shift = rng.random(width)
                extra = _lattice_uniforms(1, generator, extra_shift)
                uniforms = fnp.concatenate((uniforms, extra), axis=0)
        else:
            shift = rng.random(width)
            uniforms = _lattice_uniforms(n_samples, generator, shift)

        uniforms = fnp.clip(uniforms, _TAIL_CLIP, 1.0 - _TAIL_CLIP)
        activations = flops.stats.norm.ppf(uniforms)

        rows = []
        for layer_index, weight in enumerate(mlp.weights):
            activations = fnp.maximum(activations @ weight, 0.0)
            if layer_index == 0:
                rows.append(_exact_first_layer_mean(weight))
            else:
                rows.append(fnp.mean(activations, axis=0))

        return fnp.stack(rows, axis=0)
```
