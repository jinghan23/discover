```python
from __future__ import annotations

import flopscope.numpy as fnp
from whestbench import BaseEstimator


_BUDGET_FRACTION = 0.152
_MIN_SAMPLES = 32
_MAX_SAMPLES = 1_000_000
_ARRAY_BYTES_LIMIT = 110 * 1024 * 1024
_WORST_CASE_ITEMSIZE = 8
_SETUP_SEED_OFFSET = 3

_INV_SQRT_2PI = 0.3989422804014327
_RELU_VAR_FACTOR = 0.5 - _INV_SQRT_2PI * _INV_SQRT_2PI
_EPS = 1e-6


def _as_int(value, default=None):
    if value is None:
        return default
    try:
        return int(value)
    except Exception:
        return default


def _sample_count(budget, width: int, depth: int) -> int:
    max_k_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _WORST_CASE_ITEMSIZE, 1)

    if budget is None:
        k = min(_MAX_SAMPLES, max_k_by_bytes)
    else:
        per_sample = depth * (2 * width * width + width) + width * width
        fixed = 14 * width**3 + 12 * width * width
        k = (int(_BUDGET_FRACTION * int(budget)) - fixed) // max(per_sample, 1)
        k = int(max(_MIN_SAMPLES, min(_MAX_SAMPLES, max_k_by_bytes, k)))

    k = max(2, int(k))
    return k - (k & 1)


class Estimator(BaseEstimator):
    def setup(self, context):
        self._samples = None
        self._width = None
        self._depth = None
        self._sample_source = None
        self._sample_seed = None

        width = _as_int(getattr(context, "width", None))
        depth = _as_int(getattr(context, "depth", None))
        budget = _as_int(getattr(context, "flop_budget", None))
        seed = _as_int(getattr(context, "seed", None), 0)

        if width is not None and depth is not None:
            self._samples = self._make_samples(
                width,
                _sample_count(budget, width, depth),
                seed + _SETUP_SEED_OFFSET,
            )
            self._width = width
            self._depth = depth
            self._sample_source = "setup"
            self._sample_seed = seed + _SETUP_SEED_OFFSET

    def teardown(self):
        pass

    def _make_samples(self, width, k, seed):
        rng = fnp.random.default_rng(seed)
        half_k = max(1, k // 2)
        u = rng.standard_normal((half_k, width), dtype=fnp.float32)

        covariance = (u.T @ u) / float(half_k)
        eigenvalues, eigenvectors = fnp.linalg.eigh(covariance)
        eigenvalues = fnp.maximum(eigenvalues, _EPS)
        inverse_sqrt = (eigenvectors / fnp.sqrt(eigenvalues)) @ eigenvectors.T

        half = fnp.asarray(u @ inverse_sqrt, dtype=fnp.float32)
        return fnp.concatenate([half, -half], axis=0)

    def _ensure_samples(self, width, depth, budget, seed):
        need_samples = (
            self._samples is None
            or self._samples.shape[1] != width
            or self._width != width
            or self._depth != depth
            or (
                self._sample_source != "setup"
                and self._sample_seed is not None
                and self._sample_seed != seed
            )
        )

        if need_samples:
            self._samples = self._make_samples(
                width,
                _sample_count(_as_int(budget), width, depth),
                seed,
            )
            self._width = width
            self._depth = depth
            self._sample_source = "predict"
            self._sample_seed = seed

    def predict(self, mlp, budget):
        width = int(mlp.width)
        depth = int(mlp.depth)
        seed = int(mlp.seed)

        self._ensure_samples(width, depth, budget, seed)

        first_weight = mlp.weights[0]
        first_variance = fnp.sum(first_weight * first_weight, axis=0)
        first_mean = fnp.sqrt(fnp.maximum(first_variance, 0.0)) * _INV_SQRT_2PI

        if depth == 1:
            return fnp.asarray(first_mean[None, :], dtype=fnp.float32)

        x = fnp.maximum(self._samples @ first_weight, 0.0)

        sample_mean = fnp.mean(x, axis=0)
        sample_second = fnp.mean(x * x, axis=0)
        sample_variance = fnp.maximum(sample_second - sample_mean * sample_mean, _EPS)
        target_variance = fnp.maximum(first_variance * _RELU_VAR_FACTOR, _EPS)
        x = (x - sample_mean) * fnp.sqrt(target_variance / sample_variance) + first_mean

        for weight in mlp.weights[1:]:
            x = fnp.maximum(x @ weight, 0.0)

        final_mean = fnp.asarray(
            fnp.mean(fnp.asarray(x, dtype=fnp.float64), axis=0),
            dtype=fnp.float32,
        )

        first_row = fnp.asarray(first_mean[None, :], dtype=fnp.float32)
        if depth == 2:
            return fnp.concatenate([first_row, final_mean[None, :]], axis=0)

        middle = fnp.zeros((depth - 2, width), dtype=fnp.float32)
        return fnp.concatenate([first_row, middle, final_mean[None, :]], axis=0)
```
