from __future__ import annotations

import flopscope.numpy as fnp
from whestbench import BaseEstimator


_BUDGET_FRACTION = 0.152
_MIN_SAMPLES = 32
_MAX_SAMPLES = 1_000_000
_ARRAY_BYTES_LIMIT = 110 * 1024 * 1024
_WORST_CASE_ITEMSIZE = 8
_SETUP_SEED_OFFSET = 3
_SECOND_MATCH_MAX_WIDTH = 2048

_PI = 3.141592653589793
_INV_SQRT_2PI = 0.3989422804014327
_INV_TWO_PI = 0.15915494309189535
_RELU_VAR_FACTOR = 0.5 - _INV_SQRT_2PI * _INV_SQRT_2PI
_EPS = 1e-6

_CDF_P = 0.2316419
_CDF_B1 = 0.319381530
_CDF_B2 = -0.356563782
_CDF_B3 = 1.781477937
_CDF_B4 = -1.821255978
_CDF_B5 = 1.330274429


def _use_second_match(width: int, depth: int) -> bool:
    return depth > 1 and width <= _SECOND_MATCH_MAX_WIDTH


def _budget_value(budget, width: int, depth: int) -> int:
    try:
        value = int(budget)
        if value > 0:
            return value
    except Exception:
        pass
    return int(64 * (depth * (2 * width * width + width) + width * width))


def _sample_count(budget: int, width: int, depth: int) -> int:
    per_sample = depth * (2 * width * width + width) + width * width
    fixed = 14 * width**3 + 12 * width * width
    if _use_second_match(width, depth):
        fixed += 4 * width**3 + 16 * width * width

    k = (int(_BUDGET_FRACTION * budget) - fixed) // max(per_sample, 1)
    max_k_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _WORST_CASE_ITEMSIZE, 1)
    k = int(max(_MIN_SAMPLES, min(_MAX_SAMPLES, max_k_by_bytes, k)))
    return k - (k & 1)


def _sample_correction_weight(k: int, width: int) -> float:
    value = float(k) / float(k + 4 * max(width, 1))
    return max(0.30, min(0.95, value))


def _sample_moments(x):
    mean = fnp.mean(x, axis=0)
    second = fnp.mean(x * x, axis=0)
    variance = fnp.maximum(second - mean * mean, _EPS)
    return mean, variance


def _normal_cdf(x):
    ax = fnp.abs(x)
    t = 1.0 / (1.0 + _CDF_P * ax)
    poly = (((((_CDF_B5 * t + _CDF_B4) * t + _CDF_B3) * t + _CDF_B2) * t + _CDF_B1) * t)
    pdf = _INV_SQRT_2PI * fnp.exp(-0.5 * ax * ax)
    cdf_pos = 1.0 - pdf * poly
    return 0.5 + fnp.sign(x) * (cdf_pos - 0.5)


def _relu_gaussian_moments(mean, variance):
    variance = fnp.maximum(variance, _EPS)
    std = fnp.sqrt(variance)
    alpha = mean / std
    pdf = _INV_SQRT_2PI * fnp.exp(-0.5 * alpha * alpha)
    cdf = _normal_cdf(alpha)

    relu_mean = fnp.maximum(std * pdf + mean * cdf, 0.0)
    relu_second = (variance + mean * mean) * cdf + mean * std * pdf
    relu_variance = fnp.maximum(relu_second - relu_mean * relu_mean, _EPS)
    return relu_mean, relu_variance


def _match_marginals_from_moments(x, target_mean, target_variance, sample_mean, sample_variance):
    scale = fnp.sqrt(fnp.maximum(target_variance, _EPS) / fnp.maximum(sample_variance, _EPS))
    return (x - sample_mean) * scale + target_mean


def _match_marginals(x, target_mean, target_variance):
    sample_mean, sample_variance = _sample_moments(x)
    return _match_marginals_from_moments(
        x,
        target_mean,
        target_variance,
        sample_mean,
        sample_variance,
    )


def _activation_target(z, pre_mean, pre_variance, mean_beta):
    x = fnp.maximum(z, 0.0)
    sample_mean, sample_variance = _sample_moments(x)
    gaussian_mean, gaussian_variance = _relu_gaussian_moments(pre_mean, pre_variance)

    variance_beta = min(0.985, mean_beta + 0.20)
    target_mean = gaussian_mean + mean_beta * (sample_mean - gaussian_mean)
    target_variance = gaussian_variance + variance_beta * (sample_variance - gaussian_variance)
    target_variance = fnp.maximum(target_variance, _EPS)

    x = _match_marginals_from_moments(
        x,
        target_mean,
        target_variance,
        sample_mean,
        sample_variance,
    )
    return x, target_mean


def _first_relu_second_moment(weight):
    covariance = weight.T @ weight
    variance = fnp.maximum(fnp.sum(weight * weight, axis=0), _EPS)
    std = fnp.sqrt(variance)
    scale = std[:, None] * std[None, :]

    rho = covariance / fnp.maximum(scale, _EPS)
    rho = fnp.maximum(-1.0, fnp.minimum(1.0, rho))

    kernel = (
        fnp.sqrt(fnp.maximum(1.0 - rho * rho, 0.0))
        + (_PI - fnp.arccos(rho)) * rho
    ) * _INV_TWO_PI
    return scale * kernel


class Estimator(BaseEstimator):
    def setup(self, context):
        self._width = None
        self._depth = None
        self._samples = None
        self._setup_seed = 0

        try:
            self._setup_seed = int(context.seed)
            width = int(context.width)
            depth = int(context.depth)
            budget = _budget_value(context.flop_budget, width, depth)
        except Exception:
            return

        self._width = width
        self._depth = depth
        self._samples = self._make_samples(
            width,
            _sample_count(budget, width, depth),
            self._setup_seed + _SETUP_SEED_OFFSET,
        )

    def teardown(self):
        pass

    def _make_samples(self, width, k, seed):
        rng = fnp.random.default_rng(seed)
        half_k = max(1, k // 2)
        u = rng.standard_normal((half_k, width), dtype=fnp.float32)

        covariance = (u.T @ u) / float(half_k)
        eigenvalues, eigenvectors = fnp.linalg.eigh(covariance)
        eigenvalues = fnp.maximum(eigenvalues, _EPS)
        inverse_sqrt = (eigenvectors / fnp.sqrt(eigenvalues)) @ eigenvectors.T

        half = fnp.asarray(u @ inverse_sqrt, dtype=fnp.float32)
        return fnp.concatenate([half, -half], axis=0)

    def predict(self, mlp, budget):
        width = int(mlp.width)
        depth = int(mlp.depth)

        if (
            getattr(self, "_samples", None) is None
            or self._samples.shape[1] != width
            or getattr(self, "_depth", None) != depth
        ):
            try:
                seed = int(mlp.seed)
            except Exception:
                seed = 0
            self._samples = self._make_samples(
                width,
                _sample_count(_budget_value(budget, width, depth), width, depth),
                seed,
            )
            self._width = width
            self._depth = depth

        beta = _sample_correction_weight(int(self._samples.shape[0]), width)

        first_weight = mlp.weights[0]
        first_variance = fnp.sum(first_weight * first_weight, axis=0)
        first_mean = fnp.sqrt(fnp.maximum(first_variance, 0.0)) * _INV_SQRT_2PI
        first_activation_variance = fnp.maximum(first_variance * _RELU_VAR_FACTOR, _EPS)

        rows = [fnp.asarray(first_mean[None, :], dtype=fnp.float32)]

        if depth == 1:
            return rows[0]

        x = fnp.maximum(self._samples @ first_weight, 0.0)
        x = _match_marginals(x, first_mean, first_activation_variance)

        second_weight = mlp.weights[1]
        z = x @ second_weight

        if _use_second_match(width, depth):
            first_second = _first_relu_second_moment(first_weight)
            second_pre_mean = first_mean @ second_weight
            second_pre_second = fnp.sum(
                (first_second @ second_weight) * second_weight,
                axis=0,
            )
            second_pre_variance = fnp.maximum(
                second_pre_second - second_pre_mean * second_pre_mean,
                _EPS,
            )
            z = _match_marginals(z, second_pre_mean, second_pre_variance)
        else:
            second_pre_mean, second_pre_variance = _sample_moments(z)

        x, layer_mean = _activation_target(
            z,
            second_pre_mean,
            second_pre_variance,
            min(0.975, beta + 0.08),
        )
        rows.append(fnp.asarray(layer_mean[None, :], dtype=fnp.float32))

        for weight in mlp.weights[2:]:
            z = x @ weight
            pre_mean, pre_variance = _sample_moments(z)
            x, layer_mean = _activation_target(z, pre_mean, pre_variance, beta)
            rows.append(fnp.asarray(layer_mean[None, :], dtype=fnp.float32))

        return fnp.concatenate(rows, axis=0)
