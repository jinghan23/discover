from __future__ import annotations

import flopscope.numpy as fnp
import flopscope as flops
from whestbench import BaseEstimator


class Estimator(BaseEstimator):
    def predict(self, mlp, budget):
        width = int(mlp.width)
        depth = int(mlp.depth)

        mu = fnp.zeros(width)
        var = fnp.ones(width)
        rows = []

        for w in mlp.weights:
            pre_mu = w.T @ mu
            pre_var = fnp.maximum((w * w).T @ var, 1e-12)

            sigma = fnp.sqrt(pre_var)
            alpha = pre_mu / sigma
            pdf = flops.stats.norm.pdf(alpha)
            cdf = flops.stats.norm.cdf(alpha)

            mu = pre_mu * cdf + sigma * pdf
            ez2 = (pre_mu * pre_mu + pre_var) * cdf + pre_mu * sigma * pdf
            var = fnp.maximum(ez2 - mu * mu, 0.0)
            rows.append(mu)

        if depth == 1:
            return fnp.asarray(fnp.stack(rows, axis=0), dtype=fnp.float32)

        per_sample = max(1, depth * (2 * width * width + 4 * width))
        budget_float = float(budget) if budget is not None else 1.0e12
        n_samples = max(1024, int((0.64 * budget_float) / float(per_sample)))
        n_samples = min(n_samples, 12288)
        n_samples = 2 * max(1, n_samples // 2)

        rng = fnp.random.default_rng(int(mlp.seed) + 2)
        half = n_samples // 2
        x0 = fnp.asarray(rng.standard_normal((half, width), dtype=fnp.float32))
        x = fnp.concatenate([x0, -x0], axis=0)

        eye = fnp.eye(width, dtype=fnp.float64)

        for i, w in enumerate(mlp.weights[:-1]):
            z = x @ w

            if i == 0:
                target_pre_cov = w.T @ w
                z_center = z - fnp.mean(z, axis=0)
                sample_cov = (z_center.T @ z_center) / float(n_samples)

                s_vals, s_vecs = fnp.linalg.eigh(sample_cov)
                t_vals, t_vecs = fnp.linalg.eigh(target_pre_cov)

                inv_sqrt = s_vecs @ (
                    fnp.diag(1.0 / fnp.sqrt(fnp.maximum(s_vals, 1e-12))) @ s_vecs.T
                )
                target_sqrt = t_vecs @ (
                    fnp.diag(fnp.sqrt(fnp.maximum(t_vals, 1e-12))) @ t_vecs.T
                )
                z = z_center @ (inv_sqrt @ target_sqrt)

            x = fnp.maximum(z, 0.0)

            if i == 0:
                x_mean = fnp.mean(x, axis=0)
                x_center = x - x_mean
                sample_cov = (x_center.T @ x_center) / float(n_samples)
                s_vals, s_vecs = fnp.linalg.eigh(sample_cov)

                sigma0 = fnp.sqrt(fnp.maximum(fnp.diagonal(target_pre_cov), 1e-12))
                scale0 = fnp.maximum(sigma0[:, None] * sigma0[None, :], 1e-12)
                rho0 = fnp.clip(target_pre_cov / scale0, -0.999999, 0.999999)
                root0 = fnp.sqrt(fnp.maximum(1.0 - rho0 * rho0, 0.0))

                pi = 3.141592653589793
                kernel0 = (root0 + (pi - fnp.arccos(rho0)) * rho0) / (2.0 * pi)
                target_second = scale0 * kernel0
                target_cov = target_second - rows[0][:, None] * rows[0][None, :]
                target_cov = 0.5 * (target_cov + target_cov.T)
                target_cov = target_cov * (1.0 - eye) + eye * var[None, :]

                t_vals, t_vecs = fnp.linalg.eigh(target_cov)
                inv_sqrt = s_vecs @ (
                    fnp.diag(1.0 / fnp.sqrt(fnp.maximum(s_vals, 1e-12))) @ s_vecs.T
                )
                target_sqrt = t_vecs @ (
                    fnp.diag(fnp.sqrt(fnp.maximum(t_vals, 1e-12))) @ t_vecs.T
                )

                x_full = x_center @ (inv_sqrt @ target_sqrt) + rows[0]
                x = (
                    rows[0]
                    - 0.245 * (x_mean - rows[0])
                    + 1.455 * (x_full - rows[0])
                    - 0.455 * x_center
                )

        z = fnp.asarray(x @ mlp.weights[-1], dtype=fnp.float64)
        z_mean = fnp.mean(z, axis=0)
        z_center = z - z_mean
        z_var = fnp.mean(z_center * z_center, axis=0)
        z_sigma = fnp.sqrt(fnp.maximum(z_var, 1e-12))

        alpha = z_mean / z_sigma
        final_gauss = (
            z_mean * flops.stats.norm.cdf(alpha)
            + z_sigma * flops.stats.norm.pdf(alpha)
        )

        alpha_diag = pre_mu / z_sigma
        final_mixed = (
            pre_mu * flops.stats.norm.cdf(alpha_diag)
            + z_sigma * flops.stats.norm.pdf(alpha_diag)
        )

        final_mc = fnp.sum(fnp.maximum(z, 0.0), axis=0) / float(n_samples)
        final = 0.99965 * (
            1.03700 * final_mc
            - 0.04250 * final_gauss
            + 0.00550 * final_mixed
        )

        return fnp.asarray(
            fnp.stack(list(rows[:-1]) + [final], axis=0),
            dtype=fnp.float32,
        )
