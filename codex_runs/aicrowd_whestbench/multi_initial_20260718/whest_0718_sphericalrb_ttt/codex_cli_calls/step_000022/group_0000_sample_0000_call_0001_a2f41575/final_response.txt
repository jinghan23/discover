```python
from __future__ import annotations

import math

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


_TARGET_FLOP_FRACTION = 0.075
_DEFAULT_BUDGET = 2_800_000_000
_MAX_WORKING_SET_BYTES = 72 * 1024 * 1024
_FLOAT64_ITEMSIZE = 8
_TAIL_CLIP = 1e-11
_MIN_ANTITHETIC_SAMPLES = 8
_VAR_FLOOR = 1e-20
_PRIME_CACHE: dict[int, tuple[int, ...]] = {}


def _first_primes(count: int) -> tuple[int, ...]:
    cached = _PRIME_CACHE.get(count)
    if cached is not None:
        return cached

    primes = []
    candidate = 2
    while len(primes) < count:
        is_prime = True
        for prime in primes:
            if prime * prime > candidate:
                break
            if candidate % prime == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(candidate)
        candidate += 1

    result = tuple(primes)
    _PRIME_CACHE[count] = result
    return result


def _mean_chi(dimension: int) -> float:
    half_dimension = 0.5 * float(dimension)
    return math.sqrt(2.0) * math.exp(
        math.lgamma(half_dimension + 0.5) - math.lgamma(half_dimension)
    )


def _sample_count(budget: int | None, width: int, depth: int) -> int:
    if budget is None:
        budget = _DEFAULT_BUDGET

    usable = int(_TARGET_FLOP_FRACTION * int(budget))
    per_sample = max(1, 2 * depth * width * width + 16 * depth * width)
    fixed = 6 * depth * width * width + 16 * depth * width
    count = (usable - fixed) // per_sample

    memory_limited = _MAX_WORKING_SET_BYTES // max(width * _FLOAT64_ITEMSIZE, 1)
    count = int(max(_MIN_ANTITHETIC_SAMPLES, min(count, memory_limited)))
    count -= count & 1
    return max(2, count)


def _lattice_normal_samples(n_samples: int, width: int, rng) -> fnp.ndarray:
    roots = fnp.sqrt(fnp.asarray(_first_primes(width), dtype=fnp.float64))
    generator = roots - fnp.floor(roots)
    shift = rng.random(width)

    indices = fnp.arange(n_samples, dtype=fnp.float64)[:, None]
    uniforms = indices * generator[None, :] + shift[None, :]
    uniforms = uniforms - fnp.floor(uniforms)
    uniforms = fnp.clip(uniforms, _TAIL_CLIP, 1.0 - _TAIL_CLIP)
    return flops.stats.norm.ppf(uniforms)


def _qmc_directions(n_samples: int, width: int, rng) -> fnp.ndarray:
    normals = _lattice_normal_samples(n_samples, width, rng)
    radii = fnp.sqrt(fnp.maximum(fnp.sum(normals * normals, axis=1), 1e-24))
    return normals / radii[:, None]


def _normal_relu_moments(mean: fnp.ndarray, var: fnp.ndarray) -> tuple[fnp.ndarray, fnp.ndarray]:
    std = fnp.sqrt(fnp.maximum(var, _VAR_FLOOR))
    alpha = mean / std
    cdf = flops.stats.norm.cdf(alpha)
    pdf = fnp.exp(-0.5 * alpha * alpha) / math.sqrt(2.0 * math.pi)

    relu_mean = std * pdf + mean * cdf
    relu_second = (mean * mean + var) * cdf + mean * std * pdf
    return fnp.maximum(relu_mean, 0.0), fnp.maximum(relu_second, 0.0)


def _match_moments(samples: fnp.ndarray, target_mean: fnp.ndarray, target_var: fnp.ndarray) -> fnp.ndarray:
    sample_mean = fnp.mean(samples, axis=0)
    centered = samples - sample_mean[None, :]
    sample_var = fnp.mean(centered * centered, axis=0)
    scale = fnp.sqrt(fnp.maximum(target_var, 0.0) / fnp.maximum(sample_var, _VAR_FLOOR))
    scale = fnp.minimum(scale, 8.0)
    return target_mean[None, :] + centered * scale[None, :]


def _first_layer_mean_and_var(weights: fnp.ndarray, width: int) -> tuple[fnp.ndarray, fnp.ndarray]:
    norm_sq = fnp.sum(weights * weights, axis=0)
    mean = fnp.sqrt(norm_sq) / math.sqrt(2.0 * math.pi)

    radial = _mean_chi(width)
    second = norm_sq * (radial * radial) / (2.0 * float(width))
    var = fnp.maximum(second - mean * mean, _VAR_FLOOR)
    return mean, var


def _maybe_whitened_first_preactivations(
    directions: fnp.ndarray, weights: fnp.ndarray, width: int
) -> fnp.ndarray:
    half_samples = directions.shape[0]

    if half_samples < width:
        return directions @ weights

    second = (directions.T @ directions) * (float(width) / float(half_samples))
    eigenvalues, eigenvectors = fnp.linalg.eigh(second)
    eigenvalues = fnp.maximum(eigenvalues, 1e-8)
    whitening = (eigenvectors / fnp.sqrt(eigenvalues)[None, :]) @ eigenvectors.T
    return directions @ (whitening @ weights)


class Estimator(BaseEstimator):
    def predict(self, mlp, budget: int | None) -> fnp.ndarray:
        width = mlp.width
        depth = mlp.depth
        n_samples = _sample_count(budget, width, depth)
        half_samples = n_samples // 2

        rng = fnp.random.default_rng(mlp.seed)
        half_directions = _qmc_directions(half_samples, width, rng)
        radial = _mean_chi(width)

        first_half = _maybe_whitened_first_preactivations(
            half_directions, mlp.weights[0], width
        ) * radial
        pre0 = fnp.concatenate((first_half, -first_half), axis=0)

        first_mean, layer_var = _first_layer_mean_and_var(mlp.weights[0], width)
        activations = _match_moments(fnp.maximum(pre0, 0.0), first_mean, layer_var)

        rows = [first_mean]
        previous_mean = first_mean
        previous_var = layer_var

        sample_weight = float(n_samples) / float(n_samples + 64)

        for weights in mlp.weights[1:]:
            linear_mean = previous_mean @ weights
            diag_var = previous_var @ (weights * weights)

            pre = activations @ weights
            pre_mean = fnp.mean(pre, axis=0)
            pre = pre - pre_mean[None, :] + linear_mean[None, :]

            centered = pre - linear_mean[None, :]
            sample_var = fnp.mean(centered * centered, axis=0)
            pre_var = fnp.maximum(
                (1.0 - sample_weight) * diag_var + sample_weight * sample_var,
                _VAR_FLOOR,
            )

            normal_mean, normal_second = _normal_relu_moments(linear_mean, pre_var)

            raw = fnp.maximum(pre, 0.0)
            sample_mean = fnp.mean(raw, axis=0)
            sample_second = fnp.mean(raw * raw, axis=0)

            layer_mean = fnp.maximum(
                (1.0 - sample_weight) * normal_mean + sample_weight * sample_mean,
                0.0,
            )
            layer_second = fnp.maximum(
                (1.0 - sample_weight) * normal_second + sample_weight * sample_second,
                layer_mean * layer_mean,
            )
            layer_var = fnp.maximum(layer_second - layer_mean * layer_mean, _VAR_FLOOR)

            activations = _match_moments(raw, layer_mean, layer_var)

            rows.append(layer_mean)
            previous_mean = layer_mean
            previous_var = layer_var

        return fnp.asarray(fnp.stack(rows, axis=0), dtype=fnp.float32)
```
