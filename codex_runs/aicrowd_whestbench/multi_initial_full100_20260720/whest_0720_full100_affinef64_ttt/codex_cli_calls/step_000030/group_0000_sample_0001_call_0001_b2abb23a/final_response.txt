from __future__ import annotations

import flopscope.numpy as fnp
from whestbench import BaseEstimator


_INV_SQRT_2PI = 0.3989422804014327
_MIN_VAR = 1e-12


def _to_int(value, default=0):
    try:
        if value is None:
            return default
        return int(value)
    except Exception:
        return default


def _sample_count(budget: int, width: int, depth: int) -> int:
    per_sample = max(1, depth * (2 * width * width + width))
    k = int(0.085 * max(0, budget)) // per_sample
    k = max(32, min(262144, k))
    return k - (k & 1)


class Estimator(BaseEstimator):
    def setup(self, context):
        self._samples = None
        self._width = None
        self._depth = None
        self._seed = _to_int(getattr(context, "seed", 0)) + 17

        width = getattr(context, "width", None)
        depth = getattr(context, "depth", None)
        budget = _to_int(getattr(context, "flop_budget", 0))
        if width is not None and depth is not None and budget > 0:
            self._width = int(width)
            self._depth = int(depth)
            self._samples = self._make_samples(
                self._width,
                _sample_count(budget, self._width, self._depth),
                self._seed,
            )

    def teardown(self):
        self._samples = None

    def _make_samples(self, width, k, seed):
        rng = fnp.random.default_rng(seed)
        half = rng.standard_normal((k // 2, width), dtype=fnp.float32)

        mean = fnp.mean(half, axis=0)
        second = fnp.mean(half * half, axis=0)
        var = fnp.maximum(second - mean * mean, 1e-6)
        half = fnp.asarray((half - mean) / fnp.sqrt(var), dtype=fnp.float32)

        return fnp.concatenate([half, -half], axis=0)

    def predict(self, mlp, budget):
        width = int(mlp.width)
        depth = int(mlp.depth)

        if (
            self._samples is None
            or self._samples.shape[1] != width
            or self._depth != depth
        ):
            self._width = width
            self._depth = depth
            self._samples = self._make_samples(
                width,
                _sample_count(_to_int(budget), width, depth),
                _to_int(getattr(mlp, "seed", 0)) + 17,
            )

        x = self._samples
        outputs = []

        for weight in mlp.weights:
            z = x @ weight
            x = fnp.maximum(z, 0.0)
            mean = fnp.asarray(
                fnp.mean(fnp.asarray(x, dtype=fnp.float64), axis=0),
                dtype=fnp.float32,
            )
            outputs.append(mean)

        return fnp.asarray(fnp.stack(outputs, axis=0), dtype=fnp.float32)
