from __future__ import annotations

import flopscope.numpy as fnp
from whestbench import BaseEstimator


_BUDGET_FRACTION = 0.146
_MIN_SAMPLES = 32
_MAX_SAMPLES = 1_000_000
_ARRAY_BYTES_LIMIT = 100 * 1024 * 1024
_WORST_CASE_ITEMSIZE = 8
_SETUP_SEED_OFFSET = 3

_INV_SQRT_2PI = 0.3989422804014327
_INV_TWO_PI = 0.15915494309189535
_PI = 3.141592653589793

_HIDDEN_GAUSS_BLEND = 0.25
_SECOND_LAYER_EXACT_BLEND = 0.42
_FINAL_GAUSS_BLEND = 0.30
_FINAL_EXACT_BLEND = 0.48
_EXACT_COV_WIDTH_LIMIT = 2048


def _to_int(value, default=0):
    try:
        if value is None:
            return default
        return int(value)
    except Exception:
        return default


def _sample_count(budget: int, width: int, depth: int) -> int:
    per_sample = depth * (2 * width * width + width) + width * width
    fixed = 13 * width**3
    k = (int(_BUDGET_FRACTION * budget) - fixed) // max(per_sample, 1)
    max_k_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _WORST_CASE_ITEMSIZE, 1)
    k = int(max(_MIN_SAMPLES, min(_MAX_SAMPLES, max_k_by_bytes, k)))
    return k - (k & 1)


def _normal_cdf(x):
    ax = fnp.abs(x)
    t = 1.0 / (1.0 + 0.2316419 * ax)
    poly = (((((1.330274429 * t - 1.821255978) * t + 1.781477937) * t
              - 0.356563782) * t + 0.319381530) * t)
    phi = _INV_SQRT_2PI * fnp.exp(-0.5 * ax * ax)
    c = 1.0 - phi * poly
    c = fnp.where(x >= 0.0, c, 1.0 - c)
    return fnp.maximum(0.0, fnp.minimum(1.0, c))


def _relu_normal_moments(mu, var):
    var = fnp.maximum(var, 1e-12)
    sigma = fnp.sqrt(var)
    a = mu / sigma
    phi = _INV_SQRT_2PI * fnp.exp(-0.5 * a * a)
    cdf = _normal_cdf(a)
    mean = sigma * phi + mu * cdf
    second = (mu * mu + var) * cdf + mu * sigma * phi
    return mean, fnp.maximum(second - mean * mean, 1e-8)


def _blend_match(y, target_mean, target_var, blend):
    sample_mean = fnp.mean(y, axis=0)
    sample_second = fnp.mean(y * y, axis=0)
    sample_var = fnp.maximum(sample_second - sample_mean * sample_mean, 1e-8)

    mean = sample_mean + blend * (target_mean - sample_mean)
    var = sample_var + blend * (target_var - sample_var)
    var = fnp.maximum(var, 1e-8)

    y = (y - sample_mean) * fnp.sqrt(var / sample_var) + mean
    return y, fnp.asarray(mean, dtype=fnp.float32)


def _first_relu_covariance(weight, first_var, first_mean):
    std = fnp.sqrt(fnp.maximum(first_var, 1e-12))
    gram = weight.T @ weight
    denom = fnp.maximum(std[:, None] * std[None, :], 1e-12)
    rho = fnp.maximum(-1.0, fnp.minimum(1.0, gram / denom))

    root = fnp.sqrt(fnp.maximum(0.0, 1.0 - rho * rho))
    kernel = (root + (_PI - fnp.arccos(rho)) * rho) * _INV_TWO_PI
    second = denom * kernel
    return second - first_mean[:, None] * first_mean[None, :]


class Estimator(BaseEstimator):
    def setup(self, context):
        self._samples = None
        self._width = None
        self._depth = None
        self._setup_seed = _to_int(getattr(context, "seed", 0)) + _SETUP_SEED_OFFSET

        width = getattr(context, "width", None)
        depth = getattr(context, "depth", None)
        flop_budget = _to_int(getattr(context, "flop_budget", 0))

        if width is not None and depth is not None and flop_budget > 0:
            self._width = int(width)
            self._depth = int(depth)
            self._samples = self._make_samples(
                self._width,
                _sample_count(flop_budget, self._width, self._depth),
                self._setup_seed,
            )

    def teardown(self):
        self._samples = None

    def _make_samples(self, width, k, seed):
        rng = fnp.random.default_rng(seed)
        u = rng.standard_normal((k // 2, width), dtype=fnp.float32)

        covariance = (u.T @ u) / float(k // 2)
        eigenvalues, eigenvectors = fnp.linalg.eigh(covariance)
        eigenvalues = fnp.maximum(eigenvalues, 1e-6)
        inverse_sqrt = (eigenvectors / fnp.sqrt(eigenvalues)) @ eigenvectors.T

        half = fnp.asarray(u @ inverse_sqrt, dtype=fnp.float32)
        return fnp.concatenate([half, -half], axis=0)

    def _layer2_exact_preact_moments(self, first_weight, second_weight, first_var, first_mean, budget):
        width = int(first_weight.shape[0])
        if width > _EXACT_COV_WIDTH_LIMIT:
            return None, None
        if budget > 0 and 4 * width**3 > int(0.10 * budget):
            return None, None

        cov = _first_relu_covariance(first_weight, first_var, first_mean)
        mean = first_mean @ second_weight
        temp = cov @ second_weight
        var = fnp.maximum(fnp.sum(second_weight * temp, axis=0), 1e-9)
        return mean, var

    def predict(self, mlp, budget):
        width = int(mlp.width)
        depth = int(mlp.depth)
        budget_int = _to_int(budget)

        if (
            not hasattr(self, "_samples")
            or self._samples is None
            or self._samples.shape[1] != width
            or getattr(self, "_depth", depth) != depth
        ):
            self._samples = self._make_samples(
                width,
                _sample_count(budget_int, width, depth),
                int(mlp.seed) + _SETUP_SEED_OFFSET,
            )
            self._width = width
            self._depth = depth

        first_weight = mlp.weights[0]
        first_var = fnp.sum(first_weight * first_weight, axis=0)
        first_mean = fnp.sqrt(first_var) * _INV_SQRT_2PI
        first_post_var = first_var * (0.5 - _INV_SQRT_2PI * _INV_SQRT_2PI)

        outputs = [fnp.asarray(first_mean, dtype=fnp.float32)[None, :]]
        if depth == 1:
            return outputs[0]

        layer2_mu, layer2_var = self._layer2_exact_preact_moments(
            first_weight, mlp.weights[1], first_var, first_mean, budget_int
        )

        x = fnp.maximum(self._samples @ first_weight, 0.0)
        x, _ = _blend_match(x, first_mean, first_post_var, 1.0)

        for layer_index in range(1, depth):
            weight = mlp.weights[layer_index]
            z = x @ weight
            y = fnp.maximum(z, 0.0)

            if layer_index == 1 and layer2_mu is not None:
                target_mean, target_var = _relu_normal_moments(layer2_mu, layer2_var)
                blend = _FINAL_EXACT_BLEND if layer_index == depth - 1 else _SECOND_LAYER_EXACT_BLEND
            else:
                z_mean = fnp.mean(z, axis=0)
                z_second = fnp.mean(z * z, axis=0)
                z_var = fnp.maximum(z_second - z_mean * z_mean, 1e-9)
                target_mean, target_var = _relu_normal_moments(z_mean, z_var)
                blend = _FINAL_GAUSS_BLEND if layer_index == depth - 1 else _HIDDEN_GAUSS_BLEND

            if layer_index == depth - 1:
                raw_mean = fnp.mean(fnp.asarray(y, dtype=fnp.float64), axis=0)
                final_mean = raw_mean + blend * (target_mean - raw_mean)
                outputs.append(fnp.asarray(final_mean, dtype=fnp.float32)[None, :])
            else:
                x, layer_mean = _blend_match(y, target_mean, target_var, blend)
                outputs.append(layer_mean[None, :])

        return fnp.concatenate(outputs, axis=0)
