```python
from __future__ import annotations

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


_TARGET_FLOP_FRACTION = 0.94
_TAIL_CLIP = 1e-12
_ARRAY_BYTES_LIMIT = 128 * 1024 * 1024
_FLOAT64_ITEMSIZE = 8


def _exact_first_layer_mean(weight: fnp.ndarray) -> fnp.ndarray:
    sd = fnp.sqrt(fnp.sum(weight * weight, axis=0))
    return sd / fnp.sqrt(2.0 * fnp.pi)


def _sample_count(budget: int, width: int, depth: int) -> int:
    per_sample = depth * (2 * width * width + width) + depth * width + 64 * width
    fixed = 3 * width * width + 64 * width
    count = (int(_TARGET_FLOP_FRACTION * budget) - fixed) // max(per_sample, 1)
    max_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _FLOAT64_ITEMSIZE, 1)
    return int(max(1, min(count, max_by_bytes)))


def _normal_samples(rng, n_samples: int, width: int) -> fnp.ndarray:
    half = (n_samples + 1) // 2
    u = rng.random((half, width))
    u = fnp.clip(u, _TAIL_CLIP, 1.0 - _TAIL_CLIP)
    z = flops.stats.norm.ppf(u)
    z = fnp.concatenate((z, -z), axis=0)
    return z[:n_samples]


def _analytic_diagonal_rows(mlp) -> fnp.ndarray:
    width = mlp.width
    mean = fnp.zeros(width, dtype=fnp.float64)
    second = fnp.ones(width, dtype=fnp.float64)
    rows = []

    for weight in mlp.weights:
        pre_mean = mean @ weight
        pre_second = second @ (weight * weight)
        pre_var = fnp.maximum(pre_second - pre_mean * pre_mean, 1e-30)
        sd = fnp.sqrt(pre_var)
        alpha = pre_mean / sd
        pdf = fnp.exp(-0.5 * alpha * alpha) / fnp.sqrt(2.0 * fnp.pi)
        cdf = flops.stats.norm.cdf(alpha)

        relu_mean = sd * pdf + pre_mean * cdf
        relu_second = (pre_var + pre_mean * pre_mean) * cdf + pre_mean * sd * pdf

        rows.append(relu_mean)
        mean = relu_mean
        second = relu_second

    return fnp.stack(rows, axis=0)


class Estimator(BaseEstimator):
    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth

        analytic = _analytic_diagonal_rows(mlp)

        n_samples = _sample_count(budget, width, depth)
        rng = fnp.random.default_rng(mlp.seed)
        activations = _normal_samples(rng, n_samples, width)

        rows = []
        for layer_index, weight in enumerate(mlp.weights):
            activations = fnp.maximum(activations @ weight, 0.0)
            if layer_index == 0:
                rows.append(_exact_first_layer_mean(weight))
            else:
                mc_mean = fnp.mean(activations, axis=0)
                rows.append(mc_mean)

        mc = fnp.stack(rows, axis=0)

        if n_samples < 64:
            return analytic

        blend = fnp.minimum(0.35, 8.0 / fnp.sqrt(fnp.array(float(n_samples), dtype=fnp.float64)))
        layer_scale = fnp.arange(depth, dtype=fnp.float64) / fnp.maximum(float(depth - 1), 1.0)
        layer_scale = layer_scale[:, None]
        return (1.0 - blend * layer_scale) * mc + (blend * layer_scale) * analytic
```
