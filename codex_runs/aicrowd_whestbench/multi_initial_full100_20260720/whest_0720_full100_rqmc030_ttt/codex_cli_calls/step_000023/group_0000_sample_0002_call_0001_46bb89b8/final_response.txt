from __future__ import annotations

import math
from functools import lru_cache

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


_INV_SQRT_2PI = 0.39894228040143267794
_INV_2PI = 0.15915494309189533577
_PI = 3.14159265358979323846
_TAIL_CLIP = 1e-12
_ARRAY_BYTES_LIMIT = 112 * 1024 * 1024
_FLOAT64_ITEMSIZE = 8
_TARGET_FLOP_FRACTION = 0.116


@lru_cache(maxsize=None)
def _first_primes(count: int) -> tuple[int, ...]:
    primes = []
    candidate = 2
    while len(primes) < count:
        ok = True
        for p in primes:
            if p * p > candidate:
                break
            if candidate % p == 0:
                ok = False
                break
        if ok:
            primes.append(candidate)
        candidate += 1
    return tuple(primes)


@lru_cache(maxsize=None)
def _mean_chi(dim: int) -> float:
    return math.sqrt(2.0) * math.exp(
        math.lgamma(0.5 * (dim + 1.0)) - math.lgamma(0.5 * dim)
    )


def _lattice_generator(width: int) -> fnp.ndarray:
    roots = fnp.sqrt(fnp.array(_first_primes(width), dtype=fnp.float64))
    return roots - fnp.floor(roots)


def _sphere_lattice_inputs(n_samples: int, width: int, seed: int) -> fnp.ndarray:
    half = max(1, n_samples // 2)
    rng = fnp.random.default_rng(seed)
    generator = _lattice_generator(width)
    shift = rng.random(width)

    indices = fnp.arange(half, dtype=fnp.float64)[:, None]
    uniforms = indices * generator[None, :] + shift[None, :]
    uniforms = uniforms - fnp.floor(uniforms)
    uniforms = fnp.clip(uniforms, _TAIL_CLIP, 1.0 - _TAIL_CLIP)

    base = flops.stats.norm.ppf(uniforms)
    norm = fnp.sqrt(fnp.sum(base * base, axis=1, keepdims=True))
    base = (_mean_chi(width) * base) / fnp.maximum(norm, 1e-30)

    return fnp.concatenate((base, -base), axis=0)


def _pair_average(values: fnp.ndarray) -> fnp.ndarray:
    half = values.shape[0] // 2
    return 0.5 * (values[:half] + values[half : 2 * half])


def _exact_first_layer_mean(weight: fnp.ndarray) -> fnp.ndarray:
    sigma = fnp.sqrt(fnp.sum(weight * weight, axis=0))
    return sigma * _INV_SQRT_2PI


def _exact_first_layer_square_mean(weight: fnp.ndarray, width: int) -> fnp.ndarray:
    scale = 0.5 * (_mean_chi(width) * _mean_chi(width)) / width
    return scale * fnp.sum(weight * weight, axis=0)


def _relu_product_kernel(gram: fnp.ndarray) -> fnp.ndarray:
    diag = fnp.maximum(fnp.diag(gram), 0.0)
    sigma = fnp.sqrt(diag)
    denom = fnp.maximum(sigma[:, None] * sigma[None, :], 1e-30)
    rho = fnp.clip(gram / denom, -1.0, 1.0)

    root = fnp.sqrt(fnp.maximum(1.0 - rho * rho, 0.0))
    theta = fnp.arccos(rho)
    return (sigma[:, None] * sigma[None, :]) * (
        root + (_PI - theta) * rho
    ) * _INV_2PI


def _first_layer_sphere_second(weight: fnp.ndarray, width: int) -> fnp.ndarray:
    radius = _mean_chi(width)
    return (radius * radius / width) * _relu_product_kernel(weight.T @ weight)


def _exact_second_preact_square_mean(
    first_weight: fnp.ndarray,
    second_weight: fnp.ndarray,
    width: int,
) -> fnp.ndarray:
    first_second = _first_layer_sphere_second(first_weight, width)
    projected = first_second @ second_weight
    return fnp.sum(second_weight * projected, axis=0)


def _use_quadratic_cv(budget: int, width: int, depth: int) -> bool:
    if depth < 2:
        return False
    target = int(_TARGET_FLOP_FRACTION * budget)
    return target > 20 * width * width * width + 48 * width * width


def _sample_count(budget: int, width: int, depth: int, quadratic_cv: bool) -> int:
    forward_per_sample = depth * (2 * width * width + width)
    sphere_per_sample = 4 * width
    ppf_per_sample = 45 * width
    mean_per_sample = max(depth - 1, 0) * width

    cv_per_sample = 7 * width * width
    cv_fixed = 7 * width * width * width + 18 * width * width + 10 * width

    if quadratic_cv:
        cv_per_sample += 8 * width * width
        cv_fixed += 10 * width * width * width + 24 * width * width

    per_sample = (
        forward_per_sample
        + sphere_per_sample
        + ppf_per_sample
        + mean_per_sample
        + cv_per_sample
    )

    count = (int(_TARGET_FLOP_FRACTION * budget) - cv_fixed) // max(per_sample, 1)
    max_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _FLOAT64_ITEMSIZE, 1)
    count = int(max(8, min(count, max_by_bytes)))
    count -= count % 2
    return max(2, count)


def _ridge_for(cov: fnp.ndarray, width: int, n_samples: int, base: float, scale: float):
    diag_scale = fnp.mean(fnp.diag(cov))
    return fnp.maximum(diag_scale, 1e-30) * (
        base + scale * width / fnp.maximum(float(n_samples), 1.0)
    )


def _ridge_for_centered(
    centered: fnp.ndarray,
    width: int,
    n_samples: int,
    base: float,
    scale: float,
):
    diag_scale = fnp.sum(centered * centered) / fnp.maximum(
        float(n_samples * width), 1.0
    )
    return fnp.maximum(diag_scale, 1e-30) * (
        base + scale * width / fnp.maximum(float(n_samples), 1.0)
    )


def _final_control_variate_primal(
    first_activations: fnp.ndarray,
    exact_first_mean: fnp.ndarray,
    exact_first_square_mean: fnp.ndarray,
    final_activations: fnp.ndarray,
    second_preactivations: fnp.ndarray | None,
    exact_second_square_mean: fnp.ndarray | None,
) -> fnp.ndarray:
    y = _pair_average(final_activations)
    y_mean = fnp.mean(y, axis=0)
    residual_y = y - y_mean
    prediction = y_mean

    n_pairs = y.shape[0]
    width = y.shape[1]
    previous = []

    def apply_group(features, exact_mean, ridge_base, ridge_scale):
        nonlocal residual_y, prediction, previous

        c = _pair_average(features)
        c_mean = fnp.mean(c, axis=0)
        centered = c - c_mean
        delta = c_mean - exact_mean

        for prev_centered, prev_system, prev_delta in previous:
            cross = (prev_centered.T @ centered) / n_pairs
            gamma = fnp.linalg.solve(prev_system, cross)
            centered = centered - prev_centered @ gamma
            delta = delta - prev_delta @ gamma

        cov = (centered.T @ centered) / n_pairs
        rhs = (centered.T @ residual_y) / n_pairs
        ridge = _ridge_for(cov, width, n_pairs, ridge_base, ridge_scale)
        system = cov + ridge * fnp.eye(width, dtype=fnp.float64)

        beta = fnp.linalg.solve(system, rhs)
        prediction = prediction - delta @ beta
        residual_y = residual_y - centered @ beta
        previous.append((centered, system, delta))

    apply_group(first_activations, exact_first_mean, 0.012, 0.8)
    apply_group(first_activations * first_activations, exact_first_square_mean, 0.026, 1.2)

    if second_preactivations is not None and exact_second_square_mean is not None:
        apply_group(
            second_preactivations * second_preactivations,
            exact_second_square_mean,
            0.045,
            1.6,
        )

    return fnp.maximum(prediction, 0.0)


def _final_control_variate_dual(
    first_activations: fnp.ndarray,
    exact_first_mean: fnp.ndarray,
    exact_first_square_mean: fnp.ndarray,
    final_activations: fnp.ndarray,
    second_preactivations: fnp.ndarray | None,
    exact_second_square_mean: fnp.ndarray | None,
) -> fnp.ndarray:
    y = _pair_average(final_activations)
    y_mean = fnp.mean(y, axis=0)
    residual_y = y - y_mean
    prediction = y_mean

    n_pairs = y.shape[0]
    width = y.shape[1]
    previous = []

    def apply_group(features, exact_mean, ridge_base, ridge_scale):
        nonlocal residual_y, prediction, previous

        c = _pair_average(features)
        c_mean = fnp.mean(c, axis=0)
        centered = c - c_mean
        delta = c_mean - exact_mean

        for prev_gram, prev_system, prev_leverage in previous:
            solved = fnp.linalg.solve(prev_system, centered)
            centered = centered - prev_gram @ solved
            delta = delta - prev_leverage @ solved

        gram = centered @ centered.T
        ridge = _ridge_for_centered(centered, width, n_pairs, ridge_base, ridge_scale)
        system = gram + (n_pairs * ridge) * fnp.eye(n_pairs, dtype=fnp.float64)

        solved_y = fnp.linalg.solve(system, residual_y)
        leverage = delta @ centered.T
        prediction = prediction - leverage @ solved_y
        residual_y = residual_y - gram @ solved_y
        previous.append((gram, system, leverage))

    apply_group(first_activations, exact_first_mean, 0.012, 0.8)
    apply_group(first_activations * first_activations, exact_first_square_mean, 0.026, 1.2)

    if second_preactivations is not None and exact_second_square_mean is not None:
        apply_group(
            second_preactivations * second_preactivations,
            exact_second_square_mean,
            0.045,
            1.6,
        )

    return fnp.maximum(prediction, 0.0)


def _final_control_variate(
    first_activations: fnp.ndarray,
    exact_first_mean: fnp.ndarray,
    exact_first_square_mean: fnp.ndarray,
    final_activations: fnp.ndarray,
    second_preactivations: fnp.ndarray | None,
    exact_second_square_mean: fnp.ndarray | None,
) -> fnp.ndarray:
    n_pairs = first_activations.shape[0] // 2
    width = first_activations.shape[1]
    if n_pairs <= width:
        return _final_control_variate_dual(
            first_activations,
            exact_first_mean,
            exact_first_square_mean,
            final_activations,
            second_preactivations,
            exact_second_square_mean,
        )
    return _final_control_variate_primal(
        first_activations,
        exact_first_mean,
        exact_first_square_mean,
        final_activations,
        second_preactivations,
        exact_second_square_mean,
    )


class Estimator(BaseEstimator):
    def setup(self, context):
        self._setup_seed = context.seed

    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth
        budget = int(budget)

        quadratic_cv = _use_quadratic_cv(budget, width, depth)
        n_samples = _sample_count(budget, width, depth, quadratic_cv)

        activations = _sphere_lattice_inputs(n_samples, width, mlp.seed)
        rows = []

        first_activations = None
        second_preactivations = None
        exact_first_mean = None
        exact_first_square_mean = None
        exact_second_square_mean = None

        for layer_index, weight in enumerate(mlp.weights):
            preactivations = activations @ weight

            if layer_index == 1 and quadratic_cv:
                second_preactivations = preactivations

            activations = fnp.maximum(preactivations, 0.0)

            if layer_index == 0:
                exact_first_mean = _exact_first_layer_mean(weight)
                exact_first_square_mean = _exact_first_layer_square_mean(weight, width)
                first_activations = activations
                rows.append(exact_first_mean)

                if quadratic_cv:
                    try:
                        exact_second_square_mean = _exact_second_preact_square_mean(
                            weight,
                            mlp.weights[1],
                            width,
                        )
                    except Exception:
                        exact_second_square_mean = None
                        second_preactivations = None
            else:
                rows.append(fnp.mean(activations, axis=0))

        if (
            depth > 1
            and first_activations is not None
            and exact_first_mean is not None
            and exact_first_square_mean is not None
        ):
            try:
                rows[-1] = _final_control_variate(
                    first_activations,
                    exact_first_mean,
                    exact_first_square_mean,
                    activations,
                    second_preactivations,
                    exact_second_square_mean,
                )
            except Exception:
                rows[-1] = fnp.mean(activations, axis=0)

        return fnp.stack(rows, axis=0)
