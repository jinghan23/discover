```python
from __future__ import annotations

import math

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


_INV_SQRT_2PI = 0.39894228040143267794
_INV_2PI = 0.15915494309189533577
_TAIL_CLIP = 1e-12
_ARRAY_BYTES_LIMIT = 120 * 1024 * 1024
_FLOAT64_ITEMSIZE = 8
_TARGET_FLOP_FRACTION = 0.072


def _first_primes(count: int) -> tuple[int, ...]:
    primes = []
    candidate = 2
    while len(primes) < count:
        ok = True
        for prime in primes:
            if prime * prime > candidate:
                break
            if candidate % prime == 0:
                ok = False
                break
        if ok:
            primes.append(candidate)
        candidate += 1
    return tuple(primes)


def _lattice_generator(width: int) -> fnp.ndarray:
    roots = fnp.sqrt(fnp.array(_first_primes(width), dtype=fnp.float64))
    return roots - fnp.floor(roots)


def _mean_chi(dim: int) -> float:
    return math.sqrt(2.0) * math.exp(
        math.lgamma(0.5 * (dim + 1.0)) - math.lgamma(0.5 * dim)
    )


def _sample_count(budget: int, width: int, depth: int) -> int:
    forward_per_sample = depth * (2 * width * width + width)
    sphere_per_sample = 4 * width
    ppf_per_sample = 45 * width
    mean_per_sample = max(depth - 1, 0) * width
    cv_per_sample = 4 * width * width

    per_sample = (
        forward_per_sample
        + sphere_per_sample
        + ppf_per_sample
        + mean_per_sample
        + cv_per_sample
    )
    fixed = 5 * width * width * width + 14 * width * width + 10 * width
    count = (int(_TARGET_FLOP_FRACTION * budget) - fixed) // max(per_sample, 1)

    max_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _FLOAT64_ITEMSIZE, 1)
    count = int(max(8, min(count, max_by_bytes)))
    count -= count % 2
    return max(2, count)


def _exact_first_layer_mean(weight: fnp.ndarray) -> fnp.ndarray:
    sigma = fnp.sqrt(fnp.sum(weight * weight, axis=0))
    return sigma * _INV_SQRT_2PI


def _relu_gaussian_mean(mu: fnp.ndarray, var: fnp.ndarray) -> fnp.ndarray:
    sigma = fnp.sqrt(fnp.maximum(var, 1e-30))
    alpha = mu / sigma
    return sigma * flops.stats.norm.pdf(alpha) + mu * flops.stats.norm.cdf(alpha)


def _normal_relu_cov_from_cov(cov: fnp.ndarray) -> fnp.ndarray:
    diag = fnp.maximum(fnp.diag(cov), 1e-30)
    sigma = fnp.sqrt(diag)
    denom = sigma[:, None] * sigma[None, :]
    rho = fnp.clip(cov / fnp.maximum(denom, 1e-30), -1.0, 1.0)
    theta = fnp.arccos(rho)
    relu_second = denom * (
        fnp.sqrt(fnp.maximum(1.0 - rho * rho, 0.0))
        + (math.pi - theta) * rho
    ) * _INV_2PI
    mean = sigma * _INV_SQRT_2PI
    return relu_second - mean[:, None] * mean[None, :]


def _gaussian_moment_rows(weights: tuple[fnp.ndarray, ...]) -> list[fnp.ndarray]:
    rows = []
    cov = None
    mean = None

    for layer_index, weight in enumerate(weights):
        if layer_index == 0:
            pre_cov = weight.T @ weight
            mean = _exact_first_layer_mean(weight)
            cov = _normal_relu_cov_from_cov(pre_cov)
            rows.append(mean)
        else:
            pre_mean = mean @ weight
            pre_cov = weight.T @ cov @ weight
            pre_var = fnp.diag(pre_cov)
            mean = _relu_gaussian_mean(pre_mean, pre_var)
            rows.append(mean)

            second = pre_cov + pre_mean[:, None] * pre_mean[None, :]
            cov = _normal_relu_cov_from_cov(second)
    return rows


def _lattice_sphere_inputs(n_samples: int, width: int, seed: int) -> fnp.ndarray:
    half = max(1, n_samples // 2)
    rng = fnp.random.default_rng(seed)
    generator = _lattice_generator(width)
    shift = rng.random(width)

    indices = fnp.arange(half, dtype=fnp.float64)[:, None]
    uniforms = indices * generator[None, :] + shift[None, :]
    uniforms = uniforms - fnp.floor(uniforms)
    uniforms = fnp.clip(uniforms, _TAIL_CLIP, 1.0 - _TAIL_CLIP)

    base = flops.stats.norm.ppf(uniforms)
    norm = fnp.sqrt(fnp.sum(base * base, axis=1, keepdims=True))
    base = (_mean_chi(width) * base) / fnp.maximum(norm, 1e-30)
    return fnp.concatenate((base, -base), axis=0)


def _ridge_solve_cv(
    control_samples: fnp.ndarray,
    control_exact: fnp.ndarray,
    target_samples: fnp.ndarray,
    ridge_base: float,
) -> fnp.ndarray:
    n_samples = control_samples.shape[0]
    width = control_samples.shape[1]

    c_mean = fnp.mean(control_samples, axis=0)
    y_mean = fnp.mean(target_samples, axis=0)

    c_centered = control_samples - c_mean
    y_centered = target_samples - y_mean

    cov_cc = (c_centered.T @ c_centered) / n_samples
    cov_cy = (c_centered.T @ y_centered) / n_samples

    scale = fnp.mean(fnp.diag(cov_cc))
    ridge = fnp.maximum(scale, 1e-30) * (
        ridge_base + 0.75 * width / fnp.maximum(float(n_samples), 1.0)
    )
    beta = fnp.linalg.solve(cov_cc + ridge * fnp.eye(width, dtype=fnp.float64), cov_cy)
    return y_mean - (c_mean - control_exact) @ beta


def _two_control_variate(
    first_samples: fnp.ndarray,
    first_exact: fnp.ndarray,
    penult_samples: fnp.ndarray,
    penult_analytic: fnp.ndarray,
    final_samples: fnp.ndarray,
) -> fnp.ndarray:
    controls = fnp.concatenate((first_samples, penult_samples), axis=1)
    exact = fnp.concatenate((first_exact, penult_analytic), axis=0)
    n_samples = controls.shape[0]
    dim = controls.shape[1]

    c_mean = fnp.mean(controls, axis=0)
    y_mean = fnp.mean(final_samples, axis=0)

    c_centered = controls - c_mean
    y_centered = final_samples - y_mean

    cov_cc = (c_centered.T @ c_centered) / n_samples
    cov_cy = (c_centered.T @ y_centered) / n_samples

    scale = fnp.mean(fnp.diag(cov_cc))
    ridge = fnp.maximum(scale, 1e-30) * (
        0.018 + 0.9 * dim / fnp.maximum(float(n_samples), 1.0)
    )
    beta = fnp.linalg.solve(cov_cc + ridge * fnp.eye(dim, dtype=fnp.float64), cov_cy)
    return y_mean - (c_mean - exact) @ beta


class Estimator(BaseEstimator):
    def setup(self, context):
        self._setup_seed = context.seed

    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth

        analytic_rows = _gaussian_moment_rows(tuple(mlp.weights))
        n_samples = _sample_count(budget, width, depth)

        activations = _lattice_sphere_inputs(n_samples, width, mlp.seed)
        rows = []
        first_samples = None
        penult_samples = None

        for layer_index, weight in enumerate(mlp.weights):
            activations = fnp.maximum(activations @ weight, 0.0)

            if layer_index == 0:
                first_samples = activations
                rows.append(analytic_rows[0])
            else:
                rows.append(fnp.mean(activations, axis=0))

            if layer_index == depth - 2:
                penult_samples = activations

        if depth > 1 and first_samples is not None:
            try:
                if depth > 2 and penult_samples is not None:
                    rows[-1] = _two_control_variate(
                        first_samples,
                        analytic_rows[0],
                        penult_samples,
                        analytic_rows[-2],
                        activations,
                    )
                else:
                    rows[-1] = _ridge_solve_cv(
                        first_samples,
                        analytic_rows[0],
                        activations,
                        0.012,
                    )
            except Exception:
                try:
                    rows[-1] = _ridge_solve_cv(
                        first_samples,
                        analytic_rows[0],
                        activations,
                        0.02,
                    )
                except Exception:
                    rows[-1] = fnp.mean(activations, axis=0)

        return fnp.stack(rows, axis=0)
```
