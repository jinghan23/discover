from __future__ import annotations

import flopscope as flops
import flopscope.numpy as fnp
from whestbench import BaseEstimator


class Estimator(BaseEstimator):
    def predict(self, mlp, budget):
        width = int(mlp.width)
        depth = int(mlp.depth)

        mus = []
        mu = fnp.zeros(width)
        var = fnp.ones(width)

        for w in mlp.weights:
            pre_mu = w.T @ mu
            pre_var = fnp.maximum((w * w).T @ var, 1e-12)
            pre_std = fnp.sqrt(pre_var)
            a = pre_mu / pre_std
            pdf = flops.stats.norm.pdf(a)
            cdf = flops.stats.norm.cdf(a)

            mu = pre_mu * cdf + pre_std * pdf
            second = (pre_mu * pre_mu + pre_var) * cdf + pre_mu * pre_std * pdf
            var = fnp.maximum(second - mu * mu, 0.0)
            mus.append(mu)

        diag = fnp.asarray(fnp.stack(mus, axis=0), dtype=fnp.float32)
        if depth <= 1:
            return diag

        if budget is None:
            n = 8192
        else:
            per_sample = max(1, depth * (2 * width * width + 6 * width))
            n = int((0.085 * float(budget)) / float(per_sample))
            n = max(1024, min(n, 8192))

        n = max(2, 2 * (n // 2))
        half = n // 2

        rng = fnp.random.default_rng(int(mlp.seed))
        base = fnp.asarray(rng.standard_normal((half, width), dtype=fnp.float32))
        x = fnp.concatenate((base, -base), axis=0)
        x = x - fnp.mean(x, axis=0)

        cov = (x.T @ x) / float(n)
        vals, vecs = fnp.linalg.eigh(cov)
        inv_sqrt = (vecs * (1.0 / fnp.sqrt(fnp.maximum(vals, 1e-12)))[None, :]) @ vecs.T
        x = x @ inv_sqrt

        diag_vars = []
        mu2 = fnp.zeros(width)
        var2 = fnp.ones(width)
        for w in mlp.weights:
            pre_mu = w.T @ mu2
            pre_var = fnp.maximum((w * w).T @ var2, 1e-12)
            pre_std = fnp.sqrt(pre_var)
            a = pre_mu / pre_std
            pdf = flops.stats.norm.pdf(a)
            cdf = flops.stats.norm.cdf(a)
            mu2 = pre_mu * cdf + pre_std * pdf
            second = (pre_mu * pre_mu + pre_var) * cdf + pre_mu * pre_std * pdf
            var2 = fnp.maximum(second - mu2 * mu2, 0.0)
            diag_vars.append(var2)

        for i, w in enumerate(mlp.weights[:-1]):
            x = fnp.maximum(x @ w, 0.0)

            if i < 4:
                xm = fnp.mean(x, axis=0)
                xc = x - xm
                xv = fnp.maximum(fnp.mean(xc * xc, axis=0), 1e-12)
                matched = xc * fnp.sqrt(fnp.maximum(diag_vars[i], 1e-12) / xv) + diag[i]

                if i == 0:
                    x = -0.50 * x + 1.50 * matched
                elif i == 1:
                    x = 0.83 * x + 0.17 * matched
                elif i == 3:
                    x = 1.035 * x - 0.035 * matched

        z = x @ mlp.weights[-1]

        final_mc = fnp.asarray(fnp.mean(fnp.maximum(z, 0.0), axis=0), dtype=fnp.float32)

        z_mu = fnp.mean(z, axis=0)
        zc = z - z_mu
        z_var = fnp.maximum(fnp.mean(zc * zc, axis=0), 1e-12)
        z_std = fnp.sqrt(z_var)
        za = z_mu / z_std
        final_norm = fnp.asarray(
            z_mu * flops.stats.norm.cdf(za) + z_std * flops.stats.norm.pdf(za),
            dtype=fnp.float32,
        )

        final = fnp.asarray(
            0.955 * final_mc + 0.040 * final_norm + 0.005 * diag[-1],
            dtype=fnp.float32,
        )

        return fnp.asarray(
            fnp.concatenate((diag[:-1], final[None, :]), axis=0),
            dtype=fnp.float32,
        )
