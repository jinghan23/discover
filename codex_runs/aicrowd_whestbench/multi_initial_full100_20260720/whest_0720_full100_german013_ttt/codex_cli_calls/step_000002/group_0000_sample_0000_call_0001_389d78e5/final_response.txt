```python
from __future__ import annotations

import flopscope.numpy as fnp
from whestbench import BaseEstimator


_BUDGET_FRACTION = 0.145
_MIN_SAMPLES = 64
_MAX_SAMPLES = 1_000_000
_ARRAY_BYTES_LIMIT = 120 * 1024 * 1024
_WORST_CASE_ITEMSIZE = 8
_INV_SQRT_2PI = 0.3989422804014327
_SQRT_2_OVER_PI = 0.7978845608028654


def _sample_count(budget: int, width: int, depth: int) -> int:
    per_sample = depth * (2 * width * width + 5 * width) + width * width
    fixed = 14 * width**3 + depth * width
    k = (int(_BUDGET_FRACTION * budget) - fixed) // max(per_sample, 1)
    max_k_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _WORST_CASE_ITEMSIZE, 1)
    k = int(max(_MIN_SAMPLES, min(_MAX_SAMPLES, max_k_by_bytes, k)))
    return k - (k & 1)


def _normal_cdf_approx(x):
    return 0.5 * (1.0 + fnp.tanh(_SQRT_2_OVER_PI * (x + 0.044715 * x * x * x)))


def _gaussian_relu_mean(mean, variance):
    variance = fnp.maximum(variance, 1e-12)
    std = fnp.sqrt(variance)
    a = mean / std
    phi = _INV_SQRT_2PI * fnp.exp(-0.5 * a * a)
    return std * phi + mean * _normal_cdf_approx(a)


class Estimator(BaseEstimator):
    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth
        weights = mlp.weights
        k = _sample_count(budget, width, depth)

        rng = fnp.random.default_rng(mlp.seed)
        u = rng.standard_normal((k // 2, width), dtype=fnp.float32)

        covariance = (u.T @ u) / float(k // 2)
        eigenvalues, eigenvectors = fnp.linalg.eigh(covariance)
        eigenvalues = fnp.maximum(eigenvalues, 1e-6)
        inverse_sqrt = (eigenvectors / fnp.sqrt(eigenvalues)) @ eigenvectors.T

        x = fnp.concatenate([u, -u], axis=0)
        z = x @ (inverse_sqrt @ weights[0])
        x = fnp.maximum(z, 0.0)

        first_mean = (
            fnp.sqrt(fnp.sum(weights[0] * weights[0], axis=0)) * _INV_SQRT_2PI
        )
        out = [first_mean[None, :]]

        for weight in weights[1:]:
            z = x @ weight
            raw = fnp.maximum(z, 0.0)

            z_mean = fnp.mean(z, axis=0)
            z_second = fnp.mean(z * z, axis=0)
            z_var = z_second - z_mean * z_mean
            moment_mean = _gaussian_relu_mean(z_mean, z_var)

            sample_mean = fnp.mean(raw, axis=0)
            out.append((0.65 * sample_mean + 0.35 * moment_mean)[None, :])
            x = raw

        return fnp.concatenate(out, axis=0)
```
