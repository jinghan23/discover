from __future__ import annotations

import flopscope.numpy as fnp
from whestbench import BaseEstimator


_BUDGET_FRACTION = 0.24
_MIN_SAMPLES = 64
_MAX_SAMPLES = 1_000_000
_ARRAY_BYTES_LIMIT = 192 * 1024 * 1024
_WORST_CASE_ITEMSIZE = 8

_EPS = 1e-6
_TINY = 1e-12
_INV_SQRT_2PI = 0.3989422804014327
_HALF_MINUS_INV_2PI = 0.3408450569081046


def _sample_count(budget, width: int, depth: int) -> int:
    max_k_by_bytes = _ARRAY_BYTES_LIMIT // max(width * _WORST_CASE_ITEMSIZE, 1)

    if budget is None:
        k = min(_MAX_SAMPLES, max_k_by_bytes)
    else:
        per_sample = depth * (2 * width * width + width) + width * width
        fixed = 11 * width**3 + 8 * width * width
        k = (int(_BUDGET_FRACTION * int(budget)) - fixed) // max(per_sample, 1)
        k = int(max(_MIN_SAMPLES, min(_MAX_SAMPLES, max_k_by_bytes, k)))

    return max(2, int(k) - (int(k) & 1))


def _normal_cdf(x):
    ax = fnp.abs(x)
    t = 1.0 / (1.0 + 0.2316419 * ax)
    poly = t * (
        0.319381530
        + t
        * (
            -0.356563782
            + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429))
        )
    )
    pdf = _INV_SQRT_2PI * fnp.exp(-0.5 * ax * ax)
    cdf_pos = 1.0 - pdf * poly
    return fnp.where(x >= 0.0, cdf_pos, 1.0 - cdf_pos)


def _gaussian_relu_mean(mu, var):
    var = fnp.maximum(var, _EPS)
    sigma = fnp.sqrt(var)
    alpha = mu / sigma
    phi = _INV_SQRT_2PI * fnp.exp(-0.5 * alpha * alpha)
    Phi = _normal_cdf(alpha)
    return sigma * phi + mu * Phi


class Estimator(BaseEstimator):
    def setup(self, context):
        self._setup_seed = context.seed

    def predict(self, mlp, budget):
        width = mlp.width
        depth = mlp.depth
        k = _sample_count(budget, width, depth)

        rng = fnp.random.default_rng(mlp.seed)
        half = k // 2
        u = rng.standard_normal((half, width), dtype=fnp.float32)

        covariance = (u.T @ u) / float(half)
        eigenvalues, eigenvectors = fnp.linalg.eigh(covariance)
        eigenvalues = fnp.maximum(eigenvalues, _EPS)
        inv_sqrt = (eigenvectors / fnp.sqrt(eigenvalues)) @ eigenvectors.T

        x = fnp.concatenate([u, -u], axis=0)
        x = x @ inv_sqrt
        del u, covariance, eigenvalues, eigenvectors, inv_sqrt

        means = []

        for layer, weight in enumerate(mlp.weights):
            z = x @ weight
            x = fnp.maximum(z, 0.0)

            raw_mean = fnp.mean(x, axis=0)

            if layer == 0:
                q = fnp.sum(weight * weight, axis=0)
                exact_mean = fnp.sqrt(fnp.maximum(q, 0.0)) * _INV_SQRT_2PI

                raw_second = fnp.mean(x * x, axis=0)
                raw_var = fnp.maximum(raw_second - raw_mean * raw_mean, _EPS)
                exact_var = fnp.maximum(q * _HALF_MINUS_INV_2PI, _EPS)

                scale = fnp.sqrt(exact_var / raw_var)
                scale = fnp.minimum(2.0, fnp.maximum(0.5, scale))
                x = (x - raw_mean) * scale + exact_mean

                mean = exact_mean
            else:
                z_mean = fnp.mean(z, axis=0)
                z_second = fnp.mean(z * z, axis=0)
                z_var = fnp.maximum(z_second - z_mean * z_mean, _EPS)
                gaussian_mean = _gaussian_relu_mean(z_mean, z_var)

                x_second = fnp.mean(x * x, axis=0)
                x_var = fnp.maximum(x_second - raw_mean * raw_mean, _EPS)
                se2 = x_var / float(k)

                diff = raw_mean - gaussian_mean
                bias2 = fnp.maximum(diff * diff - se2, 0.0)
                direct_weight = bias2 / (bias2 + se2 + _TINY)

                mean = direct_weight * raw_mean + (1.0 - direct_weight) * gaussian_mean

                if layer + 1 < depth:
                    x = x + (mean - raw_mean)

            means.append(mean)

        return fnp.stack(means, axis=0)
