"""
Outgoing TriMul forward.

Computes input LayerNorm in Triton, packs the five input projections into one
GEMM, gates/reorders left and right into hidden-major batched matrices, uses
cuBLAS batched GEMM for the N^3 triangle contraction, then applies hidden
LayerNorm, output gate, and output projection. A fused Triton epilogue is used
for the common dim=hidden=128 small-N path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}


@triton.jit
def _ln_to_half_kernel(
    x, y, norm_w, norm_b,
    total: tl.constexpr, dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)
    mask = (offs_m[:, None] < total) & (offs_d[None, :] < dim)

    v = tl.load(x + offs_m[:, None] * dim + offs_d[None, :], mask=mask, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) / dim
    c = tl.where(offs_d[None, :] < dim, v - mean[:, None], 0.0)
    var = tl.sum(c * c, axis=1) / dim
    rs = tl.rsqrt(var + 1.0e-5)

    w = tl.load(norm_w + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)
    b = tl.load(norm_b + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)
    o = c * rs[:, None] * w[None, :] + b[None, :]

    tl.store(y + offs_m[:, None] * dim + offs_d[None, :], o.to(tl.float16), mask=mask)


@triton.jit
def _gate_reorder_kernel(
    proj, mask, left_h, right_h,
    total: tl.constexpr, n2: tl.constexpr, hidden: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (p[:, None] < total) & (h[None, :] < hidden)

    base = p[:, None] * (5 * hidden) + h[None, :]
    l = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(proj + base + hidden, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * hidden, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * hidden, mask=valid, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)

    if has_mask:
        m = tl.load(mask + p, mask=p < total, other=0.0).to(tl.float32)
        l = l * m[:, None]
        r = r * m[:, None]

    b = p // n2
    pos = p - b * n2
    dst = b[:, None] * (hidden * n2) + h[None, :] * n2 + pos[:, None]

    tl.store(left_h + dst, l.to(tl.float16), mask=valid)
    tl.store(right_h + dst, r.to(tl.float16), mask=valid)


@triton.jit
def _norm_gate_kernel(
    tri_h, proj, normed, norm_w, norm_b,
    total: tl.constexpr, n2: tl.constexpr, hidden: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, BLOCK_H)

    b = p // n2
    pos = p - b * n2
    src = b[None, :] * (hidden * n2) + h[:, None] * n2 + pos[None, :]
    valid = (p[None, :] < total) & (h[:, None] < hidden)

    v = tl.load(tri_h + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    c = v - mean[None, :]
    var = tl.sum(c * c, axis=0) / hidden
    rs = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w + h, mask=h < hidden, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + h, mask=h < hidden, other=0.0).to(tl.float32)
    g = tl.load(
        proj + p[None, :] * (5 * hidden) + 4 * hidden + h[:, None],
        mask=valid,
        other=0.0,
    ).to(tl.float32)

    y = ((c * rs[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(g)
    tl.store(
        normed + p[:, None] * hidden + h[None, :],
        tl.trans(y).to(tl.float16),
        mask=(p[:, None] < total) & (h[None, :] < hidden),
    )


@triton.jit
def _norm_gate_out128_kernel(
    tri_h, proj, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, n2: tl.constexpr, hidden: tl.constexpr, dim: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, BLOCK_H)
    d = tl.arange(0, BLOCK_D)

    b = p // n2
    pos = p - b * n2
    src = b[None, :] * (hidden * n2) + h[:, None] * n2 + pos[None, :]
    valid = (p[None, :] < total) & (h[:, None] < hidden)

    v = tl.load(tri_h + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    c = v - mean[None, :]
    var = tl.sum(c * c, axis=0) / hidden
    rs = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w + h, mask=h < hidden, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + h, mask=h < hidden, other=0.0).to(tl.float32)
    g = tl.load(
        proj + p[None, :] * (5 * hidden) + 4 * hidden + h[:, None],
        mask=valid,
        other=0.0,
    ).to(tl.float32)

    y = tl.trans((((c * rs[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(g)).to(tl.float16))
    w = tl.load(
        w_out_t + h[:, None] * dim + d[None, :],
        mask=(h[:, None] < hidden) & (d[None, :] < dim),
        other=0.0,
    )
    acc = tl.dot(y, w, out_dtype=tl.float32)

    tl.store(
        out + p[:, None] * dim + d[None, :],
        acc,
        mask=(p[:, None] < total) & (d[None, :] < dim),
    )


def _cached_weights(weights):
    keys = (
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        weights["to_out.weight"].data_ptr(),
        weights["left_proj.weight"].shape,
        weights["to_out.weight"].shape,
    )
    cached = _WEIGHT_CACHE.get(keys)
    if cached is not None:
        return cached

    w_proj = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).contiguous().to(torch.float16)

    w_out = weights["to_out.weight"].contiguous().to(torch.float16)
    w_out_t = weights["to_out.weight"].t().contiguous().to(torch.float16)

    _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[keys] = (w_proj, w_out, w_out_t)
    return w_proj, w_out, w_out_t


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n

    torch.backends.cuda.matmul.allow_tf32 = True
    w_proj, w_out, w_out_t = _cached_weights(weights)

    xh = torch.empty((total, dim), device=input_tensor.device, dtype=torch.float16)

    if dim == 128:
        block_d, block_m, warps_ln = 128, 16, 4
    elif dim == 384:
        block_d, block_m, warps_ln = 512, 8, 8
    else:
        block_d, block_m, warps_ln = 1024, 4, 8

    _ln_to_half_kernel[(triton.cdiv(total, block_m),)](
        input_tensor,
        xh,
        weights["norm.weight"],
        weights["norm.bias"],
        total,
        dim,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=warps_ln,
    )

    proj = F.linear(xh, w_proj)

    left_h = torch.empty((bsz * hidden, n, n), device=input_tensor.device, dtype=torch.float16)
    right_h = torch.empty_like(left_h)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_flat = mask.reshape(total) if has_mask else input_tensor.reshape(-1)

    _gate_reorder_kernel[(triton.cdiv(total, 16), triton.cdiv(hidden, 32))](
        proj,
        mask_flat,
        left_h,
        right_h,
        total,
        n2,
        hidden,
        has_mask,
        BLOCK_P=16,
        BLOCK_H=32,
        num_warps=4,
    )

    tri_h = torch.bmm(left_h, right_h.transpose(1, 2))

    if dim == 128 and hidden == 128 and n <= 256:
        out = torch.empty((total, dim), device=input_tensor.device, dtype=torch.float32)
        _norm_gate_out128_kernel[(triton.cdiv(total, 16),)](
            tri_h,
            proj,
            out,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            w_out_t,
            total,
            n2,
            hidden,
            dim,
            BLOCK_P=16,
            BLOCK_H=128,
            BLOCK_D=128,
            num_warps=8,
        )
    else:
        normed = torch.empty((total, hidden), device=input_tensor.device, dtype=torch.float16)
        _norm_gate_kernel[(triton.cdiv(total, 16),)](
            tri_h,
            proj,
            normed,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            total,
            n2,
            hidden,
            BLOCK_P=16,
            BLOCK_H=128,
            num_warps=8,
        )
        out = F.linear(normed, w_out).to(torch.float32)

    return out.reshape(bsz, n, n, dim)
