"""
Outgoing Triangle Multiplicative Update forward pass.

The implementation uses Triton for row-wise LayerNorm, gate/mask fusion, and
the final output normalization/projection. The N^3 triangle contraction is
kept as a strided batched GEMM over hidden channels, which maps well to cuBLAS
on SM80. We specialize the hot dim=128/hidden=128 path by fusing input
LayerNorm with all five small projections, and use cuBLAS for wider input
projection shapes where one large GEMM is faster.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_MODE = 0
_MM_OUT_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    t = _WORKSPACE_CACHE.get(key)
    if t is None:
        if len(_WORKSPACE_CACHE) >= 32:
            _WORKSPACE_CACHE.clear()
        t = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = t
    return t


@triton.jit
def _ln128_half(x, y, nw, nb, total: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BM + tl.arange(0, BM)
    d = tl.arange(0, 128)
    ok = m < total
    v = tl.load(x + m[:, None] * 128 + d[None, :], mask=ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    ss = tl.sum(v * v, axis=1) * 0.0078125
    r = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)
    w = tl.load(nw + d).to(tl.float32)
    b = tl.load(nb + d).to(tl.float32)
    o = (v - mean[:, None]) * r[:, None] * w[None, :] + b[None, :]
    tl.store(y + m[:, None] * 128 + d[None, :], o.to(tl.float16), mask=ok[:, None])


@triton.jit
def _ln384_half(x, y, nw, nb, total: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BM + tl.arange(0, BM)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    ok = m < total

    v0 = tl.load(x + m[:, None] * 384 + d0[None, :], mask=ok[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 384 + 256 + d1[None, :], mask=ok[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0026041666666666665
    r = tl.rsqrt(tl.maximum(ss * 0.0026041666666666665 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(nw + d0).to(tl.float32)
    b0 = tl.load(nb + d0).to(tl.float32)
    w1 = tl.load(nw + 256 + d1).to(tl.float32)
    b1 = tl.load(nb + 256 + d1).to(tl.float32)

    tl.store(y + m[:, None] * 384 + d0[None, :], ((v0 - mean[:, None]) * r[:, None] * w0[None, :] + b0[None, :]).to(tl.float16), mask=ok[:, None])
    tl.store(y + m[:, None] * 384 + 256 + d1[None, :], ((v1 - mean[:, None]) * r[:, None] * w1[None, :] + b1[None, :]).to(tl.float16), mask=ok[:, None])


@triton.jit
def _ln768_half(x, y, nw, nb, total: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)

    mean = s * 0.0013020833333333333

    w0 = tl.load(nw + d0).to(tl.float32)

    tl.store(y + m[:, None] * 768 + 512 + d1[None, :], ((v1 - mean[:, None]) * r[:, None] * w1[None, :] + b1[None, :]).to(tl.float16), mask=ok[:, None])


@triton.jit
def _fused_ln128_proj3_nomask(x, proj_h, wproj, nw, nb, total: tl.constexpr, BM: tl.constexpr, BH: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)
    m = pm * BM + tl.arange(0, BM)
    h = ph * BH + tl.arange(0, BH)
    d = tl.arange(0, 128)
    ok = m < total

    v = tl.load(x + m[:, None] * 128 + d[None, :], mask=ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    ss = tl.sum(v * v, axis=1) * 0.0078125
    r = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)
    w = tl.load(nw + d).to(tl.float32)
    b = tl.load(nb + d).to(tl.float32)
    xn = ((v - mean[:, None]) * r[:, None] * w[None, :] + b[None, :]).to(tl.float16)

    wh = h[None, :] * 128 + d[:, None]
    hm = h[None, :] < 128
    sm = (h[:, None] < 128) & (m[None, :] < total)
    base = h[:, None] * total + m[None, :]

    wl = tl.load(wproj + wh, mask=hm, other=0.0)
    wg = tl.load(wproj + 2 * 128 * 128 + wh, mask=hm, other=0.0)
    left = tl.dot(xn, wl, out_dtype=tl.float32)
    lg = tl.dot(xn, wg, out_dtype=tl.float32)
    tl.store(proj_h + base, tl.trans((left * tl.sigmoid(lg)).to(tl.float16)), mask=sm)

    wr = tl.load(wproj + 128 * 128 + wh, mask=hm, other=0.0)
    wrg = tl.load(wproj + 3 * 128 * 128 + wh, mask=hm, other=0.0)
    right = tl.dot(xn, wr, out_dtype=tl.float32)
    rg = tl.dot(xn, wrg, out_dtype=tl.float32)
    tl.store(proj_h + 128 * total + base, tl.trans((right * tl.sigmoid(rg)).to(tl.float16)), mask=sm)

    wog = tl.load(wproj + 4 * 128 * 128 + wh, mask=hm, other=0.0)
    og = tl.dot(xn, wog, out_dtype=tl.float32)
    tl.store(proj_h + 256 * total + base, tl.trans(og.to(tl.float16)), mask=sm)

    og = tl.dot(xn, wog, out_dtype=tl.float32)

@triton.jit
def _gate_nomask(proj_h, total: tl.constexpr, BP: tl.constexpr, BH: tl.constexpr):
    pp = tl.program_id(0)
    ph = tl.program_id(1)
    p = pp * BP + tl.arange(0, BP)
    h = ph * BH + tl.arange(0, BH)
    ok = p[None, :] < total
    base = h[:, None] * total + p[None, :]

    l = tl.load(proj_h + base, mask=ok, other=0.0).to(tl.float32)
    r = tl.load(proj_h + 128 * total + base, mask=ok, other=0.0).to(tl.float32)
    lg = tl.load(proj_h + 256 * total + base, mask=ok, other=0.0).to(tl.float32)
    rg = tl.load(proj_h + 384 * total + base, mask=ok, other=0.0).to(tl.float32)

    tl.store(proj_h + base, (l * tl.sigmoid(lg)).to(tl.float16), mask=ok)
    tl.store(proj_h + 128 * total + base, (r * tl.sigmoid(rg)).to(tl.float16), mask=ok)

    pp = tl.program_id(0)

    l = tl.load(proj_h + base, mask=ok, other=0.0).to(tl.float32)

@triton.jit
def _norm_gate_out_generic(tri_h, proj_h, out, nw, nb, wout, total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr, hidden: tl.constexpr, dim: tl.constexpr, BP: tl.constexpr, BH: tl.constexpr, BD: tl.constexpr):
    pp = tl.program_id(0)
    pd = tl.program_id(1)
    p = pp * BP + tl.arange(0, BP)
    h = tl.arange(0, BH)
    d = pd * BD + tl.arange(0, BD)

    if bsz == 1:
        src = h[:, None] * total + p[None, :]
    else:
        bid = p // n2
        pos = p - bid * n2
        src = ((h[:, None] * bsz + bid[None, :]) * n2) + pos[None, :]

    ok = (p[None, :] < total) & (h[:, None] < hidden)
    v = tl.load(tri_h + src, mask=ok, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    ss = tl.sum(v * v, axis=0) / hidden
    r = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    w = tl.load(nw + h, mask=h < hidden, other=0.0).to(tl.float32)
    b = tl.load(nb + h, mask=h < hidden, other=0.0).to(tl.float32)
    g = tl.load(proj_h + (4 * hidden + h[:, None]) * total + p[None, :], mask=ok, other=0.0).to(tl.float32)
    y = (((v - mean[None, :]) * r[None, :]) * w[:, None] + b[:, None]) * tl.sigmoid(g)

    wo = tl.load(wout + h[:, None] * dim + d[None, :], mask=(h[:, None] < hidden) & (d[None, :] < dim), other=0.0)
    acc = tl.dot(tl.trans(y.to(tl.float16)), wo, out_dtype=tl.float32)
    tl.store(out + p[:, None] * dim + d[None, :], acc, mask=(p[:, None] < total) & (d[None, :] < dim))


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    wproj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    wout = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (wproj, wout)
    return wproj, wout


def _mm_half_float(a, b, out=None):
    global _MM_MODE, _MM_OUT_MODE

    if out is not None and _MM_OUT_MODE != 2:
        try:
            y = torch.mm(a, b, out=out, out_dtype=torch.float32)
            _MM_OUT_MODE = 1
            return y
        except (TypeError, RuntimeError):
            _MM_OUT_MODE = 2

    if _MM_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        y = torch.mm(a, b, out_dtype=torch.float32)
        _MM_MODE = 1
        return y
    except (TypeError, RuntimeError):
        pass

    try:
        y = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_MODE = 2
        return y
    except (TypeError, RuntimeError):
        _MM_MODE = 3
        return torch.mm(a, b).to(torch.float32)


@torch.inference_mode()
def custom_kernel(data):
    x, mask, weights, config = data
    bsz, n, _, dim = x.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n
    device = x.device

    wproj, wout = _cached_weights(weights)
    has_mask = mask is not None and not bool(config.get("nomask", False))

    if dim == 128 and hidden == 128 and total <= 262144:
        proj_h = _workspace("proj3_h", (3 * hidden, total), device, torch.float16)
        if has_mask:
            _fused_ln128_proj3_mask[(triton.cdiv(total, 32), 2)](
                x, proj_h, wproj, weights["norm.weight"], weights["norm.bias"], mask.reshape(total),
                total, BM=32, BH=64, num_warps=4,
            )
        else:
            _fused_ln128_proj3_nomask[(triton.cdiv(total, 32), 2)](
                x, proj_h, wproj, weights["norm.weight"], weights["norm.bias"],
                total, BM=32, BH=64, num_warps=4,
            )
        gate_base = 256
    else:
        xh = _workspace("xh", (total, dim), device, torch.float16)
        if dim == 128:
            _ln128_half[(triton.cdiv(total, 16),)](
                x, xh, weights["norm.weight"], weights["norm.bias"],
                total, BM=16, num_warps=4,
            )
        elif dim == 384:
            _ln384_half[(triton.cdiv(total, 8),)](
                x, xh, weights["norm.weight"], weights["norm.bias"],
                total, BM=8, num_warps=8,
            )
        elif dim == 768:
            _ln768_half[(triton.cdiv(total, 4),)](
                x, xh, weights["norm.weight"], weights["norm.bias"],
                total, BM=4, num_warps=8,
            )
        else:
            xh.copy_(torch.nn.functional.layer_norm(
                x.reshape(total, dim), (dim,),
                weights["norm.weight"], weights["norm.bias"], 1.0e-5,
            ).to(torch.float16))

        proj_h = _workspace("proj_h", (5 * hidden, total), device, torch.float16)
        torch.mm(wproj, xh.t(), out=proj_h)

        if has_mask:
            _gate_mask[(triton.cdiv(total, 64), 4)](
                proj_h, mask.reshape(total), total,
                BP=64, BH=32, num_warps=4,
            )
        else:
            _gate_nomask[(triton.cdiv(total, 64), 4)](
                proj_h, total,
                BP=64, BH=32, num_warps=4,
            )
        gate_base = 512

    left = proj_h[:hidden].view(hidden * bsz, n, n)
    right = proj_h[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri_h = _workspace("tri_h", (hidden * bsz, n, n), device, torch.float16)
    torch.bmm(left, right.transpose(1, 2), out=tri_h)

    if dim == 128 and hidden == 128:
        out = torch.empty((total, 128), device=device, dtype=torch.float32)
        if bsz == 2:
            _norm_gate_out128_b2[(triton.cdiv(n2, 32), 2)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout,
                n2, gate_base, BP=32, num_warps=8,
            )
        else:
            _norm_gate_out128_b1[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout,
                total, gate_base, BP=32, num_warps=8,
            )
    elif dim == 384 and hidden == 128 and total <= 262144:
        out = torch.empty((total, 384), device=device, dtype=torch.float32)
        if bsz == 2:
            _norm_gate_out384_b2[(triton.cdiv(n2, 32), 2)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout,
                n2, BP=32, num_warps=8,
            )
        else:
            _norm_gate_out384_b1[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout,
                total, BP=32, num_warps=8,
            )
    elif dim == 768 and hidden == 128 and total <= 131072:
        out = torch.empty((total, dim), device=device, dtype=torch.float32)
        _norm_gate_out_generic[(triton.cdiv(total, 32), triton.cdiv(dim, 128))](
            tri_h, proj_h, out,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout,
            total, n2, bsz, hidden, dim,
            BP=32, BH=128, BD=128, num_warps=8,
        )
    else:
        normed = _workspace("normed", (total, hidden), device, torch.float16)
        if bsz == 2:
            _norm_gate_hb_b2[(triton.cdiv(n2, 32), 2)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                n2, BP=32, num_warps=8,
            )
        else:
            _norm_gate_hb_b1[(triton.cdiv(total, 32),)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, BP=32, num_warps=8,
            )

        out_buf = None
        if _MM_OUT_MODE != 2:
            out_buf = torch.empty((total, dim), device=device, dtype=torch.float32)
        out = _mm_half_float(normed, wout, out_buf)

    return out.view(bsz, n, n, dim)
