"""
Outgoing Triangle Multiplicative Update forward pass.

LayerNorm is computed in fp32 and cast to fp16 for tensor-core projections.
The five input projections are materialized channel-major, input gates and mask
are fused in Triton, the triangle product uses grouped cuBLAS BMM over hidden
channels, and output LayerNorm + output gate + final projection are fused for
the hidden=128 path.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_MM_OUT_DTYPE_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


@triton.jit
def _ln128_to_half_kernel(x, y, norm_w, norm_b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, 128)
    m = offs_m < total

    vals = tl.load(x + offs_m[:, None] * 128 + offs_d[None, :], mask=m[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    ss = tl.sum(vals * vals, axis=1) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    w = tl.load(norm_w + offs_d).to(tl.float32)
    b = tl.load(norm_b + offs_d).to(tl.float32)
    out = (vals - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]
    tl.store(y + offs_m[:, None] * 128 + offs_d[None, :], out.to(tl.float16), mask=m[:, None])


@triton.jit
def _ln384_to_half_kernel(x, y, norm_w, norm_b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    m = offs_m < total

    v0 = tl.load(x + offs_m[:, None] * 384 + d0[None, :], mask=m[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + offs_m[:, None] * 384 + 256 + d1[None, :], mask=m[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 384.0
    rstd = tl.rsqrt(tl.maximum(ss / 384.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(norm_w + d0).to(tl.float32)
    b0 = tl.load(norm_b + d0).to(tl.float32)
    w1 = tl.load(norm_w + 256 + d1).to(tl.float32)
    b1 = tl.load(norm_b + 256 + d1).to(tl.float32)

    tl.store(
        y + offs_m[:, None] * 384 + d0[None, :],
        ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
        mask=m[:, None],
    )
    tl.store(
        y + offs_m[:, None] * 384 + 256 + d1[None, :],
        ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
        mask=m[:, None],
    )


@triton.jit
def _ln768_to_half_kernel(x, y, norm_w, norm_b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    m = offs_m < total

    v0 = tl.load(x + offs_m[:, None] * 768 + d0[None, :], mask=m[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + offs_m[:, None] * 768 + 512 + d1[None, :], mask=m[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 768.0
    rstd = tl.rsqrt(tl.maximum(ss / 768.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(norm_w + d0).to(tl.float32)
    b0 = tl.load(norm_b + d0).to(tl.float32)
    w1 = tl.load(norm_w + 512 + d1).to(tl.float32)
    b1 = tl.load(norm_b + 512 + d1).to(tl.float32)

    tl.store(
        y + offs_m[:, None] * 768 + d0[None, :],
        ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
        mask=m[:, None],
    )
    tl.store(
        y + offs_m[:, None] * 768 + 512 + d1[None, :],
        ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
        mask=m[:, None],
    )


@triton.jit
def _gate_h_inplace_nomask_kernel(
    proj_h,
    total: tl.constexpr, hidden: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    base = offs_h[:, None] * total + offs_p[None, :]

    left = tl.load(proj_h + base, mask=valid, other=0.0).to(tl.float32)
    right = tl.load(proj_h + hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj_h + 2 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj_h + 3 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)

    tl.store(proj_h + base, (left * tl.sigmoid(lg)).to(tl.float16), mask=valid)
    tl.store(proj_h + hidden * total + base, (right * tl.sigmoid(rg)).to(tl.float16), mask=valid)


@triton.jit
def _gate_h_inplace_mask_kernel(
    proj_h, mask,
    total: tl.constexpr, hidden: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    base = offs_h[:, None] * total + offs_p[None, :]

    left = tl.load(proj_h + base, mask=valid, other=0.0).to(tl.float32)
    right = tl.load(proj_h + hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj_h + 2 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj_h + 3 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    m = tl.load(mask + offs_p, mask=offs_p < total, other=0.0).to(tl.float32)

    tl.store(proj_h + base, (left * tl.sigmoid(lg) * m[None, :]).to(tl.float16), mask=valid)
    tl.store(proj_h + hidden * total + base, (right * tl.sigmoid(rg) * m[None, :]).to(tl.float16), mask=valid)


@triton.jit
def _norm_gate_out128_b1_kernel(
    tri_h, proj_h, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, hidden: tl.constexpr, dim: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, BLOCK_H)
    offs_d = tl.arange(0, BLOCK_D)

    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    vals = tl.load(tri_h + offs_h[:, None] * total + offs_p[None, :], mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) / hidden
    ss = tl.sum(vals * vals, axis=0) / hidden
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj_h + (4 * hidden + offs_h[:, None]) * total + offs_p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    w = tl.load(
        w_out_t + offs_h[:, None] * dim + offs_d[None, :],
        mask=(offs_h[:, None] < hidden) & (offs_d[None, :] < dim),
        other=0.0,
    )
    acc = tl.dot(tl.trans(y.to(tl.float16)), w, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * dim + offs_d[None, :], acc, mask=(offs_p[:, None] < total) & (offs_d[None, :] < dim))


@triton.jit
def _norm_gate_out128_hb_kernel(
    tri_h, proj_h, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    hidden: tl.constexpr, dim: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, BLOCK_H)
    offs_d = tl.arange(0, BLOCK_D)

    b = offs_p // n2
    pos = offs_p - b * n2
    tri_src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    vals = tl.load(tri_h + tri_src, mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) / hidden
    ss = tl.sum(vals * vals, axis=0) / hidden
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj_h + (4 * hidden + offs_h[:, None]) * total + offs_p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    w = tl.load(
        w_out_t + offs_h[:, None] * dim + offs_d[None, :],
        mask=(offs_h[:, None] < hidden) & (offs_d[None, :] < dim),
        other=0.0,
    )
    acc = tl.dot(tl.trans(y.to(tl.float16)), w, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * dim + offs_d[None, :], acc, mask=(offs_p[:, None] < total) & (offs_d[None, :] < dim))

def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_proj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (w_proj, w_out_t)
    return w_proj, w_out_t


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_MODE

    if _MM_OUT_DTYPE_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_OUT_DTYPE_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_OUT_DTYPE_MODE = 1
        return out
    except (TypeError, RuntimeError):
        pass

    try:
        out = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_OUT_DTYPE_MODE = 2
        return out
    except (TypeError, RuntimeError):
        _MM_OUT_DTYPE_MODE = 3
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n

    w_proj, w_out_t = _cached_weights(weights)
    xh = torch.empty((total, dim), device=input_tensor.device, dtype=torch.float16)

    if dim == 128:
        _ln128_to_half_kernel[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=16, num_warps=4,
        )
    elif dim == 384:
        _ln384_to_half_kernel[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=8, num_warps=8,
        )
    else:
        _ln768_to_half_kernel[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=4, num_warps=8,
        )

    proj_h = torch.mm(w_proj, xh.t())

    has_mask = mask is not None and not bool(config.get("nomask", False))
    grid_gate = (triton.cdiv(total, 64), triton.cdiv(hidden, 32))
    if has_mask:
        _gate_h_inplace_mask_kernel[grid_gate](
            proj_h, mask.reshape(total), total, hidden,
            BLOCK_P=64, BLOCK_H=32, num_warps=4,
        )
    else:
        _gate_h_inplace_nomask_kernel[grid_gate](
            proj_h, total, hidden,
            BLOCK_P=64, BLOCK_H=32, num_warps=4,
        )

    left_h = proj_h[:hidden].view(hidden * bsz, n, n)
    right_h = proj_h[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri_h = torch.bmm(left_h, right_h.transpose(1, 2))

    if dim == 128 and hidden == 128:
        out = torch.empty((total, dim), device=input_tensor.device, dtype=torch.float32)
        if bsz == 1:
            _norm_gate_out128_b1_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, hidden, dim,
                BLOCK_P=32, BLOCK_H=128, BLOCK_D=128, num_warps=8,
            )
        else:
            _norm_gate_out128_hb_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, n2, bsz, hidden, dim,
                BLOCK_P=32, BLOCK_H=128, BLOCK_D=128, num_warps=8,
            )
    else:
        normed = torch.empty((total, hidden), device=input_tensor.device, dtype=torch.float16)
        if bsz == 1:
            _norm_gate_h_b1_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, hidden,
                BLOCK_P=32, BLOCK_H=128, num_warps=8,
            )
        else:
            _norm_gate_h_hb_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, n2, bsz, hidden,
                BLOCK_P=32, BLOCK_H=128, num_warps=8,
            )
        out = _mm_half_to_float(normed, w_out_t)

    return out.reshape(bsz, n, n, dim)
