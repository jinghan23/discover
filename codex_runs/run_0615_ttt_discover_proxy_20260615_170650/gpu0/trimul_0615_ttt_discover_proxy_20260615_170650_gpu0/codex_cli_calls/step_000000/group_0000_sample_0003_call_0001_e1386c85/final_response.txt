"""
Outgoing Triangle Multiplicative Update.

The implementation keeps the large dense linear projections in cuBLAS, then uses
Triton kernels to apply gates/masks while converting pair-major projections into
hidden-major matrices for batched triangular products. The N^3 update is executed
as B*H independent half-precision bmm operations with FP32 accumulation on tensor
cores. A second Triton kernel performs the hidden-dimension LayerNorm and output
gate, then cuBLAS applies the final projection. The returned tensor is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}


@triton.jit
def _gate_reorder_kernel(
    proj,
    mask,
    left_h,
    right_h,
    out_gate_h,
    total: tl.constexpr,
    n2: tl.constexpr,
    hidden: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    valid = (offs_p[:, None] < total) & (offs_h[None, :] < hidden)

    base = offs_p[:, None] * (5 * hidden) + offs_h[None, :]
    left = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    right = tl.load(proj + base + hidden, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * hidden, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * hidden, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(proj + base + 4 * hidden, mask=valid, other=0.0).to(tl.float32)

    left = left * (1.0 / (1.0 + tl.exp(-lg)))
    right = right * (1.0 / (1.0 + tl.exp(-rg)))
    out_gate = 1.0 / (1.0 + tl.exp(-og))

    if has_mask:
        m = tl.load(mask + offs_p, mask=offs_p < total, other=0.0).to(tl.float32)
        left = left * m[:, None]
        right = right * m[:, None]

    b = offs_p // n2
    pos = offs_p - b * n2
    dst = b[:, None] * (hidden * n2) + offs_h[None, :] * n2 + pos[:, None]

    tl.store(left_h + dst, left, mask=valid)
    tl.store(right_h + dst, right, mask=valid)
    tl.store(out_gate_h + dst, out_gate, mask=valid)


@triton.jit
def _norm_gate_kernel(
    tri_h,
    out_gate_h,
    normed,
    norm_w,
    norm_b,
    total: tl.constexpr,
    n2: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, BLOCK_H)

    b = offs_p // n2
    pos = offs_p - b * n2

    src = b[None, :] * (hidden * n2) + offs_h[:, None] * n2 + pos[None, :]
    mask = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)

    vals = tl.load(tri_h + src, mask=mask, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    centered = vals - mean[None, :]
    var = tl.sum(centered * centered, axis=0) / hidden
    rstd = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(out_gate_h + src, mask=mask, other=0.0).to(tl.float32)

    y = (centered * rstd[None, :]) * nw[:, None] + nb[:, None]
    y = y * gate

    dst = offs_p[:, None] * hidden + offs_h[None, :]
    tl.store(
        normed + dst,
        tl.trans(y),
        mask=(offs_p[:, None] < total) & (offs_h[None, :] < hidden),
    )


def _cached_weights(weights):
    keys = (
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        weights["to_out.weight"].data_ptr(),
        weights["left_proj.weight"].shape,
        weights["to_out.weight"].shape,
    )
    cached = _WEIGHT_CACHE.get(keys)
    if cached is not None:
        return cached

    w_proj = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).contiguous().to(torch.float16)

    w_out = weights["to_out.weight"].contiguous().to(torch.float16)
    _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[keys] = (w_proj, w_out)
    return w_proj, w_out


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n

    torch.backends.cuda.matmul.allow_tf32 = True

    w_proj, w_out = _cached_weights(weights)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weights["norm.weight"],
        weights["norm.bias"],
        1.0e-5,
    )

    xh = x.reshape(total, dim).to(torch.float16)
    proj = F.linear(xh, w_proj)

    left_h = torch.empty((bsz * hidden, n, n), device=input_tensor.device, dtype=torch.float16)
    right_h = torch.empty_like(left_h)
    out_gate_h = torch.empty_like(left_h)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_flat = mask.reshape(total) if has_mask else input_tensor.reshape(-1)

    grid_gate = (triton.cdiv(total, 16), triton.cdiv(hidden, 32))
    _gate_reorder_kernel[grid_gate](
        proj,
        mask_flat,
        left_h,
        right_h,
        out_gate_h,
        total,
        n2,
        hidden,
        has_mask,
        BLOCK_P=16,
        BLOCK_H=32,
        num_warps=4,
    )

    tri_h = torch.bmm(left_h, right_h.transpose(1, 2))

    normed = torch.empty((total, hidden), device=input_tensor.device, dtype=torch.float16)
    grid_norm = (triton.cdiv(total, 16),)
    _norm_gate_kernel[grid_norm](
        tri_h,
        out_gate_h,
        normed,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total,
        n2,
        hidden,
        BLOCK_P=16,
        BLOCK_H=128,
        num_warps=8,
    )

    out = F.linear(normed, w_out).to(torch.float32)
    return out.reshape(bsz, n, n, dim)
