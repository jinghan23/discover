"""
Outgoing Triangle Multiplicative Update forward pass.

This version keeps the high-flop triangle product on cuBLAS batched tensor-core
GEMM, while Triton handles the layout-sensitive LayerNorm/gating steps. It avoids
the large 5H projection GEMM by projecting only left/right first, computing the
triangle product, and projecting the output gate later. This reduces peak
workspace and removes one always-on hidden gate block before the dominant bmm.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    t = _WORKSPACE_CACHE.get(key)
    if t is None:
        if len(_WORKSPACE_CACHE) >= 64:
            _WORKSPACE_CACHE.clear()
        t = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = t
    return t


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_lr = torch.cat(tensors[:2], dim=0).contiguous().to(torch.float16)
    w_lrg = torch.cat(tensors[:4], dim=0).contiguous().to(torch.float16)
    w_og = tensors[4].contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 16:
        _WEIGHT_CACHE.clear()
    cached = (w_lr, w_lrg, w_og, w_out_t)
    _WEIGHT_CACHE[key] = cached
    return cached


def _mm_half_to_float(a, b):
    global _MM_MODE
    if _MM_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_MODE = 1
        return out
    except Exception:
        pass
    try:
        out = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_MODE = 2
        return out
    except Exception:
        _MM_MODE = 3
        return torch.mm(a, b).to(torch.float32)


@triton.jit
def _ln128_kernel(x, y, nw, nb, total: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BM + tl.arange(0, BM)
    d = tl.arange(0, 128)
    ok = m < total
    v = tl.load(x + m[:, None] * 128 + d[None, :], mask=ok[:, None], other=0.).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    var = tl.sum(v * v, axis=1) * 0.0078125 - mean * mean
    r = tl.rsqrt(tl.maximum(var, 0.) + 1.0e-5)
    w = tl.load(nw + d).to(tl.float32)
    b = tl.load(nb + d).to(tl.float32)
    o = (v - mean[:, None]) * r[:, None] * w[None, :] + b[None, :]
    tl.store(y + m[:, None] * 128 + d[None, :], o.to(tl.float16), mask=ok[:, None])


@triton.jit
def _ln384_kernel(x, y, nw, nb, total: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BM + tl.arange(0, BM)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    ok = m < total
    v0 = tl.load(x + m[:, None] * 384 + d0[None, :], mask=ok[:, None], other=0.).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 384 + 256 + d1[None, :], mask=ok[:, None], other=0.).to(tl.float32)
    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 384.0
    r = tl.rsqrt(tl.maximum(ss / 384.0 - mean * mean, 0.) + 1.0e-5)
    w0 = tl.load(nw + d0).to(tl.float32)
    b0 = tl.load(nb + d0).to(tl.float32)
    w1 = tl.load(nw + 256 + d1).to(tl.float32)
    b1 = tl.load(nb + 256 + d1).to(tl.float32)
    tl.store(y + m[:, None] * 384 + d0[None, :],
             ((v0 - mean[:, None]) * r[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=ok[:, None])
    tl.store(y + m[:, None] * 384 + 256 + d1[None, :],
             ((v1 - mean[:, None]) * r[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=ok[:, None])


@triton.jit
def _ln768_kernel(x, y, nw, nb, total: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BM + tl.arange(0, BM)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    ok = m < total
    v0 = tl.load(x + m[:, None] * 768 + d0[None, :], mask=ok[:, None], other=0.).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 768 + 512 + d1[None, :], mask=ok[:, None], other=0.).to(tl.float32)
    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 768.0
    r = tl.rsqrt(tl.maximum(ss / 768.0 - mean * mean, 0.) + 1.0e-5)
    w0 = tl.load(nw + d0).to(tl.float32)
    b0 = tl.load(nb + d0).to(tl.float32)
    w1 = tl.load(nw + 512 + d1).to(tl.float32)
    b1 = tl.load(nb + 512 + d1).to(tl.float32)
    tl.store(y + m[:, None] * 768 + d0[None, :],
             ((v0 - mean[:, None]) * r[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=ok[:, None])
    tl.store(y + m[:, None] * 768 + 512 + d1[None, :],
             ((v1 - mean[:, None]) * r[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=ok[:, None])


@triton.jit
def _gate4_kernel(p, mask, total: tl.constexpr, hidden: tl.constexpr,
                  has_mask: tl.constexpr, BP: tl.constexpr, BH: tl.constexpr):
    pp = tl.program_id(0) * BP + tl.arange(0, BP)
    hh = tl.program_id(1) * BH + tl.arange(0, BH)
    ok = (pp[None, :] < total) & (hh[:, None] < hidden)
    base = hh[:, None] * total + pp[None, :]

    l = tl.load(p + base, mask=ok, other=0.).to(tl.float32)
    r = tl.load(p + hidden * total + base, mask=ok, other=0.).to(tl.float32)
    lg = tl.load(p + 2 * hidden * total + base, mask=ok, other=0.).to(tl.float32)
    rg = tl.load(p + 3 * hidden * total + base, mask=ok, other=0.).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)

    if has_mask:
        mv = tl.load(mask + pp, mask=pp < total, other=0.).to(tl.float32)
        l = l * mv[None, :]
        r = r * mv[None, :]

    tl.store(p + base, l.to(tl.float16), mask=ok)
    tl.store(p + hidden * total + base, r.to(tl.float16), mask=ok)


@triton.jit
def _gate2_nomask_kernel(p, total: tl.constexpr, hidden: tl.constexpr,
                         BP: tl.constexpr, BH: tl.constexpr):
    pp = tl.program_id(0) * BP + tl.arange(0, BP)
    hh = tl.program_id(1) * BH + tl.arange(0, BH)
    ok = (pp[None, :] < total) & (hh[:, None] < hidden)
    base = hh[:, None] * total + pp[None, :]

    l = tl.load(p + base, mask=ok, other=0.).to(tl.float32)
    r = tl.load(p + hidden * total + base, mask=ok, other=0.).to(tl.float32)
    lg = tl.load(p + 2 * hidden * total + base, mask=ok, other=0.).to(tl.float32)
    rg = tl.load(p + 3 * hidden * total + base, mask=ok, other=0.).to(tl.float32)
    tl.store(p + base, (l * tl.sigmoid(lg)).to(tl.float16), mask=ok)
    tl.store(p + hidden * total + base, (r * tl.sigmoid(rg)).to(tl.float16), mask=ok)


@triton.jit
def _norm_gate_store_kernel(tri, og, normed, nw, nb,
                            total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
                            hidden: tl.constexpr, BP: tl.constexpr, BH: tl.constexpr):
    pid = tl.program_id(0)
    pp = pid * BP + tl.arange(0, BP)
    hh = tl.arange(0, BH)
    b = pp // n2
    pos = pp - b * n2
    src = ((hh[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    ok = (pp[None, :] < total) & (hh[:, None] < hidden)

    v = tl.load(tri + src, mask=ok, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    var = tl.sum(v * v, axis=0) / hidden - mean * mean
    r = tl.rsqrt(tl.maximum(var, 0.) + 1.0e-5)
    w = tl.load(nw + hh, mask=hh < hidden, other=0.).to(tl.float32)
    b0 = tl.load(nb + hh, mask=hh < hidden, other=0.).to(tl.float32)
    g = tl.load(og + hh[:, None] * total + pp[None, :], mask=ok, other=0.).to(tl.float32)
    y = ((v - mean[None, :]) * r[None, :] * w[:, None] + b0[:, None]) * tl.sigmoid(g)

    tl.store(normed + pp[:, None] * hidden + hh[None, :],
             tl.trans(y).to(tl.float16),
             mask=(pp[:, None] < total) & (hh[None, :] < hidden))


@triton.jit
def _norm_gate_store_b1_kernel(tri, og, normed, nw, nb,
                               total: tl.constexpr, hidden: tl.constexpr,
                               BP: tl.constexpr, BH: tl.constexpr):
    pid = tl.program_id(0)
    pp = pid * BP + tl.arange(0, BP)
    hh = tl.arange(0, BH)
    ok = (pp[None, :] < total) & (hh[:, None] < hidden)
    src = hh[:, None] * total + pp[None, :]

    v = tl.load(tri + src, mask=ok, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    var = tl.sum(v * v, axis=0) / hidden - mean * mean
    r = tl.rsqrt(tl.maximum(var, 0.) + 1.0e-5)
    w = tl.load(nw + hh, mask=hh < hidden, other=0.).to(tl.float32)
    b0 = tl.load(nb + hh, mask=hh < hidden, other=0.).to(tl.float32)
    g = tl.load(og + hh[:, None] * total + pp[None, :], mask=ok, other=0.).to(tl.float32)
    y = ((v - mean[None, :]) * r[None, :] * w[:, None] + b0[:, None]) * tl.sigmoid(g)

    tl.store(normed + pp[:, None] * hidden + hh[None, :],
             tl.trans(y).to(tl.float16),
             mask=(pp[:, None] < total) & (hh[None, :] < hidden))


@triton.jit
def _norm_gate_out128_kernel(tri, og, out, nw, nb, wout,
                             total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
                             BP: tl.constexpr):
    pid = tl.program_id(0)
    pp = pid * BP + tl.arange(0, BP)
    hh = tl.arange(0, 128)
    dd = tl.arange(0, 128)
    b = pp // n2
    pos = pp - b * n2
    src = ((hh[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    ok = pp[None, :] < total

    v = tl.load(tri + src, mask=ok, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=0) * 0.0078125
    var = tl.sum(v * v, axis=0) * 0.0078125 - mean * mean
    r = tl.rsqrt(tl.maximum(var, 0.) + 1.0e-5)

    w = tl.load(nw + hh).to(tl.float32)
    b0 = tl.load(nb + hh).to(tl.float32)
    g = tl.load(og + hh[:, None] * total + pp[None, :], mask=ok, other=0.).to(tl.float32)
    y = ((v - mean[None, :]) * r[None, :] * w[:, None] + b0[:, None]) * tl.sigmoid(g)

    wo = tl.load(wout + hh[:, None] * 128 + dd[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), wo, out_dtype=tl.float32)
    tl.store(out + pp[:, None] * 128 + dd[None, :], acc, mask=pp[:, None] < total)


@triton.jit
def _norm_gate_out128_b1_kernel(tri, og, out, nw, nb, wout,
                                total: tl.constexpr, BP: tl.constexpr):
    pid = tl.program_id(0)
    pp = pid * BP + tl.arange(0, BP)
    hh = tl.arange(0, 128)
    dd = tl.arange(0, 128)
    ok = pp[None, :] < total
    src = hh[:, None] * total + pp[None, :]

    v = tl.load(tri + src, mask=ok, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=0) * 0.0078125
    var = tl.sum(v * v, axis=0) * 0.0078125 - mean * mean
    r = tl.rsqrt(tl.maximum(var, 0.) + 1.0e-5)

    w = tl.load(nw + hh).to(tl.float32)
    b0 = tl.load(nb + hh).to(tl.float32)
    g = tl.load(og + hh[:, None] * total + pp[None, :], mask=ok, other=0.).to(tl.float32)
    y = ((v - mean[None, :]) * r[None, :] * w[:, None] + b0[:, None]) * tl.sigmoid(g)

    wo = tl.load(wout + hh[:, None] * 128 + dd[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), wo, out_dtype=tl.float32)
    tl.store(out + pp[:, None] * 128 + dd[None, :], acc, mask=pp[:, None] < total)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n
    device = input_tensor.device

    w_lr, w_lrg, w_og, w_out_t = _cached_weights(weights)
    xh = _workspace("xh", (total, dim), device, torch.float16)

    if dim == 128:
        _ln128_kernel[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BM=16, num_warps=4,
        )
    elif dim == 384:
        _ln384_kernel[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BM=8, num_warps=8,
        )
    else:
        _ln768_kernel[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BM=4, num_warps=8,
        )

    has_mask = mask is not None and not bool(config.get("nomask", False))

    if has_mask:
        proj = _workspace("proj4", (4 * hidden, total), device, torch.float16)
        torch.mm(w_lrg, xh.t(), out=proj)
        _gate4_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
            proj, mask.reshape(total), total, hidden, True,
            BP=64, BH=32, num_warps=4,
        )
    else:
        proj = _workspace("proj4", (4 * hidden, total), device, torch.float16)
        torch.mm(w_lrg, xh.t(), out=proj)
        _gate2_nomask_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
            proj, total, hidden, BP=64, BH=32, num_warps=4,
        )

    left_h = proj[:hidden].view(hidden * bsz, n, n)
    right_h = proj[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri_h = _workspace("tri_h", (hidden * bsz, n, n), device, torch.float16)
    torch.bmm(left_h, right_h.transpose(1, 2), out=tri_h)

    og = _workspace("og", (hidden, total), device, torch.float16)
    torch.mm(w_og, xh.t(), out=og)

    if dim == 128 and hidden == 128:
        out = torch.empty((total, dim), device=device, dtype=torch.float32)
        if bsz == 1:
            _norm_gate_out128_b1_kernel[(triton.cdiv(total, 32),)](
                tri_h, og, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, BP=32, num_warps=8,
            )
        else:
            _norm_gate_out128_kernel[(triton.cdiv(total, 32),)](
                tri_h, og, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, n2, bsz, BP=32, num_warps=8,
            )
    else:
        normed = _workspace("normed", (total, hidden), device, torch.float16)
        if bsz == 1:
            _norm_gate_store_b1_kernel[(triton.cdiv(total, 32),)](
                tri_h, og, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, hidden, BP=32, BH=128, num_warps=8,
            )
        else:
            _norm_gate_store_kernel[(triton.cdiv(total, 32),)](
                tri_h, og, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, n2, bsz, hidden, BP=32, BH=128, num_warps=8,
            )
        out = _mm_half_to_float(normed, w_out_t)

    return out.reshape(bsz, n, n, dim)
