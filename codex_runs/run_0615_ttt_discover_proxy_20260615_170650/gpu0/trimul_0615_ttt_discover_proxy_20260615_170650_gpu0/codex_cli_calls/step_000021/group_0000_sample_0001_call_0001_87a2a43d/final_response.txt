"""
Outgoing TriMul forward: layer-normalize pair features, compute the five input
projections as one tensor-core GEMM, apply input gates/mask in Triton, evaluate
the per-hidden outgoing triangle product with batched GEMM, then fuse output
LayerNorm, output gate, and the final projection where possible. Workspaces and
fp16 weight views are cached by tensor identity and shape.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    out = _WORKSPACE_CACHE.get(key)
    if out is None:
        if len(_WORKSPACE_CACHE) >= 20:
            _WORKSPACE_CACHE.clear()
        out = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = out
    return out


@triton.jit
def _ln128_kernel(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    om = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    od = tl.arange(0, 128)
    mask = om < total

    v = tl.load(x + om[:, None] * 128 + od[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    var = tl.sum(v * v, axis=1) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    ww = tl.load(w + od).to(tl.float32)
    bb = tl.load(b + od).to(tl.float32)
    o = (v - mean[:, None]) * rstd[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + om[:, None] * 128 + od[None, :], o.to(tl.float16), mask=mask[:, None])


@triton.jit
def _ln384_kernel(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    om = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    mask = om < total

    v0 = tl.load(x + om[:, None] * 384 + d0[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + om[:, None] * 384 + 256 + d1[None, :], mask=mask[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 384.0
    rstd = tl.rsqrt(tl.maximum(ss / 384.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 256 + d1).to(tl.float32)
    b1 = tl.load(b + 256 + d1).to(tl.float32)

    tl.store(y + om[:, None] * 384 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=mask[:, None])
    tl.store(y + om[:, None] * 384 + 256 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=mask[:, None])


@triton.jit
def _ln768_kernel(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    om = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    mask = om < total

    v0 = tl.load(x + om[:, None] * 768 + d0[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + om[:, None] * 768 + 512 + d1[None, :], mask=mask[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 768.0
    rstd = tl.rsqrt(tl.maximum(ss / 768.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 512 + d1).to(tl.float32)
    b1 = tl.load(b + 512 + d1).to(tl.float32)

    tl.store(y + om[:, None] * 768 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=mask[:, None])
    tl.store(y + om[:, None] * 768 + 512 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=mask[:, None])


@triton.jit
def _gate_kernel(proj, mask, total: tl.constexpr, hidden: tl.constexpr,
                 has_mask: tl.constexpr, BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr):
    pp = tl.program_id(0) * BLOCK_P + tl.arange(0, BLOCK_P)
    hh = tl.program_id(1) * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (pp[None, :] < total) & (hh[:, None] < hidden)
    base = hh[:, None] * total + pp[None, :]

    l = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(proj + hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + 2 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + 3 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)

    if has_mask:
        m = tl.load(mask + pp, mask=pp < total, other=0.0).to(tl.float32)
        l = l * m[None, :]
        r = r * m[None, :]

    tl.store(proj + base, l.to(tl.float16), mask=valid)
    tl.store(proj + hidden * total + base, r.to(tl.float16), mask=valid)


@triton.jit
def _final128_b1_kernel(tri, proj, out, nw, nb, wt,
                        total: tl.constexpr, BLOCK_P: tl.constexpr):
    pp = tl.program_id(0) * BLOCK_P + tl.arange(0, BLOCK_P)
    hh = tl.arange(0, 128)
    dd = tl.arange(0, 128)
    valid = pp[None, :] < total

    vals = tl.load(tri + hh[:, None] * total + pp[None, :], mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    var = tl.sum(vals * vals, axis=0) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    w = tl.load(nw + hh).to(tl.float32)
    b = tl.load(nb + hh).to(tl.float32)
    g = tl.load(proj + (512 + hh[:, None]) * total + pp[None, :], mask=valid, other=0.0).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * w[:, None] + b[:, None]) * tl.sigmoid(g)

    wo = tl.load(wt + hh[:, None] * 128 + dd[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), wo, out_dtype=tl.float32)
    tl.store(out + pp[:, None] * 128 + dd[None, :], acc, mask=pp[:, None] < total)


@triton.jit
def _final128_kernel(tri, proj, out, nw, nb, wt,
                     total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
                     BLOCK_P: tl.constexpr):
    pp = tl.program_id(0) * BLOCK_P + tl.arange(0, BLOCK_P)
    hh = tl.arange(0, 128)
    dd = tl.arange(0, 128)

    bidx = pp // n2
    pos = pp - bidx * n2
    src = ((hh[:, None] * bsz + bidx[None, :]) * n2) + pos[None, :]
    valid = pp[None, :] < total

    vals = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    var = tl.sum(vals * vals, axis=0) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    w = tl.load(nw + hh).to(tl.float32)
    b = tl.load(nb + hh).to(tl.float32)
    g = tl.load(proj + (512 + hh[:, None]) * total + pp[None, :], mask=valid, other=0.0).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * w[:, None] + b[:, None]) * tl.sigmoid(g)

    wo = tl.load(wt + hh[:, None] * 128 + dd[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), wo, out_dtype=tl.float32)
    tl.store(out + pp[:, None] * 128 + dd[None, :], acc, mask=pp[:, None] < total)


@triton.jit
def _norm_gate_b1_kernel(tri, proj, yout, nw, nb,
                         total: tl.constexpr, hidden: tl.constexpr,
                         BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr):
    pp = tl.program_id(0) * BLOCK_P + tl.arange(0, BLOCK_P)
    hh = tl.arange(0, BLOCK_H)
    valid = (pp[None, :] < total) & (hh[:, None] < hidden)

    vals = tl.load(tri + hh[:, None] * total + pp[None, :], mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    var = tl.sum(vals * vals, axis=0) / hidden - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    w = tl.load(nw + hh, mask=hh < hidden, other=0.0).to(tl.float32)
    b = tl.load(nb + hh, mask=hh < hidden, other=0.0).to(tl.float32)
    g = tl.load(proj + (4 * hidden + hh[:, None]) * total + pp[None, :], mask=valid, other=0.0).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * w[:, None] + b[:, None]) * tl.sigmoid(g)

    tl.store(yout + pp[:, None] * hidden + hh[None, :],
             tl.trans(y).to(tl.float16),
             mask=(pp[:, None] < total) & (hh[None, :] < hidden))


@triton.jit
def _norm_gate_kernel(tri, proj, yout, nw, nb,
                      total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
                      hidden: tl.constexpr, BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr):
    pp = tl.program_id(0) * BLOCK_P + tl.arange(0, BLOCK_P)
    hh = tl.arange(0, BLOCK_H)
    bidx = pp // n2
    pos = pp - bidx * n2
    src = ((hh[:, None] * bsz + bidx[None, :]) * n2) + pos[None, :]
    valid = (pp[None, :] < total) & (hh[:, None] < hidden)

    vals = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    var = tl.sum(vals * vals, axis=0) / hidden - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    w = tl.load(nw + hh, mask=hh < hidden, other=0.0).to(tl.float32)
    b = tl.load(nb + hh, mask=hh < hidden, other=0.0).to(tl.float32)
    g = tl.load(proj + (4 * hidden + hh[:, None]) * total + pp[None, :], mask=valid, other=0.0).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * w[:, None] + b[:, None]) * tl.sigmoid(g)

    tl.store(yout + pp[:, None] * hidden + hh[None, :],
             tl.trans(y).to(tl.float16),
             mask=(pp[:, None] < total) & (hh[None, :] < hidden))


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_proj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)
    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (w_proj, w_out_t)
    return w_proj, w_out_t


def _mm_float(a, b):
    global _MM_MODE
    if _MM_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_MODE = 1
        return out
    except Exception:
        pass
    try:
        out = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_MODE = 2
        return out
    except Exception:
        _MM_MODE = 3
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n
    device = input_tensor.device

    w_proj, w_out_t = _cached_weights(weights)
    xh = _workspace("xh", (total, dim), device, torch.float16)

    if dim == 128:
        _ln128_kernel[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=16, num_warps=4,
        )
    elif dim == 384:
        _ln384_kernel[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=8, num_warps=8,
        )
    else:
        _ln768_kernel[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=4, num_warps=8,
        )

    proj = _workspace("proj", (5 * hidden, total), device, torch.float16)
    torch.mm(w_proj, xh.t(), out=proj)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.reshape(total) if has_mask else input_tensor
    _gate_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
        proj, mask_arg, total, hidden, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4,
    )

    left = proj[:hidden].view(hidden * bsz, n, n)
    right = proj[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri = _workspace("tri", (hidden * bsz, n, n), device, torch.float16)
    torch.bmm(left, right.transpose(1, 2), out=tri)

    if dim == 128 and hidden == 128:
        out = torch.empty((total, 128), device=device, dtype=torch.float32)
        if bsz == 1:
            _final128_b1_kernel[(triton.cdiv(total, 32),)](
                tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, BLOCK_P=32, num_warps=8,
            )
        else:
            _final128_kernel[(triton.cdiv(total, 32),)](
                tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, n2, bsz, BLOCK_P=32, num_warps=8,
            )
    else:
        normed = _workspace("normed", (total, hidden), device, torch.float16)
        if bsz == 1:
            _norm_gate_b1_kernel[(triton.cdiv(total, 32),)](
                tri, proj, normed, weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, hidden, BLOCK_P=32, BLOCK_H=128, num_warps=8,
            )
        else:
            _norm_gate_kernel[(triton.cdiv(total, 32),)](
                tri, proj, normed, weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, n2, bsz, hidden, BLOCK_P=32, BLOCK_H=128, num_warps=8,
            )
        out = _mm_float(normed, w_out_t)

    return out.reshape(bsz, n, n, dim)
