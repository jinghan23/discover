"""
Outgoing Triangle Multiplicative Update forward pass.

Computes input LayerNorm in Triton, packs the five input projections into one
tensor-core GEMM, applies left/right gates and masks in-place, evaluates the
outgoing triangle contraction as 128 strided batched GEMMs, then fuses output
LayerNorm, output gate, and final projection for the common output dimensions.
All cached workspaces and fp16 weight packs are keyed by shape/device to avoid
steady-state allocation overhead.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    out = _WORKSPACE_CACHE.get(key)
    if out is None:
        if len(_WORKSPACE_CACHE) >= 32:
            _WORKSPACE_CACHE.clear()
        out = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = out
    return out


@triton.jit
def _ln128(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d = tl.arange(0, 128)
    mask = m < total
    v = tl.load(x + m[:, None] * 128 + d[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    ss = tl.sum(v * v, axis=1) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)
    ww = tl.load(w + d).to(tl.float32)
    bb = tl.load(b + d).to(tl.float32)
    o = (v - mean[:, None]) * rstd[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + m[:, None] * 128 + d[None, :], o.to(tl.float16), mask=mask[:, None])


@triton.jit
def _ln384(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    mask = m < total

    v0 = tl.load(x + m[:, None] * 384 + d0[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 384 + 256 + d1[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    q = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0026041666666666665
    rstd = tl.rsqrt(tl.maximum(q * 0.0026041666666666665 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 256 + d1).to(tl.float32)
    b1 = tl.load(b + 256 + d1).to(tl.float32)

    tl.store(y + m[:, None] * 384 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=mask[:, None])
    tl.store(y + m[:, None] * 384 + 256 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=mask[:, None])


@triton.jit
def _ln768(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    mask = m < total

    v0 = tl.load(x + m[:, None] * 768 + d0[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 768 + 512 + d1[None, :], mask=mask[:, None], other=0.0).to(tl.float32)
    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    q = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0013020833333333333
    rstd = tl.rsqrt(tl.maximum(q * 0.0013020833333333333 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 512 + d1).to(tl.float32)
    b1 = tl.load(b + 512 + d1).to(tl.float32)

    tl.store(y + m[:, None] * 768 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=mask[:, None])
    tl.store(y + m[:, None] * 768 + 512 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=mask[:, None])


@triton.jit
def _gate_lr(proj, mask, total: tl.constexpr, has_mask: tl.constexpr,
             BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)
    p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (p[None, :] < total) & (h[:, None] < 128)
    base = h[:, None] * total + p[None, :]

    l = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(proj + 128 * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + 256 * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + 384 * total + base, mask=valid, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)
    if has_mask:
        mm = tl.load(mask + p, mask=p < total, other=0.0).to(tl.float32)
        l = l * mm[None, :]
        r = r * mm[None, :]

    tl.store(proj + base, l.to(tl.float16), mask=valid)
    tl.store(proj + 128 * total + base, r.to(tl.float16), mask=valid)


@triton.jit
def _final128(tri, proj, out, nw, nb, wt, total: tl.constexpr, n2: tl.constexpr,
              bsz: tl.constexpr, BLOCK_P: tl.constexpr):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    d = tl.arange(0, 128)

    if bsz == 1:
        src = h[:, None] * total + p[None, :]
    else:
        bb = p // n2
        pos = p - bb * n2
        src = ((h[:, None] * bsz + bb[None, :]) * n2) + pos[None, :]

    valid = p[None, :] < total
    v = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=0) * 0.0078125
    ss = tl.sum(v * v, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    y = (v - mean[None, :]) * rstd[None, :]
    y = y * tl.load(nw + h)[:, None] + tl.load(nb + h)[:, None]
    g = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = y * tl.sigmoid(g)

    w = tl.load(wt + h[:, None] * 128 + d[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), w, out_dtype=tl.float32)
    tl.store(out + p[:, None] * 128 + d[None, :], acc, mask=p[:, None] < total)


@triton.jit
def _final384(tri, proj, out, nw, nb, wt, total: tl.constexpr, n2: tl.constexpr,
              bsz: tl.constexpr, BLOCK_P: tl.constexpr):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    d = tl.arange(0, 128)

    if bsz == 1:
        src = h[:, None] * total + p[None, :]
    else:
        bb = p // n2
        pos = p - bb * n2
        src = ((h[:, None] * bsz + bb[None, :]) * n2) + pos[None, :]

    valid = p[None, :] < total
    v = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=0) * 0.0078125
    ss = tl.sum(v * v, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    y = (v - mean[None, :]) * rstd[None, :]
    y = y * tl.load(nw + h)[:, None] + tl.load(nb + h)[:, None]
    g = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=valid, other=0.0).to(tl.float32)
    yt = tl.trans((y * tl.sigmoid(g)).to(tl.float16))
    sm = p[:, None] < total

    w0 = tl.load(wt + h[:, None] * 384 + d[None, :])
    a0 = tl.dot(yt, w0, out_dtype=tl.float32)
    tl.store(out + p[:, None] * 384 + d[None, :], a0, mask=sm)

    w1 = tl.load(wt + h[:, None] * 384 + (d + 128)[None, :])
    a1 = tl.dot(yt, w1, out_dtype=tl.float32)
    tl.store(out + p[:, None] * 384 + (d + 128)[None, :], a1, mask=sm)

    w2 = tl.load(wt + h[:, None] * 384 + (d + 256)[None, :])
    a2 = tl.dot(yt, w2, out_dtype=tl.float32)
    tl.store(out + p[:, None] * 384 + (d + 256)[None, :], a2, mask=sm)


@triton.jit
def _final_generic(tri, proj, out, nw, nb, wt, total: tl.constexpr, n2: tl.constexpr,
                   bsz: tl.constexpr, dim: tl.constexpr, BLOCK_P: tl.constexpr,
                   BLOCK_D: tl.constexpr):
    pid_p = tl.program_id(0)
    pid_d = tl.program_id(1)
    p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    if bsz == 1:
        src = h[:, None] * total + p[None, :]
    else:
        bb = p // n2
        pos = p - bb * n2
        src = ((h[:, None] * bsz + bb[None, :]) * n2) + pos[None, :]

    valid = p[None, :] < total
    v = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=0) * 0.0078125
    ss = tl.sum(v * v, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    y = (v - mean[None, :]) * rstd[None, :]
    y = y * tl.load(nw + h)[:, None] + tl.load(nb + h)[:, None]
    g = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = y * tl.sigmoid(g)

    ww = tl.load(wt + h[:, None] * dim + d[None, :],
                 mask=d[None, :] < dim, other=0.0)
    acc = tl.dot(tl.trans(y.to(tl.float16)), ww, out_dtype=tl.float32)
    tl.store(out + p[:, None] * dim + d[None, :], acc,
             mask=(p[:, None] < total) & (d[None, :] < dim))


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    hit = _WEIGHT_CACHE.get(key)
    if hit is not None:
        return hit
    wproj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    wout_t = tensors[5].t().contiguous().to(torch.float16)
    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (wproj, wout_t)
    return wproj, wout_t


def _mm(a, b):
    global _MM_MODE
    if _MM_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_MODE == 2:
        return torch.mm(a, b).to(torch.float32)
    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_MODE = 1
        return out
    except Exception:
        _MM_MODE = 2
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    total = bsz * n * n
    n2 = n * n
    dev = input_tensor.device

    wproj, wout_t = _cached_weights(weights)
    xh = _workspace("xh", (total, dim), dev, torch.float16)

    if dim == 128:
        _ln128[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=16, num_warps=4
        )
    elif dim == 384:
        _ln384[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=8, num_warps=8
        )
    else:
        _ln768[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=4, num_warps=8
        )

    proj = _workspace("proj", (640, total), dev, torch.float16)
    torch.mm(wproj, xh.t(), out=proj)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_flat = mask.reshape(total) if has_mask else input_tensor
    _gate_lr[(triton.cdiv(total, 64), 4)](
        proj, mask_flat, total, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4
    )

    left = proj[:128].view(128 * bsz, n, n)
    right = proj[128:256].view(128 * bsz, n, n)
    tri = _workspace("tri", (128 * bsz, n, n), dev, torch.float16)
    torch.bmm(left, right.transpose(1, 2), out=tri)

    out = torch.empty((total, dim), device=dev, dtype=torch.float32)
    if dim == 128:
        _final128[(triton.cdiv(total, 32),)](
            tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout_t,
            total, n2, bsz, BLOCK_P=32, num_warps=8
        )
    elif dim == 384:
        _final384[(triton.cdiv(total, 32),)](
            tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout_t,
            total, n2, bsz, BLOCK_P=32, num_warps=8
        )
    else:
        _final_generic[(triton.cdiv(total, 32), triton.cdiv(dim, 128))](
            tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout_t,
            total, n2, bsz, dim, BLOCK_P=32, BLOCK_D=128, num_warps=8
        )

    return out.reshape(bsz, n, n, dim)
