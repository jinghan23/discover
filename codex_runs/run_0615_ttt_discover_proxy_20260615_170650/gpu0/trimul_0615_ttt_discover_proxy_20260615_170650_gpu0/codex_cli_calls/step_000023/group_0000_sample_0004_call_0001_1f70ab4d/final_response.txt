"""
Outgoing TriMul forward pass using fused Triton preprocessing/postprocessing and
Ampere tensor-core GEMMs. The implementation normalizes inputs to fp16, computes
all five projections in one GEMM, applies gates/masks in-place, evaluates the
per-channel outgoing triangle product with strided batched GEMM, then fuses the
128-hidden output LayerNorm, output gate, and final projection.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    out = _WORKSPACE_CACHE.get(key)
    if out is None:
        if len(_WORKSPACE_CACHE) >= 20:
            _WORKSPACE_CACHE.clear()
        out = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = out
    return out


@triton.jit
def _ln128(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d = tl.arange(0, 128)
    valid = m < total
    v = tl.load(x + m[:, None] * 128 + d[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    var = tl.sum(v * v, axis=1) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)
    ww = tl.load(w + d).to(tl.float32)
    bb = tl.load(b + d).to(tl.float32)
    o = (v - mean[:, None]) * rstd[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + m[:, None] * 128 + d[None, :], o.to(tl.float16), mask=valid[:, None])


@triton.jit
def _ln384(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    valid = m < total

    v0 = tl.load(x + m[:, None] * 384 + d0[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 384 + 256 + d1[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 384.0
    rstd = tl.rsqrt(tl.maximum(ss / 384.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 256 + d1).to(tl.float32)
    b1 = tl.load(b + 256 + d1).to(tl.float32)

    tl.store(y + m[:, None] * 384 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=valid[:, None])
    tl.store(y + m[:, None] * 384 + 256 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=valid[:, None])


@triton.jit
def _ln768(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    valid = m < total

    v0 = tl.load(x + m[:, None] * 768 + d0[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + m[:, None] * 768 + 512 + d1[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 768.0
    rstd = tl.rsqrt(tl.maximum(ss / 768.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 512 + d1).to(tl.float32)
    b1 = tl.load(b + 512 + d1).to(tl.float32)

    tl.store(y + m[:, None] * 768 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=valid[:, None])
    tl.store(y + m[:, None] * 768 + 512 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=valid[:, None])


@triton.jit
def _gate_inplace(proj, mask, total: tl.constexpr, has_mask: tl.constexpr,
                  BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)
    p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (p[None, :] < total) & (h[:, None] < 128)
    base = h[:, None] * total + p[None, :]

    l = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(proj + 128 * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + 256 * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + 384 * total + base, mask=valid, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)
    if has_mask:
        mm = tl.load(mask + p, mask=p < total, other=0.0).to(tl.float32)
        l = l * mm[None, :]
        r = r * mm[None, :]

    tl.store(proj + base, l.to(tl.float16), mask=valid)
    tl.store(proj + 128 * total + base, r.to(tl.float16), mask=valid)


@triton.jit
def _post_b1(tri, proj, out, nw, nb, wo,
             total: tl.constexpr, dim: tl.constexpr,
             BLOCK_P: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    d = tl.arange(0, BLOCK_D)

    vals = tl.load(tri + h[:, None] * total + p[None, :], mask=p[None, :] < total, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    var = tl.sum(vals * vals, axis=0) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    gate = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=p[None, :] < total, other=0.0).to(tl.float32)
    ww = tl.load(nw + h).to(tl.float32)
    bb = tl.load(nb + h).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)

    wout = tl.load(wo + h[:, None] * dim + d[None, :],
                   mask=d[None, :] < dim, other=0.0)
    acc = tl.dot(tl.trans(y.to(tl.float16)), wout, out_dtype=tl.float32)
    tl.store(out + p[:, None] * dim + d[None, :], acc,
             mask=(p[:, None] < total) & (d[None, :] < dim))


@triton.jit
def _post_b2(tri, proj, out, nw, nb, wo,
             total: tl.constexpr, n2: tl.constexpr, dim: tl.constexpr,
             BLOCK_P: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    d = tl.arange(0, BLOCK_D)

    bidx = p // n2
    pos = p - bidx * n2
    src = ((h[:, None] * 2 + bidx[None, :]) * n2) + pos[None, :]
    vals = tl.load(tri + src, mask=p[None, :] < total, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) * 0.0078125
    var = tl.sum(vals * vals, axis=0) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    gate = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=p[None, :] < total, other=0.0).to(tl.float32)
    ww = tl.load(nw + h).to(tl.float32)
    bb = tl.load(nb + h).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)

    wout = tl.load(wo + h[:, None] * dim + d[None, :],
                   mask=d[None, :] < dim, other=0.0)
    acc = tl.dot(tl.trans(y.to(tl.float16)), wout, out_dtype=tl.float32)
    tl.store(out + p[:, None] * dim + d[None, :], acc,
             mask=(p[:, None] < total) & (d[None, :] < dim))


@triton.jit
def _post_norm_gate_b1(tri, proj, normed, nw, nb,
                       total: tl.constexpr, BLOCK_P: tl.constexpr):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    vals = tl.load(tri + h[:, None] * total + p[None, :], mask=p[None, :] < total, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    var = tl.sum(vals * vals, axis=0) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)
    gate = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=p[None, :] < total, other=0.0).to(tl.float32)
    ww = tl.load(nw + h).to(tl.float32)
    bb = tl.load(nb + h).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)
    tl.store(normed + p[:, None] * 128 + h[None, :], tl.trans(y).to(tl.float16), mask=p[:, None] < total)


@triton.jit
def _post_norm_gate_b2(tri, proj, normed, nw, nb,
                       total: tl.constexpr, n2: tl.constexpr, BLOCK_P: tl.constexpr):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, 128)
    bidx = p // n2
    pos = p - bidx * n2
    src = ((h[:, None] * 2 + bidx[None, :]) * n2) + pos[None, :]
    vals = tl.load(tri + src, mask=p[None, :] < total, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    var = tl.sum(vals * vals, axis=0) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)
    gate = tl.load(proj + (512 + h[:, None]) * total + p[None, :], mask=p[None, :] < total, other=0.0).to(tl.float32)
    ww = tl.load(nw + h).to(tl.float32)
    bb = tl.load(nb + h).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)
    tl.store(normed + p[:, None] * 128 + h[None, :], tl.trans(y).to(tl.float16), mask=p[:, None] < total)


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    val = _WEIGHT_CACHE.get(key)
    if val is not None:
        return val
    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    wproj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    wout_t = tensors[5].t().contiguous().to(torch.float16)
    val = (wproj, wout_t)
    _WEIGHT_CACHE[key] = val
    return val


def _mm_fp32(a, b):
    global _MM_MODE
    if _MM_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_MODE == 2:
        return torch.mm(a, b).to(torch.float32)
    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_MODE = 1
        return out
    except Exception:
        _MM_MODE = 2
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    total = bsz * n * n
    n2 = n * n
    device = input_tensor.device

    wproj, wout_t = _cached_weights(weights)
    xh = _workspace("xh", (total, dim), device, torch.float16)

    if dim == 128:
        _ln128[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=16, num_warps=4
        )
    elif dim == 384:
        _ln384[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=8, num_warps=8
        )
    else:
        _ln768[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=4, num_warps=8
        )

    proj = _workspace("proj", (640, total), device, torch.float16)
    torch.mm(wproj, xh.t(), out=proj)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.reshape(total) if has_mask else input_tensor
    _gate_inplace[(triton.cdiv(total, 64), 4)](
        proj, mask_arg, total, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4
    )

    left = proj[:128].view(128 * bsz, n, n)
    right = proj[128:256].view(128 * bsz, n, n)
    tri = _workspace("tri", (128 * bsz, n, n), device, torch.float16)
    torch.bmm(left, right.transpose(1, 2), out=tri)

    if dim == 128:
        out = torch.empty((total, 128), device=device, dtype=torch.float32)
        if bsz == 1:
            _post_b1[(triton.cdiv(total, 32),)](
                tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout_t,
                total, 128, BLOCK_P=32, BLOCK_D=128, num_warps=8
            )
        else:
            _post_b2[(triton.cdiv(total, 32),)](
                tri, proj, out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout_t,
                total, n2, 128, BLOCK_P=32, BLOCK_D=128, num_warps=8
            )
    else:
        normed = _workspace("normed", (total, 128), device, torch.float16)
        if bsz == 1:
            _post_norm_gate_b1[(triton.cdiv(total, 32),)](
                tri, proj, normed, weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, BLOCK_P=32, num_warps=8
            )
        else:
            _post_norm_gate_b2[(triton.cdiv(total, 32),)](
                tri, proj, normed, weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, n2, BLOCK_P=32, num_warps=8
            )
        out = _mm_fp32(normed, wout_t)

    return out.reshape(bsz, n, n, dim)
