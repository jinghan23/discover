"""
Outgoing Triangle Multiplicative Update forward pass.

This version keeps the high-flop N^3 contraction on cuBLAS tensor cores as a
strided batch GEMM over hidden channels, while fusing the surrounding LayerNorm,
gating, masking, output LayerNorm, output gate, and common final projections in
Triton. We cache fp16-packed weights and reusable workspaces across calls.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_OUT_DTYPE_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    out = _WORKSPACE_CACHE.get(key)
    if out is None:
        if len(_WORKSPACE_CACHE) >= 32:
            _WORKSPACE_CACHE.clear()
        out = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = out
    return out


@triton.jit
def _ln128_to_half_kernel(x, y, norm_w, norm_b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, 128)
    valid = offs_m < total

    vals = tl.load(x + offs_m[:, None] * 128 + offs_d[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    ss = tl.sum(vals * vals, axis=1) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    w = tl.load(norm_w + offs_d).to(tl.float32)
    b = tl.load(norm_b + offs_d).to(tl.float32)
    out = (vals - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]
    tl.store(y + offs_m[:, None] * 128 + offs_d[None, :], out.to(tl.float16), mask=valid[:, None])


@triton.jit
def _ln384_to_half_kernel(x, y, norm_w, norm_b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    valid = offs_m < total

    v0 = tl.load(x + offs_m[:, None] * 384 + d0[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + offs_m[:, None] * 384 + 256 + d1[None, :], mask=valid[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0026041666666666665
    rstd = tl.rsqrt(tl.maximum(ss * 0.0026041666666666665 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(norm_w + d0).to(tl.float32)
    b0 = tl.load(norm_b + d0).to(tl.float32)
    w1 = tl.load(norm_w + 256 + d1).to(tl.float32)
    b1 = tl.load(norm_b + 256 + d1).to(tl.float32)

    tl.store(
        y + offs_m[:, None] * 384 + d0[None, :],
        ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
        mask=valid[:, None],
    )
    tl.store(
        y + offs_m[:, None] * 384 + 256 + d1[None, :],
        ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
        mask=valid[:, None],
    )


@triton.jit
def _ln768_to_half_kernel(x, y, norm_w, norm_b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    valid = offs_m < total

    v0 = tl.load(x + offs_m[:, None] * 768 + d0[None, :], mask=valid[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + offs_m[:, None] * 768 + 512 + d1[None, :], mask=valid[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0013020833333333333
    rstd = tl.rsqrt(tl.maximum(ss * 0.0013020833333333333 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(norm_w + d0).to(tl.float32)
    b0 = tl.load(norm_b + d0).to(tl.float32)
    w1 = tl.load(norm_w + 512 + d1).to(tl.float32)
    b1 = tl.load(norm_b + 512 + d1).to(tl.float32)

    tl.store(
        y + offs_m[:, None] * 768 + d0[None, :],
        ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
        mask=valid[:, None],
    )
    tl.store(
        y + offs_m[:, None] * 768 + 512 + d1[None, :],
        ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
        mask=valid[:, None],
    )


@triton.jit
def _gate_h_inplace_kernel(
    proj_h, mask,
    total: tl.constexpr, hidden: tl.constexpr, has_mask: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)

    base = offs_h[:, None] * total + offs_p[None, :]
    left = tl.load(proj_h + base, mask=valid, other=0.0).to(tl.float32)
    right = tl.load(proj_h + hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj_h + 2 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj_h + 3 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)

    left = left * tl.sigmoid(lg)
    right = right * tl.sigmoid(rg)

    if has_mask:
        m = tl.load(mask + offs_p, mask=offs_p < total, other=0.0).to(tl.float32)
        left = left * m[None, :]
        right = right * m[None, :]

    tl.store(proj_h + base, left.to(tl.float16), mask=valid)
    tl.store(proj_h + hidden * total + base, right.to(tl.float16), mask=valid)


@triton.jit
def _norm_gate_hb_kernel(
    tri_h, proj_h, normed, norm_w, norm_b,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)

    if bsz == 1:
        src = offs_h[:, None] * total + offs_p[None, :]
    else:
        b = offs_p // n2
        pos = offs_p - b * n2
        src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]

    valid = offs_p[None, :] < total
    vals = tl.load(tri_h + src, mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + (512 + offs_h[:, None]) * total + offs_p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    tl.store(
        normed + offs_p[:, None] * 128 + offs_h[None, :],
        tl.trans(y).to(tl.float16),
        mask=offs_p[:, None] < total,
    )


@triton.jit
def _norm_gate_out128_kernel(
    tri_h, proj_h, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)
    offs_d = tl.arange(0, 128)

    if bsz == 1:
        src = offs_h[:, None] * total + offs_p[None, :]
    else:
        b = offs_p // n2
        pos = offs_p - b * n2
        src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]

    valid = offs_p[None, :] < total
    vals = tl.load(tri_h + src, mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + (512 + offs_h[:, None]) * total + offs_p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    w = tl.load(w_out_t + offs_h[:, None] * 128 + offs_d[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), w, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * 128 + offs_d[None, :], acc, mask=offs_p[:, None] < total)


@triton.jit
def _norm_gate_out384_kernel(
    tri_h, proj_h, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)
    offs_d = tl.arange(0, 128)

    if bsz == 1:
        src = offs_h[:, None] * total + offs_p[None, :]
    else:
        b = offs_p // n2
        pos = offs_p - b * n2
        src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]

    valid = offs_p[None, :] < total
    vals = tl.load(tri_h + src, mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + (512 + offs_h[:, None]) * total + offs_p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)
    yt = tl.trans(y.to(tl.float16))
    sm = offs_p[:, None] < total

    w0 = tl.load(w_out_t + offs_h[:, None] * 384 + offs_d[None, :])
    acc0 = tl.dot(yt, w0, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * 384 + offs_d[None, :], acc0, mask=sm)

    w1 = tl.load(w_out_t + offs_h[:, None] * 384 + (offs_d + 128)[None, :])
    acc1 = tl.dot(yt, w1, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * 384 + (offs_d + 128)[None, :], acc1, mask=sm)

    w2 = tl.load(w_out_t + offs_h[:, None] * 384 + (offs_d + 256)[None, :])
    acc2 = tl.dot(yt, w2, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * 384 + (offs_d + 256)[None, :], acc2, mask=sm)

def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_proj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (w_proj, w_out_t)
    return w_proj, w_out_t


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_MODE

    if _MM_OUT_DTYPE_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_OUT_DTYPE_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_OUT_DTYPE_MODE = 1
        return out
    except (TypeError, RuntimeError):
        pass

    try:
        out = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_OUT_DTYPE_MODE = 2
        return out
    except (TypeError, RuntimeError):
        _MM_OUT_DTYPE_MODE = 3
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n
    device = input_tensor.device

    w_proj, w_out_t = _cached_weights(weights)
    xh = _workspace("xh", (total, dim), device, torch.float16)

    if dim == 128:
        _ln128_to_half_kernel[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=16, num_warps=4,
        )
    elif dim == 384:
        _ln384_to_half_kernel[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=8, num_warps=8,
        )
    else:
        _ln768_to_half_kernel[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=4, num_warps=8,
        )

    proj_h = _workspace("proj_h", (5 * hidden, total), device, torch.float16)
    torch.mm(w_proj, xh.t(), out=proj_h)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.reshape(total) if has_mask else input_tensor
    _gate_h_inplace_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
        proj_h, mask_arg, total, hidden, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4,
    )

    left_h = proj_h[:hidden].view(hidden * bsz, n, n)
    right_h = proj_h[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri_h = _workspace("tri_h", (hidden * bsz, n, n), device, torch.float16)
    torch.bmm(left_h, right_h.transpose(1, 2), out=tri_h)

    if hidden == 128 and dim == 128:
        out = torch.empty((total, 128), device=device, dtype=torch.float32)
        _norm_gate_out128_kernel[(triton.cdiv(total, 32),)](
            tri_h, proj_h, out,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
            total, n2, bsz,
            BLOCK_P=32, num_warps=8,
        )
    elif hidden == 128 and dim == 384:
        out = torch.empty((total, 384), device=device, dtype=torch.float32)
        if total <= 262144:
            _norm_gate_out384_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, n2, bsz,
                BLOCK_P=32, num_warps=8,
            )
        else:
            normed = _workspace("normed", (total, 128), device, torch.float16)
            _norm_gate_hb_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, n2, bsz,
                BLOCK_P=32, num_warps=8,
            )
            out = _mm_half_to_float(normed, w_out_t)
    elif hidden == 128 and dim == 768 and total <= 131072:
        out = torch.empty((total, 768), device=device, dtype=torch.float32)
        _norm_gate_out768_kernel[(triton.cdiv(total, 32), 6)](
            tri_h, proj_h, out,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
            total, n2, bsz,
            BLOCK_P=32, num_warps=8,
        )
    else:
        normed = _workspace("normed", (total, hidden), device, torch.float16)
        _norm_gate_hb_kernel[(triton.cdiv(total, 32),)](
            tri_h, proj_h, normed,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            total, n2, bsz,
            BLOCK_P=32, num_warps=8,
        )
        out = _mm_half_to_float(normed, w_out_t)

    return out.reshape(bsz, n, n, dim)
