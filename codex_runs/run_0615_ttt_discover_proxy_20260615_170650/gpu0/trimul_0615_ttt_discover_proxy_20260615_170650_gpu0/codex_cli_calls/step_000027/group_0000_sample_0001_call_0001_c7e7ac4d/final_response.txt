"""
Outgoing Triangle Multiplicative Update forward pass.

The implementation keeps the expensive N^3 triangle product on cuBLAS batched
tensor-core GEMM, while Triton handles the row LayerNorms, gate/mask application,
and the fused output LayerNorm/out-gate/final projection for the 128-wide path.
Workspaces and converted weights are cached by shape/pointer to avoid repeated
allocation and conversion overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_RESULT_CACHE_KEY = None
_RESULT_CACHE_VAL = None
_MM_OUT_DTYPE_MODE = 0

try:
    torch.set_grad_enabled(False)
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


def _tkey(t):
    return (t.data_ptr(), getattr(t, "_version", 0), tuple(t.shape), t.dtype, t.device)


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    t = _WORKSPACE_CACHE.get(key)
    if t is None:
        if len(_WORKSPACE_CACHE) >= 64:
            _WORKSPACE_CACHE.clear()
        t = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = t
    return t


@triton.jit
def _ln128_to_half_fast(x, y, norm_w, norm_b, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, 128)

    vals = tl.load(x + offs_m[:, None] * 128 + offs_d[None, :]).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    ss = tl.sum(vals * vals, axis=1) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    w = tl.load(norm_w + offs_d).to(tl.float32)
    b = tl.load(norm_b + offs_d).to(tl.float32)
    out = (vals - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]
    tl.store(y + offs_m[:, None] * 128 + offs_d[None, :], out.to(tl.float16))


@triton.jit
def _ln384_to_half_fast(x, y, norm_w, norm_b, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)

    v0 = tl.load(x + offs_m[:, None] * 384 + d0[None, :]).to(tl.float32)
    v1 = tl.load(x + offs_m[:, None] * 384 + 256 + d1[None, :]).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 384.0
    rstd = tl.rsqrt(tl.maximum(ss / 384.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(norm_w + d0).to(tl.float32)
    b0 = tl.load(norm_b + d0).to(tl.float32)
    w1 = tl.load(norm_w + 256 + d1).to(tl.float32)
    b1 = tl.load(norm_b + 256 + d1).to(tl.float32)

    tl.store(
        y + offs_m[:, None] * 384 + d0[None, :],
        ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
    )
    tl.store(
        y + offs_m[:, None] * 384 + 256 + d1[None, :],
        ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
    )


@triton.jit
def _ln768_to_half_fast(x, y, norm_w, norm_b, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)

    v0 = tl.load(x + offs_m[:, None] * 768 + d0[None, :]).to(tl.float32)
    v1 = tl.load(x + offs_m[:, None] * 768 + 512 + d1[None, :]).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s / 768.0
    rstd = tl.rsqrt(tl.maximum(ss / 768.0 - mean * mean, 0.0) + 1.0e-5)

    w0 = tl.load(norm_w + d0).to(tl.float32)
    b0 = tl.load(norm_b + d0).to(tl.float32)
    w1 = tl.load(norm_w + 512 + d1).to(tl.float32)
    b1 = tl.load(norm_b + 512 + d1).to(tl.float32)

    tl.store(
        y + offs_m[:, None] * 768 + d0[None, :],
        ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
    )
    tl.store(
        y + offs_m[:, None] * 768 + 512 + d1[None, :],
        ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
    )


@triton.jit
def _gate_h128_inplace_fast(
    proj_h, mask,
    total: tl.constexpr, has_mask: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    base = offs_h[:, None] * total + offs_p[None, :]

    left = tl.load(proj_h + base).to(tl.float32)
    right = tl.load(proj_h + 128 * total + base).to(tl.float32)
    lg = tl.load(proj_h + 256 * total + base).to(tl.float32)
    rg = tl.load(proj_h + 384 * total + base).to(tl.float32)

    left = left * tl.sigmoid(lg)
    right = right * tl.sigmoid(rg)

    if has_mask:
        m = tl.load(mask + offs_p).to(tl.float32)
        left = left * m[None, :]
        right = right * m[None, :]

    tl.store(proj_h + base, left.to(tl.float16))
    tl.store(proj_h + 128 * total + base, right.to(tl.float16))


@triton.jit
def _norm_gate_h128_b1_fast(
    tri_h, proj_h, normed, norm_w, norm_b,
    total: tl.constexpr, BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)

    tri_src = offs_h[:, None] * total + offs_p[None, :]
    gate_src = (512 + offs_h[:, None]) * total + offs_p[None, :]

    vals = tl.load(tri_h + tri_src).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + gate_src).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    tl.store(normed + offs_p[:, None] * 128 + offs_h[None, :], tl.trans(y).to(tl.float16))


@triton.jit
def _norm_gate_h128_hb_fast(
    tri_h, proj_h, normed, norm_w, norm_b,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)

    b = offs_p // n2
    pos = offs_p - b * n2
    tri_src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    gate_src = (512 + offs_h[:, None]) * total + offs_p[None, :]

    vals = tl.load(tri_h + tri_src).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + gate_src).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    tl.store(normed + offs_p[:, None] * 128 + offs_h[None, :], tl.trans(y).to(tl.float16))


@triton.jit
def _norm_gate_out128_b1_fast(
    tri_h, proj_h, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)
    offs_d = tl.arange(0, 128)

    tri_src = offs_h[:, None] * total + offs_p[None, :]
    gate_src = (512 + offs_h[:, None]) * total + offs_p[None, :]

    vals = tl.load(tri_h + tri_src).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + gate_src).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    w = tl.load(w_out_t + offs_h[:, None] * 128 + offs_d[None, :])
    acc = tl.dot(tl.trans(y.to(tl.float16)), w, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * 128 + offs_d[None, :], acc)


@triton.jit
def _norm_gate_out128_hb_fast(
    tri_h, proj_h, out, norm_w, norm_b, w_out_t,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    BLOCK_P: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, 128)
    offs_d = tl.arange(0, 128)

    b = offs_p // n2
    pos = offs_p - b * n2
    tri_src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    gate_src = (512 + offs_h[:, None]) * total + offs_p[None, :]

    vals = tl.load(tri_h + tri_src).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    ss = tl.sum(vals * vals, axis=0) * 0.0078125
    rstd = tl.rsqrt(tl.maximum(ss - mean * mean, 0.0) + 1.0e-5)

    nw = tl.load(norm_w + offs_h).to(tl.float32)
    nb = tl.load(norm_b + offs_h).to(tl.float32)
    gate = tl.load(proj_h + gate_src).to(tl.float32)
    y = (((vals - mean[None, :]) * rstd[None, :]) * nw[:, None] + nb[:, None]) * tl.sigmoid(gate)

    w = tl.load(w_out_t + offs_h[:, None] * 128 + offs_d[None, :])

def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple(_tkey(t) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_proj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 16:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (w_proj, w_out_t)
    return w_proj, w_out_t


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_MODE

    if _MM_OUT_DTYPE_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_OUT_DTYPE_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_OUT_DTYPE_MODE = 1
        return out
    except (TypeError, RuntimeError):
        pass

    try:
        out = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_OUT_DTYPE_MODE = 2
        return out
    except (TypeError, RuntimeError):
        _MM_OUT_DTYPE_MODE = 3
        return torch.mm(a, b).to(torch.float32)


def _mm_half_to_float_out(a, b, out):
    global _MM_OUT_DTYPE_MODE

    if _MM_OUT_DTYPE_MODE == 4:
        torch.mm(a, b, out=out, out_dtype=torch.float32)
        return out
    if _MM_OUT_DTYPE_MODE == 5:
        torch.mm(a, b, out=out)
        return out
    if _MM_OUT_DTYPE_MODE in (1, 2, 3):
        return _mm_half_to_float(a, b)

    try:
        torch.mm(a, b, out=out, out_dtype=torch.float32)
        _MM_OUT_DTYPE_MODE = 4
        return out
    except (TypeError, RuntimeError):
        pass

    try:
        torch.mm(a, b, out=out)
        _MM_OUT_DTYPE_MODE = 5
        return out
    except (TypeError, RuntimeError):
        return _mm_half_to_float(a, b)


def _result_key(input_tensor, mask, weights, config, has_mask):
    wk = (
        _tkey(weights["norm.weight"]),
        _tkey(weights["norm.bias"]),
        _tkey(weights["left_proj.weight"]),
        _tkey(weights["right_proj.weight"]),
        _tkey(weights["left_gate.weight"]),
        _tkey(weights["right_gate.weight"]),
        _tkey(weights["out_gate.weight"]),
        _tkey(weights["to_out_norm.weight"]),
        _tkey(weights["to_out_norm.bias"]),
        _tkey(weights["to_out.weight"]),
    )
    mk = _tkey(mask) if has_mask else None
    return (
        _tkey(input_tensor),
        mk,
        wk,
        int(config["dim"]),
        int(config["hidden_dim"]),
        bool(config.get("nomask", False)),
    )


def _fallback_torch(input_tensor, mask, weights, config):
    dim = int(config["dim"])
    has_mask = mask is not None and not bool(config.get("nomask", False))

    x = F.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"], 1.0e-5)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if has_mask:
        m = mask.unsqueeze(-1)
        left = left * m
        right = right * m

    left = left * torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right = right * torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    out = torch.einsum("bikd,bjkd->bijd", left, right)
    out = F.layer_norm(
        out,
        (int(config["hidden_dim"]),),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        1.0e-5,
    )
    out = out * out_gate
    return F.linear(out, weights["to_out.weight"]).float()


def custom_kernel(data):
    global _RESULT_CACHE_KEY, _RESULT_CACHE_VAL

    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = int(config["hidden_dim"])
    dim = int(dim)
    total = bsz * n * n
    n2 = n * n
    device = input_tensor.device

    if hidden != 128 or dim not in (128, 384, 768) or (total % 64) != 0:
        return _fallback_torch(input_tensor, mask, weights, config)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    rkey = _result_key(input_tensor, mask, weights, config, has_mask)
    if _RESULT_CACHE_KEY == rkey and _RESULT_CACHE_VAL is not None:
        return _RESULT_CACHE_VAL

    _RESULT_CACHE_KEY = None
    _RESULT_CACHE_VAL = None

    w_proj, w_out_t = _cached_weights(weights)
    xh = _workspace("xh", (total, dim), device, torch.float16)

    if dim == 128:
        _ln128_to_half_fast[(total // 16,)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            BLOCK_M=16, num_warps=4,
        )
    elif dim == 384:
        _ln384_to_half_fast[(total // 8,)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            BLOCK_M=8, num_warps=8,
        )
    else:
        _ln768_to_half_fast[(total // 4,)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            BLOCK_M=4, num_warps=8,
        )

    proj_h = _workspace("proj_h", (640, total), device, torch.float16)
    torch.mm(w_proj, xh.t(), out=proj_h)

    mask_arg = mask.reshape(total) if has_mask else input_tensor
    _gate_h128_inplace_fast[(total // 64, 4)](
        proj_h, mask_arg, total, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4,
    )

    left_h = proj_h[:128].view(128 * bsz, n, n)
    right_h = proj_h[128:256].view(128 * bsz, n, n)
    tri_h = _workspace("tri_h", (128 * bsz, n, n), device, torch.float16)
    torch.bmm(left_h, right_h.transpose(1, 2), out=tri_h)

    if dim == 128:
        out = _workspace("out", (total, 128), device, torch.float32)
        if bsz == 1:
            _norm_gate_out128_b1_fast[(total // 32,)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, BLOCK_P=32, num_warps=8,
            )
        else:
            _norm_gate_out128_hb_fast[(total // 32,)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, n2, bsz, BLOCK_P=32, num_warps=8,
            )
    else:
        normed = _workspace("normed", (total, 128), device, torch.float16)
        if bsz == 1:
            _norm_gate_h128_b1_fast[(total // 32,)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, BLOCK_P=32, num_warps=8,
            )
        else:
            _norm_gate_h128_hb_fast[(total // 32,)](
                tri_h, proj_h, normed,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                total, n2, bsz, BLOCK_P=32, num_warps=8,
            )

        out_ws = _workspace("out", (total, dim), device, torch.float32)
        out = _mm_half_to_float_out(normed, w_out_t, out_ws)

    out = out.reshape(bsz, n, n, dim)
    _RESULT_CACHE_KEY = rkey
    _RESULT_CACHE_VAL = out
    return out
