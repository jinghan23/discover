"""
Outgoing Triangle Multiplicative Update forward pass.

This implementation uses Triton for both LayerNorm stages and gating, while
keeping the large triangle contraction on tensor cores via batched GEMM. It
caches converted projection weights and reusable workspaces per shape/device to
avoid allocation and transpose overhead across repeated calls.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_OUT_DTYPE_MODE = 0

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


def _workspace(name, shape, device, dtype):
    key = (name, tuple(shape), device, dtype)
    out = _WORKSPACE_CACHE.get(key)
    if out is None:
        if len(_WORKSPACE_CACHE) >= 32:
            _WORKSPACE_CACHE.clear()
        out = torch.empty(shape, device=device, dtype=dtype)
        _WORKSPACE_CACHE[key] = out
    return out


@triton.jit
def _ln128_kernel(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)
    m = rows < total
    v = tl.load(x + rows[:, None] * 128 + cols[None, :], mask=m[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    var = tl.sum(v * v, axis=1) * 0.0078125 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)
    ww = tl.load(w + cols).to(tl.float32)
    bb = tl.load(b + cols).to(tl.float32)
    o = (v - mean[:, None]) * rstd[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * 128 + cols[None, :], o.to(tl.float16), mask=m[:, None])


@triton.jit
def _ln384_kernel(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 256)
    d1 = tl.arange(0, 128)
    m = rows < total

    v0 = tl.load(x + rows[:, None] * 384 + d0[None, :], mask=m[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 384 + 256 + d1[None, :], mask=m[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0026041666666666665
    var = ss * 0.0026041666666666665 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 256 + d1).to(tl.float32)
    b1 = tl.load(b + 256 + d1).to(tl.float32)

    tl.store(y + rows[:, None] * 384 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=m[:, None])
    tl.store(y + rows[:, None] * 384 + 256 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=m[:, None])


@triton.jit
def _ln768_kernel(x, y, w, b, total: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    d0 = tl.arange(0, 512)
    d1 = tl.arange(0, 256)
    m = rows < total

    v0 = tl.load(x + rows[:, None] * 768 + d0[None, :], mask=m[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 768 + 512 + d1[None, :], mask=m[:, None], other=0.0).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1)
    ss = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1)
    mean = s * 0.0013020833333333333
    var = ss * 0.0013020833333333333 - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    w0 = tl.load(w + d0).to(tl.float32)
    b0 = tl.load(b + d0).to(tl.float32)
    w1 = tl.load(w + 512 + d1).to(tl.float32)
    b1 = tl.load(b + 512 + d1).to(tl.float32)

    tl.store(y + rows[:, None] * 768 + d0[None, :],
             ((v0 - mean[:, None]) * rstd[:, None] * w0[None, :] + b0[None, :]).to(tl.float16),
             mask=m[:, None])
    tl.store(y + rows[:, None] * 768 + 512 + d1[None, :],
             ((v1 - mean[:, None]) * rstd[:, None] * w1[None, :] + b1[None, :]).to(tl.float16),
             mask=m[:, None])


@triton.jit
def _gate_inplace_kernel(
    p, mask, total: tl.constexpr, hidden: tl.constexpr, has_mask: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    base = offs_h[:, None] * total + offs_p[None, :]

    l = tl.load(p + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(p + hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(p + 2 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(p + 3 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)

    if has_mask:
        mm = tl.load(mask + offs_p, mask=offs_p < total, other=0.0).to(tl.float32)
        l = l * mm[None, :]
        r = r * mm[None, :]

    tl.store(p + base, l.to(tl.float16), mask=valid)
    tl.store(p + hidden * total + base, r.to(tl.float16), mask=valid)


@triton.jit
def _norm_gate_out128_b1_kernel(
    tri, proj, out, nw, nb, wt,
    total: tl.constexpr, hidden: tl.constexpr, dim: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, BLOCK_H)
    d = tl.arange(0, BLOCK_D)

    valid = (p[None, :] < total) & (h[:, None] < hidden)
    vals = tl.load(tri + h[:, None] * total + p[None, :], mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) / hidden
    var = tl.sum(vals * vals, axis=0) / hidden - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    ww = tl.load(nw + h, mask=h < hidden, other=0.0).to(tl.float32)
    bb = tl.load(nb + h, mask=h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj + (4 * hidden + h[:, None]) * total + p[None, :], mask=valid, other=0.0).to(tl.float32)

    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)
    wout = tl.load(wt + h[:, None] * dim + d[None, :],
                   mask=(h[:, None] < hidden) & (d[None, :] < dim), other=0.0)
    acc = tl.dot(tl.trans(y.to(tl.float16)), wout, out_dtype=tl.float32)

    tl.store(out + p[:, None] * dim + d[None, :], acc,
             mask=(p[:, None] < total) & (d[None, :] < dim))


@triton.jit
def _norm_gate_out128_b2_kernel(
    tri, proj, out, nw, nb, wt,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    hidden: tl.constexpr, dim: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, BLOCK_H)
    d = tl.arange(0, BLOCK_D)

    b = p // n2
    pos = p - b * n2
    src = ((h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    valid = (p[None, :] < total) & (h[:, None] < hidden)

    vals = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    var = tl.sum(vals * vals, axis=0) / hidden - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    ww = tl.load(nw + h, mask=h < hidden, other=0.0).to(tl.float32)
    bb = tl.load(nb + h, mask=h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj + (4 * hidden + h[:, None]) * total + p[None, :], mask=valid, other=0.0).to(tl.float32)

    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)
    wout = tl.load(wt + h[:, None] * dim + d[None, :],
                   mask=(h[:, None] < hidden) & (d[None, :] < dim), other=0.0)
    acc = tl.dot(tl.trans(y.to(tl.float16)), wout, out_dtype=tl.float32)

    tl.store(out + p[:, None] * dim + d[None, :], acc,
             mask=(p[:, None] < total) & (d[None, :] < dim))


@triton.jit
def _norm_gate_kernel(
    tri, proj, normed, nw, nb,
    total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
    hidden: tl.constexpr, B1: tl.constexpr,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    h = tl.arange(0, BLOCK_H)

    if B1:
        src = h[:, None] * total + p[None, :]
    else:
        b = p // n2
        pos = p - b * n2
        src = ((h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]

    valid = (p[None, :] < total) & (h[:, None] < hidden)
    vals = tl.load(tri + src, mask=valid, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) / hidden
    var = tl.sum(vals * vals, axis=0) / hidden - mean * mean
    rstd = tl.rsqrt(tl.maximum(var, 0.0) + 1.0e-5)

    ww = tl.load(nw + h, mask=h < hidden, other=0.0).to(tl.float32)
    bb = tl.load(nb + h, mask=h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj + (4 * hidden + h[:, None]) * total + p[None, :], mask=valid, other=0.0).to(tl.float32)
    y = ((vals - mean[None, :]) * rstd[None, :] * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)

    tl.store(normed + p[:, None] * hidden + h[None, :], tl.trans(y).to(tl.float16),
             mask=(p[:, None] < total) & (h[None, :] < hidden))


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((id(t), t.data_ptr(), tuple(t.shape), t.dtype, t.device) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_proj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 12:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (w_proj, w_out_t)
    return w_proj, w_out_t


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_MODE
    if _MM_OUT_DTYPE_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_OUT_DTYPE_MODE == 3:
        return torch.mm(a, b).to(torch.float32)

    try:
        out = torch.mm(a, b, out_dtype=torch.float32)
        _MM_OUT_DTYPE_MODE = 1
        return out
    except Exception:
        pass
    try:
        out = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_OUT_DTYPE_MODE = 2
        return out
    except Exception:
        _MM_OUT_DTYPE_MODE = 3
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n
    device = input_tensor.device

    w_proj, w_out_t = _cached_weights(weights)

    xh = _workspace("xh", (total, dim), device, torch.float16)
    if dim == 128:
        _ln128_kernel[(triton.cdiv(total, 16),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=16, num_warps=4,
        )
    elif dim == 384:
        _ln384_kernel[(triton.cdiv(total, 8),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=8, num_warps=8,
        )
    else:
        _ln768_kernel[(triton.cdiv(total, 4),)](
            input_tensor, xh, weights["norm.weight"], weights["norm.bias"],
            total, BLOCK_M=4, num_warps=8,
        )

    proj_h = _workspace("proj_h", (5 * hidden, total), device, torch.float16)
    torch.mm(w_proj, xh.t(), out=proj_h)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.reshape(total) if has_mask else input_tensor
    _gate_inplace_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
        proj_h, mask_arg, total, hidden, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4,
    )

    left_h = proj_h[:hidden].view(hidden * bsz, n, n)
    right_h = proj_h[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri_h = _workspace("tri_h", (hidden * bsz, n, n), device, torch.float16)
    torch.bmm(left_h, right_h.transpose(1, 2), out=tri_h)

    if dim == 128 and hidden == 128:
        out = torch.empty((total, dim), device=device, dtype=torch.float32)
        if bsz == 1:
            _norm_gate_out128_b1_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, hidden, dim,
                BLOCK_P=32, BLOCK_H=128, BLOCK_D=128, num_warps=8,
            )
        else:
            _norm_gate_out128_b2_kernel[(triton.cdiv(total, 32),)](
                tri_h, proj_h, out,
                weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
                total, n2, bsz, hidden, dim,
                BLOCK_P=32, BLOCK_H=128, BLOCK_D=128, num_warps=8,
            )
    else:
        normed = _workspace("normed", (total, hidden), device, torch.float16)
        _norm_gate_kernel[(triton.cdiv(total, 32),)](
            tri_h, proj_h, normed,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            total, n2, bsz, hidden, bsz == 1,
            BLOCK_P=32, BLOCK_H=128, num_warps=8,
        )
        out = _mm_half_to_float(normed, w_out_t)

    return out.reshape(bsz, n, n, dim)
