"""
Outgoing TriMul forward pass with fused Triton normalization/gating and channel-major
batched triangle products. The code keeps the expensive N^3 update in cuBLAS BMM,
packs the five input projections into one GEMM, caches converted weights, and fuses
the output normalization/gate/projection for the common dim=128 path.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_MM_MODE = 0


@triton.jit
def _ln_to_half_kernel(x, y, nw, nb, total:tl.constexpr, dim:tl.constexpr,
                       BLOCK_M:tl.constexpr, BLOCK_D:tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)
    mask = (offs_m[:, None] < total) & (offs_d[None, :] < dim)

    v = tl.load(x + offs_m[:, None] * dim + offs_d[None, :], mask=mask, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=1) / dim
    c = tl.where(offs_d[None, :] < dim, v - mean[:, None], 0.)
    var = tl.sum(c * c, axis=1) / dim
    r = tl.rsqrt(var + 1.0e-5)

    w = tl.load(nw + offs_d, mask=offs_d < dim, other=0.).to(tl.float32)
    b = tl.load(nb + offs_d, mask=offs_d < dim, other=0.).to(tl.float32)
    o = c * r[:, None] * w[None, :] + b[None, :]
    tl.store(y + offs_m[:, None] * dim + offs_d[None, :], o.to(tl.float16), mask=mask)


@triton.jit
def _gate_inplace_kernel(p, mask_ptr, total:tl.constexpr, hidden:tl.constexpr,
                         has_mask:tl.constexpr, BLOCK_P:tl.constexpr, BLOCK_H:tl.constexpr):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    base = offs_h[:, None] * total + offs_p[None, :]

    l = tl.load(p + base, mask=valid, other=0.).to(tl.float32)
    r = tl.load(p + hidden * total + base, mask=valid, other=0.).to(tl.float32)
    lg = tl.load(p + 2 * hidden * total + base, mask=valid, other=0.).to(tl.float32)
    rg = tl.load(p + 3 * hidden * total + base, mask=valid, other=0.).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)
    if has_mask:
        m = tl.load(mask_ptr + offs_p, mask=offs_p < total, other=0.).to(tl.float32)
        l = l * m[None, :]
        r = r * m[None, :]

    tl.store(p + base, l.to(tl.float16), mask=valid)
    tl.store(p + hidden * total + base, r.to(tl.float16), mask=valid)


@triton.jit
def _norm_gate_kernel(tri, proj, normed, nw, nb,
                      total:tl.constexpr, n2:tl.constexpr, bsz:tl.constexpr,
                      hidden:tl.constexpr, BLOCK_P:tl.constexpr, BLOCK_H:tl.constexpr):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, BLOCK_H)

    b = offs_p // n2
    pos = offs_p - b * n2
    src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    gsrc = (4 * hidden + offs_h[:, None]) * total + offs_p[None, :]
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)

    v = tl.load(tri + src, mask=valid, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    c = v - mean[None, :]
    var = tl.sum(c * c, axis=0) / hidden
    r = tl.rsqrt(var + 1.0e-5)

    w = tl.load(nw + offs_h, mask=offs_h < hidden, other=0.).to(tl.float32)
    bb = tl.load(nb + offs_h, mask=offs_h < hidden, other=0.).to(tl.float32)
    g = tl.load(proj + gsrc, mask=valid, other=0.).to(tl.float32)

    y = ((c * r[None, :]) * w[:, None] + bb[:, None]) * tl.sigmoid(g)
    tl.store(normed + offs_p[:, None] * hidden + offs_h[None, :],
             tl.trans(y).to(tl.float16),
             mask=(offs_p[:, None] < total) & (offs_h[None, :] < hidden))


@triton.jit
def _norm_gate_out128_kernel(tri, proj, out, nw, nb, wout_t,
                             total:tl.constexpr, n2:tl.constexpr, bsz:tl.constexpr,
                             hidden:tl.constexpr, dim:tl.constexpr,
                             BLOCK_P:tl.constexpr, BLOCK_H:tl.constexpr, BLOCK_D:tl.constexpr):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, BLOCK_H)
    offs_d = tl.arange(0, BLOCK_D)

    b = offs_p // n2
    pos = offs_p - b * n2
    src = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    gsrc = (4 * hidden + offs_h[:, None]) * total + offs_p[None, :]
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)

    v = tl.load(tri + src, mask=valid, other=0.).to(tl.float32)
    mean = tl.sum(v, axis=0) / hidden
    c = v - mean[None, :]
    var = tl.sum(c * c, axis=0) / hidden
    r = tl.rsqrt(var + 1.0e-5)

    w = tl.load(nw + offs_h, mask=offs_h < hidden, other=0.).to(tl.float32)
    bb = tl.load(nb + offs_h, mask=offs_h < hidden, other=0.).to(tl.float32)
    g = tl.load(proj + gsrc, mask=valid, other=0.).to(tl.float32)
    y = ((c * r[None, :]) * w[:, None] + bb[:, None]) * tl.sigmoid(g)

    wt = tl.load(wout_t + offs_h[:, None] * dim + offs_d[None, :],
                 mask=(offs_h[:, None] < hidden) & (offs_d[None, :] < dim), other=0.)
    acc = tl.dot(tl.trans(y.to(tl.float16)), wt, out_dtype=tl.float32)
    tl.store(out + offs_p[:, None] * dim + offs_d[None, :], acc,
             mask=(offs_p[:, None] < total) & (offs_d[None, :] < dim))


def _cached_weights(weights):
    ts = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((t.data_ptr(), tuple(t.shape), t.dtype) for t in ts)
    v = _WEIGHT_CACHE.get(key)
    if v is not None:
        return v

    wproj = torch.cat(ts[:5], dim=0).contiguous().to(torch.float16)
    wout_t = ts[5].t().contiguous().to(torch.float16)
    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (wproj, wout_t)
    return wproj, wout_t


def _mm_fp32(a, b):
    global _MM_MODE
    if _MM_MODE == 1:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_MODE == 2:
        return torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
    if _MM_MODE == 3:
        return torch.mm(a, b).to(torch.float32)
    try:
        o = torch.mm(a, b, out_dtype=torch.float32)
        _MM_MODE = 1
        return o
    except Exception:
        pass
    try:
        o = torch.bmm(a.unsqueeze(0), b.unsqueeze(0), out_dtype=torch.float32).squeeze(0)
        _MM_MODE = 2
        return o
    except Exception:
        _MM_MODE = 3
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    x, mask, weights, config = data
    bsz, n, _, dim = x.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n

    torch.backends.cuda.matmul.allow_tf32 = True
    wproj, wout_t = _cached_weights(weights)

    xh = torch.empty((total, dim), device=x.device, dtype=torch.float16)
    if dim == 128:
        bm, bd, nw = 16, 128, 4
    elif dim == 384:
        bm, bd, nw = 8, 512, 8
    else:
        bm, bd, nw = 4, 1024, 8

    _ln_to_half_kernel[(triton.cdiv(total, bm),)](
        x, xh, weights["norm.weight"], weights["norm.bias"],
        total, dim, BLOCK_M=bm, BLOCK_D=bd, num_warps=nw
    )

    proj = torch.mm(wproj, xh.t())

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.reshape(total) if has_mask else x
    _gate_inplace_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
        proj, mask_arg, total, hidden, has_mask,
        BLOCK_P=64, BLOCK_H=32, num_warps=4
    )

    left = proj[:hidden].view(hidden * bsz, n, n)
    right = proj[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri = torch.bmm(left, right.transpose(1, 2))

    if dim == 128 and hidden == 128 and n <= 512:
        out = torch.empty((total, dim), device=x.device, dtype=torch.float32)
        bp = 16 if n <= 256 else 8
        _norm_gate_out128_kernel[(triton.cdiv(total, bp),)](
            tri, proj, out,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"], wout_t,
            total, n2, bsz, hidden, dim,
            BLOCK_P=bp, BLOCK_H=128, BLOCK_D=128,
            num_warps=8
        )
    else:
        normed = torch.empty((total, hidden), device=x.device, dtype=torch.float16)
        _norm_gate_kernel[(triton.cdiv(total, 16),)](
            tri, proj, normed,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            total, n2, bsz, hidden,
            BLOCK_P=16, BLOCK_H=128,
            num_warps=8
        )
        out = _mm_fp32(normed, wout_t)

    return out.reshape(bsz, n, n, dim)
