"""
Outgoing TriMul forward pass.

Hybrid implementation:
1. Triton computes input LayerNorm directly to fp16.
2. A packed fp16 GEMM computes left/right projections and gates.
3. Triton applies sigmoid gates and optional mask in hidden-major layout.
4. cuBLAS batched GEMM performs the N^3 outgoing triangle product per hidden channel.
5. Triton fuses output LayerNorm with out_gate; for dim=hidden=128 it also fuses final projection.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_OUT_DTYPE_OK = None


@triton.jit
def _ln_to_half_kernel(x, y, w, b, total: tl.constexpr, dim: tl.constexpr,
                       BM: tl.constexpr, BD: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BM + tl.arange(0, BM)
    offs_d = tl.arange(0, BD)

    mask = (offs_m[:, None] < total) & (offs_d[None, :] < dim)
    vals = tl.load(x + offs_m[:, None] * dim + offs_d[None, :], mask=mask, other=0.0).to(tl.float32)

    active = offs_d[None, :] < dim
    mean = tl.sum(tl.where(active, vals, 0.0), axis=1) / dim
    centered = tl.where(active, vals - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / dim
    rstd = tl.rsqrt(var + 1.0e-5)

    ww = tl.load(w + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)
    bb = tl.load(b + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)
    out = centered * rstd[:, None] * ww[None, :] + bb[None, :]

    tl.store(y + offs_m[:, None] * dim + offs_d[None, :], out.to(tl.float16), mask=mask)


@triton.jit
def _gate_inplace_kernel(proj_h, mask_ptr, total: tl.constexpr, hidden: tl.constexpr,
                         has_mask: tl.constexpr, BP: tl.constexpr, BH: tl.constexpr):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_p = pid_p * BP + tl.arange(0, BP)
    offs_h = pid_h * BH + tl.arange(0, BH)
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)
    base = offs_h[:, None] * total + offs_p[None, :]

    left = tl.load(proj_h + base, mask=valid, other=0.0).to(tl.float32)
    right = tl.load(proj_h + hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj_h + 2 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj_h + 3 * hidden * total + base, mask=valid, other=0.0).to(tl.float32)

    left = left * tl.sigmoid(lg)
    right = right * tl.sigmoid(rg)

    if has_mask:
        m = tl.load(mask_ptr + offs_p, mask=offs_p < total, other=0.0).to(tl.float32)
        left = left * m[None, :]
        right = right * m[None, :]

    tl.store(proj_h + base, left.to(tl.float16), mask=valid)
    tl.store(proj_h + hidden * total + base, right.to(tl.float16), mask=valid)


@triton.jit
def _norm_gate_kernel(tri_h, proj_h, y, nw, nb,
                      total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
                      hidden: tl.constexpr, BP: tl.constexpr, BH: tl.constexpr):
    pid = tl.program_id(0)
    offs_p = pid * BP + tl.arange(0, BP)
    offs_h = tl.arange(0, BH)

    b = offs_p // n2
    pos = offs_p - b * n2
    valid = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)

    tri_idx = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    gate_idx = (4 * hidden + offs_h[:, None]) * total + offs_p[None, :]

    vals = tl.load(tri_h + tri_idx, mask=valid, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    centered = vals - mean[None, :]
    var = tl.sum(centered * centered, axis=0) / hidden
    rstd = tl.rsqrt(var + 1.0e-5)

    ww = tl.load(nw + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    bb = tl.load(nb + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj_h + gate_idx, mask=valid, other=0.0).to(tl.float32)

    out = ((centered * rstd[None, :]) * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)

    tl.store(
        y + offs_p[:, None] * hidden + offs_h[None, :],
        tl.trans(out).to(tl.float16),
        mask=(offs_p[:, None] < total) & (offs_h[None, :] < hidden),
    )


@triton.jit
def _norm_gate_out128_kernel(tri_h, proj_h, out, nw, nb, wout_t,
                             total: tl.constexpr, n2: tl.constexpr, bsz: tl.constexpr,
                             hidden: tl.constexpr, dim: tl.constexpr,
                             BP: tl.constexpr, BH: tl.constexpr, BD: tl.constexpr):
    pid = tl.program_id(0)
    offs_p = pid * BP + tl.arange(0, BP)
    offs_h = tl.arange(0, BH)
    offs_d = tl.arange(0, BD)

    b = offs_p // n2
    pos = offs_p - b * n2
    valid_h = (offs_p[None, :] < total) & (offs_h[:, None] < hidden)

    tri_idx = ((offs_h[:, None] * bsz + b[None, :]) * n2) + pos[None, :]
    gate_idx = (4 * hidden + offs_h[:, None]) * total + offs_p[None, :]

    vals = tl.load(tri_h + tri_idx, mask=valid_h, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    centered = vals - mean[None, :]
    var = tl.sum(centered * centered, axis=0) / hidden
    rstd = tl.rsqrt(var + 1.0e-5)

    ww = tl.load(nw + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    bb = tl.load(nb + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    gate = tl.load(proj_h + gate_idx, mask=valid_h, other=0.0).to(tl.float32)

    y = ((centered * rstd[None, :]) * ww[:, None] + bb[:, None]) * tl.sigmoid(gate)
    yh = tl.trans(y.to(tl.float16))

    wt = tl.load(
        wout_t + offs_h[:, None] * dim + offs_d[None, :],
        mask=(offs_h[:, None] < hidden) & (offs_d[None, :] < dim),
        other=0.0,
    )
    acc = tl.dot(yh, wt, out_dtype=tl.float32)

    tl.store(
        out + offs_p[:, None] * dim + offs_d[None, :],
        acc,
        mask=(offs_p[:, None] < total) & (offs_d[None, :] < dim),
    )


def _cached_weights(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
        weights["to_out.weight"],
    )
    key = tuple((t.data_ptr(), tuple(t.shape), t.dtype) for t in tensors)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    w_proj = torch.cat(tensors[:5], dim=0).contiguous().to(torch.float16)
    w_out_t = tensors[5].t().contiguous().to(torch.float16)

    if len(_WEIGHT_CACHE) >= 8:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = (w_proj, w_out_t)
    return w_proj, w_out_t


def _matmul_out(normed, w_out_t):
    global _OUT_DTYPE_OK
    if _OUT_DTYPE_OK is not False:
        try:
            out = torch.mm(normed, w_out_t, out_dtype=torch.float32)
            _OUT_DTYPE_OK = True
            return out
        except (TypeError, RuntimeError):
            _OUT_DTYPE_OK = False
    return torch.mm(normed, w_out_t).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    total = bsz * n * n
    n2 = n * n

    torch.backends.cuda.matmul.allow_tf32 = True
    w_proj, w_out_t = _cached_weights(weights)

    xh = torch.empty((total, dim), device=input_tensor.device, dtype=torch.float16)

    if dim <= 128:
        bd, bm, nw = 128, 16, 4
    elif dim <= 384:
        bd, bm, nw = 512, 8, 8
    else:
        bd, bm, nw = 1024, 4, 8

    _ln_to_half_kernel[(triton.cdiv(total, bm),)](
        input_tensor, xh,
        weights["norm.weight"], weights["norm.bias"],
        total, dim,
        BM=bm, BD=bd,
        num_warps=nw,
    )

    proj_h = torch.mm(w_proj, xh.t())

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.reshape(total) if has_mask else input_tensor

    _gate_inplace_kernel[(triton.cdiv(total, 64), triton.cdiv(hidden, 32))](
        proj_h, mask_arg,
        total, hidden, has_mask,
        BP=64, BH=32,
        num_warps=4,
    )

    left_h = proj_h[:hidden].view(hidden * bsz, n, n)
    right_h = proj_h[hidden:2 * hidden].view(hidden * bsz, n, n)
    tri_h = torch.bmm(left_h, right_h.transpose(1, 2))

    if dim == 128 and hidden == 128:
        out = torch.empty((total, dim), device=input_tensor.device, dtype=torch.float32)
        bp = 16 if n <= 256 else 8
        _norm_gate_out128_kernel[(triton.cdiv(total, bp),)](
            tri_h, proj_h, out,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"], w_out_t,
            total, n2, bsz, hidden, dim,
            BP=bp, BH=128, BD=128,
            num_warps=8,
        )
    else:
        normed = torch.empty((total, hidden), device=input_tensor.device, dtype=torch.float16)
        _norm_gate_kernel[(triton.cdiv(total, 16),)](
            tri_h, proj_h, normed,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            total, n2, bsz, hidden,
            BP=16, BH=128,
            num_warps=8,
        )
        out = _matmul_out(normed, w_out_t)

    return out.reshape(bsz, n, n, dim)
