"""TriMul outgoing forward using PyTorch layer-norm/GEMM, a Triton split-gate-mask kernel, cuBLAS batched GEMM for the triangle product, then fused PyTorch output norm/gate/projection."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_PROJ_CACHE = {}


@triton.jit
def _split_gate_mask_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    TOTAL: tl.constexpr,
    N2: tl.constexpr,
    H: tl.constexpr,
    USE_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < TOTAL

    h = offs % H
    row = offs // H
    base = row * (5 * H) + h

    lp = tl.load(proj_ptr + base, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(proj_ptr + base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(proj_ptr + base + 4 * H, mask=valid, other=0.0).to(tl.float32)

    if USE_MASK:
        m = tl.load(mask_ptr + row, mask=valid, other=0.0).to(tl.float32)
    else:
        m = 1.0

    b = row // N2
    ij = row - b * N2
    mat_off = (b * H + h) * N2 + ij

    tl.store(left_ptr + mat_off, lp * tl.sigmoid(lg) * m, mask=valid)
    tl.store(right_ptr + mat_off, rp * tl.sigmoid(rg) * m, mask=valid)
    tl.store(out_gate_ptr + row * H + h, tl.sigmoid(og), mask=valid)


def _cat_projection_weight(weights):
    lw = weights["left_proj.weight"]
    rw = weights["right_proj.weight"]
    lgw = weights["left_gate.weight"]
    rgw = weights["right_gate.weight"]
    ogw = weights["out_gate.weight"]

    key = (
        lw.data_ptr(), rw.data_ptr(), lgw.data_ptr(), rgw.data_ptr(), ogw.data_ptr(),
        lw.shape, lw.dtype, lw.device,
    )
    cached = _PROJ_CACHE.get(key)
    if cached is not None:
        return cached

    if len(_PROJ_CACHE) > 8:
        _PROJ_CACHE.clear()

    proj_weight = torch.cat((lw, rw, lgw, rgw, ogw), dim=0).contiguous()
    _PROJ_CACHE[key] = proj_weight
    return proj_weight


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    eps = float(config.get("eps", 1.0e-5))

    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True

    with torch.no_grad():
        x = input_tensor
        if x.dtype is not torch.float32:
            x = x.float()

        bs, seq_len, _, _ = x.shape
        rows = bs * seq_len * seq_len
        n2 = seq_len * seq_len

        x = F.layer_norm(
            x,
            (dim,),
            weights["norm.weight"],
            weights["norm.bias"],
            eps,
        )

        proj = F.linear(x, _cat_projection_weight(weights)).contiguous()

        left = torch.empty(
            (bs * hidden_dim, seq_len, seq_len),
            device=x.device,
            dtype=proj.dtype,
        )
        right = torch.empty_like(left)
        out_gate = torch.empty(
            (bs, seq_len, seq_len, hidden_dim),
            device=x.device,
            dtype=proj.dtype,
        )

        use_mask = (mask is not None) and (not bool(config.get("nomask", False)))
        mask_arg = mask.contiguous() if use_mask else input_tensor

        total = rows * hidden_dim
        block = 256
        _split_gate_mask_kernel[(triton.cdiv(total, block),)](
            proj,
            mask_arg,
            left,
            right,
            out_gate,
            TOTAL=total,
            N2=n2,
            H=hidden_dim,
            USE_MASK=use_mask,
            BLOCK=block,
            num_warps=4,
        )

        tri = torch.bmm(left, right.transpose(1, 2))
        tri = tri.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
        tri = tri.view(rows, hidden_dim)

        tri = F.layer_norm(
            tri,
            (hidden_dim,),
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            eps,
        )
        tri.mul_(out_gate.view(rows, hidden_dim))

        out = F.linear(tri, weights["to_out.weight"])
        return out.view(bs, seq_len, seq_len, dim).float()
