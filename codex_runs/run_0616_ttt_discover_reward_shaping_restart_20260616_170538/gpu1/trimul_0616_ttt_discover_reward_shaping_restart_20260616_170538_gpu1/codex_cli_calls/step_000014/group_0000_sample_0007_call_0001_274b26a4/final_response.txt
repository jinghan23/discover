"""
Fused AlphaFold-style outgoing Triangle Multipative Update.

The implementation keeps the expensive N^3 contraction on highly optimized
torch.bmm calls by folding batch and hidden channels, while using Triton for
the hidden-dimension output LayerNorm fused with the output gate. Input
LayerNorm and dense projections use PyTorch/cuBLAS, with projection weights
cached as one concatenated matrix to avoid repeated small setup work.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


_PROJ_WEIGHT_CACHE = {}


@triton.jit
def _out_norm_gate_h128_kernel(y_ptr, gate_ptr, norm_w_ptr, norm_b_ptr, out_ptr):
    pid = tl.program_id(0)
    offs = tl.arange(0, 128)

    vals = tl.load(y_ptr + pid * 128 + offs).to(tl.float32)
    mean = tl.sum(vals, axis=0) * 0.0078125
    centered = vals - mean
    var = tl.sum(centered * centered, axis=0) * 0.0078125
    inv = tl.rsqrt(var + 1.0e-5)

    w = tl.load(norm_w_ptr + offs).to(tl.float32)
    b = tl.load(norm_b_ptr + offs).to(tl.float32)
    g = tl.load(gate_ptr + pid * 128 + offs).to(tl.float32)

    tl.store(out_ptr + pid * 128 + offs, (centered * inv * w + b) * g)


@triton.jit
def _out_norm_gate_kernel(
    y_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    H: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = tl.arange(0, BLOCK_H)
    mask = offs < H

    vals = tl.load(y_ptr + pid * H + offs, mask=mask, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / H
    centered = vals - mean
    var = tl.sum(centered * centered, axis=0) / H
    inv = tl.rsqrt(var + 1.0e-5)

    w = tl.load(norm_w_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(norm_b_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + pid * H + offs, mask=mask, other=0.0).to(tl.float32)

    tl.store(out_ptr + pid * H + offs, (centered * inv * w + b) * g, mask=mask)


def _projection_weight(weights):
    names = (
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    )
    tensors = tuple(weights[name] for name in names)
    key = tuple(
        (
            id(t),
            int(t.data_ptr()),
            tuple(t.shape),
            tuple(t.stride()),
            str(t.dtype),
            str(t.device),
            int(getattr(t, "_version", 0)),
        )
        for t in tensors
    )

    cached = _PROJ_WEIGHT_CACHE.get(key)
    if cached is None:
        if len(_PROJ_WEIGHT_CACHE) > 16:
            _PROJ_WEIGHT_CACHE.clear()
        cached = torch.cat(tensors, dim=0).contiguous()
        _PROJ_WEIGHT_CACHE[key] = cached
    return cached


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    with torch.no_grad():
        x = input_tensor.contiguous()
        bsz, n, _, _ = x.shape

        x_norm = F.layer_norm(
            x,
            (dim,),
            weights["norm.weight"],
            weights["norm.bias"],
            1.0e-5,
        )

        proj = F.linear(x_norm, _projection_weight(weights))
        del x_norm

        left, right, left_gate, right_gate, out_gate = proj.split(hidden_dim, dim=-1)

        left = left * torch.sigmoid(left_gate)
        right = right * torch.sigmoid(right_gate)
        out_gate = torch.sigmoid(out_gate).contiguous()
        del proj, left_gate, right_gate

        if mask is not None and not bool(config.get("nomask", False)):
            m = mask.to(dtype=left.dtype).unsqueeze(-1)
            left = left * m
            right = right * m

        left = left.contiguous()
        right = right.contiguous()

        left_h = left.permute(0, 3, 1, 2).reshape(bsz * hidden_dim, n, n)
        right_h = right.permute(0, 3, 1, 2).reshape(bsz * hidden_dim, n, n)
        del left, right

        tri = torch.bmm(left_h, right_h.transpose(1, 2))
        del left_h, right_h

        tri = tri.reshape(bsz, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()

        gated_hidden = torch.empty_like(tri, dtype=torch.float32)
        rows = bsz * n * n

        if hidden_dim == 128:
            _out_norm_gate_h128_kernel[(rows,)](
                tri,
                out_gate,
                weights["to_out_norm.weight"],
                weights["to_out_norm.bias"],
                gated_hidden,
                num_warps=4,
            )
        else:
            _out_norm_gate_kernel[(rows,)](
                tri,
                out_gate,
                weights["to_out_norm.weight"],
                weights["to_out_norm.bias"],
                gated_hidden,
                hidden_dim,
                triton.next_power_of_2(hidden_dim),
                num_warps=4,
            )

        return F.linear(gated_hidden, weights["to_out.weight"]).to(torch.float32)
