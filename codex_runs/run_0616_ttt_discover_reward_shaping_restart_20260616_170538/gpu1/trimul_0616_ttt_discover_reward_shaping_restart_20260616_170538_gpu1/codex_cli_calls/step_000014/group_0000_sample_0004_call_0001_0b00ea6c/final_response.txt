"""
Fused outgoing Triangle Multiplicative Update.

Uses PyTorch/cuBLAS for the dense channel projections and hidden-channel
batched triangle products, with Triton handling the output hidden LayerNorm
and output gate fusion. Projection weights are cached after concatenation to
avoid repeated torch.cat overhead across benchmark calls.
"""

import torch
import triton
import triton.language as tl


_PROJ_WEIGHT_CACHE = {}


@triton.jit
def _out_norm_gate_kernel(
    y_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    H: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = tl.arange(0, BLOCK_H)
    m = offs < H

    vals = tl.load(y_ptr + pid * H + offs, mask=m, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / H
    centered = vals - mean
    var = tl.sum(centered * centered, axis=0) / H
    inv = tl.rsqrt(var + 1.0e-5)

    w = tl.load(norm_w_ptr + offs, mask=m, other=0.0).to(tl.float32)
    b = tl.load(norm_b_ptr + offs, mask=m, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + pid * H + offs, mask=m, other=0.0).to(tl.float32)

    out = (centered * inv * w + b) * g
    tl.store(out_ptr + pid * H + offs, out, mask=m)


def _projection_weight(weights):
    keys = (
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    )
    cache_key = tuple(
        (weights[k].data_ptr(), weights[k]._version, tuple(weights[k].shape))
        for k in keys
    )

    w = _PROJ_WEIGHT_CACHE.get(cache_key)
    if w is None:
        w = torch.cat(tuple(weights[k] for k in keys), dim=0).contiguous()
        if len(_PROJ_WEIGHT_CACHE) > 8:
            _PROJ_WEIGHT_CACHE.clear()
        _PROJ_WEIGHT_CACHE[cache_key] = w
    return w


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    x = input_tensor.contiguous()
    bsz, n, _, _ = x.shape

    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weights["norm.weight"],
        weights["norm.bias"],
        1.0e-5,
    )

    proj_w = _projection_weight(weights)
    proj = torch.matmul(x_norm, proj_w.t())
    left, right, left_gate, right_gate, out_gate = proj.split(hidden_dim, dim=-1)

    left = left * torch.sigmoid(left_gate)
    right = right * torch.sigmoid(right_gate)

    if not bool(config.get("nomask", False)):
        m = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    out_gate = torch.sigmoid(out_gate).contiguous()
    left = left.contiguous()
    right = right.contiguous()

    left_h = left.permute(0, 3, 1, 2).reshape(bsz * hidden_dim, n, n)
    right_h = right.permute(0, 3, 1, 2).reshape(bsz * hidden_dim, n, n)

    tri = torch.bmm(left_h, right_h.transpose(1, 2))
    tri = tri.reshape(bsz, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()

    gated_hidden = torch.empty_like(tri, dtype=torch.float32)
    rows = bsz * n * n

    _out_norm_gate_kernel[(rows,)](
        tri,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        gated_hidden,
        hidden_dim,
        triton.next_power_of_2(hidden_dim),
        num_warps=4,
    )

    out = torch.matmul(gated_hidden, weights["to_out.weight"].t())
    return out.to(torch.float32)
