"""
Outgoing TriMul forward.

The implementation normalizes the pair tensor with PyTorch LayerNorm, computes the
five input projections in one fused GEMM, packs gated left/right activations into
channel-major matrices with Triton, uses cuBLAS batched GEMM for the N^3 outgoing
triangle contraction, then fuses output LayerNorm, output gate, and final linear
projection in a Triton kernel. Small static weight transforms are cached by tensor
identity/version to avoid repeated concat/transpose overhead during benchmarking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}


def _tensor_key(t):
    return (t.data_ptr(), tuple(t.shape), tuple(t.stride()), str(t.dtype), getattr(t, "_version", 0))


def _get_proj_weight(weights):
    names = (
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    )
    key = ("proj",) + tuple(_tensor_key(weights[n]) for n in names)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached[0]

    src = tuple(weights[n] for n in names)
    out = torch.cat(src, dim=0).contiguous()
    _WEIGHT_CACHE[key] = (out, src)
    return out


def _get_to_out_weight_t(weights):
    w = weights["to_out.weight"]
    key = ("to_out_t", _tensor_key(w))
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached[0]

    out = w.t().contiguous()
    _WEIGHT_CACHE[key] = (out, w)
    return out


@triton.jit
def _pack_trimul_inputs(
    proj,
    mask,
    left_cm,
    right_t_cm,
    out_gate,
    TOTAL: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < TOTAL

    h = offs % H
    row = offs // H
    q = row % N
    p = (row // N) % N
    b = row // (N * N)

    base = row * (5 * H) + h

    lp = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(proj + base + 4 * H, mask=valid, other=0.0).to(tl.float32)

    if HAS_MASK:
        m = tl.load(mask + row, mask=valid, other=0.0).to(tl.float32)
    else:
        m = 1.0

    left_val = lp * (1.0 / (1.0 + tl.exp(-lg))) * m
    right_val = rp * (1.0 / (1.0 + tl.exp(-rg))) * m
    og_val = 1.0 / (1.0 + tl.exp(-og))

    nn = N * N
    cm_base = (b * H + h) * nn

    tl.store(left_cm + cm_base + p * N + q, left_val, mask=valid)
    tl.store(right_t_cm + cm_base + q * N + p, right_val, mask=valid)
    tl.store(out_gate + row * H + h, og_val, mask=valid)


@triton.jit
def _final_norm_gate_linear(
    out_cm,
    out_gate,
    norm_w,
    norm_b,
    to_out_w_t,
    y,
    M: tl.constexpr,
    NN: tl.constexpr,
    H: tl.constexpr,
    D: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)
    ds = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    r = rows % NN
    b = rows // NN

    mask_h = hs < H
    mask_mh = (rows[:, None] < M) & mask_h[None, :]

    vals = tl.load(
        out_cm + (b[:, None] * H + hs[None, :]) * NN + r[:, None],
        mask=mask_mh,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    centered = vals - mean[:, None]
    var = tl.sum(centered * centered, axis=1) / H
    rstd = tl.rsqrt(var + EPS)

    scale = tl.load(norm_w + hs, mask=mask_h, other=0.0).to(tl.float32)
    bias = tl.load(norm_b + hs, mask=mask_h, other=0.0).to(tl.float32)
    gate = tl.load(
        out_gate + rows[:, None] * H + hs[None, :],
        mask=mask_mh,
        other=0.0,
    ).to(tl.float32)

    z = (centered * rstd[:, None] * scale[None, :] + bias[None, :]) * gate
    z = tl.where(mask_h[None, :], z, 0.0)

    w = tl.load(
        to_out_w_t + hs[:, None] * D + ds[None, :],
        mask=mask_h[:, None] & (ds[None, :] < D),
        other=0.0,
    ).to(tl.float32)

    acc = tl.dot(z, w, input_precision="tf32")

    tl.store(
        y + rows[:, None] * D + ds[None, :],
        acc,
        mask=(rows[:, None] < M) & (ds[None, :] < D),
    )


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    torch.backends.cuda.matmul.allow_tf32 = True

    B, N, _, D = input_tensor.shape
    H = int(config["hidden_dim"])
    device = input_tensor.device

    x_in = input_tensor if input_tensor.dtype == torch.float32 else input_tensor.float()

    x = F.layer_norm(
        x_in,
        (D,),
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )

    proj_w = _get_proj_weight(weights)
    proj = F.linear(x, proj_w).contiguous()

    left_cm = torch.empty((B, H, N, N), device=device, dtype=torch.float32)
    right_t_cm = torch.empty((B, H, N, N), device=device, dtype=torch.float32)
    out_gate = torch.empty((B, N, N, H), device=device, dtype=torch.float32)

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    mask_arg = mask.contiguous() if has_mask else input_tensor

    total = B * N * N * H
    _pack_trimul_inputs[(triton.cdiv(total, 256),)](
        proj,
        mask_arg,
        left_cm,
        right_t_cm,
        out_gate,
        total,
        N,
        H,
        has_mask,
        BLOCK=256,
        num_warps=4,
    )

    out_cm = torch.bmm(
        left_cm.reshape(B * H, N, N),
        right_t_cm.reshape(B * H, N, N),
    ).reshape(B, H, N, N)

    out = torch.empty((B, N, N, D), device=device, dtype=torch.float32)

    block_h = triton.next_power_of_2(H)
    block_m = 8
    block_d = 128

    _final_norm_gate_linear[(triton.cdiv(B * N * N, block_m), triton.cdiv(D, block_d))](
        out_cm,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        _get_to_out_weight_t(weights),
        out,
        B * N * N,
        N * N,
        H,
        D,
        BLOCK_M=block_m,
        BLOCK_H=block_h,
        BLOCK_D=block_d,
        EPS=1e-5,
        num_warps=4,
        num_stages=3,
    )

    return out
