"""
Outgoing TriMul forward using one fused PyTorch projection, Triton packing for
gated left/right/out-gate tensors, channel-batched GEMM for the N^3 contraction,
and a Triton epilogue that fuses output LayerNorm, output gate, and final linear.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _pack_trimul_inputs(
    proj,
    mask,
    left_cm,
    right_t_cm,
    out_gate,
    TOTAL: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < TOTAL

    h = offs % H
    row = offs // H
    q = row % N
    p = (row // N) % N
    b = row // (N * N)

    base = row * (5 * H) + h

    lp = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(proj + base + 4 * H, mask=valid, other=0.0).to(tl.float32)

    if HAS_MASK:
        m = tl.load(mask + row, mask=valid, other=0.0).to(tl.float32)
    else:
        m = 1.0

    left_val = lp * (1.0 / (1.0 + tl.exp(-lg))) * m
    right_val = rp * (1.0 / (1.0 + tl.exp(-rg))) * m
    og_val = 1.0 / (1.0 + tl.exp(-og))

    nn = N * N
    cm_base = (b * H + h) * nn

    tl.store(left_cm + cm_base + p * N + q, left_val, mask=valid)
    tl.store(right_t_cm + cm_base + q * N + p, right_val, mask=valid)
    tl.store(out_gate + row * H + h, og_val, mask=valid)


@triton.jit
def _final_norm_gate_linear(
    out_cm,
    out_gate,
    norm_w,
    norm_b,
    to_out_w,
    y,
    M: tl.constexpr,
    NN: tl.constexpr,
    H: tl.constexpr,
    D: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)
    ds = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    r = rows % NN
    b = rows // NN

    mh_mask = (rows[:, None] < M) & (hs[None, :] < H)
    vals = tl.load(
        out_cm + (b[:, None] * H + hs[None, :]) * NN + r[:, None],
        mask=mh_mask,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    centered = vals - mean[:, None]
    var = tl.sum(centered * centered, axis=1) / H
    rstd = tl.rsqrt(var + EPS)

    scale = tl.load(norm_w + hs, mask=hs < H, other=0.0).to(tl.float32)
    bias = tl.load(norm_b + hs, mask=hs < H, other=0.0).to(tl.float32)
    gate = tl.load(
        out_gate + rows[:, None] * H + hs[None, :],
        mask=mh_mask,
        other=0.0,
    ).to(tl.float32)

    z = (centered * rstd[:, None] * scale[None, :] + bias[None, :]) * gate
    z = tl.where(hs[None, :] < H, z, 0.0)

    w = tl.load(
        to_out_w + ds[None, :] * H + hs[:, None],
        mask=(ds[None, :] < D) & (hs[:, None] < H),
        other=0.0,
    ).to(tl.float32)

    acc = tl.dot(z, w, input_precision="tf32")
    tl.store(
        y + rows[:, None] * D + ds[None, :],
        acc,
        mask=(rows[:, None] < M) & (ds[None, :] < D),
    )


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    torch.backends.cuda.matmul.allow_tf32 = True

    B, N, _, D = input_tensor.shape
    H = int(config["hidden_dim"])
    device = input_tensor.device

    x_in = input_tensor if input_tensor.dtype == torch.float32 else input_tensor.float()

    proj_w = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).contiguous()

    x = F.layer_norm(
        x_in,
        (D,),
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )
    proj = F.linear(x, proj_w).contiguous()

    left_cm = torch.empty((B, H, N, N), device=device, dtype=torch.float32)
    right_t_cm = torch.empty((B, H, N, N), device=device, dtype=torch.float32)
    out_gate = torch.empty((B, N, N, H), device=device, dtype=torch.float32)

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    mask_arg = mask.contiguous() if has_mask else input_tensor

    total = B * N * N * H
    _pack_trimul_inputs[(triton.cdiv(total, 256),)](
        proj,
        mask_arg,
        left_cm,
        right_t_cm,
        out_gate,
        total,
        N,
        H,
        has_mask,
        BLOCK=256,
        num_warps=4,
    )

    out_cm = torch.bmm(
        left_cm.reshape(B * H, N, N),
        right_t_cm.reshape(B * H, N, N),
    ).reshape(B, H, N, N)

    out = torch.empty((B, N, N, D), device=device, dtype=torch.float32)

    _final_norm_gate_linear[(triton.cdiv(B * N * N, 16), triton.cdiv(D, 64))](
        out_cm,
        out_gate,
        weights["to_out_norm.weight"].contiguous(),
        weights["to_out_norm.bias"].contiguous(),
        weights["to_out.weight"].contiguous(),
        out,
        B * N * N,
        N * N,
        H,
        D,
        BLOCK_M=16,
        BLOCK_H=triton.next_power_of_2(H),
        BLOCK_D=64,
        EPS=1e-5,
        num_warps=4,
        num_stages=3,
    )

    return out
