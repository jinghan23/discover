"""
Hybrid outgoing TriMul: PyTorch/cuBLAS handles the large dense projections and
hidden-channel batched GEMM, while Triton fuses projection gating, optional masking,
layout conversion, and post-triangle LayerNorm plus output gating.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    torch.set_float32_matmul_precision("high")
except Exception:
    pass


_PROJ_NAMES = (
    "left_proj.weight",
    "right_proj.weight",
    "left_gate.weight",
    "right_gate.weight",
    "out_gate.weight",
)
_PROJ_CACHE = {}


def _proj_cache_key(weights):
    key = []
    for name in _PROJ_NAMES:
        w = weights[name]
        try:
            version = w._version
        except Exception:
            version = -1
        key.append(
            (
                w.data_ptr(),
                version,
                tuple(w.shape),
                tuple(w.stride()),
                str(w.dtype),
                w.device.type,
                w.device.index,
            )
        )
    return tuple(key)


def _get_proj_weight(weights):
    key = _proj_cache_key(weights)
    cached = _PROJ_CACHE.get(key)
    if cached is None:
        cached = torch.cat(tuple(weights[name].float() for name in _PROJ_NAMES), dim=0).contiguous()
        if len(_PROJ_CACHE) >= 8:
            _PROJ_CACHE.clear()
        _PROJ_CACHE[key] = cached
    return cached


@triton.jit
def _prep_lr_gate_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    N: tl.constexpr,
    H: tl.constexpr,
    TOTAL: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    m_ok = offs_m < TOTAL
    h_ok = offs_h < H
    active = m_ok[:, None] & h_ok[None, :]

    pair_stride = 5 * H
    base = offs_m[:, None] * pair_stride + offs_h[None, :]

    if HAS_MASK:
        mv = tl.load(mask_ptr + offs_m, mask=m_ok, other=0.0).to(tl.float32)
    else:
        mv = tl.full((BLOCK_M,), 1.0, tl.float32)

    n2 = N * N
    b = offs_m // n2
    ij = offs_m - b * n2
    bhij = (b[:, None] * H + offs_h[None, :]) * n2 + ij[:, None]

    l = tl.load(proj_ptr + base, mask=active, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=active, other=0.0).to(tl.float32)
    tl.store(left_ptr + bhij, l * tl.sigmoid(lg) * mv[:, None], mask=active)

    r = tl.load(proj_ptr + base + H, mask=active, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=active, other=0.0).to(tl.float32)
    tl.store(right_ptr + bhij, r * tl.sigmoid(rg) * mv[:, None], mask=active)

    og = tl.load(proj_ptr + base + 4 * H, mask=active, other=0.0).to(tl.float32)
    tl.store(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :], tl.sigmoid(og), mask=active)


@triton.jit
def _norm_gate_kernel(
    tri_ptr,
    out_gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    y_ptr,
    N: tl.constexpr,
    H: tl.constexpr,
    TOTAL: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    m_ok = offs_m < TOTAL
    h_ok = offs_h < H
    active = m_ok[:, None] & h_ok[None, :]

    n2 = N * N
    b = offs_m // n2
    ij = offs_m - b * n2

    tri_offsets = (b[:, None] * H + offs_h[None, :]) * n2 + ij[:, None]
    vals = tl.load(tri_ptr + tri_offsets, mask=active, other=0.0).to(tl.float32)

    inv_h = 1.0 / H
    mean = tl.sum(vals, axis=1) * inv_h
    diff = tl.where(h_ok[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) * inv_h

    w = tl.load(norm_w_ptr + offs_h, mask=h_ok, other=1.0).to(tl.float32)
    beta = tl.load(norm_b_ptr + offs_h, mask=h_ok, other=0.0).to(tl.float32)
    gate = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :], mask=active, other=0.0).to(tl.float32)

    y = (diff * tl.rsqrt(var[:, None] + EPS) * w[None, :] + beta[None, :]) * gate
    tl.store(y_ptr + offs_m[:, None] * H + offs_h[None, :], y, mask=active)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.inference_mode():
        x = input_tensor.float()
        bs, seq_len, _, dim = x.shape
        hidden_dim = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
        eps = float(config.get("eps", config.get("layer_norm_eps", config.get("layernorm_eps", 1e-5))))

        norm_weight = weights["norm.weight"].float().contiguous()
        norm_bias = weights["norm.bias"].float().contiguous()
        to_out_norm_weight = weights["to_out_norm.weight"].float().contiguous()
        to_out_norm_bias = weights["to_out_norm.bias"].float().contiguous()
        to_out_weight = weights["to_out.weight"].float().contiguous()
        proj_weight = _get_proj_weight(weights)

        x = F.layer_norm(x, (dim,), norm_weight, norm_bias, eps)
        proj = F.linear(x, proj_weight)

        total_pairs = bs * seq_len * seq_len
        left = torch.empty((bs * hidden_dim, seq_len, seq_len), device=x.device, dtype=torch.float32)
        right = torch.empty_like(left)
        out_gate = torch.empty((total_pairs, hidden_dim), device=x.device, dtype=torch.float32)

        has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
        mask_ptr = mask.contiguous() if has_mask else input_tensor

        prep_bm = 32
        prep_bh = 32
        _prep_lr_gate_kernel[(triton.cdiv(total_pairs, prep_bm), triton.cdiv(hidden_dim, prep_bh))](
            proj,
            mask_ptr,
            left,
            right,
            out_gate,
            N=seq_len,
            H=hidden_dim,
            TOTAL=total_pairs,
            HAS_MASK=has_mask,
            BLOCK_M=prep_bm,
            BLOCK_H=prep_bh,
            num_warps=4,
        )

        del x, proj

        tri = torch.empty((bs * hidden_dim, seq_len, seq_len), device=left.device, dtype=torch.float32)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        del left, right

        y = torch.empty((bs, seq_len, seq_len, hidden_dim), device=tri.device, dtype=torch.float32)
        norm_bh = 1 << (hidden_dim - 1).bit_length()
        norm_bm = 4

        _norm_gate_kernel[(triton.cdiv(total_pairs, norm_bm),)](
            tri,
            out_gate,
            to_out_norm_weight,
            to_out_norm_bias,
            y,
            N=seq_len,
            H=hidden_dim,
            TOTAL=total_pairs,
            EPS=eps,
            BLOCK_M=norm_bm,
            BLOCK_H=norm_bh,
            num_warps=4,
        )

        del tri, out_gate

        return F.linear(y, to_out_weight).float()
