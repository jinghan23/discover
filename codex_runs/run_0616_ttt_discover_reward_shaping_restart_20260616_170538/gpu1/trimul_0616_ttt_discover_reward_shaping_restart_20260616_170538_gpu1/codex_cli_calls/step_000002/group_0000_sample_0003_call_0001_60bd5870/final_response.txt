"""
TriMul outgoing forward pass.

Uses PyTorch for the projection-heavy LayerNorm and linear stages, then a Triton
kernel for the dominant outgoing triangle multiplication:
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
The triangle kernel treats each hidden channel as an independent tiled GEMM and
accumulates in fp32. The result is normalized, output-gated, projected back to
the input channel dimension, and returned as float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_out_kernel(
    left,
    right,
    out,
    N: tl.constexpr,
    H: tl.constexpr,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BK: tl.constexpr,
):
    pid = tl.program_id(0)

    nj = tl.cdiv(N, BN)
    ni = tl.cdiv(N, BM)
    tiles = ni * nj

    bh = pid // tiles
    t = pid - bh * tiles

    ti = t // nj
    tj = t - ti * nj

    b = bh // H
    h = bh - b * H

    oi = ti * BM + tl.arange(0, BM)
    oj = tj * BN + tl.arange(0, BN)
    ok = tl.arange(0, BK)

    base = b * N * N * H + h
    acc = tl.zeros((BM, BN), tl.float32)

    for k0 in range(0, N, BK):
        k = k0 + ok

        lp = left + base + oi[:, None] * N * H + k[None, :] * H
        rp = right + base + oj[None, :] * N * H + k[:, None] * H

        l = tl.load(lp, mask=(oi[:, None] < N) & (k[None, :] < N), other=0.0)
        r = tl.load(rp, mask=(oj[None, :] < N) & (k[:, None] < N), other=0.0)

        acc += tl.dot(l, r, input_precision="tf32")

    op = out + base + oi[:, None] * N * H + oj[None, :] * H
    tl.store(op, acc, mask=(oi[:, None] < N) & (oj[None, :] < N))


def _layer_norm(x, weight, bias, eps=1e-5):
    xf = x.float()
    mean = xf.mean(dim=-1, keepdim=True)
    xc = xf - mean
    var = (xc * xc).mean(dim=-1, keepdim=True)
    return xc * torch.rsqrt(var + eps) * weight + bias


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, _ = input_tensor.shape
    M = B * N * N

    x = _layer_norm(input_tensor, norm_weight, norm_bias)
    x2 = x.reshape(M, dim)

    proj_w = torch.cat(
        (
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
        ),
        dim=0,
    )
    proj = torch.matmul(x2, proj_w.t()).reshape(B, N, N, 5, hidden_dim)

    left = proj[:, :, :, 0, :]
    right = proj[:, :, :, 1, :]
    left_gate = torch.sigmoid(proj[:, :, :, 2, :])
    right_gate = torch.sigmoid(proj[:, :, :, 3, :])
    out_gate = torch.sigmoid(proj[:, :, :, 4, :])

    left = left * left_gate
    right = right * right_gate

    if mask is not None:
        m = mask.unsqueeze(-1)
        left = left * m
        right = right * m

    tri = torch.empty((B, N, N, hidden_dim), device=input_tensor.device, dtype=torch.float32)

    block = 16 if N >= 768 else 32
    grid = (B * hidden_dim * triton.cdiv(N, block) * triton.cdiv(N, block),)

    _tri_out_kernel[grid](
        left,
        right,
        tri,
        N,
        hidden_dim,
        BM=block,
        BN=block,
        BK=32,
        num_warps=4,
        num_stages=4,
    )

    tri = _layer_norm(tri, to_out_norm_weight, to_out_norm_bias)
    tri = tri * out_gate.float()

    out = torch.matmul(tri.reshape(M, hidden_dim), to_out_weight.t())
    return out.reshape(B, N, N, dim).float()
