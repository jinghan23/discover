"""
TriMul outgoing forward pass.

Uses fused PyTorch dense kernels for layer norm and projections, a Triton kernel to
split the packed projections, apply sigmoid gates, optional mask, and write
left/right tensors directly in batched-GEMM layout. The triangle multiplication is
then computed with strided batched GEMM, followed by output LayerNorm, output gate,
and final projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

torch.backends.cuda.matmul.allow_tf32 = True
torch.backends.cudnn.allow_tf32 = True


@triton.jit
def _split_gate_kernel(
    proj,
    mask,
    left_bh,
    right_bh,
    out_gate,
    TOTAL_POS: tl.constexpr,
    N2: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_P: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_p = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_p = pid_p * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    p = offs_p[:, None]
    h = offs_h[None, :]
    valid = (offs_p[:, None] < TOTAL_POS) & (offs_h[None, :] < H)

    base = p * (5 * H) + h

    m = tl.full((BLOCK_P, 1), 1.0, tl.float32)
    if HAS_MASK:
        m = tl.load(mask + offs_p, mask=offs_p < TOTAL_POS, other=0.0)[:, None].to(tl.float32)

    l = tl.load(proj + base, mask=valid, other=0.0)
    lg = tl.sigmoid(tl.load(proj + base + 2 * H, mask=valid, other=0.0))
    l = l * lg * m

    r = tl.load(proj + base + H, mask=valid, other=0.0)
    rg = tl.sigmoid(tl.load(proj + base + 3 * H, mask=valid, other=0.0))
    r = r * rg * m

    og = tl.sigmoid(tl.load(proj + base + 4 * H, mask=valid, other=0.0))

    b = offs_p // N2
    pair = offs_p - b * N2

    hp_h = offs_h[:, None]
    hp_p = offs_p[None, :]
    hp_valid = (offs_h[:, None] < H) & (offs_p[None, :] < TOTAL_POS)

    bh_store = b[None, :] * (H * N2) + hp_h * N2 + pair[None, :]
    gate_store = hp_p * H + hp_h

    tl.store(left_bh + bh_store, tl.trans(l), mask=hp_valid)
    tl.store(right_bh + bh_store, tl.trans(r), mask=hp_valid)
    tl.store(out_gate + gate_store, tl.trans(og), mask=hp_valid)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    B, N, _, _ = input_tensor.shape
    total_pos = B * N * N
    n2 = N * N

    with torch.no_grad():
        x = F.layer_norm(
            input_tensor.float(),
            (dim,),
            weights["norm.weight"],
            weights["norm.bias"],
            1e-5,
        )
        x2 = x.reshape(total_pos, dim)

        proj_weight = torch.cat(
            (
                weights["left_proj.weight"],
                weights["right_proj.weight"],
                weights["left_gate.weight"],
                weights["right_gate.weight"],
                weights["out_gate.weight"],
            ),
            dim=0,
        )
        proj = F.linear(x2, proj_weight)

        left_bh = torch.empty((B * hidden_dim, N, N), device=input_tensor.device, dtype=proj.dtype)
        right_bh = torch.empty_like(left_bh)
        out_gate = torch.empty((B, N, N, hidden_dim), device=input_tensor.device, dtype=proj.dtype)

        has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
        if has_mask and not mask.is_contiguous():
            mask = mask.contiguous()
        mask_ptr = mask if has_mask else input_tensor

        grid = (triton.cdiv(total_pos, 32), triton.cdiv(hidden_dim, 32))
        _split_gate_kernel[grid](
            proj,
            mask_ptr,
            left_bh,
            right_bh,
            out_gate,
            total_pos,
            n2,
            hidden_dim,
            HAS_MASK=has_mask,
            BLOCK_P=32,
            BLOCK_H=32,
            num_warps=4,
        )

        tri_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))
        tri = tri_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

        tri = F.layer_norm(
            tri,
            (hidden_dim,),
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            1e-5,
        )
        tri.mul_(out_gate)

        out = F.linear(tri.reshape(total_pos, hidden_dim), weights["to_out.weight"])
        return out.reshape(B, N, N, dim).float()
