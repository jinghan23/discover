"""
TriMul outgoing forward pass.

Uses PyTorch/cuBLAS for the large dense projections and the per-hidden-channel
triangle products, while Triton fuses the gate sigmoid/mask application and the
final output LayerNorm + output gate + projection.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _gate_mask_kernel(
    left,
    right,
    lg,
    rg,
    mask,
    n_elem: tl.constexpr,
    H: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    m = offs < n_elem

    lv = tl.load(left + offs, mask=m, other=0.0).to(tl.float32)
    rv = tl.load(right + offs, mask=m, other=0.0).to(tl.float32)
    lgv = tl.load(lg + offs, mask=m, other=0.0).to(tl.float32)
    rgv = tl.load(rg + offs, mask=m, other=0.0).to(tl.float32)

    lgv = 1.0 / (1.0 + tl.exp(-lgv))
    rgv = 1.0 / (1.0 + tl.exp(-rgv))

    if has_mask:
        pair = offs // H
        mv = tl.load(mask + pair, mask=m, other=0.0).to(tl.float32)
        lgv *= mv
        rgv *= mv

    tl.store(left + offs, lv * lgv, mask=m)
    tl.store(right + offs, rv * rgv, mask=m)


@triton.jit
def _out_norm_gate_proj_kernel(
    tri,
    out_gate,
    w_norm,
    b_norm,
    w_out,
    y,
    M: tl.constexpr,
    D: tl.constexpr,
    H: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)
    hmask = offs_h < H

    vals = tl.load(
        tri + offs_m[:, None] * H + offs_h[None, :],
        mask=(offs_m[:, None] < M) & hmask[None, :],
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    centered = vals - mean[:, None]
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(w_norm + offs_h, mask=hmask, other=0.0).to(tl.float32)
    nb = tl.load(b_norm + offs_h, mask=hmask, other=0.0).to(tl.float32)

    gate = tl.load(
        out_gate + offs_m[:, None] * H + offs_h[None, :],
        mask=(offs_m[:, None] < M) & hmask[None, :],
        other=0.0,
    ).to(tl.float32)
    gate = 1.0 / (1.0 + tl.exp(-gate))

    v = ((centered * inv[:, None]) * nw[None, :] + nb[None, :]) * gate

    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
    w = tl.load(
        w_out + offs_d[:, None] * H + offs_h[None, :],
        mask=(offs_d[:, None] < D) & hmask[None, :],
        other=0.0,
    ).to(tl.float32)

    acc = tl.dot(v, tl.trans(w), input_precision="tf32")

    tl.store(
        y + offs_m[:, None] * D + offs_d[None, :],
        acc,
        mask=(offs_m[:, None] < M) & (offs_d[None, :] < D),
    )


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    x = input_tensor
    B, N, _, D = x.shape
    M = B * N * N

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x2 = x.reshape(M, D)
    xn = F.layer_norm(x2, (D,), norm_weight, norm_bias, eps=1e-5)

    left = F.linear(xn, left_proj_weight).contiguous()
    right = F.linear(xn, right_proj_weight).contiguous()
    left_gate = F.linear(xn, left_gate_weight).contiguous()
    right_gate = F.linear(xn, right_gate_weight).contiguous()
    out_gate = F.linear(xn, out_gate_weight).contiguous()

    n_elem = M * hidden_dim
    has_mask = mask is not None and not bool(config.get("nomask", False))

    _gate_mask_kernel[(triton.cdiv(n_elem, 256),)](
        left,
        right,
        left_gate,
        right_gate,
        mask if mask is not None else left,
        n_elem,
        hidden_dim,
        has_mask,
        BLOCK=256,
        num_warps=4,
    )

    left4 = left.view(B, N, N, hidden_dim)
    right4 = right.view(B, N, N, hidden_dim)

    a = left4.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    b = right4.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    tri = torch.bmm(a, b)
    tri = tri.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().view(M, hidden_dim)

    out = torch.empty((M, dim), device=x.device, dtype=torch.float32)

    _out_norm_gate_proj_kernel[(triton.cdiv(M, 16), triton.cdiv(dim, 32))](
        tri,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        out,
        M,
        dim,
        hidden_dim,
        1e-5,
        BLOCK_M=16,
        BLOCK_D=32,
        BLOCK_H=128,
        num_warps=4,
    )

    return out.view(B, N, N, dim)
