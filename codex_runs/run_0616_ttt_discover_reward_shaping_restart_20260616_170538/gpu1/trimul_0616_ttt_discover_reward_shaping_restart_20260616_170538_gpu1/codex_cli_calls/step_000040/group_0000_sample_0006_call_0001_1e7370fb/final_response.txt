"""
TriMul outgoing forward pass using PyTorch GEMMs plus Triton fusion kernels.

The implementation computes the input LayerNorm and the five dim->hidden
projections as one packed GEMM, uses Triton to fuse sigmoid gates, optional mask
application, and layout conversion into [B, H, N, N], evaluates the outgoing
triangle contraction as a batched channel-wise BMM, then uses Triton to fuse the
hidden LayerNorm with the output gate before the final projection. The returned
tensor is always float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_store_bh_kernel(
    proj,
    mask,
    left,
    right,
    out_gate,
    N: tl.constexpr,
    H: tl.constexpr,
    USE_MASK: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    pid_j = tl.program_id(0)
    pid_bi = tl.program_id(1)
    pid_h = tl.program_id(2)

    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    b = pid_bi // N
    i = pid_bi - b * N
    pair = (b * N + i) * N + offs_j

    h = offs_h[:, None]
    p = pair[None, :]
    valid = (offs_h[:, None] < H) & (offs_j[None, :] < N)

    base = p * (5 * H) + h

    l = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(proj + base + 4 * H, mask=valid, other=0.0).to(tl.float32)

    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    if USE_MASK:
        m = tl.load(mask + pair, mask=offs_j < N, other=0.0).to(tl.float32)[None, :]
    else:
        m = 1.0

    out_offs = ((b * H + h) * N + i) * N + offs_j[None, :]

    tl.store(left + out_offs, l * lg * m, mask=valid)
    tl.store(right + out_offs, r * rg * m, mask=valid)
    tl.store(out_gate + out_offs, og, mask=valid)


@triton.jit
def _norm_gate_to_hidden_kernel(
    tri,
    out_gate,
    norm_w,
    norm_b,
    hidden,
    N: tl.constexpr,
    H: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    pid_j = tl.program_id(0)
    pid_bi = tl.program_id(1)

    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    b = pid_bi // N
    i = pid_bi - b * N
    pair = (b * N + i) * N + offs_j

    valid = (offs_j[:, None] < N) & (offs_h[None, :] < H)
    bhnn_offs = ((b * H + offs_h[None, :]) * N + i) * N + offs_j[:, None]

    x = tl.load(tri + bhnn_offs, mask=valid, other=0.0).to(tl.float32)

    hmask = offs_h[None, :] < H
    x_masked = tl.where(hmask, x, 0.0)
    mean = tl.sum(x_masked, axis=1) / H
    diff = tl.where(hmask, x - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / H

    y = (x - mean[:, None]) * tl.rsqrt(var[:, None] + EPS)
    w = tl.load(norm_w + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    beta = tl.load(norm_b + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    g = tl.load(out_gate + bhnn_offs, mask=valid, other=0.0).to(tl.float32)

    y = (y * w[None, :] + beta[None, :]) * g
    hidden_offs = pair[:, None] * H + offs_h[None, :]
    tl.store(hidden + hidden_offs, y, mask=valid)


def _pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


_WEIGHT_CACHE = {}


def _cached_half_cat(weights, names, dtype):
    key = tuple((weights[n].data_ptr(), weights[n]._version, dtype) for n in names)
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached
    w = torch.cat([weights[n] for n in names], dim=0).to(dtype)
    if len(_WEIGHT_CACHE) > 16:
        _WEIGHT_CACHE.clear()
    _WEIGHT_CACHE[key] = w
    return w


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B, N, _, D = input_tensor.shape
    H = int(config["hidden_dim"])
    compute_dtype = torch.float16

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = F.layer_norm(input_tensor, (D,), norm_weight, norm_bias, 1e-5)

    proj_w = _cached_half_cat(
        weights,
        (
            "left_proj.weight",
            "right_proj.weight",
            "left_gate.weight",
            "right_gate.weight",
            "out_gate.weight",
        ),
        compute_dtype,
    )
    proj = F.linear(x.to(compute_dtype), proj_w)

    left = torch.empty((B, H, N, N), device=input_tensor.device, dtype=compute_dtype)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    use_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if use_mask and not mask.is_contiguous():
        mask = mask.contiguous()
    mask_arg = mask if use_mask else input_tensor

    block_h = 32
    block_j = 32
    grid_gate = (triton.cdiv(N, block_j), B * N, triton.cdiv(H, block_h))
    _gate_store_bh_kernel[grid_gate](
        proj,
        mask_arg,
        left,
        right,
        out_gate,
        N,
        H,
        use_mask,
        BLOCK_H=block_h,
        BLOCK_J=block_j,
        num_warps=8,
    )

    tri = torch.bmm(
        left.reshape(B * H, N, N),
        right.reshape(B * H, N, N).transpose(1, 2),
    ).reshape(B, H, N, N)

    hidden = torch.empty((B, N, N, H), device=input_tensor.device, dtype=compute_dtype)
    block_h2 = _pow2(H)
    block_j2 = 8
    grid_norm = (triton.cdiv(N, block_j2), B * N)
    _norm_gate_to_hidden_kernel[grid_norm](
        tri,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        hidden,
        N,
        H,
        1e-5,
        BLOCK_H=block_h2,
        BLOCK_J=block_j2,
        num_warps=8,
    )

    out = F.linear(hidden, to_out_weight.to(compute_dtype))
    return out.to(torch.float32)
