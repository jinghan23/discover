"""
Outgoing Triangle Multiplicative Update.

The implementation keeps the numerically sensitive LayerNorms in PyTorch, uses
Triton to fuse mask application and sigmoid gates for the left/right/out paths,
and relies on cuBLAS-backed grouped matrix multiplications for the N^3 triangle
contraction. Intermediate GEMMs use mixed precision, while the returned tensor is
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    left,
    right,
    out_gate_pre,
    mask,
    left_gate_pre,
    right_gate_pre,
    left_out,
    right_out,
    out_gate_out,
    total: tl.constexpr,
    hidden: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total

    l = tl.load(left + offs, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(right + offs, mask=valid, other=0.0).to(tl.float32)

    lg = tl.load(left_gate_pre + offs, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(right_gate_pre + offs, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(out_gate_pre + offs, mask=valid, other=0.0).to(tl.float32)

    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    if has_mask:
        pair = offs // hidden
        m = tl.load(mask + pair, mask=valid, other=0.0).to(tl.float32)
        l = l * m
        r = r * m

    tl.store(left_out + offs, l * lg, mask=valid)
    tl.store(right_out + offs, r * rg, mask=valid)
    tl.store(out_gate_out + offs, og, mask=valid)


def _linear(x, w):
    return torch.matmul(x, w.t())


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    eps = float(config.get("eps", 1e-5))

    x = input_tensor
    bs, n, _, _ = x.shape

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x_norm = F.layer_norm(x, (dim,), norm_weight, norm_bias, eps)

    work_dtype = torch.float16
    xh = x_norm.to(work_dtype)

    left_pre = _linear(xh, left_proj_weight.to(work_dtype))
    right_pre = _linear(xh, right_proj_weight.to(work_dtype))
    left_gate_pre = _linear(xh, left_gate_weight.to(work_dtype))
    right_gate_pre = _linear(xh, right_gate_weight.to(work_dtype))
    out_gate_pre = _linear(xh, out_gate_weight.to(work_dtype))

    left = torch.empty_like(left_pre)
    right = torch.empty_like(right_pre)
    out_gate = torch.empty_like(out_gate_pre)

    has_mask = mask is not None
    total = bs * n * n * hidden_dim
    block = 256
    grid = (triton.cdiv(total, block),)

    if has_mask:
        mask_arg = mask.contiguous()
    else:
        mask_arg = left_pre

    _apply_gate_mask_kernel[grid](
        left_pre,
        right_pre,
        out_gate_pre,
        mask_arg,
        left_gate_pre,
        right_gate_pre,
        left,
        right,
        out_gate,
        total,
        hidden_dim,
        has_mask,
        BLOCK=block,
        num_warps=8,
    )

    left_bdhw = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, n, n)
    right_bdhw = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, n, n)

    tri = torch.bmm(left_bdhw, right_bdhw.transpose(1, 2))
    tri = tri.reshape(bs, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()

    tri = F.layer_norm(
        tri.float(),
        (hidden_dim,),
        to_out_norm_weight.float(),
        to_out_norm_bias.float(),
        eps,
    )

    tri = tri * out_gate.float()
    out = torch.matmul(tri.to(work_dtype), to_out_weight.to(work_dtype).t()).float()
    return out
