"""
Outgoing TriMul forward. The implementation keeps the expensive work on tensor
cores: normalize the pair representation with PyTorch LayerNorm, compute all five
input projections in one packed GEMM, evaluate the triangular product as batched
hidden-channel GEMMs, then use a Triton kernel for output LayerNorm plus gating
before the final tensor-core projection. Outputs are returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


torch.backends.cuda.matmul.allow_tf32 = True
try:
    torch.set_float32_matmul_precision("high")
except Exception:
    pass


_PROJ_CACHE = {}
_OUT_CACHE = {}
_BMM_OUT_DTYPE = None
_MM_OUT_DTYPE = None


def _key(t):
    return (
        int(t.data_ptr()),
        tuple(t.shape),
        str(t.dtype),
        t.device.type,
        t.device.index,
        int(getattr(t, "_version", 0)),
    )


def _projection_weight(weights):
    ws = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = tuple(_key(w) for w in ws)
    cached = _PROJ_CACHE.get(key)
    if cached is None:
        if len(_PROJ_CACHE) > 8:
            _PROJ_CACHE.clear()
        cached = torch.cat([w.detach().to(torch.float16) for w in ws], dim=0).contiguous()
        _PROJ_CACHE[key] = cached
    return cached


def _out_weight_t(weights):
    w = weights["to_out.weight"]
    key = _key(w)
    cached = _OUT_CACHE.get(key)
    if cached is None:
        if len(_OUT_CACHE) > 8:
            _OUT_CACHE.clear()
        cached = w.detach().to(torch.float16).t().contiguous()
        _OUT_CACHE[key] = cached
    return cached


def _bmm_accum_fp32(a, b):
    global _BMM_OUT_DTYPE
    if _BMM_OUT_DTYPE is not False:
        try:
            y = torch.bmm(a, b, out_dtype=torch.float32)
            _BMM_OUT_DTYPE = True
            return y
        except TypeError:
            _BMM_OUT_DTYPE = False
    return torch.bmm(a, b)


def _mm_accum_fp32(a, b):
    global _MM_OUT_DTYPE
    if _MM_OUT_DTYPE is not False:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE = True
            return y
        except TypeError:
            _MM_OUT_DTYPE = False
    return torch.mm(a.float(), b.float())


def _next_power_of_2(x):
    return 1 << (int(x) - 1).bit_length()


@triton.jit
def _norm_gate_kernel(
    tri_ptr,
    gate_ptr,
    gamma_ptr,
    beta_ptr,
    hidden_ptr,
    P: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    n2 = N * N
    b = offs_m // n2
    ij = offs_m - b * n2

    row_mask = offs_m < P
    h_mask = offs_h < H

    tri_offsets = b[:, None] * (H * n2) + offs_h[None, :] * n2 + ij[:, None]
    x = tl.load(
        tri_ptr + tri_offsets,
        mask=row_mask[:, None] & h_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(x, axis=1) * (1.0 / H)
    centered = tl.where(h_mask[None, :], x - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) * (1.0 / H)
    rstd = tl.rsqrt(var + 1.0e-5)

    gamma = tl.load(gamma_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)

    gate_offsets = offs_m[:, None] * H + offs_h[None, :]
    gate = tl.load(
        gate_ptr + gate_offsets,
        mask=row_mask[:, None] & h_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    y = (centered * rstd[:, None] * gamma[None, :] + beta[None, :]) * gate
    tl.store(hidden_ptr + gate_offsets, y, mask=row_mask[:, None] & h_mask[None, :])


@torch.no_grad()
def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B, N, _, D = input_tensor.shape
    H = int(config["hidden_dim"])
    P = B * N * N

    x_norm = F.layer_norm(
        input_tensor,
        (D,),
        weights["norm.weight"],
        weights["norm.bias"],
        1.0e-5,
    )

    x_flat_h = x_norm.reshape(P, D).to(torch.float16)
    proj = F.linear(x_flat_h, _projection_weight(weights)).reshape(B, N, N, 5 * H)
    del x_norm, x_flat_h

    left = proj[..., 0:H] * torch.sigmoid(proj[..., 2 * H : 3 * H])
    right = proj[..., H : 2 * H] * torch.sigmoid(proj[..., 3 * H : 4 * H])
    out_gate = torch.sigmoid(proj[..., 4 * H : 5 * H]).contiguous()
    del proj

    if (mask is not None) and (not bool(config.get("nomask", False))):
        m = mask.to(dtype=torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    left = left.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous().reshape(B * H, N, N)

    tri = _bmm_accum_fp32(left, right).reshape(B, H, N, N)
    del left, right

    hidden = torch.empty((P, H), device=input_tensor.device, dtype=torch.float16)

    block_m = 16
    block_h = _next_power_of_2(H)
    grid = (triton.cdiv(P, block_m),)
    _norm_gate_kernel[grid](
        tri,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        hidden,
        P,
        N,
        H,
        BLOCK_M=block_m,
        BLOCK_H=block_h,
        num_warps=4,
    )
    del tri, out_gate

    out = _mm_accum_fp32(hidden, _out_weight_t(weights))
    return out.reshape(B, N, N, D).to(torch.float32)
