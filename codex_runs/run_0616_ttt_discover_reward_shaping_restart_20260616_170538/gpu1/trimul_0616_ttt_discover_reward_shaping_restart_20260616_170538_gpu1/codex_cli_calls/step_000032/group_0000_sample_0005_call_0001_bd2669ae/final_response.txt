"""
Fused TriMul outgoing forward path.

The implementation keeps the expensive triangular multiplication on tensor cores by
materializing gated left/right activations, then using batched GEMM over the hidden
channels. Triton kernels fuse the two LayerNorm+projection+gate stages and the final
LayerNorm+gate+projection stage. Outputs are returned as float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _prep_lr_kernel(
    x, mask, norm_w, norm_b,
    lp_w, rp_w, lg_w, rg_w, og_w,
    left, right, out_gate,
    total: tl.constexpr, D: tl.constexpr, H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_d = tl.arange(0, BLOCK_D)
    x_vals = tl.load(x + pid * D + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)

    mean = tl.sum(x_vals, axis=0) / D
    xc = tl.where(offs_d < D, x_vals - mean, 0.0)
    var = tl.sum(xc * xc, axis=0) / D
    inv = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)
    xn = (x_vals - mean) * inv * nw + nb
    xn = tl.where(offs_d < D, xn, 0.0)

    base_pair = pid
    m = 1.0
    if HAS_MASK:
        m = tl.load(mask + base_pair).to(tl.float32)

    offs_h = tl.arange(0, BLOCK_H)
    for h0 in range(0, H, BLOCK_H):
        hs = h0 + offs_h

        lp = tl.load(lp_w + hs[:, None] * D + offs_d[None, :],
                     mask=(hs[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
        rp = tl.load(rp_w + hs[:, None] * D + offs_d[None, :],
                     mask=(hs[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
        lg = tl.load(lg_w + hs[:, None] * D + offs_d[None, :],
                     mask=(hs[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
        rg = tl.load(rg_w + hs[:, None] * D + offs_d[None, :],
                     mask=(hs[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
        og = tl.load(og_w + hs[:, None] * D + offs_d[None, :],
                     mask=(hs[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)

        l_val = tl.sum(lp * xn[None, :], axis=1)
        r_val = tl.sum(rp * xn[None, :], axis=1)
        lg_val = tl.sum(lg * xn[None, :], axis=1)
        rg_val = tl.sum(rg * xn[None, :], axis=1)
        og_val = tl.sum(og * xn[None, :], axis=1)

        l_gate = 1.0 / (1.0 + tl.exp(-lg_val))
        r_gate = 1.0 / (1.0 + tl.exp(-rg_val))
        o_gate = 1.0 / (1.0 + tl.exp(-og_val))

        out_mask = hs < H
        tl.store(left + pid * H + hs, (l_val * l_gate * m).to(tl.float32), mask=out_mask)
        tl.store(right + pid * H + hs, (r_val * r_gate * m).to(tl.float32), mask=out_mask)
        tl.store(out_gate + pid * H + hs, o_gate.to(tl.float32), mask=out_mask)


@triton.jit
def _final_kernel(
    tri, out_gate, ton_w, ton_b, to_w, out,
    total: tl.constexpr, H: tl.constexpr, D: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_h = tl.arange(0, BLOCK_H)
    hmask = offs_h < H
    v = tl.load(tri + pid * H + offs_h, mask=hmask, other=0.0).to(tl.float32)

    mean = tl.sum(v, axis=0) / H
    vc = tl.where(hmask, v - mean, 0.0)
    var = tl.sum(vc * vc, axis=0) / H
    inv = tl.rsqrt(var + 1.0e-5)

    gw = tl.load(ton_w + offs_h, mask=hmask, other=0.0).to(tl.float32)
    gb = tl.load(ton_b + offs_h, mask=hmask, other=0.0).to(tl.float32)
    og = tl.load(out_gate + pid * H + offs_h, mask=hmask, other=0.0).to(tl.float32)

    y = ((v - mean) * inv * gw + gb) * og
    y = tl.where(hmask, y, 0.0)

    offs_d = tl.arange(0, BLOCK_D)
    for d0 in range(0, D, BLOCK_D):
        ds = d0 + offs_d
        w = tl.load(to_w + ds[:, None] * H + offs_h[None, :],
                    mask=(ds[:, None] < D) & hmask[None, :], other=0.0).to(tl.float32)
        acc = tl.sum(w * y[None, :], axis=1)
        tl.store(out + pid * D + ds, acc, mask=ds < D)


def _ceil_block_d(dim):
    if dim <= 128:
        return 128
    if dim <= 384:
        return 512
    return 1024


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = int(config["hidden_dim"])

    x = input_tensor.contiguous()
    mask = mask.contiguous() if mask is not None else None

    total_pairs = B * N * N
    block_d = _ceil_block_d(D)

    left = torch.empty((total_pairs, H), device=x.device, dtype=torch.float32)
    right = torch.empty((total_pairs, H), device=x.device, dtype=torch.float32)
    out_gate = torch.empty((total_pairs, H), device=x.device, dtype=torch.float32)

    has_mask = bool(mask is not None and not config.get("nomask", False))

    _prep_lr_kernel[(total_pairs,)](
        x, mask if mask is not None else x,
        weights["norm.weight"].contiguous(), weights["norm.bias"].contiguous(),
        weights["left_proj.weight"].contiguous(),
        weights["right_proj.weight"].contiguous(),
        weights["left_gate.weight"].contiguous(),
        weights["right_gate.weight"].contiguous(),
        weights["out_gate.weight"].contiguous(),
        left, right, out_gate,
        total_pairs, D, H, has_mask,
        BLOCK_D=block_d, BLOCK_H=32,
        num_warps=8,
    )

    l5 = left.reshape(B, N, N, H).permute(0, 3, 1, 2).reshape(B * H, N, N)
    r5 = right.reshape(B, N, N, H).permute(0, 3, 1, 2).reshape(B * H, N, N)

    tri = torch.bmm(l5, r5.transpose(1, 2))
    tri = tri.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous().reshape(total_pairs, H)

    out = torch.empty((total_pairs, D), device=x.device, dtype=torch.float32)

    _final_kernel[(total_pairs,)](
        tri, out_gate,
        weights["to_out_norm.weight"].contiguous(),
        weights["to_out_norm.bias"].contiguous(),
        weights["to_out.weight"].contiguous(),
        out,
        total_pairs, H, D,
        BLOCK_H=128, BLOCK_D=block_d,
        num_warps=8,
    )

    return out.reshape(B, N, N, D)
