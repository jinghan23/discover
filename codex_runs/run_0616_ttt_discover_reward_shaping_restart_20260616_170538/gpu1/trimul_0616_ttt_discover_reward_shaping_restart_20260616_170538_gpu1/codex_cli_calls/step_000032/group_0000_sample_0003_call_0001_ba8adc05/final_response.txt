"""
Fused TriMul outgoing forward.

Algorithm:
1. Use PyTorch for the channel-wise LayerNorm and input projections, which are dense
   [D -> 128] linear maps and are normally handled well by cuBLAS.
2. Fuse mask application, sigmoid gates, and the outgoing triangle contraction in
   Triton. The contraction is evaluated per hidden channel as batched GEMMs:
      out[b, h, i, j] = sum_k left[b, i, k, h] * right[b, j, k, h]
3. Fuse the output LayerNorm, output gate, and final projection in a Triton kernel.
   The final result is written as float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_out_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    mask_ptr,
    total_tiles: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_j_tiles = tl.cdiv(N, BLOCK_N)
    num_i_tiles = tl.cdiv(N, BLOCK_M)
    num_h_tiles = H

    h = pid % num_h_tiles
    t = pid // num_h_tiles
    j_tile = t % num_j_tiles
    t = t // num_j_tiles
    i_tile = t % num_i_tiles
    b = t // num_i_tiles

    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), tl.float32)

    base_b = b * N * N * H
    mask_base = b * N * N

    for k0 in range(0, N, BLOCK_K):
        k = k0 + offs_k

        l_ptrs = left_ptr + base_b + (offs_i[:, None] * N + k[None, :]) * H + h
        r_ptrs = right_ptr + base_b + (offs_j[None, :] * N + k[:, None]) * H + h

        l = tl.load(l_ptrs, mask=(offs_i[:, None] < N) & (k[None, :] < N), other=0.0)
        r = tl.load(r_ptrs, mask=(offs_j[None, :] < N) & (k[:, None] < N), other=0.0)

        if HAS_MASK:
            lm = tl.load(
                mask_ptr + mask_base + offs_i[:, None] * N + k[None, :],
                mask=(offs_i[:, None] < N) & (k[None, :] < N),
                other=0.0,
            )
            rm = tl.load(
                mask_ptr + mask_base + offs_j[None, :] * N + k[:, None],
                mask=(offs_j[None, :] < N) & (k[:, None] < N),
                other=0.0,
            )
            l = l * lm
            r = r * rm

        acc += tl.dot(l, r)

    out_ptrs = out_ptr + base_b + (offs_i[:, None] * N + offs_j[None, :]) * H + h
    tl.store(out_ptrs, acc, mask=(offs_i[:, None] < N) & (offs_j[None, :] < N))


@triton.jit
def _out_norm_gate_proj_kernel(
    tri_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    proj_w_ptr,
    out_ptr,
    total_rows: tl.constexpr,
    H: tl.constexpr,
    D: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    row = tl.program_id(0)
    d_tile = tl.program_id(1)

    offs_h = tl.arange(0, BLOCK_H)
    vals = tl.load(tri_ptr + row * H + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) / H
    centered = vals - mean
    var = tl.sum(centered * centered, axis=0) / H
    inv_std = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w_ptr + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    gate = tl.load(gate_ptr + row * H + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)

    y = ((vals - mean) * inv_std * nw + nb) * gate

    offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
    acc = tl.zeros((BLOCK_D,), tl.float32)

    for h0 in range(0, H, BLOCK_H):
        h = h0 + offs_h
        yh = y
        w = tl.load(
            proj_w_ptr + offs_d[:, None] * H + h[None, :],
            mask=(offs_d[:, None] < D) & (h[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        acc += tl.sum(w * yh[None, :], axis=1)

    tl.store(out_ptr + row * D + offs_d, acc, mask=offs_d < D)


def _linear_no_bias(x, weight):
    return torch.matmul(x, weight.t())


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        norm_weight,
        norm_bias,
        eps=1.0e-5,
    )

    left = _linear_no_bias(x, left_proj_weight)
    right = _linear_no_bias(x, right_proj_weight)

    left_gate = torch.sigmoid(_linear_no_bias(x, left_gate_weight))
    right_gate = torch.sigmoid(_linear_no_bias(x, right_gate_weight))
    out_gate = torch.sigmoid(_linear_no_bias(x, out_gate_weight))

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()
    out_gate = out_gate.contiguous()

    B, N, _, _ = left.shape
    H = hidden_dim
    D = dim

    tri = torch.empty((B, N, N, H), device=input_tensor.device, dtype=torch.float32)

    has_mask = not bool(config.get("nomask", False))
    mask_arg = mask if has_mask else input_tensor

    block_m = 16
    block_n = 16
    block_k = 64

    grid_tri = (B * triton.cdiv(N, block_m) * triton.cdiv(N, block_n) * H,)
    _tri_mul_out_kernel[grid_tri](
        left,
        right,
        tri,
        mask_arg,
        B * triton.cdiv(N, block_m) * triton.cdiv(N, block_n) * H,
        N,
        H,
        has_mask,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
        num_stages=3,
    )

    out = torch.empty((B, N, N, D), device=input_tensor.device, dtype=torch.float32)

    block_d = 32
    block_h = 128
    grid_out = (B * N * N, triton.cdiv(D, block_d))
    _out_norm_gate_proj_kernel[grid_out](
        tri,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        out,
        B * N * N,
        H,
        D,
        BLOCK_D=block_d,
        BLOCK_H=block_h,
        num_warps=4,
        num_stages=4,
    )

    return out
