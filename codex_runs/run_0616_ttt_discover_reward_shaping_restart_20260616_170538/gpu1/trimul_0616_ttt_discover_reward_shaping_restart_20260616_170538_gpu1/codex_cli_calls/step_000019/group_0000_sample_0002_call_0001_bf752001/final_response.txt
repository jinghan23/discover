"""
TriMul outgoing forward pass.

The implementation keeps the projection-heavy setup in PyTorch/cuBLAS, then uses
Triton for the N^3 triangular multiplication by treating each hidden channel as
an independent GEMM: out[b, h, i, j] = sum_k left[b, i, k, h] * right[b, j, k, h].
The final normalization, output gate, and output projection are done with
PyTorch operators to keep the code robust across all requested dimensions.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    total_mats: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_n_blocks = tl.cdiv(N, BLOCK_N)
    tile_m = pid // num_n_blocks
    tile_n = pid - tile_m * num_n_blocks

    mat_id = tl.program_id(1)
    b = mat_id // H
    h = mat_id - b * H

    offs_i = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), tl.float32)

    # left layout:  [B, N, N, H], right layout: [B, N, N, H]
    # Compute left[b, i, k, h] @ right[b, j, k, h]^T
    base = ((b * N * N) * H) + h
    left_ptrs = left_ptr + base + (offs_i[:, None] * N + offs_k[None, :]) * H
    right_ptrs = right_ptr + base + (offs_j[None, :] * N + offs_k[:, None]) * H

    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        k_mask = offs_k < N
        l = tl.load(
            left_ptrs,
            mask=(offs_i[:, None] < N) & k_mask[None, :],
            other=0.0,
        )
        r = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & (offs_j[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(l, r, input_precision="tf32")
        left_ptrs += BLOCK_K * H
        right_ptrs += BLOCK_K * H

    # out layout: [B, N, N, H]
    out_ptrs = out_ptr + base + (offs_i[:, None] * N + offs_j[None, :]) * H
    tl.store(
        out_ptrs,
        acc,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
    )


def _layer_norm(x, weight, bias, eps=1e-5):
    return torch.nn.functional.layer_norm(x, (x.shape[-1],), weight, bias, eps)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = input_tensor
    B, N, _, _ = x.shape
    H = hidden_dim

    x_norm = _layer_norm(x, norm_weight, norm_bias)

    x_flat = x_norm.reshape(-1, dim)
    left = torch.nn.functional.linear(x_flat, left_proj_weight).reshape(B, N, N, H)
    right = torch.nn.functional.linear(x_flat, right_proj_weight).reshape(B, N, N, H)

    if mask is not None and not bool(config.get("nomask", False)):
        m = mask.to(left.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_weight).reshape(B, N, N, H)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_weight).reshape(B, N, N, H)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_weight).reshape(B, N, N, H)
    )

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    tri_out = torch.empty((B, N, N, H), device=x.device, dtype=torch.float32)

    bm = 16
    bn = 16
    bk = 64
    grid = (triton.cdiv(N, bm) * triton.cdiv(N, bn), B * H)
    _trimul_outgoing_kernel[grid](
        left,
        right,
        tri_out,
        B * H,
        N,
        H,
        BLOCK_M=bm,
        BLOCK_N=bn,
        BLOCK_K=bk,
        num_warps=4,
        num_stages=3,
    )

    tri_out = _layer_norm(tri_out, to_out_norm_weight, to_out_norm_bias)
    tri_out = tri_out * out_gate
    out = torch.nn.functional.linear(tri_out.reshape(-1, H), to_out_weight)
    return out.reshape(B, N, N, dim).to(torch.float32)
