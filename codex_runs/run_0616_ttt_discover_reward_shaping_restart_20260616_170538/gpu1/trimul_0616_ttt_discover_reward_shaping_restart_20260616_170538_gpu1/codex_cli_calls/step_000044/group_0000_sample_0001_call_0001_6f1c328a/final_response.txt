"""
TriMul outgoing forward pass.

Algorithm:
1. Normalize pair activations over the last dimension with PyTorch layer_norm.
2. Fuse the five input projections into one matmul, then split projection/gate branches.
3. Apply sigmoid gates and optional mask.
4. Compute the outgoing triangular multiplication as one batched grouped GEMM:
   for each batch and hidden channel d, compute L[b, :, :, d] @ R[b, :, :, d]^T.
5. Use a Triton kernel for the final hidden-dimension LayerNorm + out gate + output projection,
   producing float32 output.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _out_norm_gate_proj_kernel(
    y_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    proj_w_ptr,
    out_ptr,
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    out_dim: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_o = tl.program_id(1)

    offs_h = tl.arange(0, BLOCK_H)
    hmask = offs_h < hidden_dim

    y = tl.load(y_ptr + pid_row * hidden_dim + offs_h, mask=hmask, other=0.0).to(tl.float32)
    mean = tl.sum(y, axis=0) / hidden_dim
    yc = tl.where(hmask, y - mean, 0.0)
    var = tl.sum(yc * yc, axis=0) / hidden_dim
    inv = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w_ptr + offs_h, mask=hmask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_h, mask=hmask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + pid_row * hidden_dim + offs_h, mask=hmask, other=0.0).to(tl.float32)

    z = ((y - mean) * inv * nw + nb) * g
    z = tl.where(hmask, z, 0.0)

    offs_o = pid_o * BLOCK_O + tl.arange(0, BLOCK_O)
    omask = offs_o < out_dim

    acc = tl.zeros((BLOCK_O,), dtype=tl.float32)
    for h in range(0, BLOCK_H):
        zh = tl.load(y_ptr + pid_row * hidden_dim + h).to(tl.float32)
        # Recompute scalar normalized value because Triton cannot index dynamic vector values portably.
        zh = (zh - mean) * inv
        whn = tl.load(norm_w_ptr + h).to(tl.float32)
        bhn = tl.load(norm_b_ptr + h).to(tl.float32)
        gh = tl.load(gate_ptr + pid_row * hidden_dim + h).to(tl.float32)
        zv = (zh * whn + bhn) * gh
        pw = tl.load(proj_w_ptr + offs_o * hidden_dim + h, mask=omask, other=0.0).to(tl.float32)
        acc += zv * pw

    tl.store(out_ptr + pid_row * out_dim + offs_o, acc, mask=omask)


def _final_norm_gate_project(y, out_gate, norm_w, norm_b, to_out_w, out_dim):
    total_rows = y.numel() // y.shape[-1]
    hidden_dim = y.shape[-1]
    out = torch.empty((total_rows, out_dim), device=y.device, dtype=torch.float32)

    block_h = triton.next_power_of_2(hidden_dim)
    block_o = 64
    grid = (total_rows, triton.cdiv(out_dim, block_o))

    _out_norm_gate_proj_kernel[grid](
        y,
        out_gate,
        norm_w,
        norm_b,
        to_out_w,
        out,
        total_rows,
        hidden_dim,
        out_dim,
        BLOCK_H=block_h,
        BLOCK_O=block_o,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    x = input_tensor
    bs, n, _, _ = x.shape

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x_norm = F.layer_norm(x, (dim,), norm_weight, norm_bias, eps=1.0e-5)
    x_flat = x_norm.reshape(-1, dim)

    fused_w = torch.cat(
        (
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
        ),
        dim=0,
    )
    proj = F.linear(x_flat, fused_w).reshape(bs, n, n, 5, hidden_dim)

    left = proj[:, :, :, 0, :]
    right = proj[:, :, :, 1, :]
    left_gate = torch.sigmoid(proj[:, :, :, 2, :])
    right_gate = torch.sigmoid(proj[:, :, :, 3, :])
    out_gate = torch.sigmoid(proj[:, :, :, 4, :]).contiguous()

    left = left * left_gate
    right = right * right_gate

    if mask is not None:
        left = left * mask[..., None]
        right = right * mask[..., None]

    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    tri = torch.bmm(
        left.reshape(bs * hidden_dim, n, n),
        right.reshape(bs * hidden_dim, n, n).transpose(1, 2),
    )
    tri = tri.reshape(bs, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()

    out_flat = _final_norm_gate_project(
        tri.reshape(-1, hidden_dim),
        out_gate.reshape(-1, hidden_dim),
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        dim,
    )

    return out_flat.reshape(bs, n, n, dim)
