"""
Fused mixed-precision TriMul outgoing forward.

Algorithm:
1. Apply the input LayerNorm with PyTorch for numerical stability.
2. Compute all five input projections in one GEMM.
3. Use a Triton kernel to apply sigmoid gates, optional mask, and pack left/right
   tensors into channel-major layout for fast batched GEMM.
4. Compute the outgoing triangular multiplication as B*H independent GEMMs.
5. Use a Triton kernel for output LayerNorm + output gate, then one final GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BMM_OUT_DTYPE = None
_MM_OUT_DTYPE = None


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def _pack_trimul_inputs(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    total: tl.constexpr,
    n2: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_mask = rows < total
    h_mask = hs < H
    mask2 = row_mask[:, None] & h_mask[None, :]

    base = rows[:, None] * (5 * H) + hs[None, :]

    l = tl.load(proj_ptr + base, mask=mask2, other=0.0).to(tl.float32)
    r = tl.load(proj_ptr + base + H, mask=mask2, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=mask2, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=mask2, other=0.0).to(tl.float32)
    og = tl.load(proj_ptr + base + 4 * H, mask=mask2, other=0.0).to(tl.float32)

    m = tl.full((BLOCK_M,), 1.0, tl.float32)
    if HAS_MASK:
        m = tl.load(mask_ptr + rows, mask=row_mask, other=0.0).to(tl.float32)

    l = l * _sigmoid(lg) * m[:, None]
    r = r * _sigmoid(rg) * m[:, None]
    og = _sigmoid(og)

    b = rows // n2
    p = rows - b * n2
    packed_offsets = (b[:, None] * H + hs[None, :]) * n2 + p[:, None]

    tl.store(left_ptr + packed_offsets, l.to(tl.float16), mask=mask2)
    tl.store(right_ptr + packed_offsets, r.to(tl.float16), mask=mask2)
    tl.store(out_gate_ptr + rows[:, None] * H + hs[None, :], og.to(tl.float16), mask=mask2)


@triton.jit
def _out_norm_gate_kernel(
    tri_ptr,
    out_gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    z_ptr,
    total: tl.constexpr,
    n2: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    row_mask = rows < total
    h_mask = hs < H
    mask2 = row_mask[:, None] & h_mask[None, :]

    b = rows // n2
    p = rows - b * n2
    tri_offsets = (b[:, None] * H + hs[None, :]) * n2 + p[:, None]

    x = tl.load(tri_ptr + tri_offsets, mask=mask2, other=0.0).to(tl.float32)
    mean = tl.sum(x, axis=1) / H
    centered = x - mean[:, None]
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w_ptr + hs, mask=h_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + hs, mask=h_mask, other=0.0).to(tl.float32)
    gate = tl.load(out_gate_ptr + rows[:, None] * H + hs[None, :], mask=mask2, other=0.0).to(tl.float32)

    z = (centered * inv[:, None] * nw[None, :] + nb[None, :]) * gate
    tl.store(z_ptr + rows[:, None] * H + hs[None, :], z.to(tl.float16), mask=mask2)


def _cached_projection_weight(weights, dtype):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = ("proj", dtype, tuple((t.data_ptr(), tuple(t.shape), t.dtype, t._version) for t in tensors))
    w = _WEIGHT_CACHE.get(key)
    if w is None:
        w = torch.cat(tensors, dim=0).contiguous().to(dtype)
        _WEIGHT_CACHE[key] = w
    return w


def _cached_to_out_t(weight, dtype):
    key = ("to_out_t", dtype, weight.data_ptr(), tuple(weight.shape), weight.dtype, weight._version)
    w = _WEIGHT_CACHE.get(key)
    if w is None:
        w = weight.t().contiguous().to(dtype)
        _WEIGHT_CACHE[key] = w
    return w


def _bmm_fp32(a, b):
    global _BMM_OUT_DTYPE
    if _BMM_OUT_DTYPE is not False:
        try:
            y = torch.bmm(a, b, out_dtype=torch.float32)
            _BMM_OUT_DTYPE = True
            return y
        except TypeError:
            _BMM_OUT_DTYPE = False
    return torch.bmm(a, b)


def _mm_fp32(a, b):
    global _MM_OUT_DTYPE
    if _MM_OUT_DTYPE is not False:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE = True
            return y
        except TypeError:
            _MM_OUT_DTYPE = False
    return torch.mm(a, b).float()


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    torch.backends.cuda.matmul.allow_tf32 = True

    bs, n, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    total = bs * n * n
    n2 = n * n

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = input_tensor
    if x.dtype is not torch.float32:
        x = x.float()

    x = F.layer_norm(x, (dim,), norm_weight, norm_bias, 1.0e-5)
    x2d = x.reshape(total, dim).to(torch.float16)

    proj_w = _cached_projection_weight(weights, torch.float16)
    proj = F.linear(x2d, proj_w)

    left = torch.empty((bs * hidden_dim, n, n), device=input_tensor.device, dtype=torch.float16)
    right = torch.empty_like(left)
    out_gate = torch.empty((total, hidden_dim), device=input_tensor.device, dtype=torch.float16)

    has_mask = bool((mask is not None) and (not config.get("nomask", False)))
    mask_ptr = mask.contiguous() if has_mask else input_tensor

    grid_pack = (triton.cdiv(total, 16), triton.cdiv(hidden_dim, 64))
    _pack_trimul_inputs[grid_pack](
        proj,
        mask_ptr,
        left,
        right,
        out_gate,
        total,
        n2,
        hidden_dim,
        has_mask,
        BLOCK_M=16,
        BLOCK_H=64,
        num_warps=4,
    )

    tri = _bmm_fp32(left, right.transpose(1, 2))

    z = torch.empty((total, hidden_dim), device=input_tensor.device, dtype=torch.float16)
    grid_norm = (triton.cdiv(total, 16),)
    _out_norm_gate_kernel[grid_norm](
        tri,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        z,
        total,
        n2,
        hidden_dim,
        BLOCK_M=16,
        BLOCK_H=128,
        num_warps=4,
    )

    to_out_t = _cached_to_out_t(to_out_weight, torch.float16)
    out = _mm_fp32(z, to_out_t)
    return out.reshape(bs, n, n, dim).float()
