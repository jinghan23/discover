"""
TriMul outgoing forward: use PyTorch/cuBLAS for the large dense projections and
per-channel N x N products, while fusing the final hidden LayerNorm, output gate,
and to_out projection in one Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _final_norm_gate_out_kernel(
    y_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_w_ptr,
    out_ptr,
    dim_out: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    row = tl.program_id(0)
    offs_h = tl.arange(0, BLOCK_H)

    y = tl.load(y_ptr + row * BLOCK_H + offs_h).to(tl.float32)

    mean = tl.sum(y, axis=0) / BLOCK_H
    centered = y - mean
    var = tl.sum(centered * centered, axis=0) / BLOCK_H
    rstd = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w_ptr + offs_h).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_h).to(tl.float32)
    gate = tl.load(gate_ptr + row * BLOCK_H + offs_h).to(tl.float32)

    h = (centered * rstd * nw + nb) * gate

    offs_o = tl.arange(0, BLOCK_O)
    for o0 in range(0, dim_out, BLOCK_O):
        offs = o0 + offs_o
        w = tl.load(
            out_w_ptr + offs[:, None] * BLOCK_H + offs_h[None, :],
            mask=offs[:, None] < dim_out,
            other=0.0,
        ).to(tl.float32)
        acc = tl.sum(w * h[None, :], axis=1)
        tl.store(out_ptr + row * dim_out + offs, acc, mask=offs < dim_out)


_proj_cache = {}


def _packed_projection_weight(weights, hidden_dim):
    ws = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = tuple((w.data_ptr(), tuple(w.shape), w.dtype, w.device.index) for w in ws)
    cached = _proj_cache.get(key)
    if cached is None:
        cached = torch.cat(ws, dim=0).contiguous()
        if len(_proj_cache) > 8:
            _proj_cache.clear()
        _proj_cache[key] = cached
    return cached


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    x = input_tensor
    bsz, n, _, _ = x.shape
    rows = bsz * n * n

    x_flat = x.reshape(rows, dim)

    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1.0e-5,
    )

    proj_w = _packed_projection_weight(weights, hidden_dim)
    proj = F.linear(x_norm, proj_w)

    left, right, left_gate, right_gate, out_gate = proj.split(hidden_dim, dim=-1)

    left = left * torch.sigmoid(left_gate)
    right = right * torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate).contiguous()

    if mask is not None:
        m = mask.reshape(rows, 1).to(dtype=left.dtype)
        left = left * m
        right = right * m

    left = left.reshape(bsz, n, n, hidden_dim)
    right = right.reshape(bsz, n, n, hidden_dim)

    left_bh = left.permute(0, 3, 1, 2).reshape(bsz * hidden_dim, n, n).contiguous()
    right_bh = right.permute(0, 3, 1, 2).reshape(bsz * hidden_dim, n, n).contiguous()

    y = torch.bmm(left_bh, right_bh.transpose(1, 2))
    y = y.reshape(bsz, hidden_dim, n, n).permute(0, 2, 3, 1).contiguous()

    out = torch.empty((rows, dim), device=x.device, dtype=torch.float32)

    _final_norm_gate_out_kernel[(rows,)](
        y,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"],
        out,
        dim,
        BLOCK_H=hidden_dim,
        BLOCK_O=32,
        num_warps=4,
    )

    return out.reshape(bsz, n, n, dim)
