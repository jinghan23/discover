"""
Fused TriMul forward for AlphaFold-style outgoing triangle multiplication.

Algorithm:
1. Use Triton kernels to apply input LayerNorm and compute left/right projections
   with their sigmoid gates, materializing compact [B, N, N, H] left/right buffers.
2. Compute outgoing triangle multiplication as batched per-hidden-channel matmuls:
   out[b, :, :, h] = left[b, :, :, h] @ right[b, :, :, h].T.
   This uses torch.bmm over the hidden dimension because it maps cleanly to cuBLAS.
3. Apply output LayerNorm over H in Triton, fuse the out-gate and final projection
   back to the input channel dimension D.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def _make_lr_gate_kernel(
    x_ptr, mask_ptr,
    nw_ptr, nb_ptr,
    lw_ptr, rw_ptr, lgw_ptr, rgw_ptr, ogw_ptr,
    left_ptr, right_ptr, ogate_ptr,
    total_rows: tl.constexpr, D: tl.constexpr, H: tl.constexpr,
    N: tl.constexpr, HAS_MASK: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    row = tl.program_id(0)
    h0 = tl.program_id(1) * BLOCK_H
    offs_d = tl.arange(0, BLOCK_D)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    x = tl.load(x_ptr + row * D + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)

    inv_d = 1.0 / D
    mean = tl.sum(x, axis=0) * inv_d
    xc = tl.where(offs_d < D, x - mean, 0.0)
    var = tl.sum(xc * xc, axis=0) * inv_d
    xn = xc * tl.rsqrt(var + 1.0e-5)

    w_norm = tl.load(nw_ptr + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)
    b_norm = tl.load(nb_ptr + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)
    xn = xn * w_norm + b_norm

    lw = tl.load(lw_ptr + offs_h[:, None] * D + offs_d[None, :],
                 mask=(offs_h[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
    rw = tl.load(rw_ptr + offs_h[:, None] * D + offs_d[None, :],
                 mask=(offs_h[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
    lgw = tl.load(lgw_ptr + offs_h[:, None] * D + offs_d[None, :],
                  mask=(offs_h[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
    rgw = tl.load(rgw_ptr + offs_h[:, None] * D + offs_d[None, :],
                  mask=(offs_h[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)
    ogw = tl.load(ogw_ptr + offs_h[:, None] * D + offs_d[None, :],
                  mask=(offs_h[:, None] < H) & (offs_d[None, :] < D), other=0.0).to(tl.float32)

    l = tl.sum(lw * xn[None, :], axis=1)
    r = tl.sum(rw * xn[None, :], axis=1)
    lg = _sigmoid(tl.sum(lgw * xn[None, :], axis=1))
    rg = _sigmoid(tl.sum(rgw * xn[None, :], axis=1))
    og = _sigmoid(tl.sum(ogw * xn[None, :], axis=1))

    m = 1.0
    if HAS_MASK:
        m = tl.load(mask_ptr + row).to(tl.float32)

    vals_l = (l * lg * m).to(tl.float32)
    vals_r = (r * rg * m).to(tl.float32)

    tl.store(left_ptr + row * H + offs_h, vals_l, mask=offs_h < H)
    tl.store(right_ptr + row * H + offs_h, vals_r, mask=offs_h < H)
    tl.store(ogate_ptr + row * H + offs_h, og, mask=offs_h < H)


@triton.jit
def _out_norm_project_kernel(
    y_ptr, ogate_ptr,
    ow_ptr, ob_ptr, tow_ptr,
    out_ptr,
    total_rows: tl.constexpr, H: tl.constexpr, D: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    row = tl.program_id(0)
    d0 = tl.program_id(1) * BLOCK_D

    offs_h = tl.arange(0, BLOCK_H)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    y = tl.load(y_ptr + row * H + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)

    mean = tl.sum(y, axis=0) / H
    yc = tl.where(offs_h < H, y - mean, 0.0)
    var = tl.sum(yc * yc, axis=0) / H
    yn = yc * tl.rsqrt(var + 1.0e-5)

    nw = tl.load(ow_ptr + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    nb = tl.load(ob_ptr + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    og = tl.load(ogate_ptr + row * H + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    yn = (yn * nw + nb) * og

    tw = tl.load(tow_ptr + offs_d[:, None] * H + offs_h[None, :],
                 mask=(offs_d[:, None] < D) & (offs_h[None, :] < H), other=0.0).to(tl.float32)
    z = tl.sum(tw * yn[None, :], axis=1)
    tl.store(out_ptr + row * D + offs_d, z, mask=offs_d < D)


def _triton_lr(x, mask, weights, dim, hidden_dim, nomask):
    B, N, _, D = x.shape
    total = B * N * N

    left = torch.empty((B, N, N, hidden_dim), device=x.device, dtype=torch.float32)
    right = torch.empty_like(left)
    ogate = torch.empty_like(left)

    has_mask = (mask is not None) and (not bool(nomask))
    if mask is None:
        mask = x.new_empty((1,), dtype=torch.float32)

    block_d = triton.next_power_of_2(dim)
    grid = (total, triton.cdiv(hidden_dim, 16))

    _make_lr_gate_kernel[grid](
        x, mask,
        weights["norm.weight"], weights["norm.bias"],
        weights["left_proj.weight"], weights["right_proj.weight"],
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"],
        left, right, ogate,
        total, dim, hidden_dim, N, has_mask,
        BLOCK_D=block_d, BLOCK_H=16,
        num_warps=4,
    )
    return left, right, ogate


def _triangle_bmm(left, right):
    B, N, _, H = left.shape

    # Arrange as H batches of [N, N] matrices per input batch:
    # left_h[b,h,i,k], right_h[b,h,j,k].
    l = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    r = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    y = torch.bmm(l, r.transpose(1, 2))
    return y.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()


def _triton_out(y, ogate, weights, dim, hidden_dim):
    B, N, _, H = y.shape
    total = B * N * N
    out = torch.empty((B, N, N, dim), device=y.device, dtype=torch.float32)

    grid = (total, triton.cdiv(dim, 16))
    _out_norm_project_kernel[grid](
        y, ogate,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        weights["to_out.weight"],
        out,
        total, hidden_dim, dim,
        BLOCK_H=triton.next_power_of_2(hidden_dim), BLOCK_D=16,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    nomask = bool(config.get("nomask", False))

    x = input_tensor.contiguous()
    if mask is not None and not nomask:
        mask = mask.contiguous()

    left, right, ogate = _triton_lr(x, mask, weights, dim, hidden_dim, nomask)
    tri = _triangle_bmm(left, right)
    return _triton_out(tri, ogate, weights, dim, hidden_dim)
