"""
TriMul outgoing forward pass.

Algorithm: use PyTorch's highly tuned layer_norm/linear paths for the channel
projections and gates, then use a Triton grouped matmul kernel for the expensive
per-hidden-channel contraction:
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
The contraction is stored as [B, H, N, N] for contiguous per-channel tiles,
then reshaped for the output LayerNorm, out gate, and final projection.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _trimul_contract_kernel(
    left,
    right,
    out,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_bh = tl.program_id(1)

    pid_m = pid // tl.cdiv(N, BLOCK_N)
    pid_n = pid - pid_m * tl.cdiv(N, BLOCK_N)

    b = pid_bh // H
    h = pid_bh - b * H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    left_base = left + ((b * N + offs_m[:, None]) * N + offs_k[None, :]) * H + h
    right_base = right + ((b * N + offs_n[None, :]) * N + offs_k[:, None]) * H + h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k_idxs = k0 + offs_k
        l = tl.load(left_base, mask=(offs_m[:, None] < N) & (k_idxs[None, :] < N), other=0.0)
        r = tl.load(right_base, mask=(k_idxs[:, None] < N) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(l, r)
        left_base += BLOCK_K * H
        right_base += BLOCK_K * H

    out_ptrs = out + ((pid_bh * N + offs_m[:, None]) * N + offs_n[None, :])
    tl.store(out_ptrs, acc, mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))


def _sigmoid_linear(x, weight):
    return torch.sigmoid(F.linear(x, weight, None))


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    x = input_tensor
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    bs, seq_len, _, _ = x.shape

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = F.layer_norm(x, (dim,), norm_weight, norm_bias, 1e-5)

    left = F.linear(x, left_proj_weight, None)
    right = F.linear(x, right_proj_weight, None)

    if mask is not None and not bool(config.get("nomask", False)):
        m = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    left = left * _sigmoid_linear(x, left_gate_weight)
    right = right * _sigmoid_linear(x, right_gate_weight)
    out_gate = _sigmoid_linear(x, out_gate_weight)

    left = left.contiguous()
    right = right.contiguous()

    contract_bhnn = torch.empty(
        (bs, hidden_dim, seq_len, seq_len),
        device=x.device,
        dtype=torch.float32,
    )

    if seq_len <= 256:
        bm, bn, bk, warps = 16, 16, 64, 4
    else:
        bm, bn, bk, warps = 16, 16, 64, 4

    grid = (triton.cdiv(seq_len, bm) * triton.cdiv(seq_len, bn), bs * hidden_dim)
    _trimul_contract_kernel[grid](
        left,
        right,
        contract_bhnn,
        seq_len,
        hidden_dim,
        BLOCK_M=bm,
        BLOCK_N=bn,
        BLOCK_K=bk,
        num_warps=warps,
        num_stages=3,
    )

    out = contract_bhnn.permute(0, 2, 3, 1).contiguous()
    out = F.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, 1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight, None)
    return out.to(torch.float32)
