"""
TriMul outgoing forward.

Algorithm:
1. Use PyTorch's CUDA LayerNorm and one stacked GEMM for the five input projections.
2. Use a Triton packing kernel to apply sigmoid gates/mask and write left/right in [B,H,N,N]
   layout for strided batched GEMM.
3. Compute outgoing triangular multiplication as B*H independent matrix products:
   left[b,h] @ right[b,h].T.
4. Use a Triton kernel to fuse output LayerNorm, output gate, and final projection, returning fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_PROJ_W_CACHE = {}


@triton.jit
def _gate_pack_kernel(
    proj, mask, left, right, out_gate,
    TOTAL: tl.constexpr, N2: tl.constexpr,
    H: tl.constexpr, HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    valid = (rows[:, None] < TOTAL) & (hs[None, :] < H)
    proj_stride = 5 * H

    base = proj + rows[:, None] * proj_stride + hs[None, :]
    l = tl.load(base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(base + 4 * H, mask=valid, other=0.0).to(tl.float32)

    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    if HAS_MASK:
        m = tl.load(mask + rows, mask=rows < TOTAL, other=0.0).to(tl.float32)
        l = l * lg * m[:, None]
        r = r * rg * m[:, None]
    else:
        l = l * lg
        r = r * rg

    b = rows // N2
    pair = rows - b * N2
    bhnn_offsets = b[:, None] * (H * N2) + hs[None, :] * N2 + pair[:, None]

    tl.store(left + bhnn_offsets, l, mask=valid)
    tl.store(right + bhnn_offsets, r, mask=valid)
    tl.store(out_gate + rows[:, None] * H + hs[None, :], og, mask=valid)


@triton.jit
def _final_norm_gate_proj_kernel(
    tri, out_gate, norm_w, norm_b, to_out_w, out,
    TOTAL: tl.constexpr, N2: tl.constexpr,
    DIM: tl.constexpr, H: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
    hs = tl.arange(0, BLOCK_H)

    b = rows // N2
    pair = rows - b * N2

    h_mask = hs < H
    row_mask = rows < TOTAL

    tri_offsets = b[:, None] * (H * N2) + hs[None, :] * N2 + pair[:, None]
    v = tl.load(tri + tri_offsets, mask=row_mask[:, None] & h_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(v, axis=1) / H
    centered = tl.where(h_mask[None, :], v - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + hs, mask=h_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + hs, mask=h_mask, other=0.0).to(tl.float32)
    g = tl.load(
        out_gate + rows[:, None] * H + hs[None, :],
        mask=row_mask[:, None] & h_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    a = (centered * inv[:, None]) * nw[None, :] + nb[None, :]
    a = tl.where(h_mask[None, :], a * g, 0.0)

    w = tl.load(
        to_out_w + ds[None, :] * H + hs[:, None],
        mask=(ds[None, :] < DIM) & h_mask[:, None],
        other=0.0,
    ).to(tl.float32)

    acc = tl.dot(a, w, input_precision="tf32", out_dtype=tl.float32)

    tl.store(
        out + rows[:, None] * DIM + ds[None, :],
        acc,
        mask=row_mask[:, None] & (ds[None, :] < DIM),
    )


def _stacked_proj_weight(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = tuple((t.data_ptr(), tuple(t.shape), tuple(t.stride()), t.dtype, t.device) for t in tensors)
    cached = _PROJ_W_CACHE.get(key)
    if cached is None:
        if len(_PROJ_W_CACHE) > 8:
            _PROJ_W_CACHE.clear()
        cached = torch.cat(tensors, dim=0).contiguous()
        _PROJ_W_CACHE[key] = cached
    return cached


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    n2 = seq_len * seq_len
    total = bs * n2
    eps = config.get("eps", 1e-5)

    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    try:
        torch.set_float32_matmul_precision("high")
    except Exception:
        pass

    with torch.no_grad():
        x = input_tensor if input_tensor.dtype == torch.float32 else input_tensor.float()

        x = F.layer_norm(
            x,
            (dim,),
            weights["norm.weight"],
            weights["norm.bias"],
            eps,
        )
        proj = F.linear(x, _stacked_proj_weight(weights)).contiguous()

        left = torch.empty((bs, hidden_dim, seq_len, seq_len), device=input_tensor.device, dtype=torch.float32)
        right = torch.empty_like(left)
        out_gate = torch.empty((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=torch.float32)

        has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
        mask_arg = mask.contiguous() if has_mask else input_tensor

        bm_pack = 16
        bh_pack = 64
        _gate_pack_kernel[(triton.cdiv(total, bm_pack), triton.cdiv(hidden_dim, bh_pack))](
            proj, mask_arg, left, right, out_gate,
            TOTAL=total, N2=n2, H=hidden_dim, HAS_MASK=has_mask,
            BLOCK_M=bm_pack, BLOCK_H=bh_pack,
            num_warps=4,
        )

        tri = torch.bmm(
            left.reshape(bs * hidden_dim, seq_len, seq_len),
            right.reshape(bs * hidden_dim, seq_len, seq_len).transpose(1, 2),
        ).reshape(bs, hidden_dim, seq_len, seq_len)

        out = torch.empty((bs, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32)

        bm_final = 16
        bd_final = 64
        bh_final = triton.next_power_of_2(hidden_dim)
        _final_norm_gate_proj_kernel[(triton.cdiv(total, bm_final), triton.cdiv(dim, bd_final))](
            tri,
            out_gate,
            weights["to_out_norm.weight"].contiguous(),
            weights["to_out_norm.bias"].contiguous(),
            weights["to_out.weight"].contiguous(),
            out,
            TOTAL=total,
            N2=n2,
            DIM=dim,
            H=hidden_dim,
            EPS=eps,
            BLOCK_M=bm_final,
            BLOCK_D=bd_final,
            BLOCK_H=bh_final,
            num_warps=4,
        )

    return out
