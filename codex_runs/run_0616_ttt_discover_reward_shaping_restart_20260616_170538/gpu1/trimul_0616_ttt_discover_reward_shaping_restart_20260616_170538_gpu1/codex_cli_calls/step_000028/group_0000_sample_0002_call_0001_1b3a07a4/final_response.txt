"""
Hybrid outgoing TriMul forward pass: PyTorch performs the two dense linear stages and
LayerNorm over the input channel dimension, while Triton fuses projection gating,
masking, layout conversion for batched GEMM, and the post-einsum LayerNorm/out-gate.
The N^3 triangular product is executed as a cuBLAS batched matmul over hidden channels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    torch.set_float32_matmul_precision("high")
except Exception:
    pass


@triton.jit
def _prep_lr_gate_kernel(
    proj_ptr, mask_ptr, left_ptr, right_ptr, out_gate_ptr,
    N: tl.constexpr, H: tl.constexpr, TOTAL: tl.constexpr,
    HAS_MASK: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    m_ok = offs_m < TOTAL
    h_ok = offs_h < H
    active = m_ok[:, None] & h_ok[None, :]

    pair_stride = 5 * H
    base = offs_m[:, None] * pair_stride + offs_h[None, :]

    if HAS_MASK:
        mv = tl.load(mask_ptr + offs_m, mask=m_ok, other=0.0).to(tl.float32)
    else:
        mv = tl.full((BLOCK_M,), 1.0, tl.float32)

    n2 = N * N
    b = offs_m // n2
    ij = offs_m - b * n2
    bhij = (b[:, None] * H + offs_h[None, :]) * n2 + ij[:, None]

    l = tl.load(proj_ptr + base, mask=active, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=active, other=0.0).to(tl.float32)
    lv = l * (1.0 / (1.0 + tl.exp(-lg))) * mv[:, None]
    tl.store(left_ptr + bhij, lv, mask=active)

    r = tl.load(proj_ptr + base + H, mask=active, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=active, other=0.0).to(tl.float32)
    rv = r * (1.0 / (1.0 + tl.exp(-rg))) * mv[:, None]
    tl.store(right_ptr + bhij, rv, mask=active)

    og = tl.load(proj_ptr + base + 4 * H, mask=active, other=0.0).to(tl.float32)
    og = 1.0 / (1.0 + tl.exp(-og))
    tl.store(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :], og, mask=active)


@triton.jit
def _norm_gate_kernel(
    tri_ptr, out_gate_ptr, norm_w_ptr, norm_b_ptr, y_ptr,
    N: tl.constexpr, H: tl.constexpr, TOTAL: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)
    m_ok = offs_m < TOTAL
    h_ok = offs_h < H
    active = m_ok[:, None] & h_ok[None, :]

    n2 = N * N
    b = offs_m // n2
    ij = offs_m - b * n2

    tri_offsets = (b[:, None] * H + offs_h[None, :]) * n2 + ij[:, None]
    vals = tl.load(tri_ptr + tri_offsets, mask=active, other=0.0).to(tl.float32)

    inv_h = 1.0 / H
    mean = tl.sum(vals, axis=1) * inv_h
    diff = tl.where(h_ok[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) * inv_h
    normed = diff * tl.rsqrt(var[:, None] + EPS)

    w = tl.load(norm_w_ptr + offs_h, mask=h_ok, other=1.0).to(tl.float32)
    beta = tl.load(norm_b_ptr + offs_h, mask=h_ok, other=0.0).to(tl.float32)
    gate = tl.load(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :], mask=active, other=0.0).to(tl.float32)

    y = (normed * w[None, :] + beta[None, :]) * gate
    tl.store(y_ptr + offs_m[:, None] * H + offs_h[None, :], y, mask=active)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        x = input_tensor.float()
        bs, seq_len, _, dim = x.shape
        hidden_dim = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
        eps = float(config.get("eps", config.get("layer_norm_eps", config.get("layernorm_eps", 1e-5))))

        norm_weight = weights["norm.weight"].float().contiguous()
        norm_bias = weights["norm.bias"].float().contiguous()
        to_out_norm_weight = weights["to_out_norm.weight"].float().contiguous()
        to_out_norm_bias = weights["to_out_norm.bias"].float().contiguous()
        to_out_weight = weights["to_out.weight"].float().contiguous()

        proj_weight = torch.cat(
            (
                weights["left_proj.weight"].float(),
                weights["right_proj.weight"].float(),
                weights["left_gate.weight"].float(),
                weights["right_gate.weight"].float(),
                weights["out_gate.weight"].float(),
            ),
            dim=0,
        ).contiguous()

        x = F.layer_norm(x, (dim,), norm_weight, norm_bias, eps)
        proj = F.linear(x, proj_weight)

        total_pairs = bs * seq_len * seq_len
        left = torch.empty((bs * hidden_dim, seq_len, seq_len), device=x.device, dtype=torch.float32)
        right = torch.empty_like(left)
        out_gate = torch.empty((total_pairs, hidden_dim), device=x.device, dtype=torch.float32)

        has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
        mask_ptr = mask.contiguous() if has_mask else input_tensor

        prep_bm = 32
        prep_bh = 32
        grid_prep = (triton.cdiv(total_pairs, prep_bm), triton.cdiv(hidden_dim, prep_bh))
        _prep_lr_gate_kernel[grid_prep](
            proj, mask_ptr, left, right, out_gate,
            N=seq_len, H=hidden_dim, TOTAL=total_pairs,
            HAS_MASK=has_mask, BLOCK_M=prep_bm, BLOCK_H=prep_bh,
            num_warps=4,
        )

        del x, proj

        tri = torch.bmm(left, right.transpose(1, 2))
        del left, right

        y = torch.empty((bs, seq_len, seq_len, hidden_dim), device=tri.device, dtype=torch.float32)
        norm_bh = 1 << (hidden_dim - 1).bit_length()
        norm_bm = 4
        grid_norm = (triton.cdiv(total_pairs, norm_bm),)
        _norm_gate_kernel[grid_norm](
            tri, out_gate, to_out_norm_weight, to_out_norm_bias, y,
            N=seq_len, H=hidden_dim, TOTAL=total_pairs, EPS=eps,
            BLOCK_M=norm_bm, BLOCK_H=norm_bh,
            num_warps=4,
        )

        del tri, out_gate

        return F.linear(y, to_out_weight).float()
