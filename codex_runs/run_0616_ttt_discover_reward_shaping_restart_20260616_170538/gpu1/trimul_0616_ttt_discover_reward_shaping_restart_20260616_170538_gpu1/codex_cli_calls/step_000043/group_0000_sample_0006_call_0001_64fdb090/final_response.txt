"""Forward-only outgoing TriMul: normalize with PyTorch, group projection/gate GEMMs
into left/right pairs, run the per-channel triangle contraction with cuBLAS bmm,
then fuse output LayerNorm and output gate in Triton before the final projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    torch.set_float32_matmul_precision("high")
except Exception:
    pass


@triton.jit
def _out_norm_gate_kernel(
    tri_bhnn,
    gate_bijh,
    norm_w,
    norm_b,
    z_bijh,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    pid_bi = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bi // N
    i = pid_bi - b * N
    n2 = N * N

    hs = tl.arange(0, BLOCK_H)
    js = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    hmask = hs < H
    jmask = js < N
    mask = hmask[:, None] & jmask[None, :]

    tri_offsets = b * H * n2 + hs[:, None] * n2 + i * N + js[None, :]
    vals = tl.load(tri_bhnn + tri_offsets, mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=0) * (1.0 / H)
    centered = tl.where(hmask[:, None], vals - mean[None, :], 0.0)
    var = tl.sum(centered * centered, axis=0) * (1.0 / H)
    rstd = tl.rsqrt(var + 1.0e-5)

    scale = tl.load(norm_w + hs, mask=hmask, other=0.0).to(tl.float32)[:, None]
    bias = tl.load(norm_b + hs, mask=hmask, other=0.0).to(tl.float32)[:, None]

    row_base = b * n2 + i * N + js
    gate_offsets = row_base[None, :] * H + hs[:, None]
    gate = tl.load(gate_bijh + gate_offsets, mask=mask, other=0.0).to(tl.float32)

    out = (centered * rstd[None, :] * scale + bias) * gate
    tl.store(z_bijh + gate_offsets, out, mask=mask)


def _next_power_of_2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    bs, n, _, dim = input_tensor.shape
    hidden_dim = int(config["hidden_dim"])
    use_mask = mask is not None and not bool(config.get("nomask", False))

    with torch.no_grad():
        x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=1.0e-5)
        m = mask.to(dtype=x.dtype).unsqueeze(-1) if use_mask else None

        left_pair_w = torch.cat((left_proj_weight, left_gate_weight), dim=0)
        left_pair = F.linear(x, left_pair_w)
        left, gate = left_pair.split(hidden_dim, dim=-1)
        gate.sigmoid_()
        left.mul_(gate)
        if m is not None:
            left.mul_(m)
        lhs = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()
        del left_pair, left, gate, left_pair_w

        right_pair_w = torch.cat((right_proj_weight, right_gate_weight), dim=0)
        right_pair = F.linear(x, right_pair_w)
        right, gate = right_pair.split(hidden_dim, dim=-1)
        gate.sigmoid_()
        right.mul_(gate)
        if m is not None:
            right.mul_(m)
        rhs = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()
        del right_pair, right, gate, right_pair_w, m

        out_gate = F.linear(x, out_gate_weight)
        out_gate.sigmoid_()
        del x

        lhs = lhs.view(bs * hidden_dim, n, n)
        rhs = rhs.view(bs * hidden_dim, n, n)

        try:
            tri = torch.bmm(lhs, rhs.transpose(1, 2), out_dtype=torch.float32)
        except TypeError:
            tri = torch.bmm(lhs.float(), rhs.float().transpose(1, 2))

        tri = tri.view(bs, hidden_dim, n, n)
        del lhs, rhs

        z = torch.empty(
            (bs, n, n, hidden_dim),
            device=input_tensor.device,
            dtype=torch.float32,
        )

        block_h = _next_power_of_2(hidden_dim)
        block_j = 16
        grid = (bs * n, triton.cdiv(n, block_j))

        _out_norm_gate_kernel[grid](
            tri,
            out_gate,
            to_out_norm_weight,
            to_out_norm_bias,
            z,
            N=n,
            H=hidden_dim,
            BLOCK_H=block_h,
            BLOCK_J=block_j,
            num_warps=8,
        )

        del tri, out_gate
        return F.linear(z, to_out_weight).to(torch.float32)
