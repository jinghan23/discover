"""
TriMul outgoing forward pass.

Algorithm:
1. Fuse input LayerNorm, three hidden projections, sigmoid gates, optional mask, and
   out_gate materialization into one Triton kernel.
2. Compute outgoing multiplicative update as per-channel batched GEMMs:
      out[b, d] = left[b, :, :, d] @ right[b, :, :, d].T
   using torch.bmm over flattened (B * hidden_dim) channel batches.
3. Fuse hidden LayerNorm, out_gate multiply, and final projection to output dim in a
   Triton kernel. Accumulations are float32; intermediate gated factors are float16
   for fast tensor-core GEMMs; final output is float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _ln_proj_gate_kernel(
    x, mask,
    norm_w, norm_b,
    left_w, right_w, left_gate_w, right_gate_w, out_gate_w,
    left_out, right_out, out_gate_out,
    total_tokens: tl.constexpr,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_d = tl.arange(0, BLOCK_D)
    x_vals = tl.load(x + pid * dim + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)

    mean = tl.sum(x_vals, axis=0) / dim
    xc = tl.where(offs_d < dim, x_vals - mean, 0.0)
    var = tl.sum(xc * xc, axis=0) / dim
    inv = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + offs_d, mask=offs_d < dim, other=0.0).to(tl.float32)
    xn = (x_vals - mean) * inv * nw + nb
    xn = tl.where(offs_d < dim, xn, 0.0)

    if has_mask:
        m = tl.load(mask + pid).to(tl.float32)
    else:
        m = 1.0

    offs_h = tl.arange(0, BLOCK_H)
    acc_l = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_H,), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_H,), dtype=tl.float32)

    for d0 in range(0, BLOCK_D):
        xv = tl.load(x + pid * dim + d0).to(tl.float32)
        wd = tl.load(norm_w + d0).to(tl.float32)
        bd = tl.load(norm_b + d0).to(tl.float32)
        z = (xv - mean) * inv * wd + bd

        acc_l += z * tl.load(left_w + offs_h * dim + d0, mask=offs_h < hidden, other=0.0).to(tl.float32)
        acc_r += z * tl.load(right_w + offs_h * dim + d0, mask=offs_h < hidden, other=0.0).to(tl.float32)
        acc_lg += z * tl.load(left_gate_w + offs_h * dim + d0, mask=offs_h < hidden, other=0.0).to(tl.float32)
        acc_rg += z * tl.load(right_gate_w + offs_h * dim + d0, mask=offs_h < hidden, other=0.0).to(tl.float32)
        acc_og += z * tl.load(out_gate_w + offs_h * dim + d0, mask=offs_h < hidden, other=0.0).to(tl.float32)

    lg = 1.0 / (1.0 + tl.exp(-acc_lg))
    rg = 1.0 / (1.0 + tl.exp(-acc_rg))
    og = 1.0 / (1.0 + tl.exp(-acc_og))

    base = pid * hidden + offs_h
    tl.store(left_out + base, (acc_l * lg * m).to(tl.float16), mask=offs_h < hidden)
    tl.store(right_out + base, (acc_r * rg * m).to(tl.float16), mask=offs_h < hidden)
    tl.store(out_gate_out + base, og.to(tl.float16), mask=offs_h < hidden)


@triton.jit
def _out_norm_project_kernel(
    tri, out_gate,
    norm_w, norm_b, to_out_w,
    out,
    total_tokens: tl.constexpr,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_h = tl.arange(0, BLOCK_H)

    vals = tl.load(tri + pid * hidden + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=0) / hidden
    centered = tl.where(offs_h < hidden, vals - mean, 0.0)
    var = tl.sum(centered * centered, axis=0) / hidden
    inv = tl.rsqrt(var + 1.0e-5)

    nw = tl.load(norm_w + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    og = tl.load(out_gate + pid * hidden + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32)
    z = ((vals - mean) * inv * nw + nb) * og
    z = tl.where(offs_h < hidden, z, 0.0)

    offs_d = tl.arange(0, BLOCK_D)
    for d0 in range(0, BLOCK_D):
        if d0 < dim:
            acc = tl.sum(z * tl.load(to_out_w + d0 * hidden + offs_h, mask=offs_h < hidden, other=0.0).to(tl.float32), axis=0)
            tl.store(out + pid * dim + d0, acc)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    x = input_tensor.contiguous()
    bs, n, _, dim = x.shape
    hidden = config["hidden_dim"]

    total = bs * n * n
    device = x.device

    left = torch.empty((total, hidden), device=device, dtype=torch.float16)
    right = torch.empty((total, hidden), device=device, dtype=torch.float16)
    out_gate = torch.empty((total, hidden), device=device, dtype=torch.float16)

    has_mask = mask is not None
    if has_mask:
        mask_c = mask.contiguous()
    else:
        mask_c = x.new_empty((1,))

    block_d = triton.next_power_of_2(dim)
    block_h = triton.next_power_of_2(hidden)

    _ln_proj_gate_kernel[(total,)](
        x, mask_c,
        weights["norm.weight"], weights["norm.bias"],
        weights["left_proj.weight"], weights["right_proj.weight"],
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"],
        left, right, out_gate,
        total, dim, hidden, has_mask,
        BLOCK_D=block_d, BLOCK_H=block_h,
        num_warps=8,
    )

    left5 = left.view(bs, n, n, hidden).permute(0, 3, 1, 2).reshape(bs * hidden, n, n)
    right5 = right.view(bs, n, n, hidden).permute(0, 3, 1, 2).reshape(bs * hidden, n, n)

    tri = torch.bmm(left5, right5.transpose(1, 2))
    tri = tri.reshape(bs, hidden, n, n).permute(0, 2, 3, 1).contiguous()

    out = torch.empty((bs, n, n, dim), device=device, dtype=torch.float32)

    _out_norm_project_kernel[(total,)](
        tri, out_gate,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        weights["to_out.weight"],
        out,
        total, dim, hidden,
        BLOCK_H=block_h,
        BLOCK_D=triton.next_power_of_2(dim),
        num_warps=8,
    )

    return out
