"""
TriMul outgoing forward using fused projection packing plus batched GEMM.

The input is layer-normalized with PyTorch, then the five input projections are
computed as one tensor-core linear. A Triton kernel applies sigmoid gates and the
optional mask while packing left/right into channel-major batched-matmul layout.
The outgoing contraction is performed by torch.bmm over hidden channels. A second
Triton kernel fuses output LayerNorm with the output gate, followed by the final
linear projection. The returned tensor is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cudnn.allow_tf32 = True
    torch.set_float32_matmul_precision("high")
except Exception:
    pass


@triton.jit
def _pack_lr_gate_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_mask = offs_m < M
    h_mask = offs_h < H
    mask2 = row_mask[:, None] & h_mask[None, :]

    proj_stride = 5 * H
    base = offs_m[:, None] * proj_stride + offs_h[None, :]

    left = tl.load(proj_ptr + base, mask=mask2, other=0.0).to(tl.float32)
    right = tl.load(proj_ptr + base + H, mask=mask2, other=0.0).to(tl.float32)
    left_g = tl.load(proj_ptr + base + 2 * H, mask=mask2, other=0.0).to(tl.float32)
    right_g = tl.load(proj_ptr + base + 3 * H, mask=mask2, other=0.0).to(tl.float32)
    out_g = tl.load(proj_ptr + base + 4 * H, mask=mask2, other=0.0).to(tl.float32)

    left = left * tl.sigmoid(left_g)
    right = right * tl.sigmoid(right_g)
    out_g = tl.sigmoid(out_g)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_m, mask=row_mask, other=0.0).to(tl.float32)
        left = left * m[:, None]
        right = right * m[:, None]

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    row = rem // N
    col = rem - row * N

    lr_idx = b[:, None] * (H * nn) + offs_h[None, :] * nn + row[:, None] * N + col[:, None]
    gate_idx = offs_m[:, None] * H + offs_h[None, :]

    tl.store(left_ptr + lr_idx, left, mask=mask2)
    tl.store(right_ptr + lr_idx, right, mask=mask2)
    tl.store(out_gate_ptr + gate_idx, out_g, mask=mask2)


@triton.jit
def _out_norm_gate_kernel(
    tri_ptr,
    out_gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    y_ptr,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    row_mask = offs_m < M
    h_mask = offs_h < H
    mask2 = row_mask[:, None] & h_mask[None, :]

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn

    tri_idx = b[:, None] * (H * nn) + offs_h[None, :] * nn + rem[:, None]
    vals = tl.load(tri_ptr + tri_idx, mask=mask2, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    diff = tl.where(h_mask[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / H
    rstd = tl.rsqrt(var + 1.0e-5)

    w = tl.load(norm_w_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)
    bias = tl.load(norm_b_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)
    gate = tl.load(
        out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
        mask=mask2,
        other=0.0,
    ).to(tl.float32)

    y = ((vals - mean[:, None]) * rstd[:, None]) * w[None, :] + bias[None, :]
    y = y * gate

    tl.store(y_ptr + offs_m[:, None] * H + offs_h[None, :], y, mask=mask2)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim_from_x = input_tensor.shape
    dim = int(config.get("dim", dim_from_x))
    hidden_dim = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
    n = int(seq_len)
    m = bs * n * n

    fast_dtype = torch.float16
    device = input_tensor.device

    x = input_tensor
    if x.dtype is not torch.float32:
        x = x.float()

    x_norm = F.layer_norm(
        x,
        (dim,),
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1.0e-5,
    )

    proj_weight = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).to(fast_dtype)

    proj = F.linear(x_norm.reshape(m, dim).to(fast_dtype), proj_weight)
    del x_norm, proj_weight

    left = torch.empty((bs * hidden_dim, n, n), device=device, dtype=fast_dtype)
    right = torch.empty((bs * hidden_dim, n, n), device=device, dtype=fast_dtype)
    out_gate = torch.empty((m, hidden_dim), device=device, dtype=fast_dtype)

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    mask_arg = mask.reshape(-1) if has_mask else input_tensor

    grid_pack = (triton.cdiv(m, 16), triton.cdiv(hidden_dim, 64))
    _pack_lr_gate_kernel[grid_pack](
        proj,
        mask_arg,
        left,
        right,
        out_gate,
        m,
        n,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_M=16,
        BLOCK_H=64,
        num_warps=4,
    )
    del proj

    tri = torch.bmm(left, right.transpose(1, 2))
    del left, right

    y = torch.empty((m, hidden_dim), device=device, dtype=fast_dtype)
    block_h = 1 << (hidden_dim - 1).bit_length()

    grid_norm = (triton.cdiv(m, 16),)
    _out_norm_gate_kernel[grid_norm](
        tri,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        y,
        m,
        n,
        hidden_dim,
        BLOCK_M=16,
        BLOCK_H=block_h,
        num_warps=8,
    )
    del tri, out_gate

    out = F.linear(y, weights["to_out.weight"].to(fast_dtype))
    return out.to(torch.float32).reshape(bs, n, n, dim)
