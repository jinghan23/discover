"""
Fused mixed-precision TriMul outgoing forward.

The implementation normalizes the input with a Triton LayerNorm kernel into fp16,
uses one fused projection GEMM for left/right/gates, prepares channel-major
left/right tensors with a Triton sigmoid/mask kernel, evaluates the outgoing
per-channel batched matmul with tensor cores, then applies fused Triton
LayerNorm+out-gate before the final projection. The returned tensor is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_PROJ_CACHE = {}
_HALF_CACHE = {}


def _pow2(x: int) -> int:
    return 1 << (int(x) - 1).bit_length()


def _cached_half_cat(tensors):
    key = tuple((t.data_ptr(), tuple(t.shape), str(t.dtype), str(t.device)) for t in tensors)
    out = _PROJ_CACHE.get(key)
    if out is None:
        if len(_PROJ_CACHE) > 8:
            _PROJ_CACHE.clear()
        out = torch.cat([t.detach().to(torch.float16) for t in tensors], dim=0).contiguous()
        _PROJ_CACHE[key] = out
    return out


def _cached_half(t):
    key = (t.data_ptr(), tuple(t.shape), str(t.dtype), str(t.device))
    out = _HALF_CACHE.get(key)
    if out is None:
        if len(_HALF_CACHE) > 16:
            _HALF_CACHE.clear()
        out = t.detach().to(torch.float16).contiguous()
        _HALF_CACHE[key] = out
    return out


@triton.jit
def _input_ln_kernel(x, w, b, y, P: tl.constexpr, D: tl.constexpr,
                     EPS: tl.constexpr, BLOCK_P: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_d = tl.arange(0, BLOCK_D)

    mask = (offs_p[:, None] < P) & (offs_d[None, :] < D)
    vals = tl.load(x + offs_p[:, None] * D + offs_d[None, :], mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=1) / D
    diff = tl.where(offs_d[None, :] < D, vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / D
    rstd = tl.rsqrt(var + EPS)

    ww = tl.load(w + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)
    bb = tl.load(b + offs_d, mask=offs_d < D, other=0.0).to(tl.float32)
    out = diff * rstd[:, None] * ww[None, :] + bb[None, :]

    tl.store(y + offs_p[:, None] * D + offs_d[None, :], out, mask=mask)


@triton.jit
def _prep_proj_kernel(proj, mask_ptr, left, right, out_gate,
                      TOTAL: tl.constexpr, P_PER_B: tl.constexpr, H: tl.constexpr,
                      HAS_MASK: tl.constexpr, BLOCK: tl.constexpr):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < TOTAL

    p = offs // H
    h = offs - p * H
    base = p * (5 * H) + h

    l = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    r = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.sigmoid(tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32))
    rg = tl.sigmoid(tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32))
    og = tl.sigmoid(tl.load(proj + base + 4 * H, mask=valid, other=0.0).to(tl.float32))

    m = 1.0
    if HAS_MASK:
        m = tl.load(mask_ptr + p, mask=valid, other=0.0).to(tl.float32)

    batch = p // P_PER_B
    pos = p - batch * P_PER_B
    cm_off = batch * (H * P_PER_B) + h * P_PER_B + pos

    tl.store(left + cm_off, l * lg * m, mask=valid)
    tl.store(right + cm_off, r * rg * m, mask=valid)
    tl.store(out_gate + offs, og, mask=valid)


@triton.jit
def _out_ln_gate_kernel(out_t, gate, w, b, y,
                        P: tl.constexpr, P_PER_B: tl.constexpr, H: tl.constexpr,
                        EPS: tl.constexpr, BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    offs_p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    offs_h = tl.arange(0, BLOCK_H)

    batch = offs_p // P_PER_B
    pos = offs_p - batch * P_PER_B
    mask = (offs_p[:, None] < P) & (offs_h[None, :] < H)

    src = batch[:, None] * (H * P_PER_B) + offs_h[None, :] * P_PER_B + pos[:, None]
    vals = tl.load(out_t + src, mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    diff = tl.where(offs_h[None, :] < H, vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / H
    rstd = tl.rsqrt(var + EPS)

    ww = tl.load(w + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    bb = tl.load(b + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    gg = tl.load(gate + offs_p[:, None] * H + offs_h[None, :], mask=mask, other=0.0).to(tl.float32)

    out = (diff * rstd[:, None] * ww[None, :] + bb[None, :]) * gg
    tl.store(y + offs_p[:, None] * H + offs_h[None, :], out, mask=mask)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        if not input_tensor.is_contiguous():
            input_tensor = input_tensor.contiguous()

        B, N, _, D = input_tensor.shape
        H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
        P_PER_B = N * N
        P = B * P_PER_B
        device = input_tensor.device

        xh = torch.empty((B, N, N, D), device=device, dtype=torch.float16)
        block_d = _pow2(D)
        if D <= 128:
            block_p, warps = 8, 4
        elif D <= 384:
            block_p, warps = 4, 8
        else:
            block_p, warps = 2, 8

        _input_ln_kernel[(triton.cdiv(P, block_p),)](
            input_tensor, weights["norm.weight"], weights["norm.bias"], xh,
            P, D, 1.0e-5, block_p, block_d,
            num_warps=warps,
        )

        proj_w = _cached_half_cat((
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ))
        proj = F.linear(xh, proj_w)
        del xh

        left = torch.empty((B, H, N, N), device=device, dtype=torch.float16)
        right = torch.empty_like(left)
        out_gate = torch.empty((B, N, N, H), device=device, dtype=torch.float16)

        has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
        if has_mask and not mask.is_contiguous():
            mask = mask.contiguous()
        mask_arg = mask if has_mask else input_tensor

        total = P * H
        _prep_proj_kernel[(triton.cdiv(total, 256),)](
            proj, mask_arg, left, right, out_gate,
            total, P_PER_B, H, has_mask, 256,
            num_warps=4,
        )
        del proj

        left_b = left.reshape(B * H, N, N)
        right_b = right.reshape(B * H, N, N)
        tri = torch.bmm(left_b, right_b.transpose(1, 2))
        del left, right, left_b, right_b

        gated = torch.empty((B, N, N, H), device=device, dtype=torch.float16)
        block_h = _pow2(H)
        _out_ln_gate_kernel[(triton.cdiv(P, 8),)](
            tri, out_gate, weights["to_out_norm.weight"], weights["to_out_norm.bias"], gated,
            P, P_PER_B, H, 1.0e-5, 8, block_h,
            num_warps=4,
        )
        del tri, out_gate

        to_out_w = _cached_half(weights["to_out.weight"])
        return F.linear(gated, to_out_w).to(torch.float32)
