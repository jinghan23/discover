"""
Outgoing Triangle Multiplicative Update forward pass.

The implementation keeps the high-throughput parts on cuBLAS-backed GEMMs:
input LayerNorm -> packed projection GEMM -> per-channel batched triangle GEMM
-> fused Triton output LayerNorm + output gate -> final projection GEMM.
A Triton pointwise kernel fuses projection gating and optional masking before
the triangle contraction, and another Triton kernel fuses hidden-dim LayerNorm
with the output gate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}


@triton.jit
def _prep_proj_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    NUMEL: tl.constexpr,
    H: tl.constexpr,
    USE_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < NUMEL

    row = offs // H
    h = offs - row * H
    base = row * (5 * H) + h

    lp = tl.load(proj_ptr + base, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(proj_ptr + base + H, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(proj_ptr + base + 4 * H, mask=valid, other=0.0).to(tl.float32)

    m = 1.0
    if USE_MASK:
        m = tl.load(mask_ptr + row, mask=valid, other=0.0).to(tl.float32)

    tl.store(left_ptr + offs, lp * tl.sigmoid(lg) * m, mask=valid)
    tl.store(right_ptr + offs, rp * tl.sigmoid(rg) * m, mask=valid)
    tl.store(out_gate_ptr + offs, tl.sigmoid(og), mask=valid)


@triton.jit
def _out_norm_gate_kernel(
    tri_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    z_ptr,
    TOTAL: tl.constexpr,
    H: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_b = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, H)
    mask_m = offs_m < TOTAL

    tri = tl.load(
        tri_ptr + pid_b * H * TOTAL + offs_h[None, :] * TOTAL + offs_m[:, None],
        mask=mask_m[:, None],
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(tri, axis=1) * (1.0 / H)
    centered = tri - mean[:, None]
    var = tl.sum(centered * centered, axis=1) * (1.0 / H)
    rstd = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w_ptr + offs_h).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_h).to(tl.float32)
    gate = tl.load(
        gate_ptr + (pid_b * TOTAL + offs_m[:, None]) * H + offs_h[None, :],
        mask=mask_m[:, None],
        other=0.0,
    ).to(tl.float32)

    z = (centered * rstd[:, None] * nw[None, :] + nb[None, :]) * gate

    tl.store(
        z_ptr + (pid_b * TOTAL + offs_m[:, None]) * H + offs_h[None, :],
        z,
        mask=mask_m[:, None],
    )


def _tensor_key(t):
    return (
        int(t.data_ptr()),
        tuple(t.shape),
        tuple(t.stride()),
        str(t.dtype),
        str(t.device),
        int(getattr(t, "_version", 0)),
    )


def _cached_proj_weight_t(weights, dtype):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = ("proj", str(dtype), tuple(_tensor_key(t) for t in tensors))
    cached = _WEIGHT_CACHE.get(key)
    if cached is None:
        w = torch.cat(tensors, dim=0).to(dtype=dtype).t().contiguous()
        _WEIGHT_CACHE[key] = (w, tensors)
        return w
    return cached[0]


def _cached_to_out_weight_t(weights, dtype):
    t = weights["to_out.weight"]
    key = ("to_out", str(dtype), _tensor_key(t))
    cached = _WEIGHT_CACHE.get(key)
    if cached is None:
        w = t.to(dtype=dtype).t().contiguous()
        _WEIGHT_CACHE[key] = (w, (t,))
        return w
    return cached[0]


def _bmm_accum_fp32(a, b):
    try:
        return torch.bmm(a, b, out_dtype=torch.float32)
    except TypeError:
        return torch.bmm(a, b).to(torch.float32)


def _mm_accum_fp32(a, b):
    try:
        return torch.mm(a, b, out_dtype=torch.float32)
    except TypeError:
        return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cudnn.allow_tf32 = True

        x = input_tensor
        B, N, _, D = x.shape
        H = int(config["hidden_dim"])
        total = N * N
        rows = B * total
        eps = float(config.get("eps", config.get("layer_norm_eps", 1.0e-5)))
        work_dtype = torch.float16

        x_norm = F.layer_norm(
            x,
            (D,),
            weights["norm.weight"],
            weights["norm.bias"],
            eps=eps,
        )

        xh = x_norm.reshape(rows, D).to(dtype=work_dtype)
        proj_w_t = _cached_proj_weight_t(weights, work_dtype)
        proj = torch.mm(xh, proj_w_t).contiguous()

        del x_norm, xh

        left = torch.empty((B, N, N, H), device=x.device, dtype=work_dtype)
        right = torch.empty_like(left)
        out_gate = torch.empty_like(left)

        use_mask = mask is not None and not bool(config.get("nomask", False))
        mask_arg = mask.contiguous() if use_mask else x
        numel = rows * H
        block = 1024

        _prep_proj_kernel[(triton.cdiv(numel, block),)](
            proj,
            mask_arg,
            left,
            right,
            out_gate,
            NUMEL=numel,
            H=H,
            USE_MASK=use_mask,
            BLOCK=block,
            num_warps=4,
        )

        del proj

        left_bh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

        del left, right

        tri = _bmm_accum_fp32(left_bh, right_bh.transpose(1, 2))

        del left_bh, right_bh

        z = torch.empty((B, N, N, H), device=x.device, dtype=work_dtype)
        block_m = 16

        _out_norm_gate_kernel[(triton.cdiv(total, block_m), B)](
            tri,
            out_gate,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            z,
            total,
            H,
            eps,
            BLOCK_M=block_m,
            num_warps=4,
        )

        del tri, out_gate

        to_out_w_t = _cached_to_out_weight_t(weights, work_dtype)
        out = _mm_accum_fp32(z.view(rows, H), to_out_w_t)
        return out.view(B, N, N, D).to(torch.float32)
