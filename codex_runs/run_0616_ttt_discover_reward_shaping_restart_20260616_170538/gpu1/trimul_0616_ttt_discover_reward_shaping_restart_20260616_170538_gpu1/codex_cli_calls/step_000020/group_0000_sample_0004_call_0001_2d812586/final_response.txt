"""
Outgoing Triangle Multiplicative Update.

Uses PyTorch's optimized LayerNorm/GEMM for the dense channel projections, then
Triton kernels to pack gated left/right/out-gate tensors into batched channel-major
matrices, run the N^3 contraction with torch.bmm, and fuse output LayerNorm with
the output gate before the final projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_PROJ_WEIGHT_CACHE = {}


@triton.jit
def _pack_trimul_proj_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_t_ptr,
    gate_ptr,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = hs < H
    mask2 = row_mask[:, None] & h_mask[None, :]

    nn = N * N
    b = rows // nn
    rem = rows - b * nn
    i = rem // N
    j = rem - i * N

    base = rows[:, None] * (5 * H) + hs[None, :]

    l = tl.load(proj_ptr + base, mask=mask2, other=0.0).to(tl.float32)
    r = tl.load(proj_ptr + base + H, mask=mask2, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=mask2, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=mask2, other=0.0).to(tl.float32)
    og = tl.load(proj_ptr + base + 4 * H, mask=mask2, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)
    og = tl.sigmoid(og)

    if HAS_MASK:
        mv = tl.load(mask_ptr + rows, mask=row_mask, other=0.0).to(tl.float32)
        l = l * mv[:, None]
        r = r * mv[:, None]

    mat_base = (b[:, None] * H + hs[None, :]) * nn
    left_offsets = mat_base + i[:, None] * N + j[:, None]
    right_offsets = mat_base + j[:, None] * N + i[:, None]

    tl.store(left_ptr + left_offsets, l, mask=mask2)
    tl.store(right_t_ptr + right_offsets, r, mask=mask2)
    tl.store(gate_ptr + left_offsets, og, mask=mask2)


@triton.jit
def _out_norm_gate_kernel(
    tri_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = hs < H
    mask2 = row_mask[:, None] & h_mask[None, :]

    nn = N * N
    b = rows // nn
    pair = rows - b * nn

    src_offsets = (b[:, None] * H + hs[None, :]) * nn + pair[:, None]

    vals = tl.load(tri_ptr + src_offsets, mask=mask2, other=0.0).to(tl.float32)
    gates = tl.load(gate_ptr + src_offsets, mask=mask2, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    centered = tl.where(h_mask[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    rstd = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w_ptr + hs, mask=h_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + hs, mask=h_mask, other=0.0).to(tl.float32)

    out = ((vals - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]) * gates
    tl.store(out_ptr + rows[:, None] * H + hs[None, :], out, mask=mask2)


def _projection_weight(weights):
    tensors = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = tuple((t.data_ptr(), t._version) for t in tensors)
    cached = _PROJ_WEIGHT_CACHE.get(key)
    if cached is None:
        cached = torch.cat(tensors, dim=0).contiguous()
        if len(_PROJ_WEIGHT_CACHE) > 8:
            _PROJ_WEIGHT_CACHE.clear()
        _PROJ_WEIGHT_CACHE[key] = cached
    return cached


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    H = int(config["hidden_dim"])
    eps = float(config.get("eps", 1e-5))

    x = input_tensor.float()
    x = F.layer_norm(
        x,
        (dim,),
        weights["norm.weight"].float(),
        weights["norm.bias"].float(),
        eps,
    )

    proj = F.linear(x, _projection_weight(weights))
    del x

    M = B * N * N
    compute_dtype = torch.float16

    left = torch.empty((B * H, N, N), device=input_tensor.device, dtype=compute_dtype)
    right_t = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    has_mask = mask is not None and not bool(config.get("nomask", False))
    mask_arg = mask.contiguous() if has_mask else input_tensor

    block_m = 16
    block_h = 32
    grid = (triton.cdiv(M, block_m), triton.cdiv(H, block_h))

    _pack_trimul_proj_kernel[grid](
        proj,
        mask_arg,
        left,
        right_t,
        out_gate,
        M,
        N,
        H,
        has_mask,
        BLOCK_M=block_m,
        BLOCK_H=block_h,
        num_warps=4,
    )
    del proj

    try:
        tri = torch.bmm(left, right_t, out_dtype=torch.float32)
    except TypeError:
        tri = torch.bmm(left, right_t).float()

    del left, right_t

    normed = torch.empty((M, H), device=input_tensor.device, dtype=torch.float32)

    _out_norm_gate_kernel[(triton.cdiv(M, block_m),)](
        tri,
        out_gate,
        weights["to_out_norm.weight"].float(),
        weights["to_out_norm.bias"].float(),
        normed,
        M,
        N,
        H,
        eps,
        BLOCK_M=block_m,
        BLOCK_H=triton.next_power_of_2(H),
        num_warps=4,
    )

    del tri, out_gate

    out = F.linear(normed, weights["to_out.weight"].float())
    return out.reshape(B, N, N, dim).float()
