"""
Outgoing Triangle Multiplicative Update.

The implementation uses PyTorch LayerNorm/GEMM for the dense channel projections,
a Triton packing kernel to apply input gates and masks while arranging tensors as
[B * H, N, N], cuBLAS batched matmul for the N^3 triangle contraction, and a Triton
kernel for output LayerNorm plus output gating before the final projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_PROJ_NAMES = (
    "left_proj.weight",
    "right_proj.weight",
    "left_gate.weight",
    "right_gate.weight",
    "out_gate.weight",
)
_PROJ_WEIGHT_CACHE = {}


def _packed_projection_weight(weights):
    tensors = tuple(weights[name] for name in _PROJ_NAMES)
    key = tuple(
        (
            int(t.data_ptr()),
            tuple(t.shape),
            tuple(t.stride()),
            str(t.dtype),
            str(t.device),
            int(getattr(t, "_version", 0)),
        )
        for t in tensors
    )
    cached = _PROJ_WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    packed = torch.cat(tuple(t.float() for t in tensors), dim=0).contiguous()
    if len(_PROJ_WEIGHT_CACHE) >= 8:
        _PROJ_WEIGHT_CACHE.clear()
    _PROJ_WEIGHT_CACHE[key] = packed
    return packed


@triton.jit
def _pack_trimul_proj_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_t_ptr,
    gate_ptr,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = hs < H
    mask = row_mask[:, None] & h_mask[None, :]

    nn = N * N
    b = rows // nn
    rem = rows - b * nn
    i = rem // N
    j = rem - i * N

    base = rows[:, None] * (5 * H) + hs[None, :]

    l = tl.load(proj_ptr + base, mask=mask, other=0.0).to(tl.float32)
    r = tl.load(proj_ptr + base + H, mask=mask, other=0.0).to(tl.float32)
    lg = tl.load(proj_ptr + base + 2 * H, mask=mask, other=0.0).to(tl.float32)
    rg = tl.load(proj_ptr + base + 3 * H, mask=mask, other=0.0).to(tl.float32)
    og = tl.load(proj_ptr + base + 4 * H, mask=mask, other=0.0).to(tl.float32)

    l = l * tl.sigmoid(lg)
    r = r * tl.sigmoid(rg)
    og = tl.sigmoid(og)

    if HAS_MASK:
        mv = tl.load(mask_ptr + rows, mask=row_mask, other=0.0).to(tl.float32)
        l = l * mv[:, None]
        r = r * mv[:, None]

    mat_base = (b[:, None] * H + hs[None, :]) * nn
    left_offsets = mat_base + i[:, None] * N + j[:, None]
    right_offsets = mat_base + j[:, None] * N + i[:, None]

    tl.store(left_ptr + left_offsets, l, mask=mask)
    tl.store(right_t_ptr + right_offsets, r, mask=mask)
    tl.store(gate_ptr + left_offsets, og, mask=mask)


@triton.jit
def _out_norm_gate_kernel(
    tri_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = hs < H
    mask = row_mask[:, None] & h_mask[None, :]

    nn = N * N
    b = rows // nn
    pair = rows - b * nn

    src_offsets = (b[:, None] * H + hs[None, :]) * nn + pair[:, None]

    vals = tl.load(tri_ptr + src_offsets, mask=mask, other=0.0).to(tl.float32)
    gates = tl.load(gate_ptr + src_offsets, mask=mask, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    centered = tl.where(h_mask[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    rstd = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w_ptr + hs, mask=h_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + hs, mask=h_mask, other=0.0).to(tl.float32)

    out = ((vals - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]) * gates
    tl.store(out_ptr + rows[:, None] * H + hs[None, :], out, mask=mask)


def custom_kernel(data):
    with torch.no_grad():
        input_tensor, mask, weights, config = data

        B, N, _, dim = input_tensor.shape
        hidden_dim = int(config["hidden_dim"])
        eps = float(config.get("eps", 1e-5))

        norm_weight = weights["norm.weight"]
        norm_bias = weights["norm.bias"]
        to_out_norm_weight = weights["to_out_norm.weight"]
        to_out_norm_bias = weights["to_out_norm.bias"]
        to_out_weight = weights["to_out.weight"]

        x = input_tensor.float()
        x = F.layer_norm(x, (dim,), norm_weight.float(), norm_bias.float(), eps)

        proj_weight = _packed_projection_weight(weights)
        proj = F.linear(x, proj_weight)
        del x

        M = B * N * N
        compute_dtype = torch.float16

        left = torch.empty((B * hidden_dim, N, N), device=input_tensor.device, dtype=compute_dtype)
        right_t = torch.empty_like(left)
        out_gate = torch.empty_like(left)

        has_mask = mask is not None and not bool(config.get("nomask", False))
        mask_arg = mask.contiguous() if has_mask else input_tensor

        block_m = 16
        block_h = 32
        grid = (triton.cdiv(M, block_m), triton.cdiv(hidden_dim, block_h))

        _pack_trimul_proj_kernel[grid](
            proj,
            mask_arg,
            left,
            right_t,
            out_gate,
            M,
            N,
            hidden_dim,
            has_mask,
            BLOCK_M=block_m,
            BLOCK_H=block_h,
            num_warps=4,
        )
        del proj

        try:
            tri = torch.bmm(left, right_t, out_dtype=torch.float32)
        except TypeError:
            tri = torch.bmm(left, right_t).float()

        del left, right_t

        normed = torch.empty((M, hidden_dim), device=input_tensor.device, dtype=torch.float32)

        _out_norm_gate_kernel[(triton.cdiv(M, block_m),)](
            tri,
            out_gate,
            to_out_norm_weight.float(),
            to_out_norm_bias.float(),
            normed,
            M,
            N,
            hidden_dim,
            eps,
            BLOCK_M=block_m,
            BLOCK_H=triton.next_power_of_2(hidden_dim),
            num_warps=4,
        )

        del tri, out_gate

        out = F.linear(normed, to_out_weight.float())
        return out.reshape(B, N, N, dim).float()
