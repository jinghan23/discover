"""
TriMul outgoing forward with Triton fused row kernels and cuBLAS tensor-core GEMMs.

Fast path for dim=hidden_dim=128:
1. Fuse input LayerNorm, five projections, sigmoid gates, optional mask, and split/layout writes
   into one Triton kernel, avoiding the large packed projection tensor.
2. Compute outgoing triangle multiplication with strided batched GEMM.
3. Fuse hidden LayerNorm, output gate, and final projection into one Triton kernel that returns fp32.

Fallback path supports larger dims with fused Triton LayerNorm/split/final-gate kernels plus cuBLAS GEMMs.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_MM_OUT_DTYPE_OK = None


def _np2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def _cdiv(a: int, b: int) -> int:
    return triton.cdiv(a, b)


@triton.jit
def _ln_kernel(
    x, w, b, y,
    M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, BLOCK_D)

    cm = cols < D
    rm = rows < M
    mask = rm[:, None] & cm[None, :]

    vals = tl.load(x + rows[:, None] * D + cols[None, :], mask=mask, other=0.0).to(tl.float32)
    vals_z = tl.where(cm[None, :], vals, 0.0)

    mean = tl.sum(vals_z, axis=1) / D
    diff = tl.where(cm[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + cols, mask=cm, other=1.0).to(tl.float32)
    bb = tl.load(b + cols, mask=cm, other=0.0).to(tl.float32)
    out = (vals - mean[:, None]) * inv[:, None] * ww[None, :] + bb[None, :]

    tl.store(y + rows[:, None] * D + cols[None, :], out, mask=mask)


@triton.jit
def _ln_proj_split_d128_kernel(
    x, mask_ptr, norm_w, norm_b, wcat,
    left, right, out_gate,
    M: tl.constexpr, N2: tl.constexpr, EPS: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    rows = tl.program_id(0) * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)
    hs = tl.arange(0, 128)
    rm = rows < M

    xv = tl.load(x + rows[:, None] * 128 + cols[None, :], mask=rm[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(xv, axis=1) * (1.0 / 128.0)
    diff = xv - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * (1.0 / 128.0)
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + cols).to(tl.float32)
    nb = tl.load(norm_b + cols).to(tl.float32)
    xn = ((diff * inv[:, None]) * nw[None, :] + nb[None, :]).to(tl.float16)

    if HAS_MASK:
        mv = tl.load(mask_ptr + rows, mask=rm, other=0.0).to(tl.float32)
    else:
        mv = tl.full((BLOCK_M,), 1.0, tl.float32)

    bidx = rows // N2
    ij = rows - bidx * N2
    out_off = bidx[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]

    wc0 = tl.load(wcat + cols[:, None] * 640 + hs[None, :])
    wc1 = tl.load(wcat + cols[:, None] * 640 + (128 + hs[None, :]))
    lp = tl.dot(xn, wc0, out_dtype=tl.float32)
    lg = tl.dot(xn, wc1, out_dtype=tl.float32)
    lv = lp * (1.0 / (1.0 + tl.exp(-lg))) * mv[:, None]
    tl.store(left + out_off, lv, mask=rm[:, None])

    wc2 = tl.load(wcat + cols[:, None] * 640 + (256 + hs[None, :]))
    wc3 = tl.load(wcat + cols[:, None] * 640 + (384 + hs[None, :]))
    rp = tl.dot(xn, wc2, out_dtype=tl.float32)
    rg = tl.dot(xn, wc3, out_dtype=tl.float32)
    rv = rp * (1.0 / (1.0 + tl.exp(-rg))) * mv[:, None]
    tl.store(right + out_off, rv, mask=rm[:, None])

    wc4 = tl.load(wcat + cols[:, None] * 640 + (512 + hs[None, :]))
    og = tl.dot(xn, wc4, out_dtype=tl.float32)
    gv = 1.0 / (1.0 + tl.exp(-og))
    tl.store(out_gate + rows[:, None] * 128 + hs[None, :], gv, mask=rm[:, None])


@triton.jit
def _split_gate_kernel(
    p, m,
    l, r, og,
    M: tl.constexpr, N2: tl.constexpr, H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = ph * BLOCK_H + tl.arange(0, BLOCK_H)
    rm = rows < M
    hm = hs < H
    mask = rm[:, None] & hm[None, :]

    base = p + rows[:, None] * (5 * H) + hs[None, :]
    lp = tl.load(base, mask=mask, other=0.0).to(tl.float32)
    rp = tl.load(base + H, mask=mask, other=0.0).to(tl.float32)
    lg = tl.load(base + 2 * H, mask=mask, other=0.0).to(tl.float32)
    rg = tl.load(base + 3 * H, mask=mask, other=0.0).to(tl.float32)
    oo = tl.load(base + 4 * H, mask=mask, other=0.0).to(tl.float32)

    if HAS_MASK:
        mv = tl.load(m + rows, mask=rm, other=0.0).to(tl.float32)
    else:
        mv = tl.full((BLOCK_M,), 1.0, tl.float32)

    lv = lp * (1.0 / (1.0 + tl.exp(-lg))) * mv[:, None]
    rv = rp * (1.0 / (1.0 + tl.exp(-rg))) * mv[:, None]
    gv = 1.0 / (1.0 + tl.exp(-oo))

    bidx = rows // N2
    ij = rows - bidx * N2
    out_off = bidx[:, None] * (H * N2) + hs[None, :] * N2 + ij[:, None]

    tl.store(l + out_off, lv, mask=mask)
    tl.store(r + out_off, rv, mask=mask)
    tl.store(og + rows[:, None] * H + hs[None, :], gv, mask=mask)


@triton.jit
def _final_ln_gate_kernel(
    tri, gate, w, b, y,
    M: tl.constexpr, N2: tl.constexpr, H: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    rm = rows < M
    hm = hs < H
    mask = rm[:, None] & hm[None, :]

    bidx = rows // N2
    ij = rows - bidx * N2
    off = bidx[:, None] * (H * N2) + hs[None, :] * N2 + ij[:, None]

    vals = tl.load(tri + off, mask=mask, other=0.0).to(tl.float32)
    vals_z = tl.where(hm[None, :], vals, 0.0)

    mean = tl.sum(vals_z, axis=1) / H
    diff = tl.where(hm[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + hs, mask=hm, other=1.0).to(tl.float32)
    bb = tl.load(b + hs, mask=hm, other=0.0).to(tl.float32)
    gg = tl.load(gate + rows[:, None] * H + hs[None, :], mask=mask, other=0.0).to(tl.float32)

    out = ((vals - mean[:, None]) * inv[:, None] * ww[None, :] + bb[None, :]) * gg
    tl.store(y + rows[:, None] * H + hs[None, :], out, mask=mask)


@triton.jit
def _final_ln_gate_out_d128_kernel(
    tri, gate, norm_w, norm_b, wout, out,
    M: tl.constexpr, N2: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    rows = tl.program_id(0) * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    ds = tl.arange(0, 128)
    rm = rows < M

    bidx = rows // N2
    ij = rows - bidx * N2
    tri_off = bidx[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]

    vals = tl.load(tri + tri_off, mask=rm[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * (1.0 / 128.0)
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * (1.0 / 128.0)
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + hs).to(tl.float32)
    nb = tl.load(norm_b + hs).to(tl.float32)
    gg = tl.load(gate + rows[:, None] * 128 + hs[None, :], mask=rm[:, None], other=0.0).to(tl.float32)

    hidden = (((diff * inv[:, None]) * nw[None, :] + nb[None, :]) * gg).to(tl.float16)
    ww = tl.load(wout + hs[:, None] * 128 + ds[None, :])
    acc = tl.dot(hidden, ww, out_dtype=tl.float32)

    tl.store(out + rows[:, None] * 128 + ds[None, :], acc, mask=rm[:, None])


def _weights(weights):
    key = (
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        weights["to_out.weight"].data_ptr(),
        tuple(weights["left_proj.weight"].shape),
        tuple(weights["to_out.weight"].shape),
        weights["left_proj.weight"].device.index,
    )
    got = _WEIGHT_CACHE.get(key)
    if got is not None:
        return got

    if len(_WEIGHT_CACHE) > 8:
        _WEIGHT_CACHE.clear()

    wcat = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).to(torch.float16).t().contiguous()

    wout = weights["to_out.weight"].to(torch.float16).t().contiguous()
    _WEIGHT_CACHE[key] = (wcat, wout)
    return wcat, wout


def _mm32(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is True:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_OK is None:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return y
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    x, mask, weights, config = data
    B, N, _, D = x.shape
    H = int(config["hidden_dim"])
    M = B * N * N
    N2 = N * N
    eps = float(config.get("eps", 1.0e-5))

    if not x.is_contiguous():
        x = x.contiguous()

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if has_mask and not mask.is_contiguous():
        mask = mask.contiguous()

    wcat, wout = _weights(weights)

    left = torch.empty((B * H, N, N), device=x.device, dtype=torch.float16)
    right = torch.empty((B * H, N, N), device=x.device, dtype=torch.float16)
    out_gate = torch.empty((M, H), device=x.device, dtype=torch.float16)

    if D == 128 and H == 128:
        _ln_proj_split_d128_kernel[(_cdiv(M, 8),)](
            x, mask if has_mask else x,
            weights["norm.weight"], weights["norm.bias"], wcat,
            left, right, out_gate,
            M, N2, eps,
            HAS_MASK=has_mask,
            BLOCK_M=8,
            num_warps=8,
        )
    else:
        xh = torch.empty((M, D), device=x.device, dtype=torch.float16)
        bd = _np2(D)
        bm_ln = 16 if D <= 128 else (8 if D <= 384 else 4)

        _ln_kernel[(_cdiv(M, bm_ln),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, D, eps,
            BLOCK_M=bm_ln, BLOCK_D=bd,
            num_warps=8,
        )

        proj = torch.mm(xh, wcat)
        del xh

        _split_gate_kernel[(_cdiv(M, 32), _cdiv(H, 64))](
            proj, mask if has_mask else proj,
            left, right, out_gate,
            M, N2, H,
            HAS_MASK=has_mask,
            BLOCK_M=32, BLOCK_H=64,
            num_warps=4,
        )
        del proj

    tri = torch.bmm(left, right.transpose(1, 2))
    del left, right

    if D == 128 and H == 128:
        out = torch.empty((M, 128), device=x.device, dtype=torch.float32)
        _final_ln_gate_out_d128_kernel[(_cdiv(M, 8),)](
            tri, out_gate,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            wout, out,
            M, N2, eps,
            BLOCK_M=8,
            num_warps=8,
        )
        del tri, out_gate
        return out.view(B, N, N, D)

    hidden = torch.empty((M, H), device=x.device, dtype=torch.float16)
    _final_ln_gate_kernel[(_cdiv(M, 16),)](
        tri, out_gate,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        hidden,
        M, N2, H, eps,
        BLOCK_M=16, BLOCK_H=_np2(H),
        num_warps=8,
    )
    del tri, out_gate

    out = _mm32(hidden, wout)
    return out.view(B, N, N, D)
