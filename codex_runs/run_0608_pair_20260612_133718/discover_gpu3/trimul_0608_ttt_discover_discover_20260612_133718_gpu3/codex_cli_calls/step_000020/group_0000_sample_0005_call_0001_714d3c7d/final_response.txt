"""
Outgoing Triangle Multiplicative Update forward pass.

The implementation normalizes pair activations with Triton row-wise LayerNorm,
uses cuBLAS for the dense projections, writes the five projected hidden streams
directly into channel-major workspace, gates/masks the triangle operands in-place,
computes the outgoing N^3 contraction as a batched tensor-core GEMM, then applies
the output LayerNorm/gate and final projection. Workspaces and transposed fp16
weights are cached per shape to avoid repeated allocation and transpose overhead.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WORKSPACE_CACHE = {}
_MM_OUT_DTYPE_OK = None

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
except Exception:
    pass


def _cdiv(a: int, b: int) -> int:
    return triton.cdiv(int(a), int(b))


@triton.jit
def _ln_128_kernel(x, w, b, y, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)

    vals = tl.load(x + rows[:, None] * 128 + cols[None, :]).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + cols).to(tl.float32)
    bb = tl.load(b + cols).to(tl.float32)
    tl.store(y + rows[:, None] * 128 + cols[None, :],
             diff * inv[:, None] * ww[None, :] + bb[None, :])


@triton.jit
def _ln_384_kernel(x, w, b, y, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)

    v0 = tl.load(x + rows[:, None] * 384 + cols[None, :]).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 384 + (cols + 128)[None, :]).to(tl.float32)
    v2 = tl.load(x + rows[:, None] * 384 + (cols + 256)[None, :]).to(tl.float32)

    mean = (tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1)) * 0.0026041666666666665
    d0 = v0 - mean[:, None]
    d1 = v1 - mean[:, None]
    d2 = v2 - mean[:, None]
    var = (tl.sum(d0 * d0, axis=1) + tl.sum(d1 * d1, axis=1) + tl.sum(d2 * d2, axis=1)) * 0.0026041666666666665
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + cols).to(tl.float32)
    w1 = tl.load(w + cols + 128).to(tl.float32)
    w2 = tl.load(w + cols + 256).to(tl.float32)
    b0 = tl.load(b + cols).to(tl.float32)
    b1 = tl.load(b + cols + 128).to(tl.float32)
    b2 = tl.load(b + cols + 256).to(tl.float32)

    tl.store(y + rows[:, None] * 384 + cols[None, :], d0 * inv[:, None] * w0[None, :] + b0[None, :])
    tl.store(y + rows[:, None] * 384 + (cols + 128)[None, :], d1 * inv[:, None] * w1[None, :] + b1[None, :])
    tl.store(y + rows[:, None] * 384 + (cols + 256)[None, :], d2 * inv[:, None] * w2[None, :] + b2[None, :])


@triton.jit
def _ln_768_kernel(x, w, b, y, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)

    v0 = tl.load(x + rows[:, None] * 768 + cols[None, :]).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 768 + (cols + 128)[None, :]).to(tl.float32)
    v2 = tl.load(x + rows[:, None] * 768 + (cols + 256)[None, :]).to(tl.float32)
    v3 = tl.load(x + rows[:, None] * 768 + (cols + 384)[None, :]).to(tl.float32)
    v4 = tl.load(x + rows[:, None] * 768 + (cols + 512)[None, :]).to(tl.float32)
    v5 = tl.load(x + rows[:, None] * 768 + (cols + 640)[None, :]).to(tl.float32)

    mean = (
        tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1) +
        tl.sum(v3, axis=1) + tl.sum(v4, axis=1) + tl.sum(v5, axis=1)
    ) * 0.0013020833333333333

    d0 = v0 - mean[:, None]
    d1 = v1 - mean[:, None]
    d2 = v2 - mean[:, None]
    d3 = v3 - mean[:, None]
    d4 = v4 - mean[:, None]
    d5 = v5 - mean[:, None]

    var = (
        tl.sum(d0 * d0, axis=1) + tl.sum(d1 * d1, axis=1) + tl.sum(d2 * d2, axis=1) +
        tl.sum(d3 * d3, axis=1) + tl.sum(d4 * d4, axis=1) + tl.sum(d5 * d5, axis=1)
    ) * 0.0013020833333333333
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + cols).to(tl.float32)
    w1 = tl.load(w + cols + 128).to(tl.float32)
    w2 = tl.load(w + cols + 256).to(tl.float32)
    w3 = tl.load(w + cols + 384).to(tl.float32)
    w4 = tl.load(w + cols + 512).to(tl.float32)
    w5 = tl.load(w + cols + 640).to(tl.float32)
    b0 = tl.load(b + cols).to(tl.float32)
    b1 = tl.load(b + cols + 128).to(tl.float32)
    b2 = tl.load(b + cols + 256).to(tl.float32)
    b3 = tl.load(b + cols + 384).to(tl.float32)
    b4 = tl.load(b + cols + 512).to(tl.float32)
    b5 = tl.load(b + cols + 640).to(tl.float32)

    tl.store(y + rows[:, None] * 768 + cols[None, :], d0 * inv[:, None] * w0[None, :] + b0[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 128)[None, :], d1 * inv[:, None] * w1[None, :] + b1[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 256)[None, :], d2 * inv[:, None] * w2[None, :] + b2[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 384)[None, :], d3 * inv[:, None] * w3[None, :] + b3[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 512)[None, :], d4 * inv[:, None] * w4[None, :] + b4[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 640)[None, :], d5 * inv[:, None] * w5[None, :] + b5[None, :])


@triton.jit
def _gate_packed_nomask_kernel(p, TOTAL: tl.constexpr, BLOCK: tl.constexpr):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    m = offs < TOTAL

    lp = tl.load(p + offs, mask=m, other=0.0).to(tl.float32)
    rp = tl.load(p + TOTAL + offs, mask=m, other=0.0).to(tl.float32)
    lg = tl.load(p + 2 * TOTAL + offs, mask=m, other=0.0).to(tl.float32)
    rg = tl.load(p + 3 * TOTAL + offs, mask=m, other=0.0).to(tl.float32)

    tl.store(p + offs, lp * tl.sigmoid(lg), mask=m)
    tl.store(p + TOTAL + offs, rp * tl.sigmoid(rg), mask=m)


@triton.jit
def _gate_packed_mask_kernel(p, mask_ptr, N2: tl.constexpr, B: tl.constexpr,
                             TOTAL: tl.constexpr, BLOCK: tl.constexpr):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < TOTAL

    hb = offs // N2
    bidx = hb % B
    ij = offs - hb * N2
    mv = tl.load(mask_ptr + bidx * N2 + ij, mask=valid, other=0.0).to(tl.float32)

    lp = tl.load(p + offs, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(p + TOTAL + offs, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(p + 2 * TOTAL + offs, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(p + 3 * TOTAL + offs, mask=valid, other=0.0).to(tl.float32)

    tl.store(p + offs, lp * tl.sigmoid(lg) * mv, mask=valid)
    tl.store(p + TOTAL + offs, rp * tl.sigmoid(rg) * mv, mask=valid)


@triton.jit
def _final_ln_gate_kernel(tri, packed, w, b, y,
                          N2: tl.constexpr, B: tl.constexpr, EPS: tl.constexpr,
                          BLOCK_M: tl.constexpr):
    rows = tl.program_id(0) * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)

    bidx = rows // N2
    ij = rows - bidx * N2
    stride_h = B * N2
    off = hs[None, :] * stride_h + bidx[:, None] * N2 + ij[:, None]

    vals = tl.load(tri + off).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + hs).to(tl.float32)
    bb = tl.load(b + hs).to(tl.float32)
    og = tl.load(packed + 4 * 128 * B * N2 + off).to(tl.float32)

    tl.store(y + rows[:, None] * 128 + hs[None, :],
             ((diff * inv[:, None] * ww[None, :] + bb[None, :]) * tl.sigmoid(og)).to(tl.float16))


@triton.jit
def _final_ln_gate_out_128_kernel(tri, packed, w, b, wout, y,
                                  N2: tl.constexpr, B: tl.constexpr, EPS: tl.constexpr,
                                  BLOCK_M: tl.constexpr):
    rows = tl.program_id(0) * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    ds = tl.arange(0, 128)

    bidx = rows // N2
    ij = rows - bidx * N2
    stride_h = B * N2
    off = hs[None, :] * stride_h + bidx[:, None] * N2 + ij[:, None]

    vals = tl.load(tri + off).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + hs).to(tl.float32)
    bb = tl.load(b + hs).to(tl.float32)
    og = tl.load(packed + 4 * 128 * B * N2 + off).to(tl.float32)
    hidden = ((diff * inv[:, None] * ww[None, :] + bb[None, :]) * tl.sigmoid(og)).to(tl.float16)

    wo = tl.load(wout + hs[:, None] * 128 + ds[None, :])
    acc = tl.dot(hidden, wo, out_dtype=tl.float32)
    tl.store(y + rows[:, None] * 128 + ds[None, :], acc)


def _weights(weights):
    key = (
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        weights["to_out.weight"].data_ptr(),
        tuple(weights["left_proj.weight"].shape),
        tuple(weights["to_out.weight"].shape),
        weights["left_proj.weight"].device.index,
    )
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    if len(_WEIGHT_CACHE) > 8:
        _WEIGHT_CACHE.clear()

    wcat = torch.cat((
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ), dim=0).to(torch.float16).t().contiguous()
    wout = weights["to_out.weight"].to(torch.float16).t().contiguous()

    _WEIGHT_CACHE[key] = (wcat, wout)
    return wcat, wout


def _workspace(device, B, N, D, need_hidden):
    key = (device.index, B, N, D, need_hidden)
    cached = _WORKSPACE_CACHE.get(key)
    if cached is not None:
        return cached

    if len(_WORKSPACE_CACHE) > 10:
        _WORKSPACE_CACHE.clear()

    M = B * N * N
    N2 = N * N
    packed = torch.empty((5 * B * 128, N, N), device=device, dtype=torch.float16)

    ws = {
        "xh": torch.empty((M, D), device=device, dtype=torch.float16),
        "packed": packed,
        "packed_mm": torch.as_strided(packed, (M, 640), (1, B * N2)),
        "left": packed[:B * 128],
        "right": packed[B * 128:2 * B * 128],
        "tri": torch.empty((B * 128, N, N), device=device, dtype=torch.float16),
    }
    if need_hidden:
        ws["hidden"] = torch.empty((M, 128), device=device, dtype=torch.float16)

    _WORKSPACE_CACHE[key] = ws
    return ws


def _mm32(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is True:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_OK is None:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return y
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    x, mask, weights, config = data
    B, N, _, D = x.shape
    M = B * N * N
    N2 = N * N
    total = B * 128 * N2
    eps = float(config.get("eps", 1.0e-5))

    if not x.is_contiguous():
        x = x.contiguous()

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if has_mask and not mask.is_contiguous():
        mask = mask.contiguous()

    wcat, wout = _weights(weights)
    ws = _workspace(x.device, B, N, D, D != 128)

    xh = ws["xh"]
    packed = ws["packed"]
    packed_mm = ws["packed_mm"]
    left = ws["left"]
    right = ws["right"]
    tri = ws["tri"]

    if D == 128:
        _ln_128_kernel[(_cdiv(M, 16),)](
            x, weights["norm.weight"], weights["norm.bias"], xh, eps,
            BLOCK_M=16, num_warps=4,
        )
    elif D == 384:
        _ln_384_kernel[(_cdiv(M, 8),)](
            x, weights["norm.weight"], weights["norm.bias"], xh, eps,
            BLOCK_M=8, num_warps=8,
        )
    elif D == 768:
        _ln_768_kernel[(_cdiv(M, 4),)](
            x, weights["norm.weight"], weights["norm.bias"], xh, eps,
            BLOCK_M=4, num_warps=8,
        )
    else:
        xh.copy_(torch.nn.functional.layer_norm(
            x.view(M, D),
            (D,),
            weights["norm.weight"],
            weights["norm.bias"],
            eps,
        ).to(torch.float16))

    torch.mm(xh, wcat, out=packed_mm)

    if has_mask:
        _gate_packed_mask_kernel[(_cdiv(total, 1024),)](
            packed, mask, N2, B, total,
            BLOCK=1024, num_warps=4,
        )
    else:
        _gate_packed_nomask_kernel[(_cdiv(total, 1024),)](
            packed, total,
            BLOCK=1024, num_warps=4,
        )

    torch.bmm(left, right.transpose(1, 2), out=tri)

    if D == 128:
        out = torch.empty((M, 128), device=x.device, dtype=torch.float32)
        _final_ln_gate_out_128_kernel[(_cdiv(M, 16),)](
            tri, packed,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            wout, out, N2, B, eps,
            BLOCK_M=16, num_warps=4,
        )
        return out.view(B, N, N, 128)

    hidden = ws["hidden"]
    _final_ln_gate_kernel[(_cdiv(M, 16),)](
        tri, packed,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        hidden, N2, B, eps,
        BLOCK_M=16, num_warps=4,
    )

    out = _mm32(hidden, wout)
    return out.view(B, N, N, D)
