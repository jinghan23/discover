"""
Outgoing TriMul forward pass.

Fused path:
- Cache fp16 packed projection weights.
- Compute input LayerNorm, five projections, left/right sigmoid gates, mask,
  layout packing, and output gate values in Triton.
- Run the triangle update as strided batched tensor-core GEMMs.
- For dim=128, fuse hidden LayerNorm, output gate, and final projection in Triton.
- For wider output dims, fuse hidden LayerNorm + output gate, then use torch.mm
  with fp32 output for the final projection.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_WS_CACHE = {}
_MM_OUT_DTYPE_OK = None

try:
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
except Exception:
    pass


def _cdiv(a, b):
    return triton.cdiv(int(a), int(b))


@triton.jit
def _project128(x, msk, wnorm, bnorm, wcat, left, right, ogate,
                N2: tl.constexpr, EPS: tl.constexpr,
                HAS_MASK: tl.constexpr, BM: tl.constexpr, BH: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BM + tl.arange(0, BM)
    hs = ph * BH + tl.arange(0, BH)
    c = tl.arange(0, 128)

    v = tl.load(x + rows[:, None] * 128 + c[None, :]).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    q = tl.sum(v * v, axis=1) * 0.0078125
    inv = tl.rsqrt(tl.maximum(q - mean * mean, 0.0) + EPS)

    nw = tl.load(wnorm + c).to(tl.float32)
    nb = tl.load(bnorm + c).to(tl.float32)
    xn = ((v - mean[:, None]) * inv[:, None] * nw[None, :] + nb[None, :]).to(tl.float16)

    wk = c[:, None] * 640 + hs[None, :]

    wl = tl.load(wcat + wk + 0)
    wr = tl.load(wcat + wk + 128)
    wlg = tl.load(wcat + wk + 256)
    wrg = tl.load(wcat + wk + 384)
    wog = tl.load(wcat + wk + 512)

    lp = tl.dot(xn, wl, out_dtype=tl.float32)
    rp = tl.dot(xn, wr, out_dtype=tl.float32)
    lg = tl.dot(xn, wlg, out_dtype=tl.float32)
    rg = tl.dot(xn, wrg, out_dtype=tl.float32)
    og = tl.dot(xn, wog, out_dtype=tl.float32)

    bid = rows // N2
    ij = rows - bid * N2
    out = bid[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]

    mult = 1.0
    if HAS_MASK:
        mult = tl.load(msk + rows).to(tl.float32)[:, None]

    tl.store(left + out, (lp * tl.sigmoid(lg) * mult).to(tl.float16))
    tl.store(right + out, (rp * tl.sigmoid(rg) * mult).to(tl.float16))
    tl.store(ogate + rows[:, None] * 128 + hs[None, :], tl.sigmoid(og).to(tl.float16))


@triton.jit
def _project384(x, msk, wnorm, bnorm, wcat, left, right, ogate,
                N2: tl.constexpr, EPS: tl.constexpr,
                HAS_MASK: tl.constexpr, BM: tl.constexpr, BH: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BM + tl.arange(0, BM)
    hs = ph * BH + tl.arange(0, BH)
    c = tl.arange(0, 128)

    v0 = tl.load(x + rows[:, None] * 384 + c[None, :]).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 384 + (c + 128)[None, :]).to(tl.float32)
    v2 = tl.load(x + rows[:, None] * 384 + (c + 256)[None, :]).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1)
    q = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1) + tl.sum(v2 * v2, axis=1)
    mean = s * 0.0026041666666666665
    inv = tl.rsqrt(tl.maximum(q * 0.0026041666666666665 - mean * mean, 0.0) + EPS)

    w0 = tl.load(wnorm + c).to(tl.float32)
    w1 = tl.load(wnorm + c + 128).to(tl.float32)
    w2 = tl.load(wnorm + c + 256).to(tl.float32)
    b0 = tl.load(bnorm + c).to(tl.float32)
    b1 = tl.load(bnorm + c + 128).to(tl.float32)
    b2 = tl.load(bnorm + c + 256).to(tl.float32)

    x0 = ((v0 - mean[:, None]) * inv[:, None] * w0[None, :] + b0[None, :]).to(tl.float16)
    x1 = ((v1 - mean[:, None]) * inv[:, None] * w1[None, :] + b1[None, :]).to(tl.float16)
    x2 = ((v2 - mean[:, None]) * inv[:, None] * w2[None, :] + b2[None, :]).to(tl.float16)

    wk0 = c[:, None] * 640 + hs[None, :]
    wk1 = (c[:, None] + 128) * 640 + hs[None, :]
    wk2 = (c[:, None] + 256) * 640 + hs[None, :]

    lp = tl.dot(x0, tl.load(wcat + wk0 + 0), out_dtype=tl.float32)
    lp += tl.dot(x1, tl.load(wcat + wk1 + 0), out_dtype=tl.float32)
    lp += tl.dot(x2, tl.load(wcat + wk2 + 0), out_dtype=tl.float32)

    lg = tl.dot(x0, tl.load(wcat + wk0 + 256), out_dtype=tl.float32)
    lg += tl.dot(x1, tl.load(wcat + wk1 + 256), out_dtype=tl.float32)
    lg += tl.dot(x2, tl.load(wcat + wk2 + 256), out_dtype=tl.float32)

    rp = tl.dot(x0, tl.load(wcat + wk0 + 128), out_dtype=tl.float32)
    rp += tl.dot(x1, tl.load(wcat + wk1 + 128), out_dtype=tl.float32)
    rp += tl.dot(x2, tl.load(wcat + wk2 + 128), out_dtype=tl.float32)

    rg = tl.dot(x0, tl.load(wcat + wk0 + 384), out_dtype=tl.float32)
    rg += tl.dot(x1, tl.load(wcat + wk1 + 384), out_dtype=tl.float32)
    rg += tl.dot(x2, tl.load(wcat + wk2 + 384), out_dtype=tl.float32)

    og = tl.dot(x0, tl.load(wcat + wk0 + 512), out_dtype=tl.float32)
    og += tl.dot(x1, tl.load(wcat + wk1 + 512), out_dtype=tl.float32)
    og += tl.dot(x2, tl.load(wcat + wk2 + 512), out_dtype=tl.float32)

    bid = rows // N2
    ij = rows - bid * N2
    out = bid[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]

    mult = 1.0
    if HAS_MASK:
        mult = tl.load(msk + rows).to(tl.float32)[:, None]

    tl.store(left + out, (lp * tl.sigmoid(lg) * mult).to(tl.float16))
    tl.store(right + out, (rp * tl.sigmoid(rg) * mult).to(tl.float16))
    tl.store(ogate + rows[:, None] * 128 + hs[None, :], tl.sigmoid(og).to(tl.float16))


@triton.jit
def _ln768(x, w, b, y, EPS: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    r = pid * BM + tl.arange(0, BM)
    c = tl.arange(0, 128)

    v0 = tl.load(x + r[:, None] * 768 + c[None, :]).to(tl.float32)
    v1 = tl.load(x + r[:, None] * 768 + (c + 128)[None, :]).to(tl.float32)
    v2 = tl.load(x + r[:, None] * 768 + (c + 256)[None, :]).to(tl.float32)
    v3 = tl.load(x + r[:, None] * 768 + (c + 384)[None, :]).to(tl.float32)
    v4 = tl.load(x + r[:, None] * 768 + (c + 512)[None, :]).to(tl.float32)
    v5 = tl.load(x + r[:, None] * 768 + (c + 640)[None, :]).to(tl.float32)

    s = tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1) + tl.sum(v3, axis=1) + tl.sum(v4, axis=1) + tl.sum(v5, axis=1)
    q = tl.sum(v0 * v0, axis=1) + tl.sum(v1 * v1, axis=1) + tl.sum(v2 * v2, axis=1) + tl.sum(v3 * v3, axis=1) + tl.sum(v4 * v4, axis=1) + tl.sum(v5 * v5, axis=1)
    mean = s * 0.0013020833333333333
    inv = tl.rsqrt(tl.maximum(q * 0.0013020833333333333 - mean * mean, 0.0) + EPS)

    w0 = tl.load(w + c).to(tl.float32)
    w1 = tl.load(w + c + 128).to(tl.float32)
    w2 = tl.load(w + c + 256).to(tl.float32)
    w3 = tl.load(w + c + 384).to(tl.float32)
    w4 = tl.load(w + c + 512).to(tl.float32)
    w5 = tl.load(w + c + 640).to(tl.float32)
    b0 = tl.load(b + c).to(tl.float32)
    b1 = tl.load(b + c + 128).to(tl.float32)
    b2 = tl.load(b + c + 256).to(tl.float32)
    b3 = tl.load(b + c + 384).to(tl.float32)
    b4 = tl.load(b + c + 512).to(tl.float32)
    b5 = tl.load(b + c + 640).to(tl.float32)

    tl.store(y + r[:, None] * 768 + c[None, :], ((v0 - mean[:, None]) * inv[:, None] * w0[None, :] + b0[None, :]).to(tl.float16))
    tl.store(y + r[:, None] * 768 + (c + 128)[None, :], ((v1 - mean[:, None]) * inv[:, None] * w1[None, :] + b1[None, :]).to(tl.float16))
    tl.store(y + r[:, None] * 768 + (c + 256)[None, :], ((v2 - mean[:, None]) * inv[:, None] * w2[None, :] + b2[None, :]).to(tl.float16))
    tl.store(y + r[:, None] * 768 + (c + 384)[None, :], ((v3 - mean[:, None]) * inv[:, None] * w3[None, :] + b3[None, :]).to(tl.float16))
    tl.store(y + r[:, None] * 768 + (c + 512)[None, :], ((v4 - mean[:, None]) * inv[:, None] * w4[None, :] + b4[None, :]).to(tl.float16))
    tl.store(y + r[:, None] * 768 + (c + 640)[None, :], ((v5 - mean[:, None]) * inv[:, None] * w5[None, :] + b5[None, :]).to(tl.float16))


@triton.jit
def _split_from_proj(p, msk, left, right, ogate,
                     N2: tl.constexpr, HAS_MASK: tl.constexpr,
                     BM: tl.constexpr, BH: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)
    rows = pm * BM + tl.arange(0, BM)
    hs = ph * BH + tl.arange(0, BH)

    bid = rows // N2
    ij = rows - bid * N2
    out = bid[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]
    base = p + rows[:, None] * 640 + hs[None, :]

    lp = tl.load(base + 0).to(tl.float32)
    rp = tl.load(base + 128).to(tl.float32)
    lg = tl.load(base + 256).to(tl.float32)
    rg = tl.load(base + 384).to(tl.float32)
    og = tl.load(base + 512).to(tl.float32)

    mult = 1.0
    if HAS_MASK:
        mult = tl.load(msk + rows).to(tl.float32)[:, None]

    tl.store(left + out, (lp * tl.sigmoid(lg) * mult).to(tl.float16))
    tl.store(right + out, (rp * tl.sigmoid(rg) * mult).to(tl.float16))
    tl.store(ogate + rows[:, None] * 128 + hs[None, :], tl.sigmoid(og).to(tl.float16))


@triton.jit
def _final_fused128(tri, ogate, w, b, wout, y,
                    N2: tl.constexpr, EPS: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BM + tl.arange(0, BM)
    h = tl.arange(0, 128)

    bid = rows // N2
    ij = rows - bid * N2
    off = bid[:, None] * (128 * N2) + h[None, :] * N2 + ij[:, None]

    v = tl.load(tri + off).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    q = tl.sum(v * v, axis=1) * 0.0078125
    inv = tl.rsqrt(tl.maximum(q - mean * mean, 0.0) + EPS)

    ww = tl.load(w + h).to(tl.float32)
    bb = tl.load(b + h).to(tl.float32)
    gate = tl.load(ogate + rows[:, None] * 128 + h[None, :]).to(tl.float32)

    hidden = ((v - mean[:, None]) * inv[:, None] * ww[None, :] + bb[None, :]) * gate
    wo = tl.load(wout + h[:, None] * 128 + h[None, :])
    out = tl.dot(hidden.to(tl.float16), wo, out_dtype=tl.float32)
    tl.store(y + rows[:, None] * 128 + h[None, :], out)


@triton.jit
def _final_ln_gate_og(tri, ogate, w, b, hidden,
                      N2: tl.constexpr, EPS: tl.constexpr, BM: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BM + tl.arange(0, BM)
    h = tl.arange(0, 128)

    bid = rows // N2
    ij = rows - bid * N2
    off = bid[:, None] * (128 * N2) + h[None, :] * N2 + ij[:, None]

    v = tl.load(tri + off).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    q = tl.sum(v * v, axis=1) * 0.0078125
    inv = tl.rsqrt(tl.maximum(q - mean * mean, 0.0) + EPS)

    ww = tl.load(w + h).to(tl.float32)
    bb = tl.load(b + h).to(tl.float32)
    gate = tl.load(ogate + rows[:, None] * 128 + h[None, :]).to(tl.float32)

    out = ((v - mean[:, None]) * inv[:, None] * ww[None, :] + bb[None, :]) * gate
    tl.store(hidden + rows[:, None] * 128 + h[None, :], out.to(tl.float16))


def _weights(weights):
    key = (
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        weights["to_out.weight"].data_ptr(),
        tuple(weights["left_proj.weight"].shape),
        tuple(weights["to_out.weight"].shape),
        weights["left_proj.weight"].device.index,
    )
    got = _WEIGHT_CACHE.get(key)
    if got is not None:
        return got
    if len(_WEIGHT_CACHE) > 8:
        _WEIGHT_CACHE.clear()

    wcat = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).to(torch.float16).t().contiguous()
    wout = weights["to_out.weight"].to(torch.float16).t().contiguous()
    _WEIGHT_CACHE[key] = (wcat, wout)
    return wcat, wout


def _workspace(device, B, N, D, mode):
    key = (device.index, B, N, D, mode)
    got = _WS_CACHE.get(key)
    if got is not None:
        return got
    if len(_WS_CACHE) > 12:
        _WS_CACHE.clear()

    M = B * N * N
    ws = {
        "left": torch.empty((B * 128, N, N), device=device, dtype=torch.float16),
        "right": torch.empty((B * 128, N, N), device=device, dtype=torch.float16),
        "tri": torch.empty((B * 128, N, N), device=device, dtype=torch.float16),
        "ogate": torch.empty((M, 128), device=device, dtype=torch.float16),
    }
    if mode == "fallback":
        ws["xh"] = torch.empty((M, D), device=device, dtype=torch.float16)
        ws["proj"] = torch.empty((M, 640), device=device, dtype=torch.float16)
    if D != 128:
        ws["hidden"] = torch.empty((M, 128), device=device, dtype=torch.float16)

    _WS_CACHE[key] = ws
    return ws


def _mm32(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is True:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_OK is None:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return y
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    x, mask, weights, config = data
    B, N, _, D = x.shape
    M = B * N * N
    N2 = N * N
    eps = float(config.get("eps", 1.0e-5))

    if not x.is_contiguous():
        x = x.contiguous()

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if has_mask and not mask.is_contiguous():
        mask = mask.contiguous()

    wcat, wout = _weights(weights)

    mode = "fused" if D in (128, 384) else "fallback"
    ws = _workspace(x.device, B, N, D, mode)

    left = ws["left"]
    right = ws["right"]
    tri = ws["tri"]
    ogate = ws["ogate"]

    if D == 128:
        _project128[(_cdiv(M, 16), 2)](
            x, mask, weights["norm.weight"], weights["norm.bias"], wcat,
            left, right, ogate, N2, eps,
            HAS_MASK=has_mask, BM=16, BH=64, num_warps=4,
        )
    elif D == 384:
        _project384[(_cdiv(M, 8), 2)](
            x, mask, weights["norm.weight"], weights["norm.bias"], wcat,
            left, right, ogate, N2, eps,
            HAS_MASK=has_mask, BM=8, BH=64, num_warps=4,
        )
    else:
        xh = ws["xh"]
        proj = ws["proj"]
        _ln768[(_cdiv(M, 4),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            eps, BM=4, num_warps=8,
        )
        torch.mm(xh, wcat, out=proj)
        _split_from_proj[(_cdiv(M, 32), 2)](
            proj, mask, left, right, ogate, N2,
            HAS_MASK=has_mask, BM=32, BH=64, num_warps=4,
        )

    torch.bmm(left, right.transpose(1, 2), out=tri)

    if D == 128:
        out = torch.empty((M, 128), device=x.device, dtype=torch.float32)
        _final_fused128[(_cdiv(M, 16),)](
            tri, ogate,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            wout,
            out,
            N2,
            eps,
            BM=16,
            num_warps=4,
        )
        return out.view(B, N, N, D)

    hidden = ws["hidden"]
    _final_ln_gate_og[(_cdiv(M, 16),)](
        tri,
        ogate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        hidden,
        N2,
        eps,
        BM=16,
        num_warps=4,
    )
    return _mm32(hidden, wout).view(B, N, N, D)
