"""
Outgoing Triangle Multiplicative Update forward pass.

The implementation uses Triton for the row-wise LayerNorms, gate/mask packing,
and the fused final LayerNorm/out-gate/projection path for dim=128. The N^3
triangle contraction is kept as a channel-batched GEMM so tensor cores handle the
dominant work efficiently. Projection weights are cached as transposed fp16
matrices for reuse across calls.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_MM_OUT_DTYPE_OK = None


def _np2(x: int) -> int:
    return 1 << (int(x) - 1).bit_length()


def _cdiv(a: int, b: int) -> int:
    return triton.cdiv(int(a), int(b))


@triton.jit
def _ln_128_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr,
                   BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)

    vals = tl.load(x + rows[:, None] * 128 + cols[None, :]).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + cols).to(tl.float32)
    bb = tl.load(b + cols).to(tl.float32)
    tl.store(y + rows[:, None] * 128 + cols[None, :],
             diff * inv[:, None] * ww[None, :] + bb[None, :])


@triton.jit
def _ln_384_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr,
                   BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)

    v0 = tl.load(x + rows[:, None] * 384 + cols[None, :]).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 384 + (cols + 128)[None, :]).to(tl.float32)
    v2 = tl.load(x + rows[:, None] * 384 + (cols + 256)[None, :]).to(tl.float32)

    mean = (tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1)) * 0.0026041666666666665

    d0 = v0 - mean[:, None]
    d1 = v1 - mean[:, None]
    d2 = v2 - mean[:, None]
    var = (tl.sum(d0 * d0, axis=1) + tl.sum(d1 * d1, axis=1) + tl.sum(d2 * d2, axis=1)) * 0.0026041666666666665
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + cols).to(tl.float32)
    w1 = tl.load(w + cols + 128).to(tl.float32)
    w2 = tl.load(w + cols + 256).to(tl.float32)
    b0 = tl.load(b + cols).to(tl.float32)
    b1 = tl.load(b + cols + 128).to(tl.float32)
    b2 = tl.load(b + cols + 256).to(tl.float32)

    tl.store(y + rows[:, None] * 384 + cols[None, :],
             d0 * inv[:, None] * w0[None, :] + b0[None, :])
    tl.store(y + rows[:, None] * 384 + (cols + 128)[None, :],
             d1 * inv[:, None] * w1[None, :] + b1[None, :])
    tl.store(y + rows[:, None] * 384 + (cols + 256)[None, :],
             d2 * inv[:, None] * w2[None, :] + b2[None, :])


@triton.jit
def _ln_768_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr,
                   BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, 128)

    v0 = tl.load(x + rows[:, None] * 768 + cols[None, :]).to(tl.float32)
    v1 = tl.load(x + rows[:, None] * 768 + (cols + 128)[None, :]).to(tl.float32)
    v2 = tl.load(x + rows[:, None] * 768 + (cols + 256)[None, :]).to(tl.float32)
    v3 = tl.load(x + rows[:, None] * 768 + (cols + 384)[None, :]).to(tl.float32)
    v4 = tl.load(x + rows[:, None] * 768 + (cols + 512)[None, :]).to(tl.float32)
    v5 = tl.load(x + rows[:, None] * 768 + (cols + 640)[None, :]).to(tl.float32)

    mean = (
        tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1) +
        tl.sum(v3, axis=1) + tl.sum(v4, axis=1) + tl.sum(v5, axis=1)
    ) * 0.0013020833333333333

    d0 = v0 - mean[:, None]
    d1 = v1 - mean[:, None]
    d2 = v2 - mean[:, None]
    d3 = v3 - mean[:, None]
    d4 = v4 - mean[:, None]
    d5 = v5 - mean[:, None]

    var = (
        tl.sum(d0 * d0, axis=1) + tl.sum(d1 * d1, axis=1) + tl.sum(d2 * d2, axis=1) +
        tl.sum(d3 * d3, axis=1) + tl.sum(d4 * d4, axis=1) + tl.sum(d5 * d5, axis=1)
    ) * 0.0013020833333333333
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + cols).to(tl.float32)
    w1 = tl.load(w + cols + 128).to(tl.float32)
    w2 = tl.load(w + cols + 256).to(tl.float32)
    w3 = tl.load(w + cols + 384).to(tl.float32)
    w4 = tl.load(w + cols + 512).to(tl.float32)
    w5 = tl.load(w + cols + 640).to(tl.float32)
    b0 = tl.load(b + cols).to(tl.float32)
    b1 = tl.load(b + cols + 128).to(tl.float32)
    b2 = tl.load(b + cols + 256).to(tl.float32)
    b3 = tl.load(b + cols + 384).to(tl.float32)
    b4 = tl.load(b + cols + 512).to(tl.float32)
    b5 = tl.load(b + cols + 640).to(tl.float32)

    tl.store(y + rows[:, None] * 768 + cols[None, :], d0 * inv[:, None] * w0[None, :] + b0[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 128)[None, :], d1 * inv[:, None] * w1[None, :] + b1[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 256)[None, :], d2 * inv[:, None] * w2[None, :] + b2[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 384)[None, :], d3 * inv[:, None] * w3[None, :] + b3[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 512)[None, :], d4 * inv[:, None] * w4[None, :] + b4[None, :])
    tl.store(y + rows[:, None] * 768 + (cols + 640)[None, :], d5 * inv[:, None] * w5[None, :] + b5[None, :])


@triton.jit
def _ln_cols_kernel(x, w, b, y, M: tl.constexpr, D: tl.constexpr,
                    EPS: tl.constexpr, BLOCK_M: tl.constexpr,
                    BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    cols = tl.arange(0, BLOCK_D)
    cm = cols < D

    vals = tl.load(x + rows[:, None] * D + cols[None, :],
                   mask=cm[None, :], other=0.0).to(tl.float32)
    vals = tl.where(cm[None, :], vals, 0.0)

    mean = tl.sum(vals, axis=1) / D
    diff = tl.where(cm[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(diff * diff, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + cols, mask=cm, other=1.0).to(tl.float32)
    bb = tl.load(b + cols, mask=cm, other=0.0).to(tl.float32)
    out = (vals - mean[:, None]) * inv[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * D + cols[None, :], out, mask=cm[None, :])


@triton.jit
def _split_gate_nomask_128_kernel(p, l, r, M: tl.constexpr, N2: tl.constexpr,
                                  BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = ph * BLOCK_H + tl.arange(0, BLOCK_H)

    bidx = rows // N2
    ij = rows - bidx * N2
    out_off = bidx[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]
    base = p + rows[:, None] * 640 + hs[None, :]

    lp = tl.load(base).to(tl.float32)
    rp = tl.load(base + 128).to(tl.float32)
    lg = tl.load(base + 256).to(tl.float32)
    rg = tl.load(base + 384).to(tl.float32)

    tl.store(l + out_off, lp * tl.sigmoid(lg))
    tl.store(r + out_off, rp * tl.sigmoid(rg))


@triton.jit
def _split_gate_mask_128_kernel(p, m, l, r, M: tl.constexpr, N2: tl.constexpr,
                                BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = ph * BLOCK_H + tl.arange(0, BLOCK_H)

    mv = tl.load(m + rows).to(tl.float32)
    bidx = rows // N2
    ij = rows - bidx * N2
    out_off = bidx[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]
    base = p + rows[:, None] * 640 + hs[None, :]

    lp = tl.load(base).to(tl.float32)
    rp = tl.load(base + 128).to(tl.float32)
    lg = tl.load(base + 256).to(tl.float32)
    rg = tl.load(base + 384).to(tl.float32)

    tl.store(l + out_off, lp * tl.sigmoid(lg) * mv[:, None])
    tl.store(r + out_off, rp * tl.sigmoid(rg) * mv[:, None])


@triton.jit
def _final_ln_gate_out_128_kernel(tri, proj, w, b, wout, y,
                                  M: tl.constexpr, N2: tl.constexpr,
                                  EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    ds = tl.arange(0, 128)

    bidx = rows // N2
    ij = rows - bidx * N2
    off = bidx[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]

    vals = tl.load(tri + off).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + hs).to(tl.float32)
    bb = tl.load(b + hs).to(tl.float32)
    og = tl.load(proj + rows[:, None] * 640 + 512 + hs[None, :]).to(tl.float32)
    hidden = ((diff * inv[:, None] * ww[None, :] + bb[None, :]) *
              tl.sigmoid(og)).to(tl.float16)

    wo = tl.load(wout + hs[:, None] * 128 + ds[None, :])
    acc = tl.dot(hidden, wo, out_dtype=tl.float32)
    tl.store(y + rows[:, None] * 128 + ds[None, :], acc)


@triton.jit
def _final_ln_gate_kernel(tri, proj, w, b, y,
                          M: tl.constexpr, N2: tl.constexpr,
                          EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)

    bidx = rows // N2
    ij = rows - bidx * N2
    off = bidx[:, None] * (128 * N2) + hs[None, :] * N2 + ij[:, None]

    vals = tl.load(tri + off).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    diff = vals - mean[:, None]
    var = tl.sum(diff * diff, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + hs).to(tl.float32)
    bb = tl.load(b + hs).to(tl.float32)
    og = tl.load(proj + rows[:, None] * 640 + 512 + hs[None, :]).to(tl.float32)
    out = ((diff * inv[:, None] * ww[None, :] + bb[None, :]) *
           tl.sigmoid(og)).to(tl.float16)
    tl.store(y + rows[:, None] * 128 + hs[None, :], out)


def _weights(weights):
    key = (
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        weights["to_out.weight"].data_ptr(),
        tuple(weights["left_proj.weight"].shape),
        tuple(weights["to_out.weight"].shape),
        weights["left_proj.weight"].device.index,
    )
    cached = _WEIGHT_CACHE.get(key)
    if cached is not None:
        return cached

    if len(_WEIGHT_CACHE) > 8:
        _WEIGHT_CACHE.clear()

    wcat = torch.cat(
        (
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ),
        dim=0,
    ).to(torch.float16).t().contiguous()

    wout = weights["to_out.weight"].to(torch.float16).t().contiguous()
    _WEIGHT_CACHE[key] = (wcat, wout)
    return wcat, wout


def _mm32(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is True:
        return torch.mm(a, b, out_dtype=torch.float32)
    if _MM_OUT_DTYPE_OK is None:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return y
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    x, mask, weights, config = data
    B, N, _, D = x.shape
    H = int(config["hidden_dim"])
    M = B * N * N
    N2 = N * N
    eps = float(config.get("eps", 1.0e-5))

    if not x.is_contiguous():
        x = x.contiguous()

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    if has_mask and not mask.is_contiguous():
        mask = mask.contiguous()

    wcat, wout = _weights(weights)

    xh = torch.empty((M, D), device=x.device, dtype=torch.float16)

    if D == 128:
        _ln_128_kernel[(_cdiv(M, 16),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, eps,
            BLOCK_M=16,
            num_warps=4,
        )
    elif D == 384:
        _ln_384_kernel[(_cdiv(M, 8),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, eps,
            BLOCK_M=8,
            num_warps=8,
        )
    elif D == 768:
        _ln_768_kernel[(_cdiv(M, 4),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, eps,
            BLOCK_M=4,
            num_warps=8,
        )
    else:
        bd = _np2(D)
        _ln_cols_kernel[(_cdiv(M, 8),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, D, eps,
            BLOCK_M=8, BLOCK_D=bd,
            num_warps=8,
        )

    proj = torch.empty((M, 640), device=x.device, dtype=torch.float16)
    torch.mm(xh, wcat, out=proj)
    del xh

    left = torch.empty((B * H, N, N), device=x.device, dtype=torch.float16)
    right = torch.empty((B * H, N, N), device=x.device, dtype=torch.float16)

    if has_mask:
        _split_gate_mask_128_kernel[(_cdiv(M, 32), 2)](
            proj, mask, left, right,
            M, N2,
            BLOCK_M=32, BLOCK_H=64,
            num_warps=4,
        )
    else:
        _split_gate_nomask_128_kernel[(_cdiv(M, 32), 2)](
            proj, left, right,
            M, N2,
            BLOCK_M=32, BLOCK_H=64,
            num_warps=4,
        )

    tri = torch.empty((B * H, N, N), device=x.device, dtype=torch.float16)
    torch.bmm(left, right.transpose(1, 2), out=tri)
    del left, right

    if D == 128:
        out = torch.empty((M, 128), device=x.device, dtype=torch.float32)
        _final_ln_gate_out_128_kernel[(_cdiv(M, 16),)](
            tri, proj,
            weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            wout, out,
            M, N2, eps,
            BLOCK_M=16,
            num_warps=4,
        )
        return out.view(B, N, N, 128)

    hidden = torch.empty((M, 128), device=x.device, dtype=torch.float16)
    _final_ln_gate_kernel[(_cdiv(M, 16),)](
        tri, proj,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        hidden,
        M, N2, eps,
        BLOCK_M=16,
        num_warps=4,
    )
    del tri, proj

    out = _mm32(hidden, wout)
    return out.view(B, N, N, D)
