"""
TriMul outgoing forward pass.

Algorithm:
1. Use PyTorch LayerNorm and one fused GEMM to produce all five projections.
2. A Triton kernel applies input gates, optional mask, and writes left/right in
   [B, H, N, N] layout for efficient batched matrix multiply.
3. Use torch.bmm for the outgoing triangle contraction:
      out[b, h] = left[b, h] @ right[b, h].T
4. A Triton kernel performs hidden-dim LayerNorm and applies the output gate.
5. Use one final GEMM for the output projection. The returned tensor is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_transpose_kernel(
    proj,
    mask,
    left_out,
    right_out,
    out_gate,
    M: tl.constexpr,
    N2: tl.constexpr,
    H: tl.constexpr,
    NOMASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = hs < H
    mask_2d = row_mask[:, None] & h_mask[None, :]

    proj_base = rows[:, None] * (5 * H) + hs[None, :]

    lp = tl.load(proj + proj_base, mask=mask_2d, other=0.0).to(tl.float32)
    rp = tl.load(proj + proj_base + H, mask=mask_2d, other=0.0).to(tl.float32)
    lg = tl.load(proj + proj_base + 2 * H, mask=mask_2d, other=0.0).to(tl.float32)
    rg = tl.load(proj + proj_base + 3 * H, mask=mask_2d, other=0.0).to(tl.float32)
    og = tl.load(proj + proj_base + 4 * H, mask=mask_2d, other=0.0).to(tl.float32)

    lg = tl.sigmoid(lg)
    rg = tl.sigmoid(rg)
    og = tl.sigmoid(og)

    mval = tl.full((BLOCK_M,), 1.0, tl.float32)
    if not NOMASK:
        mval = tl.load(mask + rows, mask=row_mask, other=0.0).to(tl.float32)

    left_val = lp * lg * mval[:, None]
    right_val = rp * rg * mval[:, None]

    b = rows // N2
    rem = rows - b * N2
    bhnn_ptrs = (b[:, None] * H + hs[None, :]) * N2 + rem[:, None]

    tl.store(left_out + bhnn_ptrs, left_val, mask=mask_2d)
    tl.store(right_out + bhnn_ptrs, right_val, mask=mask_2d)
    tl.store(out_gate + rows[:, None] * H + hs[None, :], og, mask=mask_2d)


@triton.jit
def _out_norm_gate_kernel(
    tri_out,
    out_gate,
    norm_w,
    norm_b,
    normed,
    M: tl.constexpr,
    N2: tl.constexpr,
    H: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    row_mask = rows < M
    h_mask = hs < H
    mask_2d = row_mask[:, None] & h_mask[None, :]

    b = rows // N2
    rem = rows - b * N2

    ptrs = (b[:, None] * H + hs[None, :]) * N2 + rem[:, None]
    vals = tl.load(tri_out + ptrs, mask=mask_2d, other=0.0).to(tl.float32)

    mean = tl.sum(vals, axis=1) / H
    centered = vals - mean[:, None]
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + hs, mask=h_mask, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + hs, mask=h_mask, other=0.0).to(tl.float32)
    gate = tl.load(out_gate + rows[:, None] * H + hs[None, :], mask=mask_2d, other=0.0).to(tl.float32)

    out = (centered * inv[:, None] * nw[None, :] + nb[None, :]) * gate
    tl.store(normed + rows[:, None] * H + hs[None, :], out, mask=mask_2d)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B, N, _, D = input_tensor.shape
    H = int(config["hidden_dim"])
    M = B * N * N
    N2 = N * N
    nomask = bool(config.get("nomask", False)) or mask is None

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x = F.layer_norm(input_tensor, (D,), norm_weight, norm_bias, 1e-5)
    x = x.reshape(M, D)

    proj_w = torch.cat(
        (
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
        ),
        dim=0,
    )
    proj = x.matmul(proj_w.t())

    left = torch.empty((B, H, N, N), device=input_tensor.device, dtype=torch.float32)
    right = torch.empty((B, H, N, N), device=input_tensor.device, dtype=torch.float32)
    out_gate = torch.empty((M, H), device=input_tensor.device, dtype=torch.float32)

    block_m = 16
    block_h = min(64, triton.next_power_of_2(H))
    grid_gate = (triton.cdiv(M, block_m), triton.cdiv(H, block_h))
    _gate_transpose_kernel[grid_gate](
        proj,
        mask,
        left,
        right,
        out_gate,
        M,
        N2,
        H,
        nomask,
        BLOCK_M=block_m,
        BLOCK_H=block_h,
        num_warps=4,
    )

    tri = torch.bmm(
        left.reshape(B * H, N, N),
        right.reshape(B * H, N, N).transpose(1, 2),
    ).reshape(B, H, N, N)

    normed = torch.empty((M, H), device=input_tensor.device, dtype=torch.float32)

    block_h_full = triton.next_power_of_2(H)
    block_m_norm = 8
    _out_norm_gate_kernel[(triton.cdiv(M, block_m_norm),)](
        tri,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        normed,
        M,
        N2,
        H,
        1e-5,
        BLOCK_M=block_m_norm,
        BLOCK_H=block_h_full,
        num_warps=4,
    )

    out = normed.matmul(to_out_weight.t()).reshape(B, N, N, D)
    return out.to(torch.float32)
