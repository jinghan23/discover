"""
TriMul outgoing forward pass.

The implementation uses PyTorch for the large dense input projection, then Triton
kernels to fuse projection post-processing into channel-major matrices and to fuse
the final hidden LayerNorm, output gate, and output projection. The N^3 triangular
multiplication is dispatched as a batched GEMM over hidden channels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


_WEIGHT_CACHE = {}


@triton.jit
def _prep_trimul_kernel(
    proj_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    gate_ptr,
    total: tl.constexpr,
    nn: tl.constexpr,
    hdim: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total

    m = offs // hdim
    h = offs - m * hdim
    b = m // nn
    pair = m - b * nn
    bh_pair = (b * hdim + h) * nn + pair

    base = m * (5 * hdim) + h
    l = tl.load(proj_ptr + base, mask=valid, other=0.0)
    r = tl.load(proj_ptr + base + hdim, mask=valid, other=0.0)
    lg = tl.load(proj_ptr + base + 2 * hdim, mask=valid, other=0.0)
    rg = tl.load(proj_ptr + base + 3 * hdim, mask=valid, other=0.0)
    og = tl.load(proj_ptr + base + 4 * hdim, mask=valid, other=0.0)

    l = l / (1.0 + tl.exp(-lg))
    r = r / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    if has_mask:
        mv = tl.load(mask_ptr + m, mask=valid, other=0.0)
        l *= mv
        r *= mv

    tl.store(left_ptr + bh_pair, l, mask=valid)
    tl.store(right_ptr + bh_pair, r, mask=valid)
    tl.store(gate_ptr + offs, og, mask=valid)


@triton.jit
def _final_trimul_kernel(
    tri_ptr,
    gate_ptr,
    ln_w_ptr,
    ln_b_ptr,
    out_w_ptr,
    out_ptr,
    rows: tl.constexpr,
    nn: tl.constexpr,
    dim: tl.constexpr,
    hdim: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    b = offs_m // nn
    pair = offs_m - b * nn

    vals = tl.load(
        tri_ptr + (b[:, None] * hdim + offs_h[None, :]) * nn + pair[:, None],
        mask=(offs_m[:, None] < rows) & (offs_h[None, :] < hdim),
        other=0.0,
    )

    mean = tl.sum(vals, axis=1) / hdim
    centered = vals - mean[:, None]
    var = tl.sum(centered * centered, axis=1) / hdim
    inv = tl.rsqrt(var + 1.0e-5)

    gamma = tl.load(ln_w_ptr + offs_h, mask=offs_h < hdim, other=0.0)
    beta = tl.load(ln_b_ptr + offs_h, mask=offs_h < hdim, other=0.0)
    gate = tl.load(
        gate_ptr + offs_m[:, None] * hdim + offs_h[None, :],
        mask=(offs_m[:, None] < rows) & (offs_h[None, :] < hdim),
        other=0.0,
    )

    hidden = centered * inv[:, None]
    hidden = (hidden * gamma[None, :] + beta[None, :]) * gate

    w = tl.load(
        out_w_ptr + offs_d[None, :] * hdim + offs_h[:, None],
        mask=(offs_d[None, :] < dim) & (offs_h[:, None] < hdim),
        other=0.0,
    )

    acc = tl.dot(hidden, w, input_precision="tf32")
    tl.store(
        out_ptr + offs_m[:, None] * dim + offs_d[None, :],
        acc,
        mask=(offs_m[:, None] < rows) & (offs_d[None, :] < dim),
    )


def _cached_proj_weight(weights):
    parts = (
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    )
    key = tuple((p.data_ptr(), tuple(p.shape), p.dtype, p.device.index) for p in parts)
    wt = _WEIGHT_CACHE.get(key)
    if wt is None:
        wt = torch.cat(parts, dim=0).t().contiguous()
        _WEIGHT_CACHE.clear()
        _WEIGHT_CACHE[key] = wt
    return wt


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, n, _, dim = input_tensor.shape
    hdim = int(config["hidden_dim"])
    rows = bsz * n * n
    nn = n * n

    torch.backends.cuda.matmul.allow_tf32 = True

    x = input_tensor.float()
    x = F.layer_norm(
        x,
        (dim,),
        weights["norm.weight"].float(),
        weights["norm.bias"].float(),
        eps=1.0e-5,
    )

    proj_wt = _cached_proj_weight(weights).float()
    proj = torch.mm(x.reshape(rows, dim), proj_wt)

    left = torch.empty((bsz * hdim, n, n), device=x.device, dtype=torch.float32)
    right = torch.empty_like(left)
    gate = torch.empty((rows, hdim), device=x.device, dtype=torch.float32)

    has_mask = (mask is not None) and (not bool(config.get("nomask", False)))
    mask_arg = mask.contiguous() if has_mask else input_tensor

    total = rows * hdim
    _prep_trimul_kernel[(triton.cdiv(total, 256),)](
        proj,
        mask_arg,
        left,
        right,
        gate,
        total,
        nn,
        hdim,
        has_mask,
        BLOCK=256,
        num_warps=4,
    )

    tri = torch.bmm(left, right.transpose(1, 2))

    out = torch.empty((rows, dim), device=x.device, dtype=torch.float32)
    _final_trimul_kernel[(triton.cdiv(rows, 16), triton.cdiv(dim, 64))](
        tri,
        gate,
        weights["to_out_norm.weight"].float(),
        weights["to_out_norm.bias"].float(),
        weights["to_out.weight"].float(),
        out,
        rows,
        nn,
        dim,
        hdim,
        BLOCK_M=16,
        BLOCK_D=64,
        BLOCK_H=128,
        num_warps=4,
    )

    return out.reshape(bsz, n, n, dim).float()
