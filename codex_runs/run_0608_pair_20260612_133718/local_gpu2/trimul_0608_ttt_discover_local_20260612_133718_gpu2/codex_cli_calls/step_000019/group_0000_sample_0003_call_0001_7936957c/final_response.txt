"""
TriMul outgoing forward pass.

Uses Triton kernels for both LayerNorm stages, gate/mask packing, and output-gate
fusion. The expensive N^3 outgoing contraction is executed as grouped fp16 cuBLAS
BMM over hidden channels after packing left/right into channel-major [B*H,N,N]
layout. Projection and output weights plus scratch buffers are cached across calls.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_BACKENDS_SET = False
_MM_OUT_DTYPE_OK = None


@triton.jit
def _layernorm128_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    ok = rows < M

    v = tl.load(x + rows[:, None] * 128 + ds[None, :], mask=ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(v, axis=1) * 0.0078125
    c = v - mean[:, None]
    var = tl.sum(c * c, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds).to(tl.float32)
    bb = tl.load(b + ds).to(tl.float32)
    tl.store(y + rows[:, None] * 128 + ds[None, :], c * inv[:, None] * ww[None, :] + bb[None, :], mask=ok[:, None])


@triton.jit
def _layernorm384_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    ok = rows < M
    base = rows[:, None] * 384 + ds[None, :]

    v0 = tl.load(x + base, mask=ok[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + base + 128, mask=ok[:, None], other=0.0).to(tl.float32)
    v2 = tl.load(x + base + 256, mask=ok[:, None], other=0.0).to(tl.float32)

    mean = (tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1)) * 0.0026041666666666665
    c0 = v0 - mean[:, None]
    c1 = v1 - mean[:, None]
    c2 = v2 - mean[:, None]
    var = (tl.sum(c0 * c0, axis=1) + tl.sum(c1 * c1, axis=1) + tl.sum(c2 * c2, axis=1)) * 0.0026041666666666665
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + ds).to(tl.float32)
    w1 = tl.load(w + ds + 128).to(tl.float32)
    w2 = tl.load(w + ds + 256).to(tl.float32)
    b0 = tl.load(b + ds).to(tl.float32)
    b1 = tl.load(b + ds + 128).to(tl.float32)
    b2 = tl.load(b + ds + 256).to(tl.float32)

    tl.store(y + base, c0 * inv[:, None] * w0[None, :] + b0[None, :], mask=ok[:, None])
    tl.store(y + base + 128, c1 * inv[:, None] * w1[None, :] + b1[None, :] + 0.0, mask=ok[:, None])
    tl.store(y + base + 256, c2 * inv[:, None] * w2[None, :] + b2[None, :] + 0.0, mask=ok[:, None])


@triton.jit
def _layernorm768_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    ok = rows < M
    base = rows[:, None] * 768 + ds[None, :]

    v0 = tl.load(x + base, mask=ok[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + base + 128, mask=ok[:, None], other=0.0).to(tl.float32)
    v2 = tl.load(x + base + 256, mask=ok[:, None], other=0.0).to(tl.float32)
    v3 = tl.load(x + base + 384, mask=ok[:, None], other=0.0).to(tl.float32)
    v4 = tl.load(x + base + 512, mask=ok[:, None], other=0.0).to(tl.float32)
    v5 = tl.load(x + base + 640, mask=ok[:, None], other=0.0).to(tl.float32)

    mean = (
        tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1)
        + tl.sum(v3, axis=1) + tl.sum(v4, axis=1) + tl.sum(v5, axis=1)
    ) * 0.0013020833333333333

    c0 = v0 - mean[:, None]
    c1 = v1 - mean[:, None]
    c2 = v2 - mean[:, None]
    c3 = v3 - mean[:, None]
    c4 = v4 - mean[:, None]
    c5 = v5 - mean[:, None]

    var = (
        tl.sum(c0 * c0, axis=1) + tl.sum(c1 * c1, axis=1) + tl.sum(c2 * c2, axis=1)
        + tl.sum(c3 * c3, axis=1) + tl.sum(c4 * c4, axis=1) + tl.sum(c5 * c5, axis=1)
    ) * 0.0013020833333333333
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + ds).to(tl.float32)
    w1 = tl.load(w + ds + 128).to(tl.float32)
    w2 = tl.load(w + ds + 256).to(tl.float32)
    w3 = tl.load(w + ds + 384).to(tl.float32)
    w4 = tl.load(w + ds + 512).to(tl.float32)
    w5 = tl.load(w + ds + 640).to(tl.float32)
    b0 = tl.load(b + ds).to(tl.float32)
    b1 = tl.load(b + ds + 128).to(tl.float32)
    b2 = tl.load(b + ds + 256).to(tl.float32)
    b3 = tl.load(b + ds + 384).to(tl.float32)
    b4 = tl.load(b + ds + 512).to(tl.float32)
    b5 = tl.load(b + ds + 640).to(tl.float32)

    tl.store(y + base, c0 * inv[:, None] * w0[None, :] + b0[None, :], mask=ok[:, None])
    tl.store(y + base + 128, c1 * inv[:, None] * w1[None, :] + b1[None, :], mask=ok[:, None])
    tl.store(y + base + 256, c2 * inv[:, None] * w2[None, :] + b2[None, :], mask=ok[:, None])
    tl.store(y + base + 384, c3 * inv[:, None] * w3[None, :] + b3[None, :], mask=ok[:, None])
    tl.store(y + base + 512, c4 * inv[:, None] * w4[None, :] + b4[None, :], mask=ok[:, None])
    tl.store(y + base + 640, c5 * inv[:, None] * w5[None, :] + b5[None, :], mask=ok[:, None])


@triton.jit
def _pack_nomask_h128(proj, left, right, M: tl.constexpr, N: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    ok = rows < M

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn
    p = rows[:, None] * 640 + hs[None, :]
    o = b[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]

    l = tl.load(proj + p, mask=ok[:, None], other=0.0).to(tl.float32)
    r = tl.load(proj + p + 128, mask=ok[:, None], other=0.0).to(tl.float32)
    lg = tl.load(proj + p + 256, mask=ok[:, None], other=0.0).to(tl.float32)
    rg = tl.load(proj + p + 384, mask=ok[:, None], other=0.0).to(tl.float32)

    tl.store(left + o, tl.trans(l * tl.sigmoid(lg)), mask=ok[None, :])
    tl.store(right + o, tl.trans(r * tl.sigmoid(rg)), mask=ok[None, :])


@triton.jit
def _pack_mask_h128(proj, mask, left, right, M: tl.constexpr, N: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    ok = rows < M

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn
    p = rows[:, None] * 640 + hs[None, :]
    o = b[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    mv = tl.load(mask + rows, mask=ok, other=0.0).to(tl.float32)

    l = tl.load(proj + p, mask=ok[:, None], other=0.0).to(tl.float32)
    r = tl.load(proj + p + 128, mask=ok[:, None], other=0.0).to(tl.float32)
    lg = tl.load(proj + p + 256, mask=ok[:, None], other=0.0).to(tl.float32)
    rg = tl.load(proj + p + 384, mask=ok[:, None], other=0.0).to(tl.float32)

    tl.store(left + o, tl.trans(l * mv[:, None] * tl.sigmoid(lg)), mask=ok[None, :])
    tl.store(right + o, tl.trans(r * mv[:, None] * tl.sigmoid(rg)), mask=ok[None, :])


@triton.jit
def _post_h128(tri, proj, nw, nb, tmp, M: tl.constexpr, N: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    ok = rows < M

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn
    ptr = tri + b[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(ptr, mask=ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) * 0.0078125
    c = x - mean[:, None]
    var = tl.sum(c * c, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    w = tl.load(nw + hs).to(tl.float32)
    bias = tl.load(nb + hs).to(tl.float32)
    g = tl.load(proj + rows[:, None] * 640 + 512 + hs[None, :], mask=ok[:, None], other=0.0).to(tl.float32)
    y = (c * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(g)
    tl.store(tmp + rows[:, None] * 128 + hs[None, :], y, mask=ok[:, None])


def _tensor_key(t):
    return (
        t.data_ptr(),
        tuple(t.shape),
        str(t.dtype),
        t.device.index if t.device.type == "cuda" else -1,
        getattr(t, "_version", 0),
    )


def _get_buffer(name, shape, device, dtype):
    key = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    buf = _BUFFER_CACHE.get(key)
    if buf is None:
        buf = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = buf
    return buf


def _get_packed_proj_t(weights, dtype):
    names = (
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    )
    src = tuple(weights[n] for n in names)
    key = ("packed_proj_t", dtype) + tuple(_tensor_key(t) for t in src)
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = torch.cat(src, dim=0).to(dtype).t().contiguous()
        _WEIGHT_CACHE[key] = item
    return item


def _get_to_out_t(w, dtype):
    key = ("to_out_t", dtype, _tensor_key(w))
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = w.to(dtype).t().contiguous()
        _WEIGHT_CACHE[key] = item
    return item


def _setup_backends_once():
    global _BACKENDS_SET
    if not _BACKENDS_SET:
        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        try:
            torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass
        _BACKENDS_SET = True


def _mm_to_float(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is not False:
        try:
            out = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return out
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.inference_mode():
        _setup_backends_once()

        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()
        xh = _get_buffer("x_norm_h", (M, D), device, half)

        if D == 128:
            bm_ln = 8
            _layernorm128_kernel[(triton.cdiv(M, bm_ln),)](
                x, weights["norm.weight"], weights["norm.bias"], xh, M, eps,
                BLOCK_M=bm_ln, num_warps=4,
            )
        elif D == 384:
            bm_ln = 4
            _layernorm384_kernel[(triton.cdiv(M, bm_ln),)](
                x, weights["norm.weight"], weights["norm.bias"], xh, M, eps,
                BLOCK_M=bm_ln, num_warps=4,
            )
        else:
            bm_ln = 2
            _layernorm768_kernel[(triton.cdiv(M, bm_ln),)](
                x, weights["norm.weight"], weights["norm.bias"], xh, M, eps,
                BLOCK_M=bm_ln, num_warps=4,
            )

        packed_proj_t = _get_packed_proj_t(weights, half)
        proj = _get_buffer("proj", (M, 5 * H), device, half)
        torch.mm(xh, packed_proj_t, out=proj)

        left = _get_buffer("left", (B * H, N, N), device, half)
        right = _get_buffer("right", (B * H, N, N), device, half)

        bm_pack = 16
        bh_pack = 64
        has_mask = mask is not None and not bool(config.get("nomask", False))

        if has_mask:
            mask_flat = mask.reshape(M).contiguous()
            _pack_mask_h128[(triton.cdiv(M, bm_pack), 2)](
                proj, mask_flat, left, right, M, N,
                BLOCK_M=bm_pack, BLOCK_H=bh_pack, num_warps=4,
            )
        else:
            _pack_nomask_h128[(triton.cdiv(M, bm_pack), 2)](
                proj, left, right, M, N,
                BLOCK_M=bm_pack, BLOCK_H=bh_pack, num_warps=4,
            )

        tri = _get_buffer("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        tmp = _get_buffer("tmp", (M, H), device, half)
        bm_post = 16
        _post_h128[(triton.cdiv(M, bm_post),)](
            tri, proj,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            tmp,
            M, N, eps,
            BLOCK_M=bm_post,
            num_warps=8,
        )

        to_out_t = _get_to_out_t(weights["to_out.weight"], half)
        out = _mm_to_float(tmp, to_out_t)
        return out.reshape(B, N, N, D)
