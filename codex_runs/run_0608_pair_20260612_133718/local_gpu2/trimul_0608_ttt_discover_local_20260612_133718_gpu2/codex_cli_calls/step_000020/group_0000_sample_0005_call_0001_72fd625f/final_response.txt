"""
TriMul outgoing forward pass using cuBLAS for the large tensor-core GEMMs and
Triton for the surrounding LayerNorm, gating, layout packing, and post-normalize
work. Large-N cases split out_gate onto a side CUDA stream so that its projection
can overlap the outgoing contraction.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_STREAM_CACHE = {}
_MM_OUT_DTYPE_OUT_OK = None
_MM_OUT_DTYPE_OK = None


@triton.jit
def _layernorm_cast_kernel(
    x, w, b, y,
    M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, BLOCK_D)
    row_ok = rows < M
    d_ok = ds < D
    mask = row_ok[:, None] & d_ok[None, :]

    vals = tl.load(x + rows[:, None] * D + ds[None, :], mask=mask, other=0.0).to(tl.float32)
    vals = tl.where(d_ok[None, :], vals, 0.0)

    mean = tl.sum(vals, axis=1) / D
    centered = tl.where(d_ok[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=d_ok, other=0.0).to(tl.float32)
    bb = tl.load(b + ds, mask=d_ok, other=0.0).to(tl.float32)
    tl.store(y + rows[:, None] * D + ds[None, :], centered * inv[:, None] * ww[None, :] + bb[None, :], mask=mask)


@triton.jit
def _prep_nomask_kernel(
    proj, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, PROJ_STRIDE: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * PROJ_STRIDE + hs[None, :]
    out_ptrs = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    out_mask = h_ok[:, None] & row_ok[None, :]

    left = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(left_out + out_ptrs, tl.trans(left * tl.sigmoid(lg)), mask=out_mask)

    right = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(right_out + out_ptrs, tl.trans(right * tl.sigmoid(rg)), mask=out_mask)


@triton.jit
def _prep_mask_kernel(
    proj, mask, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, PROJ_STRIDE: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * PROJ_STRIDE + hs[None, :]
    out_ptrs = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    out_mask = h_ok[:, None] & row_ok[None, :]

    mv = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    left = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(left_out + out_ptrs, tl.trans(left * mv[:, None] * tl.sigmoid(lg)), mask=out_mask)

    right = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(right_out + out_ptrs, tl.trans(right * mv[:, None] * tl.sigmoid(rg)), mask=out_mask)


@triton.jit
def _post_norm_proj_kernel(
    tri, proj, norm_w, norm_b, tmp,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, PROJ_STRIDE: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)
    row_ok = rows < M
    h_ok = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    tri_ptrs = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(tri_ptrs, mask=h_ok[:, None] & row_ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) / H
    centered = tl.where(h_ok[None, :], x - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs, mask=h_ok, other=0.0).to(tl.float32)
    bias = tl.load(norm_b + hs, mask=h_ok, other=0.0).to(tl.float32)
    valid = row_ok[:, None] & h_ok[None, :]
    gate = tl.load(proj + rows[:, None] * PROJ_STRIDE + 4 * H + hs[None, :], mask=valid, other=0.0).to(tl.float32)

    y = (centered * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gate)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y, mask=valid)


@triton.jit
def _post_norm_gate_kernel(
    tri, gate, norm_w, norm_b, tmp,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)
    row_ok = rows < M
    h_ok = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    tri_ptrs = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(tri_ptrs, mask=h_ok[:, None] & row_ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) / H
    centered = tl.where(h_ok[None, :], x - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs, mask=h_ok, other=0.0).to(tl.float32)
    bias = tl.load(norm_b + hs, mask=h_ok, other=0.0).to(tl.float32)
    valid = row_ok[:, None] & h_ok[None, :]
    gv = tl.load(gate + rows[:, None] * H + hs[None, :], mask=valid, other=0.0).to(tl.float32)

    y = (centered * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gv)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y, mask=valid)


def _tensor_key(t):
    return (
        t.data_ptr(),
        tuple(t.shape),
        str(t.dtype),
        t.device.index if t.device.type == "cuda" else -1,
        getattr(t, "_version", 0),
    )


def _get_buffer(name, shape, device, dtype):
    key = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    buf = _BUFFER_CACHE.get(key)
    if buf is None:
        buf = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = buf
    return buf


def _get_stream(device):
    key = device.index if device.type == "cuda" else -1
    stream = _STREAM_CACHE.get(key)
    if stream is None:
        stream = torch.cuda.Stream(device=device)
        _STREAM_CACHE[key] = stream
    return stream


def _get_packed_proj_t(weights, dtype, split_gate):
    if split_gate:
        names = ("left_proj.weight", "right_proj.weight", "left_gate.weight", "right_gate.weight")
        tag = "packed_proj4_t"
    else:
        names = ("left_proj.weight", "right_proj.weight", "left_gate.weight", "right_gate.weight", "out_gate.weight")
        tag = "packed_proj5_t"
    src = tuple(weights[n] for n in names)
    key = (tag, dtype) + tuple(_tensor_key(t) for t in src)
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = (src, torch.cat(src, dim=0).to(dtype).t().contiguous())
        _WEIGHT_CACHE[key] = item
    return item[1]


def _get_linear_t(name, w, dtype):
    key = (name, dtype, _tensor_key(w))
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = (w, w.to(dtype).t().contiguous())
        _WEIGHT_CACHE[key] = item
    return item[1]


def _mm_half_to_float(a, b, shape, device):
    global _MM_OUT_DTYPE_OUT_OK, _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OUT_OK is not False:
        out = torch.empty(shape, device=device, dtype=torch.float32)
        try:
            torch.mm(a, b, out=out, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OUT_OK = True
            return out
        except (TypeError, RuntimeError):
            _MM_OUT_DTYPE_OUT_OK = False

    if _MM_OUT_DTYPE_OK is not False:
        try:
            out = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return out
        except (TypeError, RuntimeError):
            _MM_OUT_DTYPE_OK = False

    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        try:
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()

        use_split_gate = N >= 512
        packed_proj_t = _get_packed_proj_t(weights, half, use_split_gate)
        out_gate_t = _get_linear_t("out_gate_t", weights["out_gate.weight"], half) if use_split_gate else None
        to_out_t = _get_linear_t("to_out_t", weights["to_out.weight"], half)

        xh = _get_buffer("x_norm_h", (M, D), device, half)
        bd = triton.next_power_of_2(D)
        bm_ln = 8 if D <= 128 else 4
        _layernorm_cast_kernel[(triton.cdiv(M, bm_ln),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, D, eps,
            BLOCK_M=bm_ln, BLOCK_D=bd,
            num_warps=4 if D <= 128 else 8,
        )

        if use_split_gate:
            gate = _get_buffer("out_gate", (M, H), device, half)
            cur_stream = torch.cuda.current_stream(device)
            gate_stream = _get_stream(device)
            gate_stream.wait_stream(cur_stream)
            with torch.cuda.stream(gate_stream):
                torch.mm(xh, out_gate_t, out=gate)

            proj = _get_buffer("proj4", (M, 4 * H), device, half)
            torch.mm(xh, packed_proj_t, out=proj)
            proj_stride = 4 * H
        else:
            gate = None
            gate_stream = None
            cur_stream = None
            proj = _get_buffer("proj5", (M, 5 * H), device, half)
            torch.mm(xh, packed_proj_t, out=proj)
            proj_stride = 5 * H

        left = _get_buffer("left", (B * H, N, N), device, half)
        right = _get_buffer("right", (B * H, N, N), device, half)

        bm_prep = 16
        bh_prep = 64
        has_mask = mask is not None and not bool(config.get("nomask", False))
        if has_mask:
            mask_flat = mask.reshape(M).contiguous()
            _prep_mask_kernel[(triton.cdiv(M, bm_prep), triton.cdiv(H, bh_prep))](
                proj, mask_flat, left, right,
                M, N, H, proj_stride,
                BLOCK_M=bm_prep, BLOCK_H=bh_prep,
                num_warps=4,
            )
        else:
            _prep_nomask_kernel[(triton.cdiv(M, bm_prep), triton.cdiv(H, bh_prep))](
                proj, left, right,
                M, N, H, proj_stride,
                BLOCK_M=bm_prep, BLOCK_H=bh_prep,
                num_warps=4,
            )

        tri = _get_buffer("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        if use_split_gate:
            cur_stream.wait_stream(gate_stream)

        tmp = _get_buffer("tmp", (M, H), device, half)
        bm_post = 16
        bh_post = triton.next_power_of_2(H)
        if use_split_gate:
            _post_norm_gate_kernel[(triton.cdiv(M, bm_post),)](
                tri, gate, weights["to_out_norm.weight"], weights["to_out_norm.bias"], tmp,
                M, N, H, eps,
                BLOCK_M=bm_post, BLOCK_H=bh_post,
                num_warps=8,
            )
        else:
            _post_norm_proj_kernel[(triton.cdiv(M, bm_post),)](
                tri, proj, weights["to_out_norm.weight"], weights["to_out_norm.bias"], tmp,
                M, N, H, proj_stride, eps,
                BLOCK_M=bm_post, BLOCK_H=bh_post,
                num_warps=8,
            )

        out = _mm_half_to_float(tmp, to_out_t, (M, D), device)
        return out.reshape(B, N, N, D)
