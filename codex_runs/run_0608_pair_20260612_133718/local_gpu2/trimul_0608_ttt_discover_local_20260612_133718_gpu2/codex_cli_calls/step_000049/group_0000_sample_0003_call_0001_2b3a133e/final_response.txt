"""
TriMul outgoing forward using a mixed Triton/cuBLAS pipeline.

For the common dim=hidden_dim=128 path, Triton fuses input LayerNorm, the five
input projections, masking/gating, layout conversion for batched GEMM, and the
post LayerNorm/gate/final projection. The N^3 channelwise products use cuBLAS
batched GEMM. Wider input dimensions use the robust packed-GEMM fallback path.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_MM_OUT_DTYPE_OK = None
_MATMUL_FLAGS_SET = False


@triton.jit
def _ln_proj_prep_nomask_d128_h128_kernel(
    x, norm_w, norm_b, packed, left_out, right_out, out_gate_sig,
    M: tl.constexpr, N: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    vals = tl.load(x + rows[:, None] * 128 + ds[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    cen = vals - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + ds).to(tl.float32)
    nb = tl.load(norm_b + ds).to(tl.float32)
    xn = (cen * inv[:, None] * nw[None, :] + nb[None, :]).to(tl.float16)

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn
    out_ptrs = bidx[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]

    wl = tl.load(packed + ds[:, None] * 640 + hs[None, :])
    wlg = tl.load(packed + ds[:, None] * 640 + 256 + hs[None, :])
    lv = tl.dot(xn, wl)
    lgv = tl.dot(xn, wlg)
    tl.store(left_out + out_ptrs, tl.trans(lv * tl.sigmoid(lgv)), mask=row_ok[None, :])

    wr = tl.load(packed + ds[:, None] * 640 + 128 + hs[None, :])
    wrg = tl.load(packed + ds[:, None] * 640 + 384 + hs[None, :])
    rv = tl.dot(xn, wr)
    rgv = tl.dot(xn, wrg)
    tl.store(right_out + out_ptrs, tl.trans(rv * tl.sigmoid(rgv)), mask=row_ok[None, :])

    wog = tl.load(packed + ds[:, None] * 640 + 512 + hs[None, :])
    og = tl.dot(xn, wog)
    tl.store(out_gate_sig + rows[:, None] * 128 + hs[None, :], tl.sigmoid(og), mask=row_ok[:, None])


@triton.jit
def _ln_proj_prep_mask_d128_h128_kernel(
    x, mask, norm_w, norm_b, packed, left_out, right_out, out_gate_sig,
    M: tl.constexpr, N: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    vals = tl.load(x + rows[:, None] * 128 + ds[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    cen = vals - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + ds).to(tl.float32)
    nb = tl.load(norm_b + ds).to(tl.float32)
    xn = (cen * inv[:, None] * nw[None, :] + nb[None, :]).to(tl.float16)
    mv = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn
    out_ptrs = bidx[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]

    wl = tl.load(packed + ds[:, None] * 640 + hs[None, :])
    wlg = tl.load(packed + ds[:, None] * 640 + 256 + hs[None, :])
    lv = tl.dot(xn, wl)
    lgv = tl.dot(xn, wlg)
    tl.store(left_out + out_ptrs, tl.trans(lv * tl.sigmoid(lgv) * mv[:, None]), mask=row_ok[None, :])

    wr = tl.load(packed + ds[:, None] * 640 + 128 + hs[None, :])
    wrg = tl.load(packed + ds[:, None] * 640 + 384 + hs[None, :])
    rv = tl.dot(xn, wr)
    rgv = tl.dot(xn, wrg)
    tl.store(right_out + out_ptrs, tl.trans(rv * tl.sigmoid(rgv) * mv[:, None]), mask=row_ok[None, :])

    wog = tl.load(packed + ds[:, None] * 640 + 512 + hs[None, :])
    og = tl.dot(xn, wog)
    tl.store(out_gate_sig + rows[:, None] * 128 + hs[None, :], tl.sigmoid(og), mask=row_ok[:, None])


@triton.jit
def _post_out_d128_h128_kernel(
    tri, out_gate_sig, norm_w, norm_b, to_out_t, out,
    M: tl.constexpr, N: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn

    tri_ptrs = tri + bidx[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(tri_ptrs, mask=row_ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) * 0.0078125
    cen = x - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs).to(tl.float32)
    b = tl.load(norm_b + hs).to(tl.float32)
    gate = tl.load(out_gate_sig + rows[:, None] * 128 + hs[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)

    y = ((cen * inv[:, None] * w[None, :] + b[None, :]) * gate).to(tl.float16)
    ow = tl.load(to_out_t + hs[:, None] * 128 + hs[None, :])
    z = tl.dot(y, ow)
    tl.store(out + rows[:, None] * 128 + hs[None, :], z, mask=row_ok[:, None])


@triton.jit
def _layernorm128_cast_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    row_ok = rows < M

    vals = tl.load(x + rows[:, None] * 128 + ds[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    cen = vals - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds).to(tl.float32)
    bb = tl.load(b + ds).to(tl.float32)
    out = cen * inv[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * 128 + ds[None, :], out, mask=row_ok[:, None])


@triton.jit
def _layernorm_cast_kernel(
    x, w, b, y,
    M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, BLOCK_D)

    row_ok = rows < M
    d_ok = ds < D
    mask = row_ok[:, None] & d_ok[None, :]

    vals = tl.load(x + rows[:, None] * D + ds[None, :], mask=mask, other=0.0).to(tl.float32)
    vals = tl.where(d_ok[None, :], vals, 0.0)

    mean = tl.sum(vals, axis=1) / D
    cen = tl.where(d_ok[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(cen * cen, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=d_ok, other=0.0).to(tl.float32)
    bb = tl.load(b + ds, mask=d_ok, other=0.0).to(tl.float32)
    out = cen * inv[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * D + ds[None, :], out, mask=mask)


@triton.jit
def _prep_nomask_h128_kernel(
    proj, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn

    base = rows[:, None] * 640 + hs[None, :]
    out_ptrs = bidx[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]

    left = tl.load(proj + base, mask=row_ok[:, None], other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 256, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(left_out + out_ptrs, tl.trans(left * tl.sigmoid(lg)), mask=row_ok[None, :])

    right = tl.load(proj + base + 128, mask=row_ok[:, None], other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 384, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(right_out + out_ptrs, tl.trans(right * tl.sigmoid(rg)), mask=row_ok[None, :])


@triton.jit
def _prep_mask_h128_kernel(
    proj, mask, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn

    base = rows[:, None] * 640 + hs[None, :]
    out_ptrs = bidx[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    mv = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    left = tl.load(proj + base, mask=row_ok[:, None], other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 256, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(left_out + out_ptrs, tl.trans(left * mv[:, None] * tl.sigmoid(lg)), mask=row_ok[None, :])

    right = tl.load(proj + base + 128, mask=row_ok[:, None], other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 384, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(right_out + out_ptrs, tl.trans(right * mv[:, None] * tl.sigmoid(rg)), mask=row_ok[None, :])


@triton.jit
def _post_norm_gate_h128_kernel(
    tri, proj, norm_w, norm_b, tmp,
    M: tl.constexpr, N: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn

    tri_ptrs = tri + bidx[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(tri_ptrs, mask=row_ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) * 0.0078125
    cen = x - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs).to(tl.float32)
    bias = tl.load(norm_b + hs).to(tl.float32)
    gate = tl.load(proj + rows[:, None] * 640 + 512 + hs[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)

    y = (cen * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gate)
    tl.store(tmp + rows[:, None] * 128 + hs[None, :], y, mask=row_ok[:, None])


def _set_matmul_flags():
    global _MATMUL_FLAGS_SET
    if not _MATMUL_FLAGS_SET:
        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        try:
            torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass
        _MATMUL_FLAGS_SET = True


def _tensor_key(t):
    return (
        t.data_ptr(),
        tuple(t.shape),
        str(t.dtype),
        t.device.index if t.device.type == "cuda" else -1,
        getattr(t, "_version", 0),
    )


def _get_buffer(name, shape, device, dtype):
    key = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    buf = _BUFFER_CACHE.get(key)
    if buf is None:
        buf = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = buf
    return buf


def _get_packed_proj_t(weights, dtype):
    names = (
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    )
    src = tuple(weights[n] for n in names)
    key = ("packed_proj_t", dtype) + tuple(_tensor_key(t) for t in src)
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = (src, torch.cat(src, dim=0).to(dtype).t().contiguous())
        _WEIGHT_CACHE[key] = item
    return item[1]


def _get_to_out_t(w, dtype):
    key = ("to_out_t", dtype, _tensor_key(w))
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = (w, w.to(dtype).t().contiguous())
        _WEIGHT_CACHE[key] = item
    return item[1]


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is not False:
        try:
            out = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return out
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        _set_matmul_flags()

        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()
        has_mask = mask is not None and not bool(config.get("nomask", False))

        if D == 128 and H == 128:
            packed_proj_t = _get_packed_proj_t(weights, half)
            left = _get_buffer("left_fast128", (B * 128, N, N), device, half)
            right = _get_buffer("right_fast128", (B * 128, N, N), device, half)
            out_gate_sig = _get_buffer("out_gate_sig128", (M, 128), device, half)

            bm = 16
            grid = (triton.cdiv(M, bm),)
            if has_mask:
                mask_flat = mask.reshape(M).contiguous()
                _ln_proj_prep_mask_d128_h128_kernel[grid](
                    x, mask_flat,
                    weights["norm.weight"], weights["norm.bias"],
                    packed_proj_t, left, right, out_gate_sig,
                    M, N, eps, BLOCK_M=bm,
                    num_warps=4,
                )
            else:
                _ln_proj_prep_nomask_d128_h128_kernel[grid](
                    x,
                    weights["norm.weight"], weights["norm.bias"],
                    packed_proj_t, left, right, out_gate_sig,
                    M, N, eps, BLOCK_M=bm,
                    num_warps=4,
                )

            tri = _get_buffer("tri_fast128", (B * 128, N, N), device, half)
            torch.bmm(left, right.transpose(1, 2), out=tri)

            out = torch.empty((M, 128), device=device, dtype=torch.float32)
            to_out_t = _get_to_out_t(weights["to_out.weight"], half)
            _post_out_d128_h128_kernel[grid](
                tri, out_gate_sig,
                weights["to_out_norm.weight"],
                weights["to_out_norm.bias"],
                to_out_t,
                out,
                M, N, eps,
                BLOCK_M=bm,
                num_warps=4,
            )
            return out.reshape(B, N, N, D)

        xh = _get_buffer("x_norm_h", (M, D), device, half)
        if D == 128:
            bm_ln = 8
            _layernorm128_cast_kernel[(triton.cdiv(M, bm_ln),)](
                x, weights["norm.weight"], weights["norm.bias"], xh,
                M, eps, BLOCK_M=bm_ln, num_warps=4,
            )
        else:
            bd = triton.next_power_of_2(D)
            bm_ln = 4 if D >= 768 else 8
            _layernorm_cast_kernel[(triton.cdiv(M, bm_ln),)](
                x, weights["norm.weight"], weights["norm.bias"], xh,
                M, D, eps, BLOCK_M=bm_ln, BLOCK_D=bd,
                num_warps=8,
            )

        packed_proj_t = _get_packed_proj_t(weights, half)
        proj = _get_buffer("proj", (M, 5 * H), device, half)
        torch.mm(xh, packed_proj_t, out=proj)

        left = _get_buffer("left", (B * H, N, N), device, half)
        right = _get_buffer("right", (B * H, N, N), device, half)

        grid_prep = (triton.cdiv(M, 16), 2)
        if has_mask:
            mask_flat = mask.reshape(M).contiguous()
            _prep_mask_h128_kernel[grid_prep](
                proj, mask_flat, left, right, M, N,
                BLOCK_M=16, BLOCK_H=64, num_warps=4,
            )
        else:
            _prep_nomask_h128_kernel[grid_prep](
                proj, left, right, M, N,
                BLOCK_M=16, BLOCK_H=64, num_warps=4,
            )

        tri = _get_buffer("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        tmp = xh if D == H else _get_buffer("tmp", (M, H), device, half)
        _post_norm_gate_h128_kernel[(triton.cdiv(M, 16),)](
            tri, proj,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            tmp,
            M, N, eps,
            BLOCK_M=16,
            num_warps=4,
        )

        to_out_t = _get_to_out_t(weights["to_out.weight"], half)
        out = _mm_half_to_float(tmp, to_out_t)
        return out.reshape(B, N, N, D)
