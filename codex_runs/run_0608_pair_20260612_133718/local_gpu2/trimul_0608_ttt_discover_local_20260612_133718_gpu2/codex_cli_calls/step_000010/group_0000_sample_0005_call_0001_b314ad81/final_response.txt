"""
TriMul outgoing forward pass for SM80.

The input LayerNorm is computed in Triton and cast to fp16, dense projections use
cuBLAS tensor cores, and left/right activations are packed channel-major so the
TriMul contraction is a strided batched GEMM over B*H independent N x N products.
For D=H=128, the output LayerNorm, out-gate projection, gating, and final output
projection are fused into one Triton kernel to avoid materializing the out gate
and intermediate post-gate tensor. Wider output dimensions use a cuBLAS final GEMM.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_MM_OUT_DTYPE_OK = None


@triton.jit
def _layernorm_cast_kernel(
    x,
    w,
    b,
    y,
    M: tl.constexpr,
    D: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, BLOCK_D)

    row_ok = rows < M
    d_ok = ds < D
    mask = row_ok[:, None] & d_ok[None, :]

    vals = tl.load(x + rows[:, None] * D + ds[None, :], mask=mask, other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) / D
    centered = tl.where(d_ok[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=d_ok, other=0.0).to(tl.float32)
    bb = tl.load(b + ds, mask=d_ok, other=0.0).to(tl.float32)
    out = centered * inv[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * D + ds[None, :], out, mask=mask)


@triton.jit
def _prep_projected_kernel(
    proj,
    mask,
    left_out,
    right_out,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    P: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    proj_base = rows[:, None] * (P * H) + hs[None, :]
    out_ptrs = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    out_mask = h_ok[:, None] & row_ok[None, :]

    scale = tl.full((BLOCK_M,), 1.0, tl.float32)
    if HAS_MASK:
        scale = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    left = tl.load(proj + proj_base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + proj_base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(left_out + out_ptrs, tl.trans(left * scale[:, None] * tl.sigmoid(lg)), mask=out_mask)

    right = tl.load(proj + proj_base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + proj_base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(right_out + out_ptrs, tl.trans(right * scale[:, None] * tl.sigmoid(rg)), mask=out_mask)


@triton.jit
def _post_norm_gate_from_proj_kernel(
    tri_out,
    proj,
    norm_w,
    norm_b,
    tmp,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    P: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    tri_ptrs = tri_out + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    tri_t = tl.load(tri_ptrs, mask=h_ok[:, None] & row_ok[None, :], other=0.0).to(tl.float32)
    x = tl.trans(tri_t)

    mean = tl.sum(x, axis=1) / H
    centered = tl.where(h_ok[None, :], x - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs, mask=h_ok, other=0.0).to(tl.float32)
    bias = tl.load(norm_b + hs, mask=h_ok, other=0.0).to(tl.float32)

    valid = row_ok[:, None] & h_ok[None, :]
    gate_logits = tl.load(
        proj + rows[:, None] * (P * H) + 4 * H + hs[None, :],
        mask=valid,
        other=0.0,
    ).to(tl.float32)

    y = (centered * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gate_logits)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y, mask=valid)


@triton.jit
def _post_norm_gate_to_out_d128_kernel(
    tri_out,
    xh,
    out_gate_t,
    norm_w,
    norm_b,
    to_out_t,
    out,
    M: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    D: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)
    ds = tl.arange(0, BLOCK_D)

    row_ok = rows < M
    h_ok = hs < H
    d_ok = ds < D

    nn: tl.constexpr = N * N
    bidx = rows // nn
    pair = rows - bidx * nn

    tri_ptrs = tri_out + bidx[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    tri_t = tl.load(tri_ptrs, mask=h_ok[:, None] & row_ok[None, :], other=0.0).to(tl.float32)
    tri = tl.trans(tri_t)

    mean = tl.sum(tri, axis=1) / H
    centered = tl.where(h_ok[None, :], tri - mean[:, None], 0.0)
    var = tl.sum(centered * centered, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    nw = tl.load(norm_w + hs, mask=h_ok, other=0.0).to(tl.float32)
    nb = tl.load(norm_b + hs, mask=h_ok, other=0.0).to(tl.float32)
    normed = centered * inv[:, None] * nw[None, :] + nb[None, :]

    xvals = tl.load(
        xh + rows[:, None] * D + ds[None, :],
        mask=row_ok[:, None] & d_ok[None, :],
        other=0.0,
    )
    ogw = tl.load(
        out_gate_t + ds[:, None] * H + hs[None, :],
        mask=d_ok[:, None] & h_ok[None, :],
        other=0.0,
    )
    gate_logits = tl.dot(xvals, ogw)

    y = (normed * tl.sigmoid(gate_logits)).to(tl.float16)

    ow = tl.load(
        to_out_t + hs[:, None] * D + ds[None, :],
        mask=h_ok[:, None] & d_ok[None, :],
        other=0.0,
    )
    acc = tl.dot(y, ow)
    tl.store(out + rows[:, None] * D + ds[None, :], acc, mask=row_ok[:, None] & d_ok[None, :])


def _tensor_key(t):
    return (
        t.data_ptr(),
        tuple(t.shape),
        str(t.dtype),
        t.device.index if t.device.type == "cuda" else -1,
        getattr(t, "_version", 0),
    )


def _get_buffer(name, shape, device, dtype):
    key = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    buf = _BUFFER_CACHE.get(key)
    if buf is None:
        buf = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = buf
    return buf


def _get_packed_proj_t(weights, dtype, include_out_gate):
    if include_out_gate:
        names = (
            "left_proj.weight",
            "right_proj.weight",
            "left_gate.weight",
            "right_gate.weight",
            "out_gate.weight",
        )
    else:
        names = (
            "left_proj.weight",
            "right_proj.weight",
            "left_gate.weight",
            "right_gate.weight",
        )

    src = tuple(weights[n] for n in names)
    key = ("packed_proj_t", include_out_gate, dtype) + tuple(_tensor_key(t) for t in src)
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        packed_t = torch.cat(src, dim=0).to(dtype).t().contiguous()
        item = (src, packed_t)
        _WEIGHT_CACHE[key] = item
    return item[1]


def _get_transposed_weight(tag, w, dtype):
    key = (tag, dtype, _tensor_key(w))
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        wt = w.to(dtype).t().contiguous()
        item = (w, wt)
        _WEIGHT_CACHE[key] = item
    return item[1]


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is not False:
        try:
            out = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return out
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        torch.backends.cuda.matmul.allow_tf32 = True
        try:
            torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()

        xh = _get_buffer("x_norm_h", (M, D), device, half)
        bd = triton.next_power_of_2(D)
        bm_ln = 8 if D <= 128 else (4 if D <= 384 else 2)
        _layernorm_cast_kernel[(triton.cdiv(M, bm_ln),)](
            x,
            weights["norm.weight"],
            weights["norm.bias"],
            xh,
            M,
            D,
            eps,
            BLOCK_M=bm_ln,
            BLOCK_D=bd,
            num_warps=4 if D <= 128 else 8,
        )

        fused_d128 = D == 128 and H == 128
        proj_count = 4 if fused_d128 else 5
        packed_proj_t = _get_packed_proj_t(weights, half, include_out_gate=not fused_d128)
        proj = _get_buffer("proj4" if fused_d128 else "proj5", (M, proj_count * H), device, half)
        torch.mm(xh, packed_proj_t, out=proj)

        left = _get_buffer("left", (B * H, N, N), device, half)
        right = _get_buffer("right", (B * H, N, N), device, half)

        has_mask = mask is not None and not bool(config.get("nomask", False))
        mask_flat = mask.reshape(M).contiguous() if has_mask else proj

        bm_prep = 16
        bh_prep = 64 if H >= 64 else triton.next_power_of_2(H)
        _prep_projected_kernel[(triton.cdiv(M, bm_prep), triton.cdiv(H, bh_prep))](
            proj,
            mask_flat,
            left,
            right,
            M,
            N,
            H,
            proj_count,
            has_mask,
            BLOCK_M=bm_prep,
            BLOCK_H=bh_prep,
            num_warps=4,
        )

        tri = _get_buffer("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        if fused_d128:
            out = _get_buffer("out128", (M, D), device, torch.float32)
            out_gate_t = _get_transposed_weight("out_gate_t", weights["out_gate.weight"], half)
            to_out_t = _get_transposed_weight("to_out_t", weights["to_out.weight"], half)
            _post_norm_gate_to_out_d128_kernel[(triton.cdiv(M, 16),)](
                tri,
                xh,
                out_gate_t,
                weights["to_out_norm.weight"],
                weights["to_out_norm.bias"],
                to_out_t,
                out,
                M,
                N,
                H,
                D,
                eps,
                BLOCK_M=16,
                BLOCK_H=128,
                BLOCK_D=128,
                num_warps=4,
            )
            return out.reshape(B, N, N, D)

        tmp = _get_buffer("tmp", (M, H), device, half)
        bh_post = triton.next_power_of_2(H)
        _post_norm_gate_from_proj_kernel[(triton.cdiv(M, 16),)](
            tri,
            proj,
            weights["to_out_norm.weight"],
            weights["to_out_norm.bias"],
            tmp,
            M,
            N,
            H,
            proj_count,
            eps,
            BLOCK_M=16,
            BLOCK_H=bh_post,
            num_warps=8,
        )

        to_out_t = _get_transposed_weight("to_out_t", weights["to_out.weight"], half)
        out = _mm_half_to_float(tmp, to_out_t)
        return out.reshape(B, N, N, D)
