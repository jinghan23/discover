"""
TriMul outgoing forward using fused Triton staging around cuBLAS GEMMs:
input LayerNorm is cast to fp16, the five input projections are packed into
one GEMM, left/right gates and optional mask are materialized channel-major for
a strided batched N x N product, then output LayerNorm, out_gate, and the final
projection are fused where profitable. Weight packs and scratch buffers are
cached by tensor version and shape to avoid repeat allocation/transposition.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_CAP_CACHE = {}


@triton.jit
def _ln_cast_kernel(x, w, b, y,
                    M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
                    BM: tl.constexpr, BD: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BM + tl.arange(0, BM)
    ds = tl.arange(0, BD)
    m = (rows[:, None] < M) & (ds[None, :] < D)

    v = tl.load(x + rows[:, None] * D + ds[None, :], mask=m, other=0.).to(tl.float32)
    v = tl.where(ds[None, :] < D, v, 0.)
    mean = tl.sum(v, axis=1) / D
    c = tl.where(ds[None, :] < D, v - mean[:, None], 0.)
    var = tl.sum(c * c, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=ds < D, other=0.).to(tl.float32)
    bb = tl.load(b + ds, mask=ds < D, other=0.).to(tl.float32)
    tl.store(y + rows[:, None] * D + ds[None, :],
             c * inv[:, None] * ww[None, :] + bb[None, :], mask=m)


@triton.jit
def _prep_nomask_kernel(proj, left, right,
                        M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                        BM: tl.constexpr, BH: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)
    rows = pm * BM + tl.arange(0, BM)
    hs = ph * BH + tl.arange(0, BH)
    ok = (rows[:, None] < M) & (hs[None, :] < H)

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn
    base = rows[:, None] * (5 * H) + hs[None, :]
    dst = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    dm = (hs[:, None] < H) & (rows[None, :] < M)

    l = tl.load(proj + base, mask=ok, other=0.).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=ok, other=0.).to(tl.float32)
    r = tl.load(proj + base + H, mask=ok, other=0.).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=ok, other=0.).to(tl.float32)

    tl.store(left + dst, tl.trans(l * tl.sigmoid(lg)), mask=dm)
    tl.store(right + dst, tl.trans(r * tl.sigmoid(rg)), mask=dm)


@triton.jit
def _prep_mask_kernel(proj, mask, left, right,
                      M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                      BM: tl.constexpr, BH: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)
    rows = pm * BM + tl.arange(0, BM)
    hs = ph * BH + tl.arange(0, BH)
    ok = (rows[:, None] < M) & (hs[None, :] < H)

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn
    base = rows[:, None] * (5 * H) + hs[None, :]
    dst = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    dm = (hs[:, None] < H) & (rows[None, :] < M)
    mv = tl.load(mask + rows, mask=rows < M, other=0.).to(tl.float32)

    l = tl.load(proj + base, mask=ok, other=0.).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=ok, other=0.).to(tl.float32)
    r = tl.load(proj + base + H, mask=ok, other=0.).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=ok, other=0.).to(tl.float32)

    tl.store(left + dst, tl.trans(l * mv[:, None] * tl.sigmoid(lg)), mask=dm)
    tl.store(right + dst, tl.trans(r * mv[:, None] * tl.sigmoid(rg)), mask=dm)


@triton.jit
def _post_kernel(tri, proj, nw, nb, tmp,
                 M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, EPS: tl.constexpr,
                 BM: tl.constexpr, BH: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BM + tl.arange(0, BM)
    hs = tl.arange(0, BH)

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    ptr = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    t = tl.load(ptr, mask=(hs[:, None] < H) & (rows[None, :] < M), other=0.).to(tl.float32)
    x = tl.trans(t)

    mean = tl.sum(x, axis=1) / H
    c = tl.where(hs[None, :] < H, x - mean[:, None], 0.)
    var = tl.sum(c * c, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(nw + hs, mask=hs < H, other=0.).to(tl.float32)
    bb = tl.load(nb + hs, mask=hs < H, other=0.).to(tl.float32)
    gate = tl.load(proj + rows[:, None] * (5 * H) + 4 * H + hs[None, :],
                   mask=(rows[:, None] < M) & (hs[None, :] < H), other=0.).to(tl.float32)

    y = (c * inv[:, None] * w[None, :] + bb[None, :]) * tl.sigmoid(gate)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y,
             mask=(rows[:, None] < M) & (hs[None, :] < H))


@triton.jit
def _post_final_128_kernel(tri, proj, nw, nb, ow, out,
                           M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                           EPS: tl.constexpr, BM: tl.constexpr,
                           BH: tl.constexpr, BD: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_d = tl.program_id(1)
    rows = pid_m * BM + tl.arange(0, BM)
    ds = pid_d * BD + tl.arange(0, BD)
    hs = tl.arange(0, BH)

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn
    tri_ptr = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    t = tl.load(tri_ptr, mask=(hs[:, None] < H) & (rows[None, :] < M), other=0.).to(tl.float32)
    x = tl.trans(t)

    mean = tl.sum(x, axis=1) / H
    c = tl.where(hs[None, :] < H, x - mean[:, None], 0.)
    var = tl.sum(c * c, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(nw + hs, mask=hs < H, other=0.).to(tl.float32)
    bb = tl.load(nb + hs, mask=hs < H, other=0.).to(tl.float32)
    gate = tl.load(proj + rows[:, None] * (5 * H) + 4 * H + hs[None, :],
                   mask=(rows[:, None] < M) & (hs[None, :] < H), other=0.).to(tl.float32)
    y = (c * inv[:, None] * w[None, :] + bb[None, :]) * tl.sigmoid(gate)

    owv = tl.load(ow + hs[:, None] * H + ds[None, :],
                  mask=(hs[:, None] < H) & (ds[None, :] < H), other=0.).to(tl.float32)
    acc = tl.dot(y.to(tl.float16), owv.to(tl.float16), out_dtype=tl.float32)
    tl.store(out + rows[:, None] * H + ds[None, :], acc,
             mask=(rows[:, None] < M) & (ds[None, :] < H))


def _key(t):
    return (t.data_ptr(), tuple(t.shape), str(t.dtype),
            t.device.index if t.device.type == "cuda" else -1, getattr(t, "_version", 0))


def _buf(name, shape, device, dtype):
    key = (name, tuple(shape), device.index, dtype)
    v = _BUFFER_CACHE.get(key)
    if v is None:
        v = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = v
    return v


def _packed_proj_t(weights, dtype):
    names = ("left_proj.weight", "right_proj.weight", "left_gate.weight",
             "right_gate.weight", "out_gate.weight")
    src = tuple(weights[n] for n in names)
    key = ("proj", dtype) + tuple(_key(t) for t in src)
    v = _WEIGHT_CACHE.get(key)
    if v is None:
        v = torch.cat(src, 0).to(dtype).t().contiguous()
        _WEIGHT_CACHE[key] = v
    return v


def _to_out_t(w, dtype):
    key = ("out_t", dtype, _key(w))
    v = _WEIGHT_CACHE.get(key)
    if v is None:
        v = w.to(dtype).t().contiguous()
        _WEIGHT_CACHE[key] = v
    return v


def _mm(a, b, out=None):
    cap = _CAP_CACHE
    if out is not None and cap.get("mm_out_dtype_out", True):
        try:
            return torch.mm(a, b, out=out, out_dtype=torch.float32)
        except Exception:
            cap["mm_out_dtype_out"] = False
    if cap.get("mm_out_dtype", True):
        try:
            return torch.mm(a, b, out_dtype=torch.float32)
        except Exception:
            cap["mm_out_dtype"] = False
    r = torch.mm(a, b)
    return r if r.dtype is torch.float32 else r.to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    with torch.no_grad():
        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        device = input_tensor.device
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        half = torch.float16

        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        try:
            torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()

        xh = _buf("x_norm_h", (M, D), device, half)
        bd = triton.next_power_of_2(D)
        bm_ln = 8 if D <= 128 else 4
        _ln_cast_kernel[(triton.cdiv(M, bm_ln),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, D, eps, BM=bm_ln, BD=bd, num_warps=4 if D <= 128 else 8
        )

        proj = _buf("proj", (M, 5 * H), device, half)
        torch.mm(xh, _packed_proj_t(weights, half), out=proj)

        left = _buf("left", (B * H, N, N), device, half)
        right = _buf("right", (B * H, N, N), device, half)

        has_mask = mask is not None and not bool(config.get("nomask", False))
        if has_mask:
            mf = mask.reshape(M).contiguous()
            _prep_mask_kernel[(triton.cdiv(M, 16), triton.cdiv(H, 64))](
                proj, mf, left, right, M, N, H, BM=16, BH=64, num_warps=4
            )
        else:
            _prep_nomask_kernel[(triton.cdiv(M, 16), triton.cdiv(H, 64))](
                proj, left, right, M, N, H, BM=16, BH=64, num_warps=4
            )

        tri = _buf("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        out_buf = _buf("out_f32", (M, D), device, torch.float32)
        if D == H:
            _post_final_128_kernel[(triton.cdiv(M, 16), triton.cdiv(D, 32))](
                tri, proj, weights["to_out_norm.weight"], weights["to_out_norm.bias"],
                _to_out_t(weights["to_out.weight"], half), out_buf,
                M, N, H, eps, BM=16, BH=128, BD=32, num_warps=8
            )
            return out_buf.reshape(B, N, N, D)

        tmp = _buf("tmp", (M, H), device, half)
        _post_kernel[(triton.cdiv(M, 16),)](
            tri, proj, weights["to_out_norm.weight"], weights["to_out_norm.bias"],
            tmp, M, N, H, eps, BM=16, BH=128, num_warps=8
        )
        out = _mm(tmp, _to_out_t(weights["to_out.weight"], half), out_buf)
        return out.reshape(B, N, N, D)
