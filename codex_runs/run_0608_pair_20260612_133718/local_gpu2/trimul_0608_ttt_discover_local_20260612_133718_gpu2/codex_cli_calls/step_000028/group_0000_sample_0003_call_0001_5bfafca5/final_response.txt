"""
TriMul outgoing forward: fuse input LayerNorm and all five input projections,
materialize gated left/right in channel-major layout for fast grouped GEMM over
the k dimension, then fuse output LayerNorm with out_gate before final projection.
Scratch tensors and packed/transposed weights are cached per device and version.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_MM_OUT_DTYPE_OK = None


@triton.jit
def _ln_cast_kernel(x, w, b, y,
                    M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
                    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, BLOCK_D)

    rm = rows < M
    dm = ds < D
    mask = rm[:, None] & dm[None, :]

    v = tl.load(x + rows[:, None] * D + ds[None, :], mask=mask, other=0.).to(tl.float32)
    v = tl.where(dm[None, :], v, 0.)
    mean = tl.sum(v, axis=1) / D
    c = tl.where(dm[None, :], v - mean[:, None], 0.)
    var = tl.sum(c * c, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=dm, other=0.).to(tl.float32)
    bb = tl.load(b + ds, mask=dm, other=0.).to(tl.float32)
    tl.store(y + rows[:, None] * D + ds[None, :],
             c * inv[:, None] * ww[None, :] + bb[None, :],
             mask=mask)


@triton.jit
def _prep_nomask_kernel(proj, left, right,
                        M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                        BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = ph * BLOCK_H + tl.arange(0, BLOCK_H)
    rm = rows < M
    hm = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    in_ptr = rows[:, None] * (5 * H) + hs[None, :]
    out_ptr = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]

    valid = rm[:, None] & hm[None, :]
    out_mask = hm[:, None] & rm[None, :]

    l = tl.load(proj + in_ptr, mask=valid, other=0.).to(tl.float32)
    r = tl.load(proj + in_ptr + H, mask=valid, other=0.).to(tl.float32)
    lg = tl.load(proj + in_ptr + 2 * H, mask=valid, other=0.).to(tl.float32)
    rg = tl.load(proj + in_ptr + 3 * H, mask=valid, other=0.).to(tl.float32)

    tl.store(left + out_ptr, tl.trans(l * tl.sigmoid(lg)), mask=out_mask)
    tl.store(right + out_ptr, tl.trans(r * tl.sigmoid(rg)), mask=out_mask)


@triton.jit
def _prep_mask_kernel(proj, mask, left, right,
                      M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                      BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pm = tl.program_id(0)
    ph = tl.program_id(1)

    rows = pm * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = ph * BLOCK_H + tl.arange(0, BLOCK_H)
    rm = rows < M
    hm = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    in_ptr = rows[:, None] * (5 * H) + hs[None, :]
    out_ptr = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]

    valid = rm[:, None] & hm[None, :]
    out_mask = hm[:, None] & rm[None, :]

    mv = tl.load(mask + rows, mask=rm, other=0.).to(tl.float32)
    l = tl.load(proj + in_ptr, mask=valid, other=0.).to(tl.float32)
    r = tl.load(proj + in_ptr + H, mask=valid, other=0.).to(tl.float32)
    lg = tl.load(proj + in_ptr + 2 * H, mask=valid, other=0.).to(tl.float32)
    rg = tl.load(proj + in_ptr + 3 * H, mask=valid, other=0.).to(tl.float32)

    tl.store(left + out_ptr, tl.trans(l * mv[:, None] * tl.sigmoid(lg)), mask=out_mask)
    tl.store(right + out_ptr, tl.trans(r * mv[:, None] * tl.sigmoid(rg)), mask=out_mask)


@triton.jit
def _post_kernel(tri, proj, nw, nb, tmp,
                 M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, EPS: tl.constexpr,
                 BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    rm = rows < M
    hm = hs < H
    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    ptr = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(ptr, mask=hm[:, None] & rm[None, :], other=0.).to(tl.float32))

    mean = tl.sum(x, axis=1) / H
    c = tl.where(hm[None, :], x - mean[:, None], 0.)
    var = tl.sum(c * c, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(nw + hs, mask=hm, other=0.).to(tl.float32)
    b0 = tl.load(nb + hs, mask=hm, other=0.).to(tl.float32)
    valid = rm[:, None] & hm[None, :]
    g = tl.load(proj + rows[:, None] * (5 * H) + 4 * H + hs[None, :],
                mask=valid, other=0.).to(tl.float32)

    y = (c * inv[:, None] * w[None, :] + b0[None, :]) * tl.sigmoid(g)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y, mask=valid)


def _key(t):
    return (t.data_ptr(), tuple(t.shape), str(t.dtype),
            t.device.index if t.device.type == "cuda" else -1,
            getattr(t, "_version", 0))


def _buf(name, shape, device, dtype):
    k = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    v = _BUFFER_CACHE.get(k)
    if v is None:
        v = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[k] = v
    return v


def _packed_proj_t(weights, dtype):
    names = ("left_proj.weight", "right_proj.weight", "left_gate.weight",
             "right_gate.weight", "out_gate.weight")
    src = tuple(weights[n] for n in names)
    k = ("packed_proj_t", dtype) + tuple(_key(t) for t in src)
    v = _WEIGHT_CACHE.get(k)
    if v is None:
        v = torch.cat(src, dim=0).to(dtype).t().contiguous()
        _WEIGHT_CACHE[k] = v
    return v


def _to_out_t(w, dtype):
    k = ("to_out_t", dtype, _key(w))
    v = _WEIGHT_CACHE.get(k)
    if v is None:
        v = w.to(dtype).t().contiguous()
        _WEIGHT_CACHE[k] = v
    return v


def _mm_float(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is not False:
        try:
            y = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return y
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        try:
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()

        xh = _buf("x_norm_h", (M, D), device, half)
        bd = triton.next_power_of_2(D)
        bm_ln = 8 if D <= 128 else 4
        _ln_cast_kernel[(triton.cdiv(M, bm_ln),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, D, eps,
            BLOCK_M=bm_ln, BLOCK_D=bd,
            num_warps=4 if D <= 128 else 8,
        )

        proj = _buf("proj", (M, 5 * H), device, half)
        torch.mm(xh, _packed_proj_t(weights, half), out=proj)

        left = _buf("left", (B * H, N, N), device, half)
        right = _buf("right", (B * H, N, N), device, half)

        has_mask = mask is not None and not bool(config.get("nomask", False))
        if has_mask:
            mask_flat = mask.reshape(M).contiguous()
            _prep_mask_kernel[(triton.cdiv(M, 16), triton.cdiv(H, 64))](
                proj, mask_flat, left, right, M, N, H,
                BLOCK_M=16, BLOCK_H=64, num_warps=4,
            )
        else:
            _prep_nomask_kernel[(triton.cdiv(M, 16), triton.cdiv(H, 64))](
                proj, left, right, M, N, H,
                BLOCK_M=16, BLOCK_H=64, num_warps=4,
            )

        tri = _buf("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        tmp = _buf("tmp", (M, H), device, half)
        _post_kernel[(triton.cdiv(M, 16),)](
            tri, proj, weights["to_out_norm.weight"], weights["to_out_norm.bias"], tmp,
            M, N, H, eps,
            BLOCK_M=16, BLOCK_H=triton.next_power_of_2(H),
            num_warps=8,
        )

        out = _mm_float(tmp, _to_out_t(weights["to_out.weight"], half))
        return out.reshape(B, N, N, D)
