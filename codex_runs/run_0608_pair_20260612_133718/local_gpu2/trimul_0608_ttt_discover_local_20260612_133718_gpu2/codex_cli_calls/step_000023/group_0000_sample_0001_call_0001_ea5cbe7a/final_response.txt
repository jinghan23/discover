"""
TriMul outgoing forward with fused Triton preprocessing/postprocessing around
tensor-core GEMMs. The input LayerNorm is fused to fp16, all five input
projections are packed into one GEMM, gated left/right activations are stored
channel-major for strided batched GEMM over k, and output LayerNorm plus
out_gate are fused before the final projection.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_MM_OUT_DTYPE_OK = None
_MM_OUT_DTYPE_WITH_OUT_OK = None


@triton.jit
def _ln_cast_kernel(x, w, b, y, M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
                    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, BLOCK_D)
    mask = (rows[:, None] < M) & (ds[None, :] < D)

    v = tl.load(x + rows[:, None] * D + ds[None, :], mask=mask, other=0.0).to(tl.float32)
    v = tl.where(ds[None, :] < D, v, 0.0)
    mean = tl.sum(v, axis=1) / D
    c = tl.where(ds[None, :] < D, v - mean[:, None], 0.0)
    var = tl.sum(c * c, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=ds < D, other=0.0).to(tl.float32)
    bb = tl.load(b + ds, mask=ds < D, other=0.0).to(tl.float32)
    tl.store(y + rows[:, None] * D + ds[None, :],
             c * inv[:, None] * ww[None, :] + bb[None, :],
             mask=mask)


@triton.jit
def _prep_nomask_kernel(proj, left, right, M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                        BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * (5 * H) + hs[None, :]
    dst = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    dst_mask = h_ok[:, None] & row_ok[None, :]

    lp = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)

    tl.store(left + dst, tl.trans(lp * tl.sigmoid(lg)), mask=dst_mask)
    tl.store(right + dst, tl.trans(rp * tl.sigmoid(rg)), mask=dst_mask)


@triton.jit
def _prep_mask_kernel(proj, mask, left, right, M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                      BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * (5 * H) + hs[None, :]
    dst = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    dst_mask = h_ok[:, None] & row_ok[None, :]
    mv = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    lp = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)

    tl.store(left + dst, tl.trans(lp * mv[:, None] * tl.sigmoid(lg)), mask=dst_mask)
    tl.store(right + dst, tl.trans(rp * mv[:, None] * tl.sigmoid(rg)), mask=dst_mask)


@triton.jit
def _post_kernel(tri, proj, nw, nb, tmp, M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
                 EPS: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)
    row_ok = rows < M
    h_ok = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    ptr = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    xt = tl.load(ptr, mask=h_ok[:, None] & row_ok[None, :], other=0.0).to(tl.float32)
    x = tl.trans(xt)

    mean = tl.sum(x, axis=1) / H
    c = tl.where(h_ok[None, :], x - mean[:, None], 0.0)
    var = tl.sum(c * c, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(nw + hs, mask=h_ok, other=0.0).to(tl.float32)
    bias = tl.load(nb + hs, mask=h_ok, other=0.0).to(tl.float32)

    valid = row_ok[:, None] & h_ok[None, :]
    gate = tl.load(proj + rows[:, None] * (5 * H) + 4 * H + hs[None, :],
                   mask=valid, other=0.0).to(tl.float32)

    y = (c * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gate)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y, mask=valid)


def _tensor_key(t):
    return (t.data_ptr(), tuple(t.shape), str(t.dtype),
            t.device.index if t.device.type == "cuda" else -1, getattr(t, "_version", 0))


def _buf(name, shape, device, dtype):
    key = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    out = _BUFFER_CACHE.get(key)
    if out is None:
        out = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = out
    return out


def _packed_proj_t(weights, dtype):
    names = ("left_proj.weight", "right_proj.weight", "left_gate.weight",
             "right_gate.weight", "out_gate.weight")
    src = tuple(weights[n] for n in names)
    key = ("packed_proj_t", dtype) + tuple(_tensor_key(t) for t in src)
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = torch.cat(src, dim=0).to(dtype).t().contiguous()
        _WEIGHT_CACHE[key] = item
    return item


def _to_out_t(w, dtype):
    key = ("to_out_t", dtype, _tensor_key(w))
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = w.to(dtype).t().contiguous()
        _WEIGHT_CACHE[key] = item
    return item


def _mm_f32(a, b, out=None):
    global _MM_OUT_DTYPE_OK, _MM_OUT_DTYPE_WITH_OUT_OK
    if out is not None and _MM_OUT_DTYPE_WITH_OUT_OK is not False:
        try:
            torch.mm(a, b, out=out, out_dtype=torch.float32)
            _MM_OUT_DTYPE_WITH_OUT_OK = True
            return out
        except (TypeError, RuntimeError):
            _MM_OUT_DTYPE_WITH_OUT_OK = False
    if _MM_OUT_DTYPE_OK is not False:
        try:
            r = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return r
        except (TypeError, RuntimeError):
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    with torch.no_grad():
        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        torch.backends.cuda.matmul.allow_tf32 = True
        torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        try:
            torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()

        xh = _buf("x_norm_h", (M, D), device, half)
        bd = triton.next_power_of_2(D)
        bm_ln = 8 if D <= 128 else 4
        _ln_cast_kernel[(triton.cdiv(M, bm_ln),)](
            x, weights["norm.weight"], weights["norm.bias"], xh,
            M, D, eps, BLOCK_M=bm_ln, BLOCK_D=bd,
            num_warps=4 if D <= 128 else 8,
        )

        proj = _buf("proj", (M, 5 * H), device, half)
        torch.mm(xh, _packed_proj_t(weights, half), out=proj)

        left = _buf("left", (B * H, N, N), device, half)
        right = _buf("right", (B * H, N, N), device, half)

        bm_prep = 16
        bh_prep = 64
        has_mask = mask is not None and not bool(config.get("nomask", False))
        if has_mask:
            mask_flat = mask.reshape(M).contiguous()
            _prep_mask_kernel[(triton.cdiv(M, bm_prep), triton.cdiv(H, bh_prep))](
                proj, mask_flat, left, right, M, N, H,
                BLOCK_M=bm_prep, BLOCK_H=bh_prep, num_warps=4,
            )
        else:
            _prep_nomask_kernel[(triton.cdiv(M, bm_prep), triton.cdiv(H, bh_prep))](
                proj, left, right, M, N, H,
                BLOCK_M=bm_prep, BLOCK_H=bh_prep, num_warps=4,
            )

        tri = _buf("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        tmp = xh if D == H else _buf("tmp", (M, H), device, half)
        _post_kernel[(triton.cdiv(M, 16),)](
            tri, proj, weights["to_out_norm.weight"], weights["to_out_norm.bias"], tmp,
            M, N, H, eps, BLOCK_M=16, BLOCK_H=triton.next_power_of_2(H), num_warps=8,
        )

        out_buf = _buf("out_f32", (M, D), device, torch.float32)
        out = _mm_f32(tmp, _to_out_t(weights["to_out.weight"], half), out_buf)
        return out.reshape(B, N, N, D)
