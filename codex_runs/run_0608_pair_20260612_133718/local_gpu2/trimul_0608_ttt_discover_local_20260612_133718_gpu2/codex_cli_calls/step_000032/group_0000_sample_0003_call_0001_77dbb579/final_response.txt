"""
TriMul outgoing forward: Triton LayerNorm kernels materialize fp16 normalized inputs,
a packed cuBLAS GEMM computes all projections/gates, Triton applies masks and gates
into channel-major operands, cuBLAS strided batched GEMM performs the N x N x N
triangle contraction, then Triton fuses output LayerNorm with output gating before
the final cuBLAS projection to float32.
"""

import torch
import triton
import triton.language as tl


_WEIGHT_CACHE = {}
_BUFFER_CACHE = {}
_BACKENDS_READY = False
_MM_OUT_DTYPE_OK = None


@triton.jit
def _layernorm128_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    row_ok = rows < M

    vals = tl.load(x + rows[:, None] * 128 + ds[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)
    mean = tl.sum(vals, axis=1) * 0.0078125
    cen = vals - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds).to(tl.float32)
    bb = tl.load(b + ds).to(tl.float32)
    out = cen * inv[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * 128 + ds[None, :], out, mask=row_ok[:, None])


@triton.jit
def _layernorm384_kernel(x, w, b, y, M: tl.constexpr, EPS: tl.constexpr, BLOCK_M: tl.constexpr):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, 128)
    row_ok = rows < M
    base = rows[:, None] * 384 + ds[None, :]

    v0 = tl.load(x + base, mask=row_ok[:, None], other=0.0).to(tl.float32)
    v1 = tl.load(x + base + 128, mask=row_ok[:, None], other=0.0).to(tl.float32)
    v2 = tl.load(x + base + 256, mask=row_ok[:, None], other=0.0).to(tl.float32)

    mean = (tl.sum(v0, axis=1) + tl.sum(v1, axis=1) + tl.sum(v2, axis=1)) * 0.0026041666666666665
    c0 = v0 - mean[:, None]
    c1 = v1 - mean[:, None]
    c2 = v2 - mean[:, None]
    var = (tl.sum(c0 * c0, axis=1) + tl.sum(c1 * c1, axis=1) + tl.sum(c2 * c2, axis=1)) * 0.0026041666666666665
    inv = tl.rsqrt(var + EPS)

    w0 = tl.load(w + ds).to(tl.float32)
    w1 = tl.load(w + ds + 128).to(tl.float32)
    w2 = tl.load(w + ds + 256).to(tl.float32)
    b0 = tl.load(b + ds).to(tl.float32)
    b1 = tl.load(b + ds + 128).to(tl.float32)
    b2 = tl.load(b + ds + 256).to(tl.float32)

    tl.store(y + base, c0 * inv[:, None] * w0[None, :] + b0[None, :], mask=row_ok[:, None])
    tl.store(y + base + 128, c1 * inv[:, None] * w1[None, :] + b1[None, :], mask=row_ok[:, None])
    tl.store(y + base + 256, c2 * inv[:, None] * w2[None, :] + b2[None, :], mask=row_ok[:, None])


@triton.jit
def _layernorm_generic_kernel(
    x, w, b, y,
    M: tl.constexpr, D: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    ds = tl.arange(0, BLOCK_D)

    row_ok = rows < M
    d_ok = ds < D
    mask = row_ok[:, None] & d_ok[None, :]

    vals = tl.load(x + rows[:, None] * D + ds[None, :], mask=mask, other=0.0).to(tl.float32)
    vals = tl.where(d_ok[None, :], vals, 0.0)

    mean = tl.sum(vals, axis=1) / D
    cen = tl.where(d_ok[None, :], vals - mean[:, None], 0.0)
    var = tl.sum(cen * cen, axis=1) / D
    inv = tl.rsqrt(var + EPS)

    ww = tl.load(w + ds, mask=d_ok, other=0.0).to(tl.float32)
    bb = tl.load(b + ds, mask=d_ok, other=0.0).to(tl.float32)
    out = cen * inv[:, None] * ww[None, :] + bb[None, :]
    tl.store(y + rows[:, None] * D + ds[None, :], out, mask=mask)


@triton.jit
def _prep128_nomask_kernel(
    proj, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * 640 + hs[None, :]
    dst = b[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]

    left = tl.load(proj + base, mask=row_ok[:, None], other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 256, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(left_out + dst, tl.trans(left * tl.sigmoid(lg)), mask=row_ok[None, :])

    right = tl.load(proj + base + 128, mask=row_ok[:, None], other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 384, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(right_out + dst, tl.trans(right * tl.sigmoid(rg)), mask=row_ok[None, :])


@triton.jit
def _prep128_mask_kernel(
    proj, mask, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * 640 + hs[None, :]
    dst = b[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    mv = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    left = tl.load(proj + base, mask=row_ok[:, None], other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 256, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(left_out + dst, tl.trans(left * mv[:, None] * tl.sigmoid(lg)), mask=row_ok[None, :])

    right = tl.load(proj + base + 128, mask=row_ok[:, None], other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 384, mask=row_ok[:, None], other=0.0).to(tl.float32)
    tl.store(right_out + dst, tl.trans(right * mv[:, None] * tl.sigmoid(rg)), mask=row_ok[None, :])


@triton.jit
def _prep_nomask_kernel(
    proj, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * (5 * H) + hs[None, :]
    dst = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    dst_mask = h_ok[:, None] & row_ok[None, :]

    left = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(left_out + dst, tl.trans(left * tl.sigmoid(lg)), mask=dst_mask)

    right = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(right_out + dst, tl.trans(right * tl.sigmoid(rg)), mask=dst_mask)


@triton.jit
def _prep_mask_kernel(
    proj, mask, left_out, right_out,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    rows = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H
    valid = row_ok[:, None] & h_ok[None, :]

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    base = rows[:, None] * (5 * H) + hs[None, :]
    dst = b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    dst_mask = h_ok[:, None] & row_ok[None, :]

    mv = tl.load(mask + rows, mask=row_ok, other=0.0).to(tl.float32)

    left = tl.load(proj + base, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(proj + base + 2 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(left_out + dst, tl.trans(left * mv[:, None] * tl.sigmoid(lg)), mask=dst_mask)

    right = tl.load(proj + base + H, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(proj + base + 3 * H, mask=valid, other=0.0).to(tl.float32)
    tl.store(right_out + dst, tl.trans(right * mv[:, None] * tl.sigmoid(rg)), mask=dst_mask)


@triton.jit
def _post_norm_gate128_kernel(
    tri, proj, norm_w, norm_b, tmp,
    M: tl.constexpr, N: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, 128)
    row_ok = rows < M

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    tri_ptrs = tri + b[None, :] * (128 * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(tri_ptrs, mask=row_ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) * 0.0078125
    cen = x - mean[:, None]
    var = tl.sum(cen * cen, axis=1) * 0.0078125
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs).to(tl.float32)
    bias = tl.load(norm_b + hs).to(tl.float32)
    gate = tl.load(proj + rows[:, None] * 640 + 512 + hs[None, :], mask=row_ok[:, None], other=0.0).to(tl.float32)

    y = (cen * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gate)
    tl.store(tmp + rows[:, None] * 128 + hs[None, :], y, mask=row_ok[:, None])


@triton.jit
def _post_norm_gate_kernel(
    tri, proj, norm_w, norm_b, tmp,
    M: tl.constexpr, N: tl.constexpr, H: tl.constexpr, EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    hs = tl.arange(0, BLOCK_H)

    row_ok = rows < M
    h_ok = hs < H

    nn: tl.constexpr = N * N
    b = rows // nn
    pair = rows - b * nn

    tri_ptrs = tri + b[None, :] * (H * nn) + hs[:, None] * nn + pair[None, :]
    x = tl.trans(tl.load(tri_ptrs, mask=h_ok[:, None] & row_ok[None, :], other=0.0).to(tl.float32))

    mean = tl.sum(x, axis=1) / H
    cen = tl.where(h_ok[None, :], x - mean[:, None], 0.0)
    var = tl.sum(cen * cen, axis=1) / H
    inv = tl.rsqrt(var + EPS)

    w = tl.load(norm_w + hs, mask=h_ok, other=0.0).to(tl.float32)
    bias = tl.load(norm_b + hs, mask=h_ok, other=0.0).to(tl.float32)
    valid = row_ok[:, None] & h_ok[None, :]
    gate = tl.load(proj + rows[:, None] * (5 * H) + 4 * H + hs[None, :], mask=valid, other=0.0).to(tl.float32)

    y = (cen * inv[:, None] * w[None, :] + bias[None, :]) * tl.sigmoid(gate)
    tl.store(tmp + rows[:, None] * H + hs[None, :], y, mask=valid)


def _init_backends():
    global _BACKENDS_READY
    if not _BACKENDS_READY:
        torch.backends.cuda.matmul.allow_tf32 = True
        try:
            torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
        except Exception:
            pass
        try:
            torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
        except Exception:
            pass
        try:
            torch.set_float32_matmul_precision("high")
        except Exception:
            pass
        _BACKENDS_READY = True


def _tensor_key(t):
    try:
        version = t._version
    except Exception:
        version = 0
    return (
        t.data_ptr(),
        tuple(t.shape),
        str(t.dtype),
        t.device.index if t.device.type == "cuda" else -1,
        version,
    )


def _get_buffer(name, shape, device, dtype):
    key = (name, tuple(shape), device.index if device.type == "cuda" else -1, dtype)
    buf = _BUFFER_CACHE.get(key)
    if buf is None:
        buf = torch.empty(shape, device=device, dtype=dtype)
        _BUFFER_CACHE[key] = buf
    return buf


def _get_packed_proj_t(weights, dtype):
    names = (
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
    )
    src = tuple(weights[n] for n in names)
    key = ("packed_proj_t", dtype) + tuple(_tensor_key(t) for t in src)
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = (src, torch.cat(src, dim=0).to(dtype).t().contiguous())
        _WEIGHT_CACHE[key] = item
    return item[1]


def _get_to_out_t(w, dtype):
    key = ("to_out_t", dtype, _tensor_key(w))
    item = _WEIGHT_CACHE.get(key)
    if item is None:
        item = (w, w.to(dtype).t().contiguous())
        _WEIGHT_CACHE[key] = item
    return item[1]


def _mm_half_to_float(a, b):
    global _MM_OUT_DTYPE_OK
    if _MM_OUT_DTYPE_OK is not False:
        try:
            out = torch.mm(a, b, out_dtype=torch.float32)
            _MM_OUT_DTYPE_OK = True
            return out
        except TypeError:
            _MM_OUT_DTYPE_OK = False
    return torch.mm(a, b).to(torch.float32)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    with torch.no_grad():
        _init_backends()

        B, N, _, D = input_tensor.shape
        H = int(config["hidden_dim"])
        M = B * N * N
        eps = float(config.get("eps", config.get("norm_eps", 1e-5)))
        device = input_tensor.device
        half = torch.float16

        x = input_tensor if input_tensor.is_contiguous() else input_tensor.contiguous()

        xh = _get_buffer("x_norm_h", (M, D), device, half)
        if D == 128:
            _layernorm128_kernel[(triton.cdiv(M, 8),)](
                x, weights["norm.weight"], weights["norm.bias"], xh,
                M, eps, BLOCK_M=8, num_warps=4,
            )
        elif D == 384:
            _layernorm384_kernel[(triton.cdiv(M, 4),)](
                x, weights["norm.weight"], weights["norm.bias"], xh,
                M, eps, BLOCK_M=4, num_warps=4,
            )
        else:
            bd = triton.next_power_of_2(D)
            bm_ln = 2 if D >= 768 else 4
            _layernorm_generic_kernel[(triton.cdiv(M, bm_ln),)](
                x, weights["norm.weight"], weights["norm.bias"], xh,
                M, D, eps,
                BLOCK_M=bm_ln,
                BLOCK_D=bd,
                num_warps=8,
            )

        packed_proj_t = _get_packed_proj_t(weights, half)
        proj = _get_buffer("proj", (M, 5 * H), device, half)
        torch.mm(xh, packed_proj_t, out=proj)

        left = _get_buffer("left", (B * H, N, N), device, half)
        right = _get_buffer("right", (B * H, N, N), device, half)

        has_mask = mask is not None and not bool(config.get("nomask", False))
        if H == 128:
            if has_mask:
                mask_flat = mask.reshape(M)
                _prep128_mask_kernel[(triton.cdiv(M, 8),)](
                    proj, mask_flat, left, right,
                    M, N,
                    BLOCK_M=8,
                    num_warps=4,
                )
            else:
                _prep128_nomask_kernel[(triton.cdiv(M, 8),)](
                    proj, left, right,
                    M, N,
                    BLOCK_M=8,
                    num_warps=4,
                )
        else:
            if has_mask:
                mask_flat = mask.reshape(M).contiguous()
                _prep_mask_kernel[(triton.cdiv(M, 16), triton.cdiv(H, 64))](
                    proj, mask_flat, left, right,
                    M, N, H,
                    BLOCK_M=16,
                    BLOCK_H=64,
                    num_warps=4,
                )
            else:
                _prep_nomask_kernel[(triton.cdiv(M, 16), triton.cdiv(H, 64))](
                    proj, left, right,
                    M, N, H,
                    BLOCK_M=16,
                    BLOCK_H=64,
                    num_warps=4,
                )

        tri = _get_buffer("tri", (B * H, N, N), device, half)
        torch.bmm(left, right.transpose(1, 2), out=tri)

        tmp = _get_buffer("tmp", (M, H), device, half)
        if H == 128:
            _post_norm_gate128_kernel[(triton.cdiv(M, 16),)](
                tri, proj,
                weights["to_out_norm.weight"],
                weights["to_out_norm.bias"],
                tmp,
                M, N, eps,
                BLOCK_M=16,
                num_warps=8,
            )
        else:
            _post_norm_gate_kernel[(triton.cdiv(M, 16),)](
                tri, proj,
                weights["to_out_norm.weight"],
                weights["to_out_norm.bias"],
                tmp,
                M, N, H, eps,
                BLOCK_M=16,
                BLOCK_H=triton.next_power_of_2(H),
                num_warps=8,
            )

        to_out_t = _get_to_out_t(weights["to_out.weight"], half)
        out = _mm_half_to_float(tmp, to_out_t)
        return out.reshape(B, N, N, D)
